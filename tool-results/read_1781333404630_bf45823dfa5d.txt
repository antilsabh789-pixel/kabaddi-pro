    70→function getPositionCategory(position: string | null): 'raider' | 'defender' | 'all-rounder' | 'unknown' {
    71→  if (!position) return 'unknown';
    72→  const p = position.toLowerCase();
    73→  if (p.includes('raider')) return 'raider';
    74→  if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return 'defender';
    75→  if (p.includes('all')) return 'all-rounder';
    76→  return 'unknown';
    77→}

    81→function LineupValidation({ lineup, playersPerSide, teamName }: {
    82→  lineup: MatchPlayer[];
    83→  playersPerSide: number;
    84→  teamName: string;
    85→}) {
    86→  const warnings: { message: string; severity: 'error' | 'warning' | 'info' }[] = [];
    87→
    88→  if (lineup.length === 0) {
    89→    warnings.push({ message: 'No players added', severity: 'error' });
    90→  } else if (lineup.length < playersPerSide) {
    91→    warnings.push({
    92→      message: `${lineup.length}/${playersPerSide} players — need ${playersPerSide - lineup.length} more`,
    93→      severity: 'warning',
    94→    });
    95→  } else if (lineup.length > playersPerSide) {
    96→    warnings.push({
    97→      message: `${lineup.length} players — ${lineup.length - playersPerSide} extra`,
    98→      severity: 'info',
    99→    });
   100→  }
   101→
   102→  // Position balance check (only if we have position data from DB)
   103→  const balance = POSITION_BALANCE[playersPerSide];
   104→  // Note: position data might not be available for quick-added players
   105→  // We check this based on the allPlayers data which is in parent scope
   106→
   107→  if (warnings.length === 0) return null;
   108→
   109→  return (
   110→    <motion.div
   111→      initial={{ opacity: 0, y: -5 }}
   112→      animate={{ opacity: 1, y: 0 }}
   113→      className="space-y-1.5"
   114→    >
   115→      {warnings.map((w, i) => (
   116→        <div
   117→          key={i}
   118→          className={`flex items-center gap-2 px-3 py-2 rounded-lg text-xs font-medium ${
   119→            w.severity === 'error'
   120→              ? 'bg-brand-red/10 text-brand-red dark:bg-brand-red/15'
   121→              : w.severity === 'warning'
   122→                ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400 dark:bg-amber-500/15'
   123→                : 'bg-blue-500/10 text-blue-600 dark:text-blue-400 dark:bg-blue-500/15'
   124→          }`}
   125→        >
   126→          <AlertTriangle className={`w-3.5 h-3.5 shrink-0 ${w.severity === 'info' ? 'hidden' : ''}`} />
   127→          <Info className={`w-3.5 h-3.5 shrink-0 ${w.severity !== 'info' ? 'hidden' : ''}`} />
   128→          <span>{teamName}: {w.message}</span>
   129→        </div>
   130→      ))}
   131→    </motion.div>
   132→  );
   133→}

   137→function PlayerStatsTooltip({ player, allPlayers }: { player: DbPlayer; allPlayers: DbPlayer[] }) {
   138→  const dbPlayer = allPlayers.find(p => p.id === player.id);
   139→  const profile = dbPlayer?.profile;
   140→
   141→  if (!profile || !profile.totalPoints) return null;
   142→
   143→  const raidSuccessRate = profile.totalRaids && profile.totalRaids > 0
   144→    ? Math.round(((profile.successfulRaids || 0) / profile.totalRaids) * 100)
   145→    : 0;
   146→
   147→  return (
   148→    <div className="absolute z-30 bottom-full left-0 mb-1 p-2 bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-600 rounded-lg shadow-lg text-[10px] min-w-[140px]">
   149→      <div className="space-y-1">
   150→        <div className="flex justify-between">
   151→          <span className="text-warm-500 dark:text-warm-400">Total Pts</span>
   152→          <span className="font-bold text-brand-navy dark:text-brand-navy-light">{profile.totalPoints}</span>
   153→        </div>
   154→        <div className="flex justify-between">
   155→          <span className="text-warm-500 dark:text-warm-400">Raid Pts</span>
   156→          <span className="font-bold text-brand-red">{profile.raidPoints || 0}</span>
   157→        </div>
   158→        <div className="flex justify-between">
   159→          <span className="text-warm-500 dark:text-warm-400">Tackle Pts</span>
   160→          <span className="font-bold text-brand-teal">{profile.tacklePoints || 0}</span>
   161→        </div>
   162→        <div className="flex justify-between">
   163→          <span className="text-warm-500 dark:text-warm-400">Raid Success</span>
   164→          <span className="font-bold text-brand-gold">{raidSuccessRate}%</span>
   165→        </div>
   166→      </div>
   167→      {/* Tooltip arrow */}
   168→      <div className="absolute -bottom-1 left-4 w-2 h-2 bg-white dark:bg-warm-800 border-r border-b border-warm-200 dark:border-warm-600 rotate-45" />
   169→    </div>
   170→  );
   171→}

   175→function FormationVisualization({ lineup, teamColor, teamName, side }: {
   176→  lineup: MatchPlayer[];
   177→  teamColor: string;
   178→  teamName: string;
   179→  side: 'left' | 'right';
   180→}) {
   181→  const maxDisplay = Math.min(lineup.length, 7);
   182→  const displayPlayers = lineup.slice(0, maxDisplay);
   183→

   185→    <div className="flex-1 relative">
   186→      {/* Team header */}
   187→      <div className="text-center mb-2">
   188→        <div
   189→          className="w-10 h-10 rounded-xl mx-auto flex items-center justify-center text-white font-black text-sm shadow-md"
   190→          style={{ backgroundColor: teamColor }}
   191→        >
   192→          {teamName.charAt(0).toUpperCase()}
   193→        </div>
   194→        <p className="text-[10px] font-bold text-warm-600 dark:text-warm-300 mt-1 truncate max-w-[80px] mx-auto">
   195→          {teamName}
   196→        </p>
   197→      </div>
   198→
   199→      {/* Formation layout */}
   200→      <div className="relative bg-warm-100/40 dark:bg-warm-700/20 rounded-xl p-2 min-h-[90px]">
   201→        {/* Court half line */}
   202→        <div className={`absolute top-1/2 left-0 right-0 h-px bg-warm-200 dark:bg-warm-600 ${
   203→          side === 'left' ? 'bg-gradient-to-r from-transparent to-warm-200 dark:to-warm-600' : 'bg-gradient-to-l from-transparent to-warm-200 dark:to-warm-600'
   204→        }`} />
   205→
   206→        {/* Players in formation */}
   207→        <div className="grid grid-cols-3 gap-1 place-items-center">
   208→          {displayPlayers.map((player, i) => {
   209→            // Simple formation: top row = defenders, middle = all-rounders, bottom = raiders
   210→            return (
   211→              <motion.div
   212→                key={player.id}
   213→                initial={{ scale: 0, opacity: 0 }}
   214→                animate={{ scale: 1, opacity: 1 }}
   215→                transition={{ delay: i * 0.05, type: 'spring', stiffness: 300, damping: 20 }}
   216→                className="flex flex-col items-center"
   217→              >
   218→                <div
   219→                  className="w-7 h-7 rounded-full flex items-center justify-center text-white text-[8px] font-bold shadow-sm"
   220→                  style={{ backgroundColor: teamColor }}
   221→                >
   222→                  {player.jerseyNumber || i + 1}
   223→                </div>
   224→                <span className="text-[7px] text-warm-400 dark:text-warm-500 mt-0.5 truncate max-w-[32px]">
   225→                  {player.name.split(' ')[0]}
   226→                </span>
   227→              </motion.div>
   228→            );
   229→          })}
   230→          {/* Empty slots */}
   231→          {Array.from({ length: Math.max(0, 7 - displayPlayers.length) }).map((_, i) => (
   232→            <div key={`empty-${i}`} className="flex flex-col items-center">
   233→              <div className="w-7 h-7 rounded-full border-2 border-dashed border-warm-200 dark:border-warm-600 flex items-center justify-center">
   234→                <span className="text-[8px] text-warm-300 dark:text-warm-600">?</span>
   235→              </div>
   236→              <span className="text-[7px] text-warm-300 dark:text-warm-600 mt-0.5">Empty</span>
   237→            </div>
   238→          ))}
   239→        </div>
   240→      </div>
   241→    </div>
   242→  );
   243→}
