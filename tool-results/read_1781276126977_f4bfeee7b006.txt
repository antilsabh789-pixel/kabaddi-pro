     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { ChevronLeft, ChevronRight, Plus, X, Search, UserPlus, Database, Check, Clock, Users, Swords, Play, GripVertical, Shield, Zap } from 'lucide-react';
     6→import { useKabaddiStore, type MatchPlayer } from '@/lib/store';
     7→import { Button } from '@/components/ui/button';
     8→import { Input } from '@/components/ui/input';
     9→import { Badge } from '@/components/ui/badge';
    10→
    11→interface UserTeam {
    12→  id: string;
    13→  name: string;
    14→  shortName: string | null;
    15→  color: string | null;
    16→}
    17→
    18→interface MatchConfig {
    19→  gender: string;
    20→  halfDuration: number;
    21→  playersPerSide: number;
    22→  homeTeam: string;
    23→  awayTeam: string;
    24→  homeTeamColor: string;
    25→  awayTeamColor: string;
    26→  homeLineup: MatchPlayer[];
    27→  awayLineup: MatchPlayer[];
    28→}
    29→
    30→// Removed 'Type' step - Quick Score is always practice
    31→const STEPS = ['Gender', 'Settings', 'Teams', 'Lineup', 'Start'];
    32→const STEP_ICONS = [Users, Clock, Swords, Shield, Play];
    33→
    34→interface DbPlayer {
    35→  id: string;
    36→  name: string | null;
    37→  phone: string | null;
    38→  playerCode: string | null;
    39→  avatar: string | null;
    40→  gender?: string | null;
    41→  profile?: {
    42→    position: string | null;
    43→    jerseyNumber: number | null;
    44→    overallRating: number;
    45→  } | null;
    46→}
    47→
    48→export default function QuickScoreTab() {
    49→  const { initiateToss, currentUser } = useKabaddiStore();
    50→  const [step, setStep] = useState(0);
    51→  const [config, setConfig] = useState<MatchConfig>({
    52→    gender: '',
    53→    halfDuration: 20,
    54→    playersPerSide: 7,
    55→    homeTeam: '',
    56→    awayTeam: '',
    57→    homeTeamColor: '#DC2626',
    58→    awayTeamColor: '#1E293B',
    59→    homeLineup: [],
    60→    awayLineup: [],
    61→  });
    62→  const [playerSearch, setPlayerSearch] = useState('');
    63→  const [activeLineupTeam, setActiveLineupTeam] = useState<'home' | 'away'>('home');
    64→  const [searchResults, setSearchResults] = useState<DbPlayer[]>([]);
    65→  const [isSearching, setIsSearching] = useState(false);
    66→  const [showSuggestions, setShowSuggestions] = useState(false);
    67→  const [allPlayers, setAllPlayers] = useState<DbPlayer[]>([]);
    68→  const playerInputRef = useRef<HTMLDivElement>(null);
    69→  const [userTeams, setUserTeams] = useState<UserTeam[]>([]);
    70→  const [homeTeamSuggestions, setHomeTeamSuggestions] = useState<UserTeam[]>([]);
    71→  const [awayTeamSuggestions, setAwayTeamSuggestions] = useState<UserTeam[]>([]);
    72→  const [showHomeSuggestions, setShowHomeSuggestions] = useState(false);
    73→  const [showAwaySuggestions, setShowAwaySuggestions] = useState(false);
    74→  const teamInputRef = useRef<HTMLDivElement>(null);
    75→
    76→  // Close team/player suggestions when clicking outside
    77→  useEffect(() => {
    78→    const handleClickOutside = (e: MouseEvent) => {
    79→      if (teamInputRef.current && !teamInputRef.current.contains(e.target as Node)) {
    80→        setShowHomeSuggestions(false);
    81→        setShowAwaySuggestions(false);
    82→      }
    83→      if (playerInputRef.current && !playerInputRef.current.contains(e.target as Node)) {
    84→        setShowSuggestions(false);
    85→      }
    86→    };
    87→    document.addEventListener('mousedown', handleClickOutside);
    88→    return () => document.removeEventListener('mousedown', handleClickOutside);
    89→  }, []);
    90→
    91→  // Fetch all players on mount for instant filtering
    92→  useEffect(() => {
    93→    const fetchAllPlayers = async () => {
    94→      try {
    95→        const res = await fetch('/api/players?limit=100');
    96→        if (res.ok) {
    97→          const data = await res.json();
    98→          setAllPlayers(data.players || []);
    99→        }
   100→      } catch {
   101→        // silently fail
   102→      }
   103→    };
   104→    fetchAllPlayers();
   105→  }, []);
   106→
   107→  // Fetch user's teams for suggestions
   108→  useEffect(() => {
   109→    if (!currentUser?.id) return;
   110→    const fetchUserTeams = async () => {
   111→      try {
   112→        const res = await fetch('/api/teams');
   113→        if (res.ok) {
   114→          const data = await res.json();
   115→          const teams: UserTeam[] = (data.teams || [])
   116→            .filter((team: any) =>
   117→              team.members?.some((m: any) => m.userId === currentUser.id)
   118→            )
   119→            .map((team: any) => ({
   120→              id: team.id,
   121→              name: team.name,
   122→              shortName: team.shortName,
   123→              color: team.color,
   124→            }));
   125→          setUserTeams(teams);
   126→        }
   127→      } catch {
   128→        // silently fail
   129→      }
   130→    };
   131→    fetchUserTeams();
   132→  }, [currentUser?.id]);
   133→
   134→  // Filter team suggestions based on input
   135→  useEffect(() => {
   136→    const query = config.homeTeam.toLowerCase().trim();
   137→    if (query.length > 0) {
   138→      const filtered = userTeams.filter(t =>
   139→        t.name.toLowerCase().includes(query)
   140→      );
   141→      setHomeTeamSuggestions(filtered);
   142→    } else {
   143→      setHomeTeamSuggestions(userTeams);
   144→    }
   145→  }, [config.homeTeam, userTeams]);
   146→
   147→  useEffect(() => {
   148→    const query = config.awayTeam.toLowerCase().trim();
   149→    if (query.length > 0) {
   150→      const filtered = userTeams.filter(t =>
   151→        t.name.toLowerCase().includes(query)
   152→      );
   153→      setAwayTeamSuggestions(filtered);
   154→    } else {
   155→      setAwayTeamSuggestions(userTeams);
   156→    }
   157→  }, [config.awayTeam, userTeams]);
   158→
   159→  const selectHomeTeamSuggestion = (team: UserTeam) => {
   160→    setConfig({
   161→      ...config,
   162→      homeTeam: team.name,
   163→      homeTeamColor: team.color || config.homeTeamColor,
   164→    });
   165→    setShowHomeSuggestions(false);
   166→  };
   167→
   168→  const selectAwayTeamSuggestion = (team: UserTeam) => {
   169→    setConfig({
   170→      ...config,
   171→      awayTeam: team.name,
   172→      awayTeamColor: team.color || config.awayTeamColor,
   173→    });
   174→    setShowAwaySuggestions(false);
   175→  };
   176→
   177→  const canNext = () => {
   178→    switch (step) {
   179→      case 0: return config.gender !== '';
   180→      case 1: return config.halfDuration >= 1 && config.playersPerSide >= 1;
   181→      case 2: return config.homeTeam !== '' && config.awayTeam !== '';
   182→      case 3: return config.homeLineup.length > 0 && config.awayLineup.length > 0;
   183→      case 4: return true;
   184→      default: return false;
   185→    }
   186→  };
   187→
   188→  const handleNext = () => {
   189→    if (step < 4 && canNext()) setStep(step + 1);
   190→  };
   191→
   192→  const handlePrev = () => {
   193→    if (step > 0) setStep(step - 1);
   194→  };
   195→
   196→  const handleStart = () => {
   197→    initiateToss({
   198→      id: `match_${Date.now()}`,
   199→      homeTeamId: `home_${Date.now()}`,
   200→      awayTeamId: `away_${Date.now()}`,
   201→      homeTeam: config.homeTeam,
   202→      awayTeam: config.awayTeam,
   203→      homeTeamColor: config.homeTeamColor,
   204→      awayTeamColor: config.awayTeamColor,
   205→      isPractice: true, // Quick Score is always practice
   206→      gender: config.gender,
   207→      halfDuration: config.halfDuration,
   208→      playersPerSide: config.playersPerSide,
   209→      homeLineup: config.homeLineup,
   210→      awayLineup: config.awayLineup,
   211→    });
   212→  };
   213→
   214→  const addQuickPlayer = (team: 'home' | 'away') => {
   215→    const name = playerSearch.trim();
   216→    if (!name) return;
   217→    const lineup = team === 'home' ? config.homeLineup : config.awayLineup;
   218→    const newPlayer: MatchPlayer = {
   219→      id: `p_${Date.now()}_${Math.random().toString(36).slice(2, 5)}`,
   220→      name,
   221→      jerseyNumber: lineup.length + 1,
   222→      team,
   223→    };
   224→    setConfig({
   225→      ...config,
   226→      [team === 'home' ? 'homeLineup' : 'awayLineup']: [...lineup, newPlayer],
   227→    });
   228→    setPlayerSearch('');
   229→    setSearchResults([]);
   230→    setShowSuggestions(false);
   231→  };
   232→
   233→  // Highlight matching text in suggestions
   234→  const highlightMatch = (text: string, query: string) => {
   235→    if (!query.trim()) return text;
   236→    const regex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
   237→    const parts = text.split(regex);
   238→    return parts.map((part, i) =>
   239→      regex.test(part) ? (
   240→        <span key={i} className="text-brand-red font-bold">{part}</span>
   241→      ) : (
   242→        <span key={i}>{part}</span>
   243→      )
   244→    );
   245→  };
   246→
   247→  const removePlayer = (team: 'home' | 'away', playerId: string) => {
   248→    const lineup = team === 'home' ? config.homeLineup : config.awayLineup;
   249→    setConfig({
   250→      ...config,
   251→      [team === 'home' ? 'homeLineup' : 'awayLineup']: lineup.filter((p) => p.id !== playerId),
   252→    });
   253→  };
   254→
   255→  // Local filter for instant results
   256→  const getLocalFiltered = useCallback((query: string): DbPlayer[] => {
   257→    if (!query.trim()) return allPlayers.slice(0, 20);
   258→    const q = query.toLowerCase().trim();
   259→    return allPlayers.filter(p =>
   260→      (p.name && p.name.toLowerCase().includes(q)) ||
   261→      (p.playerCode && p.playerCode.toLowerCase().includes(q)) ||
   262→      (p.phone && p.phone.includes(q))
   263→    ).slice(0, 20);
   264→  }, [allPlayers]);
   265→
   266→  // Search existing players from DB (server-side for more results)
   267→  const searchPlayers = useCallback(async (query: string) => {
   268→    if (query.length < 1) {
   269→      setSearchResults(getLocalFiltered(''));
   270→      return;
   271→    }
   272→    // Show local results immediately
   273→    const localResults = getLocalFiltered(query);
   274→    setSearchResults(localResults);
   275→
   276→    // Also fetch from server for more comprehensive results
   277→    if (query.length >= 2) {
   278→      setIsSearching(true);
   279→      try {
   280→        const res = await fetch(`/api/players?search=${encodeURIComponent(query)}`);
   281→        if (res.ok) {
   282→          const data = await res.json();
   283→          const serverPlayers: DbPlayer[] = data.players || [];
   284→          // Merge: server results first, then local-only results
   285→          const serverIds = new Set(serverPlayers.map(p => p.id));
   286→          const merged = [...serverPlayers, ...localResults.filter(p => !serverIds.has(p.id))].slice(0, 20);
   287→          setSearchResults(merged);
   288→        }
   289→      } catch {
   290→        // Keep local results on error
   291→      } finally {
   292→        setIsSearching(false);
   293→      }
   294→    }
   295→  }, [getLocalFiltered]);
   296→
   297→  // Debounce server search, but local filter is instant
   298→  useEffect(() => {
   299→    const timer = setTimeout(() => {
   300→      if (showSuggestions && playerSearch.length >= 2) {
   301→        searchPlayers(playerSearch);
   302→      }
   303→    }, 300);
   304→    return () => clearTimeout(timer);
   305→  }, [playerSearch, showSuggestions, searchPlayers]);
   306→
   307→  // Instant local filter on every keystroke
   308→  useEffect(() => {
   309→    if (showSuggestions) {
   310→      setSearchResults(getLocalFiltered(playerSearch));
   311→    }
   312→  }, [playerSearch, showSuggestions, getLocalFiltered]);
   313→
   314→  const addDbPlayer = (dbPlayer: DbPlayer) => {
   315→    const lineup = activeLineupTeam === 'home' ? config.homeLineup : config.awayLineup;
   316→    // Check if already added
   317→    const alreadyAdded = [...config.homeLineup, ...config.awayLineup].some(p => p.id === dbPlayer.id);
   318→    if (alreadyAdded) return;
   319→    const newPlayer: MatchPlayer = {
   320→      id: dbPlayer.id,
   321→      name: dbPlayer.name || 'Unknown',
   322→      jerseyNumber: dbPlayer.profile?.jerseyNumber || lineup.length + 1,
   323→      playerCode: dbPlayer.playerCode || undefined,
   324→      team: activeLineupTeam,
   325→    };
   326→    setConfig({
   327→      ...config,
   328→      [activeLineupTeam === 'home' ? 'homeLineup' : 'awayLineup']: [...lineup, newPlayer],
   329→    });
   330→    setPlayerSearch('');
   331→    setSearchResults([]);
   332→    setShowSuggestions(false);
   333→  };
   334→
   335→  const teamColors = [
   336→    '#DC2626', '#1E293B', '#14B8A6', '#475569', '#9333EA',
   337→    '#F97316', '#06B6D4', '#EC4899', '#84CC16', '#6366F1',
   338→  ];
   339→
   340→  // Position icon mapping
   341→  const getPositionIcon = (position: string | null) => {
   342→    if (!position) return null;
   343→    const p = position.toLowerCase();
   344→    if (p.includes('raider')) return <Zap className="w-3 h-3" />;
   345→    if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return <Shield className="w-3 h-3" />;
   346→    if (p.includes('all')) return <Swords className="w-3 h-3" />;
   347→    return null;
   348→  };
   349→
   350→  return (
   351→    <div className="px-4 py-6 space-y-5">
   352→      {/* Step Progress Indicator */}
   353→      <div className="flex items-center justify-between px-1">
   354→        {STEPS.map((label, i) => {
   355→          const StepIcon = STEP_ICONS[i];
   356→          const isCompleted = i < step;
   357→          const isCurrent = i === step;
   358→          return (
   359→            <div key={label} className="flex flex-col items-center gap-1.5 flex-1">
   360→              <div className="flex items-center w-full">
   361→                {i > 0 && (
   362→                  <div className={`flex-1 h-0.5 rounded-full transition-all duration-500 ${
   363→                    i <= step ? 'bg-brand-red' : 'bg-warm-200 dark:bg-warm-700'
   364→                  }`} />
   365→                )}
   366→                <motion.div
   367→                  className={`w-8 h-8 rounded-full flex items-center justify-center shrink-0 transition-all duration-300 ${
   368→                    isCompleted
   369→                      ? 'bg-brand-red text-white shadow-md shadow-brand-red/30'
   370→                      : isCurrent
   371→                        ? 'bg-brand-red/15 text-brand-red ring-2 ring-brand-red/30'
   372→                        : 'bg-warm-100 dark:bg-warm-800 text-warm-400 dark:text-warm-500'
   373→                  }`}
   374→                  animate={isCurrent ? { scale: [1, 1.1, 1] } : {}}
   375→                  transition={{ duration: 2, repeat: Infinity, ease: 'easeInOut' }}
   376→                >
   377→                  {isCompleted ? (
   378→                    <Check className="w-4 h-4" />
   379→                  ) : (
   380→                    <StepIcon className="w-3.5 h-3.5" />
   381→                  )}
   382→                </motion.div>
   383→                {i < STEPS.length - 1 && (
   384→                  <div className={`flex-1 h-0.5 rounded-full transition-all duration-500 ${
   385→                    i < step ? 'bg-brand-red' : 'bg-warm-200 dark:bg-warm-700'
   386→                  }`} />
   387→                )}
   388→              </div>
   389→              <span className={`text-[9px] font-semibold tracking-wide ${
   390→                isCompleted || isCurrent ? 'text-brand-red' : 'text-warm-400 dark:text-warm-500'
   391→              }`}>
   392→                {label}
   393→              </span>
   394→            </div>
   395→          );
   396→        })}
   397→      </div>
   398→
   399→      {/* Step Content */}
   400→      <AnimatePresence mode="wait">
   401→        <motion.div
   402→          key={step}
   403→          initial={{ opacity: 0, x: 20 }}
   404→          animate={{ opacity: 1, x: 0 }}
   405→          exit={{ opacity: 0, x: -20 }}
   406→          transition={{ duration: 0.2 }}
   407→          className="min-h-[400px]"
   408→        >
   409→          {step === 0 && (
   410→            <div className="space-y-5">
   411→              <div className="text-center">
   412→                <h2 className="text-xl font-bold text-warm-800 dark:text-warm-100">Select Category</h2>
   413→                <p className="text-sm text-warm-500 dark:text-warm-400 mt-1">Choose the match category to get started</p>
   414→              </div>
   415→              <div className="grid grid-cols-2 gap-4">
   416→                <motion.button
   417→                  onClick={() => setConfig({ ...config, gender: 'male' })}
   418→                  whileTap={{ scale: 0.97 }}
   419→                  whileHover={{ scale: 1.02 }}
   420→                  className={`relative p-6 rounded-2xl border-2 transition-all duration-300 flex flex-col items-center gap-3 overflow-hidden ${
   421→                    config.gender === 'male'
   422→                      ? 'border-blue-500 bg-gradient-to-br from-blue-500/20 to-blue-600/10 shadow-lg shadow-blue-500/20'
   423→                      : 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/50 hover:border-blue-300'
   424→                  }`}
   425→                >
   426→                  {config.gender === 'male' && (
   427→                    <motion.div
   428→                      className="absolute inset-0 bg-gradient-to-br from-blue-500/10 to-transparent"
   429→                      initial={{ opacity: 0 }}
   430→                      animate={{ opacity: 1 }}
   431→                    />
   432→                  )}
   433→                  <div className={`w-16 h-16 rounded-2xl flex items-center justify-center relative ${
   434→                    config.gender === 'male'
   435→                      ? 'bg-gradient-to-br from-blue-500 to-blue-600 shadow-md'
   436→                      : 'bg-blue-50 dark:bg-blue-900/30'
   437→                  }`}>
   438→                    <span className="text-3xl">♂</span>
   439→                  </div>
   440→                  <span className={`font-bold text-lg relative ${config.gender === 'male' ? 'text-blue-600 dark:text-blue-400' : 'text-warm-600 dark:text-warm-300'}`}>
   441→                    Boys
   442→                  </span>
   443→                  {config.gender === 'male' && (
   444→                    <motion.div
   445→                      className="absolute top-2 right-2"
   446→                      initial={{ scale: 0 }}
   447→                      animate={{ scale: 1 }}
   448→                      transition={{ type: 'spring', stiffness: 500, damping: 15 }}
   449→                    >
   450→                      <div className="w-5 h-5 rounded-full bg-blue-500 flex items-center justify-center">
   451→                        <Check className="w-3 h-3 text-white" />
   452→                      </div>
   453→                    </motion.div>
   454→                  )}
   455→                </motion.button>
   456→                <motion.button
   457→                  onClick={() => setConfig({ ...config, gender: 'female' })}
   458→                  whileTap={{ scale: 0.97 }}
   459→                  whileHover={{ scale: 1.02 }}
   460→                  className={`relative p-6 rounded-2xl border-2 transition-all duration-300 flex flex-col items-center gap-3 overflow-hidden ${
   461→                    config.gender === 'female'
   462→                      ? 'border-red-500 bg-gradient-to-br from-red-500/20 to-red-600/10 shadow-lg shadow-red-500/20'
   463→                      : 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/50 hover:border-red-300'
   464→                  }`}
   465→                >
   466→                  {config.gender === 'female' && (
   467→                    <motion.div
   468→                      className="absolute inset-0 bg-gradient-to-br from-red-500/10 to-transparent"
   469→                      initial={{ opacity: 0 }}
   470→                      animate={{ opacity: 1 }}
   471→                    />
   472→                  )}
   473→                  <div className={`w-16 h-16 rounded-2xl flex items-center justify-center relative ${
   474→                    config.gender === 'female'
   475→                      ? 'bg-gradient-to-br from-red-500 to-red-600 shadow-md'
   476→                      : 'bg-red-50 dark:bg-red-900/30'
   477→                  }`}>
   478→                    <span className="text-3xl">♀</span>
   479→                  </div>
   480→                  <span className={`font-bold text-lg relative ${config.gender === 'female' ? 'text-red-600 dark:text-red-400' : 'text-warm-600 dark:text-warm-300'}`}>
   481→                    Girls
   482→                  </span>
   483→                  {config.gender === 'female' && (
   484→                    <motion.div
   485→                      className="absolute top-2 right-2"
   486→                      initial={{ scale: 0 }}
   487→                      animate={{ scale: 1 }}
   488→                      transition={{ type: 'spring', stiffness: 500, damping: 15 }}
   489→                    >
   490→                      <div className="w-5 h-5 rounded-full bg-red-500 flex items-center justify-center">
   491→                        <Check className="w-3 h-3 text-white" />
   492→                      </div>
   493→                    </motion.div>
   494→                  )}
   495→                </motion.button>
   496→              </div>
   497→            </div>
   498→          )}
   499→
   500→          {step === 1 && (
   501→            <div className="space-y-6">
   502→              <div className="text-center">
   503→                <h2 className="text-xl font-bold text-warm-800 dark:text-warm-100">Match Settings</h2>
   504→                <p className="text-sm text-warm-500 dark:text-warm-400 mt-1">Configure your practice match</p>
   505→              </div>
   506→
   507→              {/* Practice Match Indicator */}
   508→              <motion.div
   509→                initial={{ opacity: 0, y: -5 }}
   510→                animate={{ opacity: 1, y: 0 }}
   511→                className="bg-gradient-to-r from-emerald-50 to-teal-50 dark:from-emerald-900/20 dark:to-teal-900/20 border border-emerald-200 dark:border-emerald-800/30 rounded-2xl p-4 flex items-center gap-3"
   512→              >
   513→                <div className="w-10 h-10 rounded-xl bg-emerald-500/10 flex items-center justify-center shrink-0">
   514→                  <span className="text-xl">🏋️</span>
   515→                </div>
   516→                <div>
   517→                  <p className="text-sm font-bold text-emerald-700 dark:text-emerald-400">Practice Match</p>
   518→                  <p className="text-[11px] text-emerald-600/70 dark:text-emerald-400/60">Quick Score is for practice. Tournament matches are scored via Tournaments tab.</p>
   519→                </div>
   520→              </motion.div>
   521→
   522→              {/* Half Duration */}
   523→              <motion.div
   524→                initial={{ opacity: 0, y: 5 }}
   525→                animate={{ opacity: 1, y: 0 }}
   526→                transition={{ delay: 0.1 }}
   527→                className="bg-white dark:bg-warm-800/50 border border-warm-200 dark:border-warm-700 rounded-2xl p-5"
   528→              >
   529→                <div className="flex items-center gap-2 mb-3">
   530→                  <div className="w-8 h-8 rounded-lg bg-brand-red/10 flex items-center justify-center">
