    80→/** Animated status indicator dot with enhanced glow */

    82→  const config = {
    83→    ongoing: { color: 'bg-emerald-500', ring: 'ring-emerald-500/30', pulse: true, glow: 'shadow-emerald-500/50' },
    84→    upcoming: { color: 'bg-amber-500', ring: 'ring-amber-500/30', pulse: false, glow: 'shadow-amber-500/50' },
    85→    past: { color: 'bg-warm-400', ring: 'ring-warm-400/30', pulse: false, glow: 'shadow-warm-400/50' },
    86→  }[status] || { color: 'bg-warm-400', ring: 'ring-warm-400/30', pulse: false, glow: 'shadow-warm-400/50' };
    87→
    88→  return (
    89→    <span className="relative flex h-3.5 w-3.5">
    90→      {config.pulse && (

   161→/** Enhanced knockout bracket view with team colors */
   162→function BracketView({ tournament }: { tournament: Tournament }) {
   163→  const [bracketMatches, setBracketMatches] = useState<BracketMatch[]>([]);
   164→  const [loadingBracket, setLoadingBracket] = useState(false);
   165→
   166→  useEffect(() => {
   167→    const fetchBracket = async () => {
   168→      setLoadingBracket(true);
   169→      try {
   170→        const res = await fetch(`/api/tournaments/${tournament.id}/matches`);
   171→        if (res.ok) {
   172→          const data = await res.json();
   173→          const matches: BracketMatch[] = (data.matches || []).map((m: Record<string, unknown>) => ({
   174→            id: m.id as string,
   175→            round: (m.round as number) || 1,
   176→            position: (m.position as number) || 0,
   177→            team1Id: (m.team1Id as string) || null,
   178→            team2Id: (m.team2Id as string) || null,
   179→            team1Score: (m.team1Score as number) ?? null,
   180→            team2Score: (m.team2Score as number) ?? null,
   181→            winnerId: (m.winnerId as string) || null,
   182→            status: (m.status as string) || 'scheduled',
   183→          }));
   184→          setBracketMatches(matches);
   185→        }
   186→      } catch {
   187→        // silently fail
   188→      } finally {
   189→        setLoadingBracket(false);
   190→      }
   191→    };
   192→    fetchBracket();
   193→  }, [tournament.id]);
   194→
   195→  const teamsMap = new Map(tournament.teams.map(t => [t.id, t]));
   196→
   197→  // Group matches by round
   198→  const rounds = new Map<number, BracketMatch[]>();
   199→  bracketMatches.forEach(m => {
   200→    const list = rounds.get(m.round) || [];
   201→    list.push(m);
   202→    rounds.set(m.round, list);
   203→  });
   204→  const sortedRounds = Array.from(rounds.entries()).sort((a, b) => a[0] - b[0]);
   205→
   206→  // If knockout with no matches yet, show placeholder bracket
   207→  const showPlaceholder = bracketMatches.length === 0 && !loadingBracket;
   208→
   209→  // Generate placeholder bracket based on team count
   210→  const generatePlaceholderRounds = () => {
   211→    const teamCount = tournament.teams.length;
   212→    if (teamCount < 2) return [];
   213→    const roundsCount = Math.ceil(Math.log2(teamCount));
   214→    const result: { round: number; matches: { team1: string | null; team2: string | null }[] }[] = [];
   215→    let currentMatches = Math.floor(teamCount / 2);
   216→
   217→    for (let r = 1; r <= roundsCount; r++) {
   218→      const matches = [];
   219→      for (let m = 0; m < currentMatches; m++) {
   220→        const team1Idx = m * 2;
   221→        const team2Idx = m * 2 + 1;
   222→        matches.push({
   223→          team1: tournament.teams[team1Idx]?.id || null,
   224→          team2: team2Idx < tournament.teams.length ? tournament.teams[team2Idx]?.id || null : null,
   225→        });
   226→      }
   227→      result.push({ round: r, matches });
   228→      currentMatches = Math.ceil(currentMatches / 2);
   229→    }
   230→    return result;
   231→  };
   232→
   233→  const placeholderRounds = showPlaceholder ? generatePlaceholderRounds() : [];
   234→
   235→  const getRoundName = (round: number, totalRounds: number) => {
   236→    if (round === totalRounds) return 'Final';
   237→    if (round === totalRounds - 1) return 'Semi-Final';
   238→    if (round === totalRounds - 2) return 'Quarter-Final';
   239→    return `Round ${round}`;
   240→  };
   241→
   242→  return (
   243→    <motion.div
   244→      initial={{ opacity: 0, y: 20 }}
   245→      animate={{ opacity: 1, y: 0 }}
   246→      transition={{ duration: 0.5, ease: 'easeOut' }}
   247→    >
   248→      <div className="flex items-center gap-2 mb-3">
   249→        <div className="w-6 h-6 rounded-lg bg-gradient-to-br from-orange-500 to-orange-400 flex items-center justify-center shadow-lg shadow-orange-500/20">
   250→          <Swords className="w-3.5 h-3.5 text-white" />
   251→        </div>
   252→        <h4 className="text-xs font-bold text-warm-500 dark:text-warm-400 uppercase tracking-wider">
   253→          {tournament.type === 'knockout' ? 'Knockout Bracket' : tournament.type === 'league' ? 'League Fixtures' : 'Tournament Bracket'}
   254→        </h4>
   255→      </div>
   256→
   257→      {loadingBracket ? (
   258→        <div className="flex items-center justify-center py-8 gap-2">
   259→          <Loader2 className="w-5 h-5 animate-spin text-brand-red" />
   260→          <span className="text-sm text-warm-400">Loading bracket...</span>
   261→        </div>
   262→      ) : (
   263→        <div className="overflow-x-auto pb-2 -mx-1">
   264→          <div className="flex gap-6 min-w-max px-1">
   265→            {/* Real bracket from API data */}
   266→            {sortedRounds.length > 0 ? sortedRounds.map(([roundNum, matches], rIdx) => {
   267→              const totalRounds = sortedRounds.length;
   268→              return (
   269→                <motion.div
   270→                  key={roundNum}
   271→                  className="flex flex-col gap-3 min-w-[180px]"
   272→                  initial={{ opacity: 0, x: 30 }}
   273→                  animate={{ opacity: 1, x: 0 }}
   274→                  transition={{ delay: rIdx * 0.15, duration: 0.4 }}
   275→                >
   276→                  <div className="text-center">
   277→                    <Badge className="bg-gradient-to-r from-brand-red to-brand-red-light text-white text-[10px] border-0 font-bold shadow-lg shadow-brand-red/20">
   278→                      {getRoundName(roundNum, totalRounds)}
   279→                    </Badge>
   280→                  </div>
   281→                  <div className="flex flex-col justify-around flex-1 gap-2">
   282→                    {matches.sort((a, b) => a.position - b.position).map((match) => {
   283→                      const team1 = match.team1Id ? teamsMap.get(match.team1Id) : null;
   284→                      const team2 = match.team2Id ? teamsMap.get(match.team2Id) : null;
   285→                      const isLive = match.status === 'live' || match.status === 'ongoing';
   286→                      const isCompleted = match.status === 'completed';
   287→                      return (
   288→                        <motion.div
   289→                          key={match.id}
   290→                          className={`rounded-xl border p-2.5 text-xs transition-all ${
   291→                            isLive
   292→                              ? 'border-brand-red/50 bg-gradient-to-br from-brand-red/5 to-brand-red/10 dark:from-brand-red/10 dark:to-brand-red/20 ring-2 ring-brand-red/30 shadow-lg shadow-brand-red/10'
   293→                              : isCompleted
   294→                                ? 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/50 shadow-sm'
   295→                                : 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/30'
   296→                          }`}
   297→                          initial={{ opacity: 0, scale: 0.9 }}
   298→                          animate={{ opacity: 1, scale: 1 }}
   299→                          transition={{ delay: rIdx * 0.1 + 0.1 }}
   300→                        >
   301→                          {isLive && (
   302→                            <div className="flex items-center gap-1 mb-1">
   303→                              <Radio className="w-2.5 h-2.5 text-brand-red animate-pulse" />
   304→                              <span className="text-[9px] font-bold text-brand-red uppercase">Live</span>
   305→                            </div>
   306→                          )}
   307→                          {/* Team 1 */}
   308→                          <div className={`flex items-center gap-2 py-1 ${match.winnerId === match.team1Id ? 'font-bold' : ''}`}>
   309→                            {team1 ? (
   310→                              <>
   311→                                <div className="w-6 h-6 rounded-lg flex items-center justify-center text-[9px] font-bold text-white shrink-0 shadow-sm" style={{ backgroundColor: team1.color }}>
   312→                                  {team1.shortName?.charAt(0) || team1.name.charAt(0)}
   313→                                </div>
   314→                                <span className={`truncate flex-1 ${match.winnerId === match.team1Id ? 'text-warm-800 dark:text-warm-100' : 'text-warm-600 dark:text-warm-300'}`}>
   315→                                  {team1.name}
   316→                                </span>
   317→                              </>
   318→                            ) : (
   319→                              <span className="text-warm-400 italic">TBD</span>
   320→                            )}
   321→                            {match.team1Score !== null && (
   322→                              <span className={`font-mono font-bold text-xs px-1.5 py-0.5 rounded ${match.winnerId === match.team1Id ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400' : 'text-warm-500'}`}>
   323→                                {match.team1Score}
   324→                              </span>
   325→                            )}
   326→                          </div>
   327→                          <div className="border-t border-warm-100 dark:border-warm-700/50 my-0.5 flex items-center justify-center">
   328→                            <span className="text-[8px] text-warm-300 dark:text-warm-600 bg-warm-50 dark:bg-warm-800 px-1 rounded font-bold">VS</span>
   329→                          </div>
   330→                          {/* Team 2 */}
   331→                          <div className={`flex items-center gap-2 py-1 ${match.winnerId === match.team2Id ? 'font-bold' : ''}`}>
   332→                            {team2 ? (
   333→                              <>
   334→                                <div className="w-6 h-6 rounded-lg flex items-center justify-center text-[9px] font-bold text-white shrink-0 shadow-sm" style={{ backgroundColor: team2.color }}>
   335→                                  {team2.shortName?.charAt(0) || team2.name.charAt(0)}
   336→                                </div>
   337→                                <span className={`truncate flex-1 ${match.winnerId === match.team2Id ? 'text-warm-800 dark:text-warm-100' : 'text-warm-600 dark:text-warm-300'}`}>
   338→                                  {team2.name}
   339→                                </span>
   340→                              </>
   341→                            ) : (
   342→                              <span className="text-warm-400 italic">TBD</span>
   343→                            )}
   344→                            {match.team2Score !== null && (
   345→                              <span className={`font-mono font-bold text-xs px-1.5 py-0.5 rounded ${match.winnerId === match.team2Id ? 'bg-emerald-500/10 text-emerald-600 dark:text-emerald-400' : 'text-warm-500'}`}>
   346→                                {match.team2Score}
   347→                              </span>
   348→                            )}
   349→                          </div>
   350→                          {match.winnerId && (
   351→                            <div className="flex items-center gap-1 mt-1 pt-1 border-t border-warm-100 dark:border-warm-700/30">
   352→                              <Trophy className="w-2.5 h-2.5 text-brand-gold" />
   353→                              <span className="text-[9px] text-brand-gold font-bold">
   354→                                {teamsMap.get(match.winnerId)?.name || 'Winner'}
   355→                              </span>
   356→                            </div>
   357→                          )}
   358→                        </motion.div>
   359→                      );
   360→                    })}
   361→                  </div>
   362→                </motion.div>
   363→              );
   364→            }) : null}
   365→
   366→            {/* Placeholder bracket */}
   367→            {placeholderRounds.length > 0 ? placeholderRounds.map((round, rIdx) => {
   368→              const totalRounds = placeholderRounds.length;
   369→              return (
   370→                <motion.div
   371→                  key={`placeholder-${round.round}`}
   372→                  className="flex flex-col gap-3 min-w-[160px]"
   373→                  initial={{ opacity: 0, x: 30 }}
   374→                  animate={{ opacity: 1, x: 0 }}
   375→                  transition={{ delay: rIdx * 0.15, duration: 0.4 }}
   376→                >
   377→                  <div className="text-center">
   378→                    <Badge className="bg-warm-200 dark:bg-warm-700 text-warm-500 dark:text-warm-400 text-[10px] border-0 font-bold">
   379→                      {getRoundName(round.round, totalRounds)}
   380→                    </Badge>
   381→                  </div>
   382→                  <div className="flex flex-col justify-around flex-1 gap-2">
   383→                    {round.matches.map((match, mIdx) => {
   384→                      const team1 = match.team1 ? teamsMap.get(match.team1) : null;
   385→                      const team2 = match.team2 ? teamsMap.get(match.team2) : null;
   386→                      return (
   387→                        <div key={mIdx} className="rounded-xl border border-dashed border-warm-300 dark:border-warm-600 p-2.5 text-xs bg-warm-50 dark:bg-warm-800/20">
   388→                          <div className="flex items-center gap-2 py-0.5">
   389→                            {team1 ? (
   390→                              <>
   391→                                <div className="w-6 h-6 rounded-lg flex items-center justify-center text-[9px] font-bold text-white shrink-0" style={{ backgroundColor: team1.color }}>
   392→                                  {team1.shortName?.charAt(0) || team1.name.charAt(0)}
   393→                                </div>
   394→                                <span className="truncate text-warm-500 dark:text-warm-400">{team1.name}</span>
   395→                              </>
   396→                            ) : (
   397→                              <span className="text-warm-400 italic">TBD</span>
   398→                            )}
   399→                          </div>
   400→                          <div className="text-warm-300 dark:text-warm-600 text-center py-0.5 text-[10px] font-bold">VS</div>
   401→                          <div className="flex items-center gap-2 py-0.5">
   402→                            {team2 ? (
   403→                              <>
   404→                                <div className="w-6 h-6 rounded-lg flex items-center justify-center text-[9px] font-bold text-white shrink-0" style={{ backgroundColor: team2.color }}>
   405→                                  {team2.shortName?.charAt(0) || team2.name.charAt(0)}
   406→                                </div>
   407→                                <span className="truncate text-warm-500 dark:text-warm-400">{team2.name}</span>
   408→                              </>
   409→                            ) : (
   410→                              <span className="text-warm-400 italic">BYE</span>
   411→                            )}
   412→                          </div>
   413→                        </div>
   414→                      );
   415→                    })}
   416→                  </div>
   417→                </motion.div>
   418→              );
   419→            }) : null}
   420→
   421→            {/* No data at all */}
   422→            {sortedRounds.length === 0 && placeholderRounds.length === 0 && !loadingBracket && (
   423→              <div className="text-center py-6 w-full">
   424→                <div className="w-16 h-16 mx-auto mb-3 rounded-2xl bg-warm-100 dark:bg-warm-800 flex items-center justify-center">
   425→                  <Target className="w-8 h-8 text-warm-300 dark:text-warm-600" />
   426→                </div>
   427→                <p className="text-sm text-warm-400 dark:text-warm-500 font-medium">No bracket generated yet</p>
   428→                <p className="text-[10px] text-warm-400 dark:text-warm-500 mt-1">Add teams and generate a bracket to see the tournament tree</p>
   429→              </div>
   430→            )}
   431→          </div>
   432→        </div>
   433→      )}
   434→    </motion.div>
   435→  );
   436→}
