   239→   239→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   240→   240→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   241→   241→  let homeScore = 0;
   242→   242→  let awayScore = 0;
   243→   243→  let homeOutPlayers = 0;
   244→   244→  let awayOutPlayers = 0;
   245→   245→  let homeOutPlayerIds: string[] = [];
   246→   246→  let awayOutPlayerIds: string[] = [];
   247→   247→
   248→   248→  for (const evt of events) {
   249→   249→    const side = getTeamSide(match, evt.teamId);
   250→   250→    const points = evt.value;
   251→   251→    if (side === 'home') {
   252→   252→      homeScore += points;
   253→   253→    } else {
   254→   254→      awayScore += points;
   255→   255→    }
   256→   256→
   257→   257→    // Count-based out tracking (backward compat)
   258→   258→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   259→   259→      if (side === 'home') {
   260→   260→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   261→   261→      } else {
   262→   262→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   263→   263→      }
   264→   264→    }
   265→   265→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   266→   266→      if (side === 'home') {
   267→   267→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   268→   268→      } else {
   269→   269→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   270→   270→      }
   271→   271→    }
   272→   272→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   273→   273→    if (evt.eventType === 'self_out') {
   274→   274→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   275→   275→      if (side === 'home') {
   276→   276→        // Home scored from self-out → away defender went out
   277→   277→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   278→   278→        // Home revives one player
   279→   279→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   280→   280→      } else {
   281→   281→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   282→   282→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   283→   283→      }
   284→   284→    }
   285→   285→    if (evt.eventType === 'all_out') {
   286→   286→      if (side === 'home') {
   287→   287→        awayOutPlayers = 0;
   288→   288→      } else {
   289→   289→        homeOutPlayers = 0;
   290→   290→      }
   291→   291→    }
   292→   292→
   293→   293→    // ID-based out tracking
   294→   294→    let details: Record<string, unknown> = {};
   295→   295→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   296→   296→
   297→   297→    if (evt.eventType === 'raid_point') {
   298→   298→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   299→   299→      if (side === 'home') {
   300→   300→        // Home team scored raid → away defenders go out
   301→   301→        for (const tid of touchedIds) {
   302→   302→          if (!awayOutPlayerIds.includes(tid)) {
   303→   303→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   304→   304→          }
   305→   305→        }
   306→   306→        // Home team revives players (first-out-first-in)
   307→   307→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   308→   308→        if (reviveCount > 0) {
   309→   309→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   310→   310→        }
   311→   311→      } else {
   312→   312→        // Away team scored raid → home defenders go out
   313→   313→        for (const tid of touchedIds) {
   314→   314→          if (!homeOutPlayerIds.includes(tid)) {
   315→   315→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   316→   316→          }
   317→   317→        }
   318→   318→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   319→   319→        if (reviveCount > 0) {
   320→   320→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   321→   321→        }
   322→   322→      }
   323→   323→    }
   324→   324→
   325→   325→    if (evt.eventType === 'bonus_point') {
   326→   326→      // One out player from scoring team revives
   327→   327→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   328→   328→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   329→   329→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   330→   330→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   331→   331→      }
   332→   332→    }
   333→   333→
   334→   334→    // Self-out: defender steps off mat during a live raid
   335→   335→    if (evt.eventType === 'self_out') {
   336→   336→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   337→   337→      if (selfOutPlayerId) {
   338→   338→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   339→   339→        if (side === 'home') {
   340→   340→          // Home scored → away defender self-out
   341→   341→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   342→   342→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   343→   343→          }
   344→   344→          // Home revives first-out player
   345→   345→          if (homeOutPlayerIds.length > 0) {
   346→   346→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   347→   347→          }
   348→   348→        } else {
   349→   349→          // Away scored → home defender self-out
   350→   350→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   351→   351→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   352→   352→          }
   353→   353→          if (awayOutPlayerIds.length > 0) {
   354→   354→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   355→   355→          }
   356→   356→        }
   357→   357→      }
   358→   358→    }
   359→   359→
   360→   360→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   361→   361→      // Raider goes out. The tackle is scored by the DEFENDING team.
   362→   362→      // So the raider is from the OPPOSING team.
   363→   363→      const raiderId = (details.raiderId as string) || evt.playerId;
   364→   364→      if (side === 'home' && raiderId) {
   365→   365→        // Home team scored tackle → away raider was caught → away raider goes out
   366→   366→        if (!awayOutPlayerIds.includes(raiderId)) {
   367→   367→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   368→   368→        }
   369→   369→      } else if (side === 'away' && raiderId) {
   370→   370→        // Away team scored tackle → home raider was caught → home raider goes out
   371→   371→        if (!homeOutPlayerIds.includes(raiderId)) {
   372→   372→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   373→   373→        }
   374→   374→      }
   375→   375→    }
   376→   376→
   377→   377→    if (evt.eventType === 'all_out') {
   378→   378→      if (side === 'home') {
   379→   379→        awayOutPlayerIds = [];
   380→   380→      } else {
   381→   381→        homeOutPlayerIds = [];
   382→   382→      }
   383→   383→    }
   384→   384→  }
   385→   385→
   386→   386→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   387→   387→}

   389→   389→/** Event types that constitute a complete raid and should flip the turn */
   390→   390→const RAID_ENDING_EVENT_TYPES: string[] = [
   391→   391→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   392→   392→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   393→   393→];
   394→   394→
   395→   395→/** Event types that do NOT flip the turn (non-raid events) */
   396→   396→const NON_RAID_EVENT_TYPES: string[] = [
   397→   397→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   398→   398→];
   399→   399→
   400→   400→/** Helper: determine raid queue from last raid-ending event */
   401→   401→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   402→   402→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   403→   403→  if (lastRaidEvent) {
   404→   404→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   405→   405→  }
   406→   406→  return match.raidQueue;
   407→   407→}
