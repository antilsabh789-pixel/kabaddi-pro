    56→/* ── Password Strength Helper ── */
    57→function getPasswordStrength(pwd: string): { label: string; color: string; pct: number; barColor: string } {
    58→  if (!pwd) return { label: '', color: '', pct: 0, barColor: '' };
    59→  let score = 0;
    60→  if (pwd.length >= 6) score++;
    61→  if (pwd.length >= 8) score++;
    62→  if (/[A-Z]/.test(pwd)) score++;
    63→  if (/[0-9]/.test(pwd)) score++;
    64→  if (/[^A-Za-z0-9]/.test(pwd)) score++;
    65→
    66→  if (score <= 1) return { label: 'Weak', color: 'text-red-500', pct: 25, barColor: 'bg-red-500' };
    67→  if (score <= 2) return { label: 'Medium', color: 'text-amber-500', pct: 50, barColor: 'bg-amber-500' };
    68→  if (score <= 3) return { label: 'Strong', color: 'text-emerald-500', pct: 75, barColor: 'bg-emerald-500' };
    69→  return { label: 'Very Strong', color: 'text-teal-500', pct: 100, barColor: 'bg-teal-500' };
    70→}

   159→/* ── Date of Birth Picker Component ── */
   160→function DOBPicker({
   161→  day,
   162→  month,
   163→  year,
   164→  onChange,
   165→}: {
   166→  day: string;
   167→  month: string;
   168→  year: string;
   169→  onChange: (day: string, month: string, year: string) => void;
   170→}) {
   171→  const months = [
   172→    'Jan', 'Feb', 'Mar', 'Apr', 'May', 'Jun',
   173→    'Jul', 'Aug', 'Sep', 'Oct', 'Nov', 'Dec',
   174→  ];
   175→
   176→  const currentYear = new Date().getFullYear();
   177→  const daysInMonth = useMemo(() => {
   178→    const m = month ? parseInt(month) : 0;
   179→    const y = year ? parseInt(year) : currentYear;
   180→    if (m === 0) return 31;
   181→    return new Date(y, m, 0).getDate();
   182→  }, [month, year, currentYear]);
   183→
   184→  const days = useMemo(() => Array.from({ length: daysInMonth }, (_, i) => String(i + 1)), [daysInMonth]);
   185→  const years = useMemo(() => Array.from({ length: 80 }, (_, i) => String(currentYear - 10 - i)), [currentYear]);
   186→
   187→  const selectClass =
   188→    'h-12 bg-white/60 dark:bg-white/5 border-warm-300/60 dark:border-warm-600/40 text-warm-800 dark:text-warm-100 text-sm rounded-xl focus:ring-2 focus:ring-brand-red/40 focus:border-brand-red/60 transition-all appearance-none cursor-pointer text-center font-medium';
   189→
   190→  return (
   191→    <div className="flex gap-2">
   192→      {/* Day */}
   193→      <div className="relative flex-1">
   194→        <select
   195→          value={day}
   196→          onChange={(e) => onChange(e.target.value, month, year)}
   197→          className={`${selectClass} w-full ${!day ? 'text-warm-400 dark:text-warm-500' : ''}`}
   198→        >
   199→          <option value="" disabled>DD</option>
   200→          {days.map((d) => (
   201→            <option key={d} value={d}>{d.padStart(2, '0')}</option>
   202→          ))}
   203→        </select>
   204→        <div className="absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none">
   205→          <svg className="w-3 h-3 text-warm-400" fill="none" viewBox="0 0 10 6">
   206→            <path d="M1 1l4 4 4-4" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   207→          </svg>
   208→        </div>
   209→      </div>
   210→
   211→      {/* Month */}
   212→      <div className="relative flex-[1.3]">
   213→        <select
   214→          value={month}
   215→          onChange={(e) => {
   216→            const newMonth = e.target.value;
   217→            const m = newMonth ? parseInt(newMonth) : 0;
   218→            const y = year ? parseInt(year) : currentYear;
   219→            const maxDays = new Date(y, m, 0).getDate();
   220→            const newDay = day && parseInt(day) > maxDays ? String(maxDays) : day;
   221→            onChange(newDay, newMonth, year);
   222→          }}
   223→          className={`${selectClass} w-full ${!month ? 'text-warm-400 dark:text-warm-500' : ''}`}
   224→        >
   225→          <option value="" disabled>Month</option>
   226→          {months.map((m, i) => (
   227→            <option key={i} value={String(i + 1)}>{m}</option>
   228→          ))}
   229→        </select>
   230→        <div className="absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none">
   231→          <svg className="w-3 h-3 text-warm-400" fill="none" viewBox="0 0 10 6">
   232→            <path d="M1 1l4 4 4-4" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   233→          </svg>
   234→        </div>
   235→      </div>
   236→
   237→      {/* Year */}
   238→      <div className="relative flex-[1.2]">
   239→        <select
   240→          value={year}
   241→          onChange={(e) => {
   242→            const newYear = e.target.value;
   243→            const m = month ? parseInt(month) : 0;
   244→            const y = newYear ? parseInt(newYear) : currentYear;
   245→            const maxDays = new Date(y, m, 0).getDate();
   246→            const newDay = day && parseInt(day) > maxDays ? String(maxDays) : day;
   247→            onChange(newDay, month, newYear);
   248→          }}
   249→          className={`${selectClass} w-full ${!year ? 'text-warm-400 dark:text-warm-500' : ''}`}
   250→        >
   251→          <option value="" disabled>YYYY</option>
   252→          {years.map((y) => (
   253→            <option key={y} value={y}>{y}</option>
   254→          ))}
   255→        </select>
   256→        <div className="absolute right-2 top-1/2 -translate-y-1/2 pointer-events-none">
   257→          <svg className="w-3 h-3 text-warm-400" fill="none" viewBox="0 0 10 6">
   258→            <path d="M1 1l4 4 4-4" stroke="currentColor" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round" />
   259→          </svg>
   260→        </div>
   261→      </div>
   262→    </div>
   263→  );
   264→}
