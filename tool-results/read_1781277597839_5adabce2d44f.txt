    64→    64→
    65→    65→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    66→    66→  const [display, setDisplay] = useState(0);
    67→    67→  const [hasAnimated, setHasAnimated] = useState(false);
    68→    68→  const ref = useRef<HTMLSpanElement>(null);
    69→    69→  const valueRef = useRef(value);
    70→    70→
    71→    71→  useEffect(() => {
    72→    72→    valueRef.current = value;
    73→    73→  }, [value]);
    74→    74→
    75→    75→  useEffect(() => {
    76→    76→    if (hasAnimated) return;
    77→    77→    if (value === 0) return;
    78→    78→
    79→    79→    const observer = new IntersectionObserver(
    80→    80→      ([entry]) => {
    81→    81→        if (entry.isIntersecting && !hasAnimated) {
    82→    82→          setHasAnimated(true);
    83→    83→          const targetVal = valueRef.current;
    84→    84→          const duration = 800;
    85→    85→          const startTime = performance.now();
    86→    86→
    87→    87→          function step(currentTime: number) {
    88→    88→            const elapsed = currentTime - startTime;
    89→    89→            const progress = Math.min(elapsed / duration, 1);
    90→    90→            const eased = 1 - Math.pow(1 - progress, 3);
    91→    91→            const current = eased * targetVal;
    92→    92→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
    93→    93→            if (progress < 1) {
    94→    94→              requestAnimationFrame(step);
    95→    95→            }
    96→    96→          }
    97→    97→
    98→    98→          requestAnimationFrame(step);
    99→    99→        }
   100→   100→      },
   101→   101→      { threshold: 0.1 }
   102→   102→    );
   103→   103→
   104→   104→    if (ref.current) observer.observe(ref.current);
   105→   105→    return () => observer.disconnect();
   106→   106→  }, [value, decimals, hasAnimated]);
   107→   107→
   108→   108→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   109→   109→}
