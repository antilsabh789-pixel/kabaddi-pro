   254→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   255→  const [count, setCount] = useState(0);
   256→  const prevTarget = useRef(target);
   257→
   258→  useEffect(() => {
   259→    if (prevTarget.current === target && count !== 0) return;
   260→    prevTarget.current = target;
   261→
   262→    let start = 0;
   263→    const startTime = performance.now();
   264→
   265→    function step(currentTime: number) {
   266→      const elapsed = currentTime - startTime;
   267→      const progress = Math.min(elapsed / duration, 1);
   268→      // ease-out cubic
   269→      const eased = 1 - Math.pow(1 - progress, 3);
   270→      const current = Math.round(eased * target);
   271→      setCount(current);
   272→      if (progress < 1) {
   273→        requestAnimationFrame(step);
   274→      }
   275→    }
   276→
   277→    requestAnimationFrame(step);
   278→  }, [target, duration]);
   279→
   280→  return <span>{count}</span>;
   281→}

   285→function NumberTicker({ value }: { value: number }) {
   286→  const prevValue = useRef(value);
   287→  const [animating, setAnimating] = useState(false);
   288→  const [flash, setFlash] = useState(false);
   289→
   290→  useEffect(() => {
   291→    if (prevValue.current !== value) {
   292→      prevValue.current = value;
   293→      const t1 = setTimeout(() => setAnimating(true), 0);
   294→      const t2 = setTimeout(() => setAnimating(false), 400);
   295→      const t3 = setTimeout(() => setFlash(true), 0);
   296→      const t4 = setTimeout(() => setFlash(false), 600);
   297→      return () => { clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4); };
   298→    }
   299→  }, [value]);
   300→
   301→  return (
   302→    <span className={`${animating ? 'number-ticker' : ''} ${flash ? 'score-change-flash' : ''}`} key={`${value}-${animating}`}>
   303→      {value}
   304→    </span>
   305→  );
   306→}

   310→function ConfettiParticles({ trigger }: { trigger: number }) {
   311→  const [particles, setParticles] = useState<Array<{ id: number; x: number; color: string; delay: number }>>([]);
   312→  const prevTrigger = useRef(trigger);
   313→

   316→      prevTrigger.current = trigger;
   317→      const colors = ['#DC2626', '#F59E0B', '#14B8A6', '#1E293B', '#FBBF24'];
   318→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   319→        id: Date.now() + i,
   320→        x: 30 + Math.random() * 40,
   321→        color: colors[Math.floor(Math.random() * colors.length)],
   322→        delay: i * 0.05,
   323→      }));
   324→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   325→      const t2 = setTimeout(() => setParticles([]), 900);
   326→      return () => { clearTimeout(t1); clearTimeout(t2); };
   327→    }
   328→  }, [trigger]);
   329→
   330→  return (
   331→    <>
   332→      {particles.map((p) => (
   333→        <div
   334→          key={p.id}
   335→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"
   336→          style={{
   337→            left: `${p.x}%`,

   350→function AwardsConfettiSection() {
   351→  const ref = useRef<HTMLDivElement>(null);
   352→  const [shown, setShown] = useState(false);
   353→
   354→  useEffect(() => {
   355→    const el = ref.current;
   356→    if (!el) return;
   357→    const observer = new IntersectionObserver(
   358→      ([entry]) => {
   359→        if (entry.isIntersecting && !shown) {
   360→          setShown(true);
   361→        }
   362→      },
   363→      { threshold: 0.3 }
   364→    );
   365→    observer.observe(el);
   366→    return () => observer.disconnect();
   367→  }, [shown]);
   368→
   369→  const confettiColors = ['#DC2626', '#F59E0B', '#14B8A6', '#FBBF24', '#1E293B'];
   370→
   371→  return (
   372→    <div ref={ref} className="absolute inset-0 overflow-hidden pointer-events-none z-0">
   373→      {shown && [...Array(8)].map((_, i) => (
   374→        <motion.div
   375→          key={`award-confetti-${i}`}
   376→          className="absolute w-1.5 h-1.5 rounded-full"
   377→          style={{
   378→            left: `${10 + i * 12}%`,
   379→            top: '40%',
   380→            backgroundColor: confettiColors[i % confettiColors.length],
   381→          }}
   382→          initial={{ y: 0, opacity: 0, scale: 0 }}
   383→          animate={{ y: [-10, -40, -80], opacity: [0, 1, 0], scale: [0, 1.2, 0.5] }}
   384→          transition={{ duration: 1.5, delay: i * 0.1, ease: 'easeOut' }}
   385→        />
   386→      ))}
   387→    </div>
   388→  );

   441→// ─── Countdown Timer ────────────────────────────────────────────────
   442→
   443→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   444→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   445→  const [expired, setExpired] = useState(!targetDate);
   446→
   447→  useEffect(() => {
   448→    if (!targetDate) {
   449→      return;
   450→    }
   451→
   452→    function calculate() {
   453→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   454→      if (diff === 0) {
   455→        setExpired(true);
   456→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   457→        return true;
   458→      }
   459→      setExpired(false);
   460→      setTimeLeft({
   461→        d: Math.floor(diff / 86400000),
   462→        h: Math.floor((diff % 86400000) / 3600000),
   463→        m: Math.floor((diff % 3600000) / 60000),
   464→        s: Math.floor((diff % 60000) / 1000),
   465→      });
   466→      return false;
   467→    }
   468→
   469→    const done = calculate();
   470→    if (done) return;
   471→
   472→    const id = setInterval(() => {
   473→      const finished = calculate();
   474→      if (finished) clearInterval(id);
   475→    }, 1000);
   476→
   477→    return () => clearInterval(id);
   478→  }, [targetDate]);
   479→
   480→  if (expired) {
   481→    return (
   482→      <motion.div
   483→        className="flex items-center justify-center gap-1.5 py-1"
   484→        animate={{ opacity: [1, 0.5, 1] }}
   485→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   486→      >
   487→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   488→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   489→      </motion.div>
   490→    );
   491→  }
   492→
   493→  if (!timeLeft) return null;
   494→
   495→  const units = [
   496→    { value: timeLeft.d, label: 'D' },
   497→    { value: timeLeft.h, label: 'H' },
   498→    { value: timeLeft.m, label: 'M' },
   499→    { value: timeLeft.s, label: 'S' },
   500→  ];
   501→
   502→  return (
   503→    <div className="flex items-center justify-center gap-1.5">
   504→      {units.map((unit, idx) => (
   505→        <div key={unit.label} className="flex items-center gap-1.5">
   506→          <div className="flex flex-col items-center">
   507→            <motion.div
   508→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   509→              key={`${unit.label}-${unit.value}`}
   510→              initial={{ rotateX: -90, opacity: 0 }}
   511→              animate={{ rotateX: 0, opacity: 1 }}
   512→              transition={{ duration: 0.3, ease: 'easeOut' }}
   513→            >
   514→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums countdown-flip">
   515→                {String(unit.value).padStart(2, '0')}
   516→              </span>
   517→            </motion.div>
   518→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   519→              {unit.label}
   520→            </span>
   521→          </div>
   522→          {idx < units.length - 1 && (
   523→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   524→          )}
   525→        </div>
   526→      ))}
   527→    </div>
   528→  );
   529→}
