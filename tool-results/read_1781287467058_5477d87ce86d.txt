   246→// ─── Animated Counter ───────────────────────────────────────────────
   247→
   248→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   249→  const [count, setCount] = useState(0);
   250→  const prevTarget = useRef(target);
   251→
   252→  useEffect(() => {
   253→    if (prevTarget.current === target && count !== 0) return;
   254→    prevTarget.current = target;
   255→
   256→    let start = 0;
   257→    const startTime = performance.now();
   258→
   259→    function step(currentTime: number) {
   260→      const elapsed = currentTime - startTime;
   261→      const progress = Math.min(elapsed / duration, 1);
   262→      // ease-out cubic
   263→      const eased = 1 - Math.pow(1 - progress, 3);
   264→      const current = Math.round(eased * target);
   265→      setCount(current);
   266→      if (progress < 1) {
   267→        requestAnimationFrame(step);
   268→      }
   269→    }
   270→
   271→    requestAnimationFrame(step);
   272→  }, [target, duration]);
   273→
   274→  return <span>{count}</span>;
   275→}

   279→function NumberTicker({ value }: { value: number }) {
   280→  const prevValue = useRef(value);
   281→  const [animating, setAnimating] = useState(false);
   282→  const [flash, setFlash] = useState(false);
   283→
   284→  useEffect(() => {
   285→    if (prevValue.current !== value) {
   286→      prevValue.current = value;
   287→      const t1 = setTimeout(() => setAnimating(true), 0);
   288→      const t2 = setTimeout(() => setAnimating(false), 400);
   289→      const t3 = setTimeout(() => setFlash(true), 0);
   290→      const t4 = setTimeout(() => setFlash(false), 600);
   291→      return () => { clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4); };
   292→    }
   293→  }, [value]);
   294→
   295→  return (
   296→    <span className={`${animating ? 'number-ticker' : ''} ${flash ? 'score-change-flash' : ''}`} key={`${value}-${animating}`}>
   297→      {value}
   298→    </span>
   299→  );
   300→}

   304→function ConfettiParticles({ trigger }: { trigger: number }) {
   305→  const [particles, setParticles] = useState<Array<{ id: number; x: number; color: string; delay: number }>>([]);
   306→  const prevTrigger = useRef(trigger);
   307→
   308→  useEffect(() => {
   309→    if (prevTrigger.current !== trigger) {
   310→      prevTrigger.current = trigger;
   311→      const colors = ['#DC2626', '#F59E0B', '#14B8A6', '#1E293B', '#FBBF24'];
   312→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   313→        id: Date.now() + i,
   314→        x: 30 + Math.random() * 40,
   315→        color: colors[Math.floor(Math.random() * colors.length)],
   316→        delay: i * 0.05,
   317→      }));
   318→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   319→      const t2 = setTimeout(() => setParticles([]), 900);
   320→      return () => { clearTimeout(t1); clearTimeout(t2); };
   321→    }
   322→  }, [trigger]);
   323→
   324→  return (
   325→    <>
   326→      {particles.map((p) => (
   327→        <div
   328→          key={p.id}
   329→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"
   330→          style={{
   331→            left: `${p.x}%`,
   332→            top: '20%',
   333→            backgroundColor: p.color,
   334→            animationDelay: `${p.delay}s`,

   336→        />
   337→      ))}
   338→    </>
   339→  );

   344→function AwardsConfettiSection() {
   345→  const ref = useRef<HTMLDivElement>(null);
   346→  const [shown, setShown] = useState(false);
   347→
   348→  useEffect(() => {
   349→    const el = ref.current;
   350→    if (!el) return;
   351→    const observer = new IntersectionObserver(
   352→      ([entry]) => {
   353→        if (entry.isIntersecting && !shown) {
   354→          setShown(true);
   355→        }
   356→      },
   357→      { threshold: 0.3 }
   358→    );
   359→    observer.observe(el);
   360→    return () => observer.disconnect();
   361→  }, [shown]);
   362→
   363→  const confettiColors = ['#DC2626', '#F59E0B', '#14B8A6', '#FBBF24', '#1E293B'];
   364→
   365→  return (
   366→    <div ref={ref} className="absolute inset-0 overflow-hidden pointer-events-none z-0">
   367→      {shown && [...Array(8)].map((_, i) => (
   368→        <motion.div
   369→          key={`award-confetti-${i}`}
   370→          className="absolute w-1.5 h-1.5 rounded-full"
   371→          style={{
   372→            left: `${10 + i * 12}%`,
   373→            top: '40%',
   374→            backgroundColor: confettiColors[i % confettiColors.length],
   375→          }}
   376→          initial={{ y: 0, opacity: 0, scale: 0 }}
   377→          animate={{ y: [-10, -40, -80], opacity: [0, 1, 0], scale: [0, 1.2, 0.5] }}
   378→          transition={{ duration: 1.5, delay: i * 0.1, ease: 'easeOut' }}
   379→        />
   380→      ))}
   381→    </div>
   382→  );
   383→}

   435→// ─── Countdown Timer ────────────────────────────────────────────────
   436→
   437→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   438→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   439→  const [expired, setExpired] = useState(!targetDate);
   440→
   441→  useEffect(() => {
   442→    if (!targetDate) {
   443→      return;
   444→    }
   445→
   446→    function calculate() {
   447→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   448→      if (diff === 0) {
   449→        setExpired(true);
   450→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   451→        return true;
   452→      }
   453→      setExpired(false);
   454→      setTimeLeft({
   455→        d: Math.floor(diff / 86400000),
   456→        h: Math.floor((diff % 86400000) / 3600000),
   457→        m: Math.floor((diff % 3600000) / 60000),
   458→        s: Math.floor((diff % 60000) / 1000),
   459→      });
   460→      return false;
   461→    }
   462→
   463→    const done = calculate();
   464→    if (done) return;
   465→
   466→    const id = setInterval(() => {
   467→      const finished = calculate();
   468→      if (finished) clearInterval(id);
   469→    }, 1000);
   470→
   471→    return () => clearInterval(id);
   472→  }, [targetDate]);
   473→
   474→  if (expired) {
   475→    return (
   476→      <motion.div
   477→        className="flex items-center justify-center gap-1.5 py-1"
   478→        animate={{ opacity: [1, 0.5, 1] }}
   479→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   480→      >
   481→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   482→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   483→      </motion.div>
   484→    );
   485→  }
   486→
   487→  if (!timeLeft) return null;
   488→
   489→  const units = [
   490→    { value: timeLeft.d, label: 'D' },
   491→    { value: timeLeft.h, label: 'H' },
   492→    { value: timeLeft.m, label: 'M' },
   493→    { value: timeLeft.s, label: 'S' },
   494→  ];
   495→
   496→  return (
   497→    <div className="flex items-center justify-center gap-1.5">
   498→      {units.map((unit, idx) => (
   499→        <div key={unit.label} className="flex items-center gap-1.5">
   500→          <div className="flex flex-col items-center">
   501→            <motion.div
   502→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   503→              key={`${unit.label}-${unit.value}`}
   504→              initial={{ rotateX: -90, opacity: 0 }}
   505→              animate={{ rotateX: 0, opacity: 1 }}
   506→              transition={{ duration: 0.3, ease: 'easeOut' }}
   507→            >
   508→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums countdown-flip">
   509→                {String(unit.value).padStart(2, '0')}
   510→              </span>
   511→            </motion.div>
   512→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   513→              {unit.label}
   514→            </span>
   515→          </div>
   516→          {idx < units.length - 1 && (
   517→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   518→          )}
   519→        </div>
   520→      ))}
   521→    </div>
   522→  );
   523→}
