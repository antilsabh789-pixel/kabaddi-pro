    72→  const [generatingBracket, setGeneratingBracket] = useState(false);
    73→
    74→  // Tournament search by code

    76→
    77→  // New tournament form

    85→  // Add teams to tournament

    94→
    95→  // Remove team confirmation

    97→  const [removingTeam, setRemovingTeam] = useState(false);

   110→        const res = await fetch('/api/teams?limit=100');

   112→          const data = await res.json();

   116→        // silently fail

   139→      if (tournamentSearch.trim()) params.set('search', tournamentSearch.trim());

   141→      const res = await fetch(`/api/tournaments?${params}`);
   142→      if (res.ok) {
   143→        const data = await res.json();
   144→        setTournaments(data.tournaments || []);
   145→      }

   147→      setTournaments([]);

   149→      setLoading(false);
   150→    }

   154→    const timer = setTimeout(() => {

   169→    try {
   170→      const res = await fetch('/api/tournaments', {
   171→        method: 'POST',
   172→        headers: { 'Content-Type': 'application/json' },
   173→        body: JSON.stringify({
   174→          ...newTournament,
   175→          organizerId: currentUser?.id,
   176→        }),
   177→      });

   179→        const data = await res.json();

   181→          title: 'Tournament Created!',
   182→          description: `Code: ${data.tournament?.tournamentCode || 'N/A'}`,

   184→        setCreateOpen(false);
   185→        fetchTournaments();
   186→      } else {
   187→        const data = await res.json();
   188→        toast({ title: 'Error', description: data.error, variant: 'destructive' });
   189→      }
   190→    } catch {
   191→      toast({ title: 'Error', description: 'Failed to create tournament', variant: 'destructive' });

   199→      const res = await fetch('/api/tournaments/generate-bracket', {
   200→        method: 'POST',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        body: JSON.stringify({ tournamentId, teamIds }),
   203→      });
   204→      if (res.ok) {
   205→        const data = await res.json();
   206→        toast({
   207→          title: 'Bracket Generated!',
   208→          description: `${data.matchCount} matches scheduled`,
   209→        });
   210→        fetchTournaments();
   211→      } else {
   212→        const data = await res.json();
   213→        toast({ title: 'Error', description: data.error || 'Failed to generate bracket', variant: 'destructive' });
   214→      }
   215→    } catch {
   216→      toast({ title: 'Error', description: 'Failed to generate bracket', variant: 'destructive' });

   218→      setGeneratingBracket(false);

   229→      (t.shortName && t.shortName.toLowerCase().includes(q))

   249→          }
   250→        } catch {
   251→          // keep local results
   252→        } finally {
   253→          setIsSearchingTeams(false);
   254→        }

   258→  }, [teamSearch, addTeamDialogOpen, getFilteredTeams]);
   259→

   261→    const tournament = tournaments.find(t => t.id === tournamentId);
   262→    setSelectedTeamIds([]);
   263→    setTeamSearch('');
   264→    setTeamSearchResults(getFilteredTeams(''));
   265→    setAddTeamDialogOpen(tournamentId);
   266→  };

   268→  const handleAddTeams = async () => {
   269→    if (!addTeamDialogOpen || selectedTeamIds.length === 0) return;

   271→    try {
   272→      const res = await fetch(`/api/tournaments/${addTeamDialogOpen}`, {
   273→        method: 'PATCH',
   274→        headers: { 'Content-Type': 'application/json' },
   275→        body: JSON.stringify({ addTeamIds: selectedTeamIds }),

   277→      if (res.ok) {
   278→        const data = await res.json();

   280→          title: 'Teams Added!',

   283→        setAddTeamDialogOpen(null);
   284→        setSelectedTeamIds([]);
   285→        fetchTournaments();

   287→        const data = await res.json();
   288→        toast({ title: 'Error', description: data.error, variant: 'destructive' });

   291→      toast({ title: 'Error', description: 'Failed to add teams', variant: 'destructive' });
   292→    } finally {
   293→      setAddingTeams(false);
   294→    }
   295→  };
   296→

   298→    setRemovingTeam(true);
   299→    try {
   300→      const res = await fetch(`/api/tournaments/${tournamentId}`, {
   301→        method: 'PATCH',
   302→        headers: { 'Content-Type': 'application/json' },
   303→        body: JSON.stringify({ removeTeamIds: [teamId] }),
   304→      });
   305→      if (res.ok) {
   306→        toast({ title: 'Team Removed', description: 'Team removed from tournament' });
   307→        setRemoveTeamId(null);
   308→        fetchTournaments();
   309→      } else {
   310→        const data = await res.json();
   311→        toast({ title: 'Error', description: data.error, variant: 'destructive' });
   312→      }
   313→    } catch {
   314→      toast({ title: 'Error', description: 'Failed to remove team', variant: 'destructive' });
   315→    } finally {
   316→      setRemovingTeam(false);
   317→    }

   321→    try {
   322→      await navigator.clipboard.writeText(code);
   323→      setCopiedCode(code);
   324→      toast({ title: 'Copied!', description: `Code ${code} copied to clipboard` });

   326→    } catch {
   327→      // fallback
   328→    }
   329→  };

   332→    if (t.status !== statusFilter) return false;
   333→    if (genderFilter !== 'all' && t.gender !== genderFilter) return false;
   334→    return true;
   335→  });

   338→    if (!query.trim()) return text;
   339→    const regex = new RegExp(`(${query.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')})`, 'gi');
   340→    const parts = text.split(regex);
   341→    return parts.map((part, i) =>
   342→      regex.test(part) ? (
   343→        <span key={i} className="text-brand-red font-bold">{part}</span>
   344→      ) : (
   345→        <span key={i}>{part}</span>
   346→      )
   347→    );
   348→  };
   349→
   350→  return (
   351→    <div className="px-4 py-6 space-y-5">
   352→      {/* Premium Upgrade Modal */}
   353→      {showUpgrade && (
   354→        <PremiumUpgradeScreen
   355→          onClose={() => setShowUpgrade(false)}
   356→          feature="Host Tournaments"
   357→        />

   360→      {/* Header */}

   373→            }`}
   374→          >
   375→            {isPremium ? (
   376→              <>
   377→                <Plus className="w-4 h-4 mr-1" />
   378→                Create
   379→              </>
   380→            ) : (
   381→              <>
   382→                <Crown className="w-4 h-4 mr-1" />
   383→                Host
   384→                <Lock className="w-3 h-3 ml-1" />
   385→              </>
   386→            )}
   387→          </Button>
   388→        </div>
   389→
   390→        {/* Create Tournament Dialog */}

   398→                  PRO
   399→                </Badge>

   401→            </DialogHeader>
   402→            <div className="space-y-4 mt-2">

   404→                placeholder="Tournament name"

   410→                placeholder="Venue"

   416→                {(['knockout', 'league', 'hybrid'] as const).map((t) => (
   417→                  <button
   418→                    key={t}
   419→                    onClick={() => setNewTournament({ ...newTournament, type: t })}

   421→                      newTournament.type === t
   422→                        ? 'border-brand-red bg-brand-red/10 text-brand-red'
   423→                        : 'border-warm-300 text-warm-600'
   424→                    }`}
   425→                  >
   426→                    {t}
   427→                  </button>
   428→                ))}
   429→              </div>
   430→              <div className="grid grid-cols-2 gap-3">
   431→                <button
   432→                  onClick={() => setNewTournament({ ...newTournament, gender: 'male' })}
   433→                  className={`p-3 rounded-xl border-2 text-sm font-medium ${
   434→                    newTournament.gender === 'male'
   435→                      ? 'border-brand-blue bg-brand-blue/10 text-brand-blue'
   436→                      : 'border-warm-300 text-warm-600'
   437→                  }`}
   438→                >
   439→                  ♂ Boys

   441→                <button
   442→                  onClick={() => setNewTournament({ ...newTournament, gender: 'female' })}

   448→                >
   449→                  ♀ Girls
   450→                </button>
   451→              </div>
   452→              <div className="bg-warm-100 rounded-xl p-3 text-center">

   483→          </button>
   484→        )}
   485→      </div>
   486→
   487→      {/* Premium hint for non-premium users */}
   488→      {!isPremium && (
   489→        <motion.div

   494→            onClick={() => setShowUpgrade(true)}

   499→              </div>
   500→              <div className="flex-1 min-w-0">
   501→                <p className="text-xs font-semibold text-warm-800">Want to host your own tournament?</p>

   522→            {s}
