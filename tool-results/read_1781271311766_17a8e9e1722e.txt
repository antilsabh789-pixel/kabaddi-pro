  1689→  const [player, setPlayer] = useState<{
  1690→    name: string;
  1691→    avatar: string | null;
  1692→    stat: number;
  1693→    statLabel: string;
  1694→    teamNames: string[];

  1696→
  1697→  useEffect(() => {
  1698→    async function fetchPreview() {
  1699→      try {

  1701→        if (!res.ok) return;
  1702→        const data = await res.json();
  1703→        const entry = data.leaderboard?.[rank - 1];

  1706→            name: entry.name,
  1707→            avatar: entry.avatar,
