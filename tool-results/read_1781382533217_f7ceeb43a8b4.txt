    55→    55→/* ── Password Strength Helper ── */
    56→    56→function getPasswordStrength(pwd: string): { label: string; color: string; pct: number; barColor: string } {
    57→    57→  if (!pwd) return { label: '', color: '', pct: 0, barColor: '' };
    58→    58→  let score = 0;
    59→    59→  if (pwd.length >= 6) score++;
    60→    60→  if (pwd.length >= 8) score++;
    61→    61→  if (/[A-Z]/.test(pwd)) score++;
    62→    62→  if (/[0-9]/.test(pwd)) score++;
    63→    63→  if (/[^A-Za-z0-9]/.test(pwd)) score++;
    64→    64→
    65→    65→  if (score <= 1) return { label: 'Weak', color: 'text-red-500', pct: 25, barColor: 'bg-red-500' };
    66→    66→  if (score <= 2) return { label: 'Medium', color: 'text-amber-500', pct: 50, barColor: 'bg-amber-500' };
    67→    67→  if (score <= 3) return { label: 'Strong', color: 'text-emerald-500', pct: 75, barColor: 'bg-emerald-500' };
    68→    68→  return { label: 'Very Strong', color: 'text-teal-500', pct: 100, barColor: 'bg-teal-500' };
    69→    69→}

   167→   167→/* ── OTP Input Boxes ── */
   168→   168→function OTPInput({ value, onChange, length = 6 }: { value: string; onChange: (val: string) => void; length?: number }) {
   169→   169→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   170→   170→
   171→   171→  const handleChange = (index: number, digit: string) => {
   172→   172→    if (!/^\d*$/.test(digit)) return;
   173→   173→    const newVal = value.split('');
   174→   174→    newVal[index] = digit;
   175→   175→    const combined = newVal.join('');
   176→   176→    onChange(combined);
   177→   177→    if (digit && index < length - 1) {
   178→   178→      inputRefs.current[index + 1]?.focus();
   179→   179→    }
   180→   180→  };
   181→   181→
   182→   182→  const handleKeyDown = (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   183→   183→    if (e.key === 'Backspace' && !value[index] && index > 0) {
   184→   184→      inputRefs.current[index - 1]?.focus();
   185→   185→    }
   186→   186→  };
   187→   187→
   188→   188→  const handlePaste = (e: React.ClipboardEvent) => {
   189→   189→    e.preventDefault();
   190→   190→    const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, length);
   191→   191→    onChange(pasted.padEnd(length, ''));
   192→   192→    const focusIdx = Math.min(pasted.length, length - 1);
   193→   193→    inputRefs.current[focusIdx]?.focus();
   194→   194→  };
   195→   195→
   196→   196→  return (
   197→   197→    <div className="flex gap-2 justify-center" onPaste={handlePaste}>
   198→   198→      {Array.from({ length }, (_, i) => (
   199→   199→        <motion.div
   200→   200→          key={i}
   201→   201→          initial={{ opacity: 0, y: 10 }}
   202→   202→          animate={{ opacity: 1, y: 0 }}
   203→   203→          transition={{ delay: i * 0.05 }}
   204→   204→        >
   205→   205→          <input
   206→   206→            ref={(el) => { inputRefs.current[i] = el; }}
   207→   207→            type="text"
   208→   208→            inputMode="numeric"
   209→   209→            maxLength={1}
   210→   210→            value={value[i] || ''}
   211→   211→            onChange={(e) => handleChange(i, e.target.value)}
   212→   212→            onKeyDown={(e) => handleKeyDown(i, e)}
   213→   213→            className="w-11 h-13 text-center text-lg font-bold rounded-xl border-2 border-warm-300/60 dark:border-warm-600/40 bg-white/60 dark:bg-white/5 text-warm-800 dark:text-warm-100 focus:border-brand-red focus:ring-2 focus:ring-brand-red/30 outline-none transition-all"
   214→   214→            style={{ height: 52 }}
   215→   215→          />
   216→   216→        </motion.div>
   217→   217→      ))}
   218→   218→    </div>
   219→   219→  );
   220→   220→}
