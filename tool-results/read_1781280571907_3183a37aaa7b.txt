   183→   183→/** Format time as MM:SS */
   184→   184→function formatTime(seconds: number): string {
   185→   185→  const m = Math.floor(Math.max(0, seconds) / 60);
   186→   186→  const s = Math.max(0, seconds) % 60;
   187→   187→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   188→   188→}
   189→   189→
   190→   190→/** Format relative timestamp */
   191→   191→function formatEventTime(timestamp: number, matchStart: number): string {
   192→   192→  const diff = Math.max(0, Math.floor((timestamp - matchStart) / 1000));
   193→   193→  return formatTime(diff);
   194→   194→}

   198→   198→function CircularTimer({
   199→   199→  seconds,
   200→   200→  totalSeconds,
   201→   201→  teamColor,
   202→   202→}: {
   203→   203→  seconds: number;
   204→   204→  totalSeconds: number;
   205→   205→  teamColor: string;
   206→   206→}) {
   207→   207→  const radius = 28;
   208→   208→  const stroke = 3;
   209→   209→  const normalizedRadius = radius - stroke;
   210→   210→  const circumference = normalizedRadius * 2 * Math.PI;
   211→   211→  const progress = totalSeconds > 0 ? Math.max(0, seconds) / totalSeconds : 0;
   212→   212→  const strokeDashoffset = circumference * (1 - progress);
   213→   213→
   214→   214→  return (
   215→   215→    <div className="relative w-16 h-16 flex items-center justify-center">
   216→   216→      <svg height={radius * 2} width={radius * 2} className="-rotate-90">
   217→   217→        <circle
   218→   218→          stroke="currentColor"
   219→   219→          className="text-gray-200 dark:text-warm-700"
   220→   220→          fill="transparent"
   221→   221→          strokeWidth={stroke}
   222→   222→          r={normalizedRadius}
   223→   223→          cx={radius}
   224→   224→          cy={radius}
   225→   225→        />
   226→   226→        <circle
   227→   227→          stroke={teamColor}
   228→   228→          fill="transparent"
   229→   229→          strokeWidth={stroke}
   230→   230→          strokeDasharray={circumference + ' ' + circumference}
   231→   231→          style={{ strokeDashoffset }}
   232→   232→          strokeLinecap="round"
   233→   233→          r={normalizedRadius}
   234→   234→          cx={radius}
   235→   235→          cy={radius}
   236→   236→          className="transition-all duration-1000 ease-linear"
   237→   237→        />
   238→   238→      </svg>
   239→   239→      <div className="absolute inset-0 flex items-center justify-center">
   240→   240→        <span className="text-xs font-bold text-gray-800 dark:text-warm-100 tabular-nums">
   241→   241→          {formatTime(seconds)}
   242→   242→        </span>
   243→   243→      </div>
   244→   244→    </div>
   245→   245→  );
   246→   246→}

   250→   250→function LiveEventCard({
   251→   251→  event,
   252→   252→  match,
   253→   253→  matchStart,
   254→   254→  index,
   255→   255→}: {
   256→   256→  event: MatchEventAPI;
   257→   257→  match: MatchDataAPI;
   258→   258→  matchStart: number;
   259→   259→  index: number;
   260→   260→}) {
   261→   261→  const isHome = event.teamId === match.homeTeamId;
   262→   262→  const teamName = isHome ? match.homeTeam : match.awayTeam;
   263→   263→  const teamColor = isHome ? match.homeTeamColor : match.awayTeamColor;
   264→   264→  const eventType = event.eventType as EventType;
   265→   265→  const big = isBigEvent(eventType);
   266→   266→  const iconColor = getEventIconColor(eventType);
   267→   267→  const bgColor = getEventBgColor(eventType);
   268→   268→
   269→   269→  // Build commentary text
   270→   270→  let commentaryText = '';
   271→   271→  try {
   272→   272→    let extras: CommentaryExtras = {};
   273→   273→    if (event.details) {
   274→   274→      const parsed = JSON.parse(event.details);
   275→   275→      if (parsed.isSuperRaid) extras.isSuperRaid = true;
   276→   276→      if (parsed.isSuperTackle) extras.isSuperTackle = true;
   277→   277→      if (parsed.isDoOrDie) extras.isDoOrDie = true;
   278→   278→      if (parsed.isAllOut) extras.isAllOut = true;
   279→   279→      if (parsed.touchedPlayerIds) extras.defendersTouched = (parsed.touchedPlayerIds as string[]).length;
   280→   280→    }
   281→   281→    commentaryText = generateCommentary(event.eventType, event.playerId || 'Player', teamName, event.value, extras);
   282→   282→  } catch {
   283→   283→    commentaryText = `${event.playerId || 'Player'} made a play for ${teamName}`;
   284→   284→  }
   285→   285→
   286→   286→  return (
   287→   287→    <motion.div
   288→   288→      initial={{ opacity: 0, x: 40, scale: 0.95 }}
   289→   289→      animate={{ opacity: 1, x: 0, scale: 1 }}
   290→   290→      transition={{ duration: 0.35, delay: index * 0.03, type: 'spring', stiffness: 200 }}
   291→   291→    >
   292→   292→      <div
   293→   293→        className={cn(
   294→   294→          'relative flex items-start gap-3 p-3 rounded-xl border transition-all',
   295→   295→          big
   296→   296→            ? 'border-amber-300/50 dark:border-amber-600/40 bg-amber-50/80 dark:bg-amber-900/20'
   297→   297→            : 'border-gray-100 dark:border-warm-700 bg-white dark:bg-warm-800/60'
   298→   298→        )}
   299→   299→      >
   300→   300→        {/* Big event glow */}
   301→   301→        {big && (
   302→   302→          <div
   303→   303→            className="absolute inset-0 rounded-xl opacity-20 animate-pulse"
   304→   304→            style={{ boxShadow: `0 0 20px ${teamColor}40` }}
   305→   305→          />
   306→   306→        )}
   307→   307→
   308→   308→        {/* Left color bar */}
   309→   309→        <div
   310→   310→          className="w-1 rounded-full min-h-[40px] flex-shrink-0 self-stretch"
   311→   311→          style={{ backgroundColor: teamColor }}
   312→   312→        />
   313→   313→
   314→   314→        {/* Icon */}
   315→   315→        <div className={cn('w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0', bgColor)}>
   316→   316→          <EventIconComponent type={eventType} className={cn('w-4 h-4', iconColor)} />
   317→   317→        </div>
   318→   318→
   319→   319→        {/* Content */}
   320→   320→        <div className="flex-1 min-w-0">
   321→   321→          <div className="flex items-start justify-between gap-2">
   322→   322→            <div>
   323→   323→              <div className="flex items-center gap-1.5 mb-0.5">
   324→   324→                <Badge
   325→   325→                  className={cn(
   326→   326→                    'text-[9px] font-bold px-1.5 py-0 h-4 border-0',
   327→   327→                    big ? 'bg-amber-500/20 text-amber-700 dark:text-amber-400' : 'bg-gray-100 dark:bg-warm-700 text-gray-500 dark:text-warm-400'
   328→   328→                  )}
   329→   329→                >
   330→   330→                  {getEventLabel(eventType)}
   331→   331→                </Badge>
   332→   332→                {event.half === 1 ? (
   333→   333→                  <span className="text-[9px] text-gray-400">1H</span>
   334→   334→                ) : (
   335→   335→                  <span className="text-[9px] text-gray-400">2H</span>
   336→   336→                )}
   337→   337→              </div>
   338→   338→              <p className="text-xs font-medium text-gray-700 dark:text-warm-200 leading-snug line-clamp-2">
   339→   339→                {commentaryText}
   340→   340→              </p>
   341→   341→            </div>
   342→   342→            {event.value > 0 && (
   343→   343→              <motion.div
   344→   344→                initial={{ scale: 0.5 }}
   345→   345→                animate={{ scale: 1 }}
   346→   346→                className="flex-shrink-0"
   347→   347→              >
   348→   348→                <Badge
   349→   349→                  className="text-sm font-black px-2 py-0 h-6 border-0"
   350→   350→                  style={{ backgroundColor: teamColor + '20', color: teamColor }}
   351→   351→                >
   352→   352→                  +{event.value}
   353→   353→                </Badge>
   354→   354→              </motion.div>
   355→   355→            )}
   356→   356→          </div>
   357→   357→          <div className="flex items-center gap-2 mt-1">
   358→   358→            <span className="text-[10px] text-gray-400 dark:text-warm-500 font-medium">
   359→   359→              {formatEventTime(event.timestamp, matchStart)}
   360→   360→            </span>
   361→   361→            <span className="text-[10px] text-gray-300 dark:text-warm-600">·</span>
   362→   362→            <span className="text-[10px] font-medium" style={{ color: teamColor }}>
   363→   363→              {teamName}
   364→   364→            </span>
   365→   365→          </div>
   366→   366→        </div>
   367→   367→      </div>
   368→   368→    </motion.div>
   369→   369→  );
   370→   370→}
