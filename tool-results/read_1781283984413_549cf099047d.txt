    55→
    56→function timeAgo(dateStr: string): string {
    57→  try {
    58→    const diff = Date.now() - new Date(dateStr).getTime();
    59→    const seconds = Math.floor(diff / 1000);
    60→    if (seconds < 60) return 'just now';
    61→    const minutes = Math.floor(seconds / 60);
    62→    if (minutes < 60) return `${minutes}m ago`;
    63→    const hours = Math.floor(minutes / 60);
    64→    if (hours < 24) return `${hours}h ago`;
    65→    const days = Math.floor(hours / 24);
    66→    if (days < 30) return `${days}d ago`;
    67→    const months = Math.floor(days / 30);
    68→    return `${months}mo ago`;
    69→  } catch {
    70→    return '';
    71→  }
    72→}
