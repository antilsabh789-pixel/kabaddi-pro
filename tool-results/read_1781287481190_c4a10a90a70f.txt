   132→
   133→export default function MatchTimelineScreen({ onClose }: { onClose: () => void }) {
   134→  const [currentStep, setCurrentStep] = useState(MOCK_EVENTS.length - 1);
   135→  const [isPlaying, setIsPlaying] = useState(false);
   136→  const [playSpeed, setPlaySpeed] = useState<number>(1);
   137→  const [activeFilter, setActiveFilter] = useState<EventCategory>('all');
   138→  const [expandedEvent, setExpandedEvent] = useState<string | null>(null);
   139→  const timelineRef = useRef<HTMLDivElement>(null);
   140→  const eventRefs = useRef<Record<string, HTMLDivElement | null>>({});
   141→
   142→  // ── Filtered events ──────────────────────────────────────────────────────
   143→  const filteredEvents = useMemo(() => {
   144→    if (activeFilter === 'all') return MOCK_EVENTS;
   145→    return MOCK_EVENTS.filter(evt => {
   146→      const config = EVENT_CONFIG[evt.eventType];
   147→      return config?.category.includes(activeFilter) ?? false;
   148→    });
   149→  }, [activeFilter]);
   150→
   151→  // ── Compute running scores ───────────────────────────────────────────────
   152→  const getRunningScore = useCallback((step: number) => {
   153→    let homeScore = 0;
   154→    let awayScore = 0;
   155→    for (let i = 0; i <= step && i < MOCK_EVENTS.length; i++) {
   156→      const evt = MOCK_EVENTS[i];
   157→      if (evt.teamId === 'home') {
   158→        homeScore += evt.value;
   159→      } else {
   160→        awayScore += evt.value;
   161→      }
   162→    }
   163→    return { homeScore, awayScore };
   164→  }, []);
   165→
   166→  const score = getRunningScore(currentStep);
   167→
   168→  // ── Summary stats ────────────────────────────────────────────────────────
   169→  const summaryStats = useMemo(() => {
   170→    const visibleEvents = MOCK_EVENTS.slice(0, currentStep + 1);
   171→    const totalRaids = visibleEvents.filter(e => e.eventType === 'raid_point' || e.eventType === 'super_raid' || e.eventType === 'do_or_die_raid').length;
   172→    const totalTackles = visibleEvents.filter(e => e.eventType === 'tackle_point' || e.eventType === 'super_tackle').length;
   173→    const totalBonus = visibleEvents.filter(e => e.eventType === 'bonus_point').length;
   174→    const totalAllOuts = visibleEvents.filter(e => e.eventType === 'all_out').length;
   175→    return { totalRaids, totalTackles, totalBonus, totalAllOuts };
   176→  }, [currentStep]);
   177→
   178→  // ── Key moment positions (for scrubber markers) ──────────────────────────
   179→  const keyMoments = useMemo(() => {
   180→    return MOCK_EVENTS
   181→      .map((evt, idx) => {
   182→        const isKey = ['super_raid', 'super_tackle', 'all_out', 'do_or_die_raid'].includes(evt.eventType);
   183→        return isKey ? { index: idx, eventType: evt.eventType, label: EVENT_CONFIG[evt.eventType]?.label || evt.eventType } : null;
   184→      })
   185→      .filter((m): m is { index: number; eventType: string; label: string } => m !== null);
   186→  }, []);
   187→
   188→  // ── Auto-play logic ──────────────────────────────────────────────────────
   189→  useEffect(() => {
   190→    if (!isPlaying) return;
   191→    const interval = setInterval(() => {
   192→      setCurrentStep(prev => {
   193→        if (prev >= MOCK_EVENTS.length - 1) {
   194→          setIsPlaying(false);
   195→          return prev;
   196→        }
   197→        return prev + 1;
   198→      });
   199→    }, 1500 / playSpeed);
   200→    return () => clearInterval(interval);
   201→  }, [isPlaying, playSpeed]);
   202→
   203→  // ── Auto-scroll to current event during playback ─────────────────────────
   204→  useEffect(() => {
   205→    if (!isPlaying) return;
   206→    const currentEvent = MOCK_EVENTS[currentStep];
   207→    if (!currentEvent) return;
   208→    const el = eventRefs.current[currentEvent.id];
   209→    if (el) {
   210→      el.scrollIntoView({ behavior: 'smooth', block: 'center' });
   211→    }
   212→  }, [currentStep, isPlaying]);
   213→
   214→  // ── Handlers ─────────────────────────────────────────────────────────────
   215→  const handlePlayPause = () => {
   216→    if (currentStep >= MOCK_EVENTS.length - 1) {
   217→      setCurrentStep(0);
   218→    }
   219→    setIsPlaying(!isPlaying);
   220→  };
   221→
   222→  const handleReset = () => {
   223→    setCurrentStep(0);
   224→    setIsPlaying(false);
   225→  };
   226→
   227→  const handleStepForward = () => {
   228→    setIsPlaying(false);
   229→    setCurrentStep(prev => Math.min(prev + 1, MOCK_EVENTS.length - 1));
   230→  };
   231→
   232→  const handleStepBack = () => {
   233→    setIsPlaying(false);
   234→    setCurrentStep(prev => Math.max(prev - 1, 0));
   235→  };
   236→
   237→  const handleScrub = (position: number) => {
   238→    setIsPlaying(false);
   239→    const step = Math.round((position / 100) * (MOCK_EVENTS.length - 1));
   240→    setCurrentStep(step);
   241→  };
   242→
   243→  const toggleExpand = (eventId: string) => {
   244→    setExpandedEvent(prev => prev === eventId ? null : eventId);
   245→  };
   246→
   247→  // ── Progress ─────────────────────────────────────────────────────────────
   248→  const progressPercent = MOCK_EVENTS.length > 0
   249→    ? ((currentStep + 1) / MOCK_EVENTS.length) * 100
   250→    : 0;
   251→
   252→  // ── Find half transition index ───────────────────────────────────────────
   253→  const halfTwoStartIndex = useMemo(() => {
   254→    return MOCK_EVENTS.findIndex(e => e.half === 2);
   255→  }, []);
   256→
   257→  return (
   258→    <motion.div
   259→      initial={{ opacity: 0 }}
   260→      animate={{ opacity: 1 }}
   261→      exit={{ opacity: 0 }}
   262→      className="fixed inset-0 z-50 bg-warm-50 dark:bg-warm-900 overflow-y-auto"
   263→    >
   264→      {/* ─── Header ────────────────────────────────────────────────────────── */}
   265→      <div className="sticky top-0 z-20">
   266→        {/* Glass-morphism Score Header */}
   267→        <div className="bg-gradient-to-r from-brand-navy/95 to-brand-blue/95 dark:from-brand-navy-dark/95 dark:to-brand-navy/95 backdrop-blur-xl border-b border-white/10">
   268→          <div className="flex items-center justify-between px-4 py-3">
   269→            <div className="flex items-center gap-2">
   270→              <Timer className="w-5 h-5 text-brand-gold" />
   271→              <h1 className="text-lg font-bold text-white">Match Timeline</h1>
   272→            </div>
   273→            <button onClick={onClose} className="p-2 rounded-full hover:bg-white/20 transition-colors">
   274→              <X className="w-5 h-5 text-white" />
   275→            </button>
   276→          </div>
   277→
   278→          {/* Score Display */}
   279→          <div className="px-4 pb-3">
   280→            <div className="flex items-center justify-between">
   281→              {/* Home Team */}
   282→              <div className="flex flex-col items-center gap-1 flex-1">
   283→                <div
   284→                  className="w-11 h-11 rounded-full flex items-center justify-center text-xs font-bold text-white shadow-lg"
   285→                  style={{ backgroundColor: HOME_TEAM.color }}
   286→                >
   287→                  {HOME_TEAM.shortName}
   288→                </div>
   289→                <span className="text-[11px] text-warm-300 text-center truncate max-w-[90px]">
   290→                  {HOME_TEAM.name}
   291→                </span>
   292→              </div>
   293→
   294→              {/* Score */}
   295→              <div className="flex items-center gap-3 px-3">
   296→                <motion.span
   297→                  key={`home-${score.homeScore}`}
   298→                  initial={{ scale: 1.4, color: '#FCD34D' }}
   299→                  animate={{ scale: 1, color: '#FFFFFF' }}
   300→                  transition={{ duration: 0.4 }}
   301→                  className="text-3xl font-black text-white"
   302→                >
   303→                  {score.homeScore}
   304→                </motion.span>
   305→                <div className="flex flex-col items-center">
   306→                  <span className="text-warm-400 text-sm font-medium">-</span>
   307→                </div>
   308→                <motion.span
   309→                  key={`away-${score.awayScore}`}
   310→                  initial={{ scale: 1.4, color: '#FCD34D' }}
   311→                  animate={{ scale: 1, color: '#FFFFFF' }}
   312→                  transition={{ duration: 0.4 }}
   313→                  className="text-3xl font-black text-white"
   314→                >
   315→                  {score.awayScore}
   316→                </motion.span>
   317→              </div>
   318→
   319→              {/* Away Team */}
   320→              <div className="flex flex-col items-center gap-1 flex-1">
   321→                <div
   322→                  className="w-11 h-11 rounded-full flex items-center justify-center text-xs font-bold text-white shadow-lg"
   323→                  style={{ backgroundColor: AWAY_TEAM.color }}
   324→                >
   325→                  {AWAY_TEAM.shortName}
   326→                </div>
   327→                <span className="text-[11px] text-warm-300 text-center truncate max-w-[90px]">
   328→                  {AWAY_TEAM.name}
   329→                </span>
   330→              </div>
   331→            </div>
   332→
   333→            {/* Half & Time indicator */}
   334→            <div className="mt-2 flex items-center justify-center gap-2">
   335→              <Badge variant="secondary" className="text-[10px] bg-white/10 text-warm-300 border-0">
   336→                {MOCK_EVENTS[currentStep]?.half === 1 ? '1st Half' : '2nd Half'}
   337→              </Badge>
   338→              <span className="text-[10px] text-warm-400">
   339→                {formatTime(MOCK_EVENTS[currentStep]?.minute ?? 0, MOCK_EVENTS[currentStep]?.second ?? 0)}
   340→              </span>
   341→            </div>
   342→          </div>
   343→        </div>
   344→
   345→        {/* Summary Stats Bar */}
   346→        <div className="bg-white/80 dark:bg-warm-800/80 backdrop-blur-md border-b border-warm-200 dark:border-warm-700">
   347→          <div className="flex items-center justify-around py-2 px-3">
