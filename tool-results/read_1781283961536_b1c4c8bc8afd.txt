    47→    47→function getInitial(name: string | null | undefined): string {
    48→    48→  if (!name) return '?';
    49→    49→  return name.trim().charAt(0).toUpperCase();
    50→    50→}

    56→    56→function timeAgo(dateStr: string): string {
    57→    57→  try {
    58→    58→    const diff = Date.now() - new Date(dateStr).getTime();
    59→    59→    const seconds = Math.floor(diff / 1000);
    60→    60→    if (seconds < 60) return 'just now';
    61→    61→    const minutes = Math.floor(seconds / 60);
    62→    62→    if (minutes < 60) return `${minutes}m ago`;
    63→    63→    const hours = Math.floor(minutes / 60);
    64→    64→    if (hours < 24) return `${hours}h ago`;
    65→    65→    const days = Math.floor(hours / 24);
    66→    66→    if (days < 30) return `${days}d ago`;
    67→    67→    const months = Math.floor(days / 30);
    68→    68→    return `${months}mo ago`;
    69→    69→  } catch {
    70→    70→    return '';
    71→    71→  }
    72→    72→}

   192→   192→export default function FollowScreen({ onClose }: FollowScreenProps) {
   193→   193→  const currentUser = useKabaddiStore((s) => s.currentUser);
   194→   194→  const { toast } = useToast();
   195→   195→
   196→   196→  // Tab state
   197→   197→  const [activeTab, setActiveTab] = useState<TabId>('search');
   198→   198→
   199→   199→  // Counts
   200→   200→  const [followerCount, setFollowerCount] = useState(0);
   201→   201→  const [followingCount, setFollowingCount] = useState(0);
   202→   202→  const [countsLoading, setCountsLoading] = useState(true);
   203→   203→
   204→   204→  // Search
   205→   205→  const [searchQuery, setSearchQuery] = useState('');
   206→   206→  const [searchResults, setSearchResults] = useState<PlayerResult[]>([]);
   207→   207→  const [searchLoading, setSearchLoading] = useState(false);
   208→   208→
   209→   209→  // Followers
   210→   210→  const [followers, setFollowers] = useState<FollowerEntry[]>([]);
   211→   211→  const [followersLoading, setFollowersLoading] = useState(false);
   212→   212→
   213→   213→  // Following
   214→   214→  const [following, setFollowing] = useState<FollowerEntry[]>([]);
   215→   215→  const [followingLoading, setFollowingLoading] = useState(false);
   216→   216→
   217→   217→  // Follow/Unfollow action loading per player
   218→   218→  const [actionLoadingId, setActionLoadingId] = useState<string | null>(null);
   219→   219→
   220→   220→  // Track which users the current user is following (for followers tab)
   221→   221→  const [followedIds, setFollowedIds] = useState<Set<string>>(new Set());
   222→   222→
   223→   223→  // ─── Fetch counts ───────────────────────────────────────────
   224→   224→
   225→   225→  const fetchCounts = useCallback(async () => {
   226→   226→    if (!currentUser) return;
   227→   227→    setCountsLoading(true);
   228→   228→    try {
   229→   229→      const res = await fetch(
   230→   230→        `/api/follow?userId=${currentUser.id}&type=counts`
   231→   231→      );
   232→   232→      if (!res.ok) throw new Error('Failed to fetch counts');
   233→   233→      const data = await res.json();
   234→   234→      setFollowerCount(data.followerCount || 0);
   235→   235→      setFollowingCount(data.followingCount || 0);
   236→   236→    } catch (err) {
   237→   237→      console.error('Fetch counts error:', err);
   238→   238→    } finally {
   239→   239→      setCountsLoading(false);
   240→   240→    }
   241→   241→  }, [currentUser]);
   242→   242→
   243→   243→  // ─── Search players ─────────────────────────────────────────
   244→   244→
   245→   245→  const searchPlayers = useCallback(
   246→   246→    async (query: string) => {
   247→   247→      if (!currentUser) return;
   248→   248→      setSearchLoading(true);
   249→   249→      try {
   250→   250→        const params = new URLSearchParams({
   251→   251→          userId: currentUser.id,
   252→   252→          type: 'search',
   253→   253→          search: query,
   254→   254→        });
   255→   255→        const res = await fetch(`/api/follow?${params}`);
   256→   256→        if (!res.ok) throw new Error('Search failed');
   257→   257→        const data = await res.json();
   258→   258→        setSearchResults(data.players || []);
   259→   259→      } catch (err) {
   260→   260→        console.error('Search players error:', err);
   261→   261→        setSearchResults([]);
   262→   262→      } finally {
   263→   263→        setSearchLoading(false);
   264→   264→      }
   265→   265→    },
   266→   266→    [currentUser]
   267→   267→  );
   268→   268→
   269→   269→  // ─── Fetch followers ────────────────────────────────────────
   270→   270→
   271→   271→  const fetchFollowers = useCallback(async () => {
   272→   272→    if (!currentUser) return;
   273→   273→    setFollowersLoading(true);
   274→   274→    try {
   275→   275→      const res = await fetch(
   276→   276→        `/api/follow?userId=${currentUser.id}&type=followers`
   277→   277→      );
   278→   278→      if (!res.ok) throw new Error('Failed to fetch followers');
   279→   279→      const data = await res.json();
   280→   280→      const list: FollowerEntry[] = data.followers || [];
   281→   281→      setFollowers(list);
   282→   282→      // Check which followers the current user is following back
   283→   283→      if (list.length > 0) {
   284→   284→        const checkRes = await fetch(
   285→   285→          `/api/follow?userId=${currentUser.id}&type=search&search=`
   286→   286→        );
   287→   287→        if (checkRes.ok) {
   288→   288→          const checkData = await checkRes.json();
   289→   289→          const followingSet = new Set<string>(
   290→   290→            (checkData.players || [])
   291→   291→              .filter((p: PlayerResult) => p.isFollowing)
   292→   292→              .map((p: PlayerResult) => p.id)
   293→   293→          );
   294→   294→          setFollowedIds(followingSet);
   295→   295→        }
   296→   296→      }
   297→   297→    } catch (err) {
   298→   298→      console.error('Fetch followers error:', err);
   299→   299→      setFollowers([]);
   300→   300→    } finally {
   301→   301→      setFollowersLoading(false);
   302→   302→    }
   303→   303→  }, [currentUser]);
   304→   304→
   305→   305→  // ─── Fetch following ────────────────────────────────────────
   306→   306→
   307→   307→  const fetchFollowing = useCallback(async () => {
   308→   308→    if (!currentUser) return;
   309→   309→    setFollowingLoading(true);
   310→   310→    try {
   311→   311→      const res = await fetch(
   312→   312→        `/api/follow?userId=${currentUser.id}&type=following`
   313→   313→      );
   314→   314→      if (!res.ok) throw new Error('Failed to fetch following');
   315→   315→      const data = await res.json();
   316→   316→      setFollowing(data.following || []);
   317→   317→    } catch (err) {
   318→   318→      console.error('Fetch following error:', err);
   319→   319→      setFollowing([]);
   320→   320→    } finally {
   321→   321→      setFollowingLoading(false);
   322→   322→    }
   323→   323→  }, [currentUser]);
   324→   324→
   325→   325→  // ─── Follow / Unfollow action ───────────────────────────────
   326→   326→
   327→   327→  const handleFollowAction = async (
   328→   328→    targetId: string,
   329→   329→    isCurrentlyFollowing: boolean
   330→   330→  ) => {
   331→   331→    if (!currentUser || actionLoadingId) return;
   332→   332→    setActionLoadingId(targetId);
   333→   333→    try {
   334→   334→      const res = await fetch('/api/follow', {
   335→   335→        method: 'POST',
   336→   336→        headers: { 'Content-Type': 'application/json' },
   337→   337→        body: JSON.stringify({
   338→   338→          followerId: currentUser.id,
   339→   339→          followingId: targetId,
   340→   340→          action: isCurrentlyFollowing ? 'unfollow' : 'follow',
   341→   341→        }),
   342→   342→      });
   343→   343→      if (!res.ok) throw new Error('Action failed');
   344→   344→      const data = await res.json();
   345→   345→
   346→   346→      // Optimistically update UI
   347→   347→      const nowFollowing = !isCurrentlyFollowing;
   348→   348→
   349→   349→      // Update search results
   350→   350→      setSearchResults((prev) =>
   351→   351→        prev.map((p) =>
   352→   352→          p.id === targetId ? { ...p, isFollowing: nowFollowing } : p
   353→   353→        )
   354→   354→      );
   355→   355→
   356→   356→      // Update followed IDs set (for followers tab)
   357→   357→      setFollowedIds((prev) => {
   358→   358→        const next = new Set(prev);
   359→   359→        if (nowFollowing) next.add(targetId);
   360→   360→        else next.delete(targetId);
   361→   361→        return next;
   362→   362→      });
   363→   363→
   364→   364→      // Refresh counts
   365→   365→      fetchCounts();
   366→   366→
   367→   367→      // If on following tab and we unfollowed, remove from list
   368→   368→      if (!nowFollowing && activeTab === 'following') {
   369→   369→        setFollowing((prev) => prev.filter((f) => f.id !== targetId));
   370→   370→      }
   371→   371→
   372→   372→      // If on following tab and we followed someone from followers tab, refresh
   373→   373→      if (nowFollowing && activeTab === 'following') {
   374→   374→        fetchFollowing();
   375→   375→      }

   377→   377→      toast({
   378→   378→        title: nowFollowing ? 'Following!' : 'Unfollowed',
   379→   379→        description: nowFollowing
   380→   380→          ? 'You are now following this player'
   381→   381→          : 'You unfollowed this player',
   382→   382→      });
   383→   383→    } catch (err) {
   384→   384→      console.error('Follow action error:', err);
   385→   385→      toast({
   386→   386→        title: 'Error',
   387→   387→        description: 'Something went wrong. Please try again.',
   388→   388→        variant: 'destructive',
   389→   389→      });
   390→   390→    } finally {
   391→   391→      setActionLoadingId(null);
   392→   392→    }
   393→   393→  };
   394→   394→
   395→   395→  // ─── Initial fetch ──────────────────────────────────────────
   396→   396→
   397→   397→  useEffect(() => {
   398→   398→    fetchCounts();
   399→   399→  }, [fetchCounts]);
   400→   400→
   401→   401→  // ─── Search with debounce ───────────────────────────────────
   402→   402→
   403→   403→  useEffect(() => {
   404→   404→    if (activeTab !== 'search' || !currentUser) return;
   405→   405→    const timer = setTimeout(() => {
   406→   406→      searchPlayers(searchQuery);
   407→   407→    }, 300);
   408→   408→    return () => clearTimeout(timer);
   409→   409→  }, [searchQuery, activeTab, currentUser, searchPlayers]);
   410→   410→
   411→   411→  // ─── Tab change fetch ──────────────────────────────────────
   412→   412→
   413→   413→  useEffect(() => {
   414→   414→    if (activeTab === 'followers') {
   415→   415→      fetchFollowers();
   416→   416→    } else if (activeTab === 'following') {
   417→   417→      fetchFollowing();
   418→   418→    }

   422→   422→
   423→   423→  const TABS: { id: TabId; label: string; icon: typeof Search }[] = [
   424→   424→    { id: 'search', label: 'Search', icon: Search },
   425→   425→    { id: 'followers', label: 'Followers', icon: Users },
   426→   426→    { id: 'following', label: 'Following', icon: UserCheck },
   427→   427→  ];
   428→   428→
   429→   429→  // ─── Render ─────────────────────────────────────────────────
   430→   430→
   431→   431→  return (
   432→   432→    <motion.div
   433→   433→      initial={{ y: '100%' }}
   434→   434→      animate={{ y: 0 }}
   435→   435→      exit={{ y: '100%' }}
   436→   436→      transition={{ type: 'spring', damping: 28, stiffness: 300 }}
   437→   437→      className="fixed inset-0 z-50 bg-warm-50 flex flex-col"
   438→   438→    >
   439→   439→      {/* ═══ Header ═══ */}
   440→   440→      <header className="sticky top-0 z-10 bg-warm-50/90 backdrop-blur-md border-b border-warm-200/60">
   441→   441→        <div className="px-4 py-3 flex items-center justify-between">
   442→   442→          <div className="flex items-center gap-2">
   443→   443→            <button
   444→   444→              onClick={onClose}
   445→   445→              className="w-8 h-8 rounded-full bg-warm-200 flex items-center justify-center text-warm-600 hover:bg-warm-300 transition-colors"
   446→   446→            >
   447→   447→              <ArrowLeft className="w-4 h-4" />
   448→   448→            </button>
   449→   449→            <div className="flex items-center gap-2">
   450→   450→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-brand-red to-brand-red-dark flex items-center justify-center">
   451→   451→                <Sparkles className="w-4 h-4 text-white" />
   452→   452→              </div>
   453→   453→              <h1 className="text-base font-black tracking-wider text-warm-800">
   454→   454→                FOLLOW &amp; CONNECT
   455→   455→              </h1>
   456→   456→            </div>
   457→   457→          </div>
   458→   458→          <div className="flex items-center gap-3 text-xs font-semibold">
   459→   459→            {!countsLoading && (
   460→   460→              <>
   461→   461→                <span className="text-warm-500">
   462→   462→                  <span className="text-warm-800">{followerCount}</span>{' '}
   463→   463→                  {followerCount === 1 ? 'follower' : 'followers'}
   464→   464→                </span>
   465→   465→                <span className="text-warm-300">·</span>
   466→   466→                <span className="text-warm-500">
   467→   467→                  <span className="text-warm-800">{followingCount}</span>{' '}
   468→   468→                  following
   469→   469→                </span>
   470→   470→              </>
   471→   471→            )}
   472→   472→          </div>
   473→   473→        </div>
   474→   474→
   475→   475→        {/* Tab Pills */}
   476→   476→        <div className="flex gap-1.5 px-4 pb-3">
   477→   477→          {TABS.map((tab) => {
   478→   478→            const Icon = tab.icon;
   479→   479→            const isActive = activeTab === tab.id;
   480→   480→            return (
   481→   481→              <button
   482→   482→                key={tab.id}
   483→   483→                onClick={() => setActiveTab(tab.id)}
   484→   484→                className={`flex items-center gap-1.5 px-4 py-2 rounded-full text-xs font-bold transition-all whitespace-nowrap ${
   485→   485→                  isActive
   486→   486→                    ? 'bg-gradient-to-r from-brand-red to-brand-red-dark text-white shadow-sm'
   487→   487→                    : 'bg-warm-100 text-warm-500 hover:bg-warm-200'
   488→   488→                }`}
   489→   489→              >
   490→   490→                <Icon className="w-3.5 h-3.5" />
   491→   491→                {tab.label}
   492→   492→              </button>
   493→   493→            );
   494→   494→          })}
   495→   495→        </div>
   496→   496→      </header>
   497→   497→
   498→   498→      {/* ═══ Content ═══ */}
