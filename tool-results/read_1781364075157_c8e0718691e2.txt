    70→
    71→function getPositionCategory(position: string | null): 'raider' | 'defender' | 'all-rounder' | 'unknown' {
    72→  if (!position) return 'unknown';
    73→  const p = position.toLowerCase();
    74→  if (p.includes('raider')) return 'raider';
    75→  if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return 'defender';
    76→  if (p.includes('all')) return 'all-rounder';
    77→  return 'unknown';
    78→}

    82→function LineupValidation({ lineup, playersPerSide, teamName }: {
    83→  lineup: MatchPlayer[];
    84→  playersPerSide: number;
    85→  teamName: string;
    86→}) {
    87→  const warnings: { message: string; severity: 'error' | 'warning' | 'info' }[] = [];
    88→
    89→  if (lineup.length === 0) {
    90→    warnings.push({ message: 'No players added', severity: 'error' });
    91→  } else if (lineup.length < playersPerSide) {
    92→    warnings.push({
    93→      message: `${lineup.length}/${playersPerSide} players — need ${playersPerSide - lineup.length} more`,
    94→      severity: 'warning',
    95→    });
    96→  } else if (lineup.length > playersPerSide) {
    97→    warnings.push({
    98→      message: `${lineup.length} players — ${lineup.length - playersPerSide} extra`,
    99→      severity: 'info',
   100→    });
   101→  }
   102→
   103→  // Position balance check (only if we have position data from DB)
   104→  const balance = POSITION_BALANCE[playersPerSide];
   105→  // Note: position data might not be available for quick-added players
   106→  // We check this based on the allPlayers data which is in parent scope
   107→
   108→  if (warnings.length === 0) return null;
   109→
   110→  return (
   111→    <motion.div
   112→      initial={{ opacity: 0, y: -5 }}
   113→      animate={{ opacity: 1, y: 0 }}
   114→      className="space-y-1.5"
   115→    >
   116→      {warnings.map((w, i) => (
   117→        <div
   118→          key={i}
   119→          className={`flex items-center gap-2 px-3 py-2 rounded-lg text-xs font-medium ${
   120→            w.severity === 'error'
   121→              ? 'bg-brand-red/10 text-brand-red dark:bg-brand-red/15'
   122→              : w.severity === 'warning'
   123→                ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400 dark:bg-amber-500/15'
   124→                : 'bg-blue-500/10 text-blue-600 dark:text-blue-400 dark:bg-blue-500/15'
   125→          }`}
   126→        >
   127→          <AlertTriangle className={`w-3.5 h-3.5 shrink-0 ${w.severity === 'info' ? 'hidden' : ''}`} />
   128→          <Info className={`w-3.5 h-3.5 shrink-0 ${w.severity !== 'info' ? 'hidden' : ''}`} />
   129→          <span>{teamName}: {w.message}</span>
   130→        </div>
   131→      ))}
   132→    </motion.div>
   133→  );
   134→}

   138→function PlayerStatsTooltip({ player, allPlayers }: { player: DbPlayer; allPlayers: DbPlayer[] }) {
   139→  const dbPlayer = allPlayers.find(p => p.id === player.id);
   140→  const profile = dbPlayer?.profile;
   141→
   142→  if (!profile || !profile.totalPoints) return null;
   143→
   144→  const raidSuccessRate = profile.totalRaids && profile.totalRaids > 0
   145→    ? Math.round(((profile.successfulRaids || 0) / profile.totalRaids) * 100)
   146→    : 0;
   147→
   148→  return (
   149→    <div className="absolute z-30 bottom-full left-0 mb-1 p-2 bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-600 rounded-lg shadow-lg text-[10px] min-w-[140px]">
   150→      <div className="space-y-1">
   151→        <div className="flex justify-between">
   152→          <span className="text-warm-500 dark:text-warm-400">Total Pts</span>
   153→          <span className="font-bold text-brand-navy dark:text-brand-navy-light">{profile.totalPoints}</span>
   154→        </div>
   155→        <div className="flex justify-between">
   156→          <span className="text-warm-500 dark:text-warm-400">Raid Pts</span>
   157→          <span className="font-bold text-brand-red">{profile.raidPoints || 0}</span>
   158→        </div>
   159→        <div className="flex justify-between">
   160→          <span className="text-warm-500 dark:text-warm-400">Tackle Pts</span>
   161→          <span className="font-bold text-brand-teal">{profile.tacklePoints || 0}</span>
   162→        </div>
   163→        <div className="flex justify-between">
   164→          <span className="text-warm-500 dark:text-warm-400">Raid Success</span>
   165→          <span className="font-bold text-brand-gold">{raidSuccessRate}%</span>
   166→        </div>
   167→      </div>
   168→      {/* Tooltip arrow */}
   169→      <div className="absolute -bottom-1 left-4 w-2 h-2 bg-white dark:bg-warm-800 border-r border-b border-warm-200 dark:border-warm-600 rotate-45" />
   170→    </div>
   171→  );
   172→}

   176→function FormationVisualization({ lineup, teamColor, teamName, side }: {
   177→  lineup: MatchPlayer[];
   178→  teamColor: string;
   179→  teamName: string;
   180→  side: 'left' | 'right';
   181→}) {
   182→  const maxDisplay = Math.min(lineup.length, 7);
   183→  const displayPlayers = lineup.slice(0, maxDisplay);
   184→
   185→  return (
   186→    <div className="flex-1 relative">
   187→      {/* Team header */}
   188→      <div className="text-center mb-2">
   189→        <div
   190→          className="w-10 h-10 rounded-xl mx-auto flex items-center justify-center text-white font-black text-sm shadow-md"
   191→          style={{ backgroundColor: teamColor }}
   192→        >
   193→          {teamName.charAt(0).toUpperCase()}
   194→        </div>
   195→        <p className="text-[10px] font-bold text-warm-600 dark:text-warm-300 mt-1 truncate max-w-[80px] mx-auto">
   196→          {teamName}
   197→        </p>
   198→      </div>
   199→
   200→      {/* Formation layout */}
   201→      <div className="relative bg-warm-100/40 dark:bg-warm-700/20 rounded-xl p-2 min-h-[90px]">
   202→        {/* Court half line */}
   203→        <div className={`absolute top-1/2 left-0 right-0 h-px bg-warm-200 dark:bg-warm-600 ${
   204→          side === 'left' ? 'bg-gradient-to-r from-transparent to-warm-200 dark:to-warm-600' : 'bg-gradient-to-l from-transparent to-warm-200 dark:to-warm-600'
   205→        }`} />
   206→
   207→        {/* Players in formation */}
   208→        <div className="grid grid-cols-3 gap-1 place-items-center">
   209→          {displayPlayers.map((player, i) => {
   210→            // Simple formation: top row = defenders, middle = all-rounders, bottom = raiders
   211→            return (
   212→              <motion.div
   213→                key={player.id}
   214→                initial={{ scale: 0, opacity: 0 }}
   215→                animate={{ scale: 1, opacity: 1 }}
   216→                transition={{ delay: i * 0.05, type: 'spring', stiffness: 300, damping: 20 }}
   217→                className="flex flex-col items-center"
   218→              >
   219→                <div
   220→                  className="w-7 h-7 rounded-full flex items-center justify-center text-white text-[8px] font-bold shadow-sm"
   221→                  style={{ backgroundColor: teamColor }}
   222→                >
   223→                  {player.jerseyNumber || i + 1}
   224→                </div>
   225→                <span className="text-[7px] text-warm-400 dark:text-warm-500 mt-0.5 truncate max-w-[32px]">
   226→                  {player.name.split(' ')[0]}
   227→                </span>
   228→              </motion.div>
   229→            );
   230→          })}
   231→          {/* Empty slots */}
   232→          {Array.from({ length: Math.max(0, 7 - displayPlayers.length) }).map((_, i) => (
   233→            <div key={`empty-${i}`} className="flex flex-col items-center">
   234→              <div className="w-7 h-7 rounded-full border-2 border-dashed border-warm-200 dark:border-warm-600 flex items-center justify-center">
   235→                <span className="text-[8px] text-warm-300 dark:text-warm-600">?</span>
   236→              </div>
   237→              <span className="text-[7px] text-warm-300 dark:text-warm-600 mt-0.5">Empty</span>
   238→            </div>
   239→          ))}
   240→        </div>
   241→      </div>
   242→    </div>
   243→  );
   244→}
