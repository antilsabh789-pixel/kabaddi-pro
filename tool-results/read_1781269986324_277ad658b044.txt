    71→  const [showHomeSuggestions, setShowHomeSuggestions] = useState(false);

    83→        setShowSuggestions(false);
    84→      }

    89→
    90→  // Fetch all players on mount for instant filtering

    94→        const res = await fetch('/api/players?limit=100');
    95→        if (res.ok) {
    96→          const data = await res.json();
    97→          setAllPlayers(data.players || []);
    98→        }
    99→      } catch {
   100→        // silently fail
   101→      }
   102→    };
   103→    fetchAllPlayers();
   104→  }, []);
   105→
   106→  // Fetch user's teams for suggestions
   107→  useEffect(() => {

   111→        const res = await fetch('/api/teams');

   113→          const data = await res.json();

   116→              team.members?.some((m: any) => m.userId === currentUser.id)
   117→            )
   118→            .map((team: any) => ({
   119→              id: team.id,
   120→              name: team.name,
   121→              shortName: team.shortName,
   122→              color: team.color,
   123→            }));
   124→          setUserTeams(teams);
   125→        }
   126→      } catch {
   127→        // silently fail
   128→      }
   129→    };
   130→    fetchUserTeams();
   131→  }, [currentUser?.id]);
   132→
   133→  // Filter team suggestions based on input
   134→  useEffect(() => {
   135→    const query = config.homeTeam.toLowerCase().trim();

   137→      const filtered = userTeams.filter(t =>
   138→        t.name.toLowerCase().includes(query)
   139→      );
   140→      setHomeTeamSuggestions(filtered);
   141→    } else {
   142→      setHomeTeamSuggestions(userTeams);
   143→    }
   144→  }, [config.homeTeam, userTeams]);
   145→
   146→  useEffect(() => {
   147→    const query = config.awayTeam.toLowerCase().trim();
   148→    if (query.length > 0) {
   149→      const filtered = userTeams.filter(t =>
   150→        t.name.toLowerCase().includes(query)
   151→      );
   152→      setAwayTeamSuggestions(filtered);
   153→    } else {
   154→      setAwayTeamSuggestions(userTeams);
   155→    }
   156→  }, [config.awayTeam, userTeams]);
   157→
   158→  const selectHomeTeamSuggestion = (team: UserTeam) => {
   159→    setConfig({
   160→      ...config,
   161→      homeTeam: team.name,
   162→      homeTeamColor: team.color || config.homeTeamColor,
   163→    });
   164→    setShowHomeSuggestions(false);
   165→  };
   166→
   167→  const selectAwayTeamSuggestion = (team: UserTeam) => {
   168→    setConfig({
   169→      ...config,
   170→      awayTeam: team.name,
   171→      awayTeamColor: team.color || config.awayTeamColor,
   172→    });
   173→    setShowAwaySuggestions(false);
   174→  };
   175→

   177→    switch (step) {
   178→      case 0: return config.gender !== '';
   179→      case 1: return config.halfDuration >= 1 && config.playersPerSide >= 1;
   180→      case 2: return config.homeTeam !== '' && config.awayTeam !== '';
   181→      case 3: return config.homeLineup.length > 0 && config.awayLineup.length > 0;
   182→      case 4: return true;
   183→      default: return false;
   184→    }
   185→  };
   186→
   187→  const handleNext = () => {
   188→    if (step < 4 && canNext()) setStep(step + 1);
   189→  };

   192→    if (step > 0) setStep(step - 1);
   193→  };

   197→      id: `match_${Date.now()}`,
   198→      homeTeamId: `home_${Date.now()}`,
   199→      awayTeamId: `away_${Date.now()}`,
   200→      homeTeam: config.homeTeam,
   201→      awayTeam: config.awayTeam,
   202→      homeTeamColor: config.homeTeamColor,
   203→      awayTeamColor: config.awayTeamColor,
   204→      isPractice: true, // Quick Score is always practice
   205→      gender: config.gender,
   206→      halfDuration: config.halfDuration,
   207→      playersPerSide: config.playersPerSide,
   208→      homeLineup: config.homeLineup,
   209→      awayLineup: config.awayLineup,
   210→    });

   218→      id: `p_${Date.now()}_${Math.random().toString(36).slice(2, 5)}`,
   219→      name,
   220→      jerseyNumber: lineup.length + 1,
   221→      team,
   222→    };
   223→    setConfig({
   224→      ...config,
   225→      [team === 'home' ? 'homeLineup' : 'awayLineup']: [...lineup, newPlayer],
   226→    });

   228→    setSearchResults([]);
   229→    setShowSuggestions(false);
   230→  };

   233→  const highlightMatch = (text: string, query: string) => {
   234→    if (!query.trim()) return text;

   236→    const parts = text.split(regex);
   237→    return parts.map((part, i) =>
   238→      regex.test(part) ? (
   239→        <span key={i} className="text-brand-red font-bold">{part}</span>
   240→      ) : (
   241→        <span key={i}>{part}</span>
   242→      )
   243→    );
   244→  };
   245→

   247→    const lineup = team === 'home' ? config.homeLineup : config.awayLineup;
   248→    setConfig({
   249→      ...config,
   250→      [team === 'home' ? 'homeLineup' : 'awayLineup']: lineup.filter((p) => p.id !== playerId),
   251→    });
   252→  };
   253→

   255→  const getLocalFiltered = useCallback((query: string): DbPlayer[] => {
   256→    if (!query.trim()) return allPlayers.slice(0, 20);
   257→    const q = query.toLowerCase().trim();
   258→    return allPlayers.filter(p =>
   259→      (p.name && p.name.toLowerCase().includes(q)) ||
   260→      (p.playerCode && p.playerCode.toLowerCase().includes(q)) ||
   261→      (p.phone && p.phone.includes(q))
   262→    ).slice(0, 20);
   263→  }, [allPlayers]);
   264→

   266→  const searchPlayers = useCallback(async (query: string) => {
   267→    if (query.length < 1) {
   268→      setSearchResults(getLocalFiltered(''));
   269→      return;
   270→    }
   271→    // Show local results immediately
   272→    const localResults = getLocalFiltered(query);
   273→    setSearchResults(localResults);
   274→
   275→    // Also fetch from server for more comprehensive results
   276→    if (query.length >= 2) {
   277→      setIsSearching(true);
   278→      try {
   279→        const res = await fetch(`/api/players?search=${encodeURIComponent(query)}`);

   281→          const data = await res.json();

   284→          const serverIds = new Set(serverPlayers.map(p => p.id));
   285→          const merged = [...serverPlayers, ...localResults.filter(p => !serverIds.has(p.id))].slice(0, 20);
   286→          setSearchResults(merged);
   287→        }
   288→      } catch {
   289→        // Keep local results on error
   290→      } finally {
   291→        setIsSearching(false);
   292→      }
   293→    }
   294→  }, [getLocalFiltered]);
   295→
   296→  // Debounce server search, but local filter is instant
   297→  useEffect(() => {
   298→    const timer = setTimeout(() => {
   299→      if (showSuggestions && playerSearch.length >= 2) {
   300→        searchPlayers(playerSearch);
   301→      }
   302→    }, 300);
   303→    return () => clearTimeout(timer);
   304→  }, [playerSearch, showSuggestions, searchPlayers]);
   305→
   306→  // Instant local filter on every keystroke
   307→  useEffect(() => {
   308→    if (showSuggestions) {
   309→      setSearchResults(getLocalFiltered(playerSearch));
   310→    }
   311→  }, [playerSearch, showSuggestions, getLocalFiltered]);
   312→

   314→    const lineup = activeLineupTeam === 'home' ? config.homeLineup : config.awayLineup;
   315→    // Check if already added
   316→    const alreadyAdded = [...config.homeLineup, ...config.awayLineup].some(p => p.id === dbPlayer.id);
   317→    if (alreadyAdded) return;
   318→    const newPlayer: MatchPlayer = {
   319→      id: dbPlayer.id,
   320→      name: dbPlayer.name || 'Unknown',
   321→      jerseyNumber: dbPlayer.profile?.jerseyNumber || lineup.length + 1,
   322→      playerCode: dbPlayer.playerCode || undefined,
   323→      team: activeLineupTeam,
   324→    };
   325→    setConfig({
   326→      ...config,
   327→      [activeLineupTeam === 'home' ? 'homeLineup' : 'awayLineup']: [...lineup, newPlayer],
   328→    });
   329→    setPlayerSearch('');
   330→    setSearchResults([]);
   331→    setShowSuggestions(false);
   332→  };
   333→
   334→  const teamColors = [
   335→    '#DC2626', '#1E293B', '#14B8A6', '#475569', '#9333EA',
   336→    '#F97316', '#06B6D4', '#EC4899', '#84CC16', '#6366F1',
   337→  ];
   338→
   339→  return (
   340→    <div className="px-4 py-6 space-y-5">
   341→      {/* Step Indicator */}
   342→      <div className="flex items-center gap-1">
   343→        {STEPS.map((label, i) => (
   344→          <div key={label} className="flex-1 flex flex-col items-center gap-1">
   345→            <div
   346→              className={`h-1 w-full rounded-full transition-all duration-300 ${
   347→                i <= step ? 'bg-brand-red' : 'bg-warm-300'
   348→              }`}
   349→            />
   350→            <span
   351→              className={`text-[9px] font-medium ${
   352→                i <= step ? 'text-brand-red' : 'text-warm-400'
   353→              }`}
   354→            >
   355→              {label}
   356→            </span>
   357→          </div>
   358→        ))}
   359→      </div>
   360→
   361→      {/* Step Content */}
   362→      <AnimatePresence mode="wait">
   363→        <motion.div
   364→          key={step}
   365→          initial={{ opacity: 0, x: 20 }}
   366→          animate={{ opacity: 1, x: 0 }}
   367→          exit={{ opacity: 0, x: -20 }}
   368→          transition={{ duration: 0.2 }}
   369→          className="min-h-[400px]"
   370→        >
   371→          {step === 0 && (
   372→            <div className="space-y-4">
   373→              <h2 className="text-lg font-bold text-warm-800">Select Gender</h2>
   374→              <div className="grid grid-cols-2 gap-4">
   375→                <button
   376→                  onClick={() => setConfig({ ...config, gender: 'male' })}
   377→                  className={`p-8 rounded-2xl border-2 transition-all duration-200 flex flex-col items-center gap-3 ${
   378→                    config.gender === 'male'
   379→                      ? 'border-brand-blue bg-brand-blue/10'
   380→                      : 'border-warm-300 bg-white'
   381→                  }`}
   382→                >
   383→                  <span className="text-5xl">♂</span>
   384→                  <span className={`font-bold text-lg ${config.gender === 'male' ? 'text-brand-blue' : 'text-warm-600'}`}>
   385→                    Boys
   386→                  </span>
   387→                </button>
   388→                <button
   389→                  onClick={() => setConfig({ ...config, gender: 'female' })}
   390→                  className={`p-8 rounded-2xl border-2 transition-all duration-200 flex flex-col items-center gap-3 ${
   391→                    config.gender === 'female'
   392→                      ? 'border-brand-red bg-brand-red/10'
   393→                      : 'border-warm-300 bg-white'
   394→                  }`}
   395→                >
   396→                  <span className="text-5xl">♀</span>
   397→                  <span className={`font-bold text-lg ${config.gender === 'female' ? 'text-brand-red' : 'text-warm-600'}`}>
   398→                    Girls
   399→                  </span>
   400→                </button>
   401→              </div>
   402→            </div>
   403→          )}
   404→
   405→          {step === 1 && (
   406→            <div className="space-y-6">
   407→              <h2 className="text-lg font-bold text-warm-800">Match Settings</h2>
   408→
   409→              {/* Practice Match Indicator */}
   410→              <div className="bg-brand-green/10 border border-brand-green/20 rounded-xl p-3 flex items-center gap-2">
   411→                <span className="text-lg">🏋️</span>
   412→                <div>
   413→                  <p className="text-sm font-semibold text-brand-green">Practice Match</p>
   414→                  <p className="text-[10px] text-warm-500">Quick Score is for practice. Tournament matches are scored via Tournaments tab.</p>
   415→                </div>
   416→              </div>
   417→
   418→              {/* Half Duration */}
   419→              <div>
   420→                <label className="text-sm font-medium text-warm-700 mb-2 block">
   421→                  Half Duration: {config.halfDuration} min
   422→                </label>
   423→                <div className="flex items-center gap-3">
   424→                  <button
   425→                    onClick={() => setConfig({ ...config, halfDuration: Math.max(1, config.halfDuration - 1) })}
   426→                    className="w-10 h-10 rounded-xl bg-warm-200 flex items-center justify-center text-warm-700 active:bg-warm-300 transition-colors"
   427→                  >
   428→                    -
   429→                  </button>
   430→                  <div className="flex-1 relative">
   431→                    <input
   432→                      type="range"
   433→                      min={1}
   434→                      max={20}
   435→                      step={1}
   436→                      value={config.halfDuration}
   437→                      onChange={(e) => setConfig({ ...config, halfDuration: parseInt(e.target.value) })}
   438→                      className="w-full h-2 bg-warm-200 rounded-xl appearance-none cursor-pointer [&::-webkit-slider-thumb]:appearance-none [&::-webkit-slider-thumb]:w-5 [&::-webkit-slider-thumb]:h-5 [&::-webkit-slider-thumb]:rounded-full [&::-webkit-slider-thumb]:bg-brand-red [&::-webkit-slider-thumb]:shadow-md [&::-webkit-slider-thumb]:cursor-pointer [&::-webkit-slider-thumb]:border-2 [&::-webkit-slider-thumb]:border-white"
   439→                    />
   440→                  </div>

   442→                    onClick={() => setConfig({ ...config, halfDuration: Math.min(20, config.halfDuration + 1) })}
   443→                    className="w-10 h-10 rounded-xl bg-warm-200 flex items-center justify-center text-warm-700 active:bg-warm-300 transition-colors"
   444→                  >
   445→                    +
   446→                  </button>

   454→              {/* Players Per Side */}

   456→                <label className="text-sm font-medium text-warm-700 mb-2 block">

   459→                <div className="flex items-center gap-3">

   464→                    -
   465→                  </button>

   471→                  </div>
   472→                  <button
   473→                    onClick={() => setConfig({ ...config, playersPerSide: Math.min(12, config.playersPerSide + 1) })}
   474→                    className="w-10 h-10 rounded-xl bg-warm-200 flex items-center justify-center text-warm-700"
   475→                  >
   476→                    +
   477→                  </button>
   478→                </div>
   479→                <div className="flex justify-between text-xs text-warm-400 mt-1">
   480→                  <span>1</span>
   481→                  <span>12</span>

   483→              </div>

   486→
   487→          {step === 2 && (
   488→            <div className="space-y-5" ref={teamInputRef}>
   489→              <h2 className="text-lg font-bold text-warm-800">Team Setup</h2>
   490→
   491→              {/* Home Team */}
   492→              <div className="space-y-3">
   493→                <label className="text-sm font-medium text-warm-700">Team A</label>
   494→                <div className="relative">
   495→                  <Input
   496→                    placeholder="Enter team name"
   497→                    value={config.homeTeam}
   498→                    onChange={(e) => {

   506→                  {/* Team Suggestions Dropdown */}
