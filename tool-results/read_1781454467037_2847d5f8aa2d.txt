   241→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   242→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   243→  let homeScore = 0;
   244→  let awayScore = 0;
   245→  let homeOutPlayers = 0;
   246→  let awayOutPlayers = 0;
   247→  let homeOutPlayerIds: string[] = [];
   248→  let awayOutPlayerIds: string[] = [];
   249→
   250→  for (const evt of events) {
   251→    const side = getTeamSide(match, evt.teamId);
   252→    const points = evt.value;
   253→    if (side === 'home') {
   254→      homeScore += points;
   255→    } else {
   256→      awayScore += points;
   257→    }
   258→
   259→    // Count-based out tracking (backward compat)
   260→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   261→      if (side === 'home') {
   262→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   263→      } else {
   264→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   265→      }
   266→    }
   267→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   268→      if (side === 'home') {
   269→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   270→      } else {
   271→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   272→      }
   273→    }
   274→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   275→    if (evt.eventType === 'self_out') {
   276→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   277→      if (side === 'home') {
   278→        // Home scored from self-out → away defender went out
   279→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   280→        // Home revives one player
   281→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   282→      } else {
   283→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   284→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   285→      }
   286→    }
   287→    if (evt.eventType === 'all_out') {
   288→      if (side === 'home') {
   289→        awayOutPlayers = 0;
   290→      } else {
   291→        homeOutPlayers = 0;
   292→      }
   293→    }
   294→
   295→    // ID-based out tracking
   296→    let details: Record<string, unknown> = {};
   297→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   298→
   299→    if (evt.eventType === 'raid_point') {
   300→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   301→      if (side === 'home') {
   302→        // Home team scored raid → away defenders go out
   303→        for (const tid of touchedIds) {
   304→          if (!awayOutPlayerIds.includes(tid)) {
   305→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   306→          }
   307→        }
   308→        // Home team revives players (first-out-first-in)
   309→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   310→        if (reviveCount > 0) {
   311→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   312→        }
   313→      } else {
   314→        // Away team scored raid → home defenders go out
   315→        for (const tid of touchedIds) {
   316→          if (!homeOutPlayerIds.includes(tid)) {
   317→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   318→          }
   319→        }
   320→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   321→        if (reviveCount > 0) {
   322→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   323→        }
   324→      }
   325→    }
   326→
   327→    if (evt.eventType === 'bonus_point') {
   328→      // One out player from scoring team revives
   329→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   330→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   331→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   332→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   333→      }
   334→    }
   335→
   336→    // Self-out: defender steps off mat during a live raid
   337→    if (evt.eventType === 'self_out') {
   338→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   339→      if (selfOutPlayerId) {
   340→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   341→        if (side === 'home') {
   342→          // Home scored → away defender self-out
   343→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   344→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   345→          }
   346→          // Home revives first-out player
   347→          if (homeOutPlayerIds.length > 0) {
   348→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   349→          }
   350→        } else {
   351→          // Away scored → home defender self-out
   352→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   353→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   354→          }
   355→          if (awayOutPlayerIds.length > 0) {
   356→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   357→          }
   358→        }
   359→      }
   360→    }
   361→
   362→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   363→      // Raider goes out. The tackle is scored by the DEFENDING team.
   364→      // So the raider is from the OPPOSING team.
   365→      const raiderId = (details.raiderId as string) || evt.playerId;
   366→      if (side === 'home' && raiderId) {
   367→        // Home team scored tackle → away raider was caught → away raider goes out
   368→        if (!awayOutPlayerIds.includes(raiderId)) {
   369→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   370→        }
   371→      } else if (side === 'away' && raiderId) {
   372→        // Away team scored tackle → home raider was caught → home raider goes out
   373→        if (!homeOutPlayerIds.includes(raiderId)) {
   374→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   375→        }
   376→      }
   377→    }
   378→
   379→    if (evt.eventType === 'all_out') {
   380→      if (side === 'home') {
   381→        awayOutPlayerIds = [];
   382→      } else {
   383→        homeOutPlayerIds = [];
   384→      }
   385→    }
   386→  }
   387→
   388→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   389→}

   391→/** Event types that constitute a complete raid and should flip the turn */
   392→const RAID_ENDING_EVENT_TYPES: string[] = [
   393→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   394→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   395→];
   396→
   397→/** Event types that do NOT flip the turn (non-raid events) */
   398→const NON_RAID_EVENT_TYPES: string[] = [
   399→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   400→];
   401→
   402→/**
   403→ * Event types where the SCORING team is the DEFENDING team.
   404→ * In these cases, the defending team scored (they made the tackle / raider failed do-or-die),
   405→ * so they should raid NEXT — the turn goes to the SAME side that scored.
   406→ *
   407→ * For all other raid-ending events (raid_point, bonus_point, super_raid, empty_raid, self_out),
   408→ * the scoring team is the RAIDING team, so the turn goes to the OPPOSITE side.
   409→ */
   410→const DEFENDING_TEAM_SCORES: Set<string> = new Set([
   411→  'tackle_point',
   412→  'super_tackle',
   413→  'do_or_die_raid',
   414→]);
   415→
   416→/** Helper: determine raid queue from last raid-ending event */
   417→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   418→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   419→  if (lastRaidEvent) {
   420→    const side = getTeamSide(match, lastRaidEvent.teamId);
   421→    // If the defending team scored (tackle/super_tackle/do_or_die_raid),
   422→    // THEY raid next — return the SAME side as the scoring team.
   423→    if (DEFENDING_TEAM_SCORES.has(lastRaidEvent.eventType)) {
   424→      return side;
   425→    }
   426→    // If the raiding team scored (touch points, bonus, super raid, etc.)
   427→    // or it was an empty raid, the DEFENDING team raids next — return the OPPOSITE side.
   428→    return side === 'home' ? 'away' : 'home';
   429→  }
   430→  return match.raidQueue;
   431→}

   590→          const newEvent: MatchEvent = {

   592→            id: `evt_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   593→            timestamp: Date.now(),
   594→          };
   595→          const updatedEvents = [...state.activeMatch.events, newEvent];
   596→
   597→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   598→
   599→          // Only flip raidQueue for raid-ending events.
   600→          // Non-raid events (timeout, substitution, cards, standalone all_out)
   601→          // should NOT change whose turn it is to raid.
   602→          const isRaidEndingEvent = RAID_ENDING_EVENT_TYPES.includes(event.eventType);
   603→          const raidQueue = isRaidEndingEvent
   604→            ? getRaidQueueFromEvents(state.activeMatch, updatedEvents)
   605→            : state.activeMatch.raidQueue;
   606→
   607→          return {
   608→            activeMatch: {
   609→              ...state.activeMatch,
   610→              events: updatedEvents,
   611→              ...calculated,
   612→              raidQueue,
   613→            },
   614→          };
   615→        }),
   616→
   617→      addBatchEvents: (events) =>
   618→        set((state) => {
   619→          if (!state.activeMatch || events.length === 0) return {};
   620→          const newEvents: MatchEvent[] = events.map((e, i) => ({
   621→            ...e,
   622→            id: `evt_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 7)}`,
   623→            timestamp: Date.now() + i,
   624→          }));
   625→          const updatedEvents = [...state.activeMatch.events, ...newEvents];
   626→
   627→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   628→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, updatedEvents);
   629→
   630→          return {
   631→            activeMatch: {
   632→              ...state.activeMatch,
   633→              events: updatedEvents,
   634→              ...calculated,
   635→              raidQueue,
   636→            },
   637→          };
   638→        }),
   639→
   640→      undoLastEvent: () =>
   641→        set((state) => {
   642→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   643→          const events = [...state.activeMatch.events];
   644→          events.pop();
   645→
   646→          const calculated = recalculateFromEvents(state.activeMatch, events);
   647→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   648→
   649→          return {
   650→            activeMatch: {
   651→              ...state.activeMatch,
   652→              events,
   653→              ...calculated,
   654→              raidQueue,
   655→            },
   656→          };
   657→        }),
   658→
   659→      undoLastRaid: () =>
   660→        set((state) => {
   661→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   662→
   663→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   664→          // A raid batch has events with the same second in timestamp
   665→          const events = [...state.activeMatch.events];
   666→          if (events.length === 0) return {};
   667→
   668→          // Find the timestamp of the last event
   669→          const lastTimestamp = events[events.length - 1].timestamp;
   670→          // Remove all events that have the same timestamp (within 5ms for batch events)
   671→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   672→            events.pop();
   673→          }
   674→
   675→          const calculated = recalculateFromEvents(state.activeMatch, events);
   676→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   677→
   678→          return {
   679→            activeMatch: {
   680→              ...state.activeMatch,
   681→              events,
   682→              ...calculated,
   683→              raidQueue,
   684→            },
   685→          };
   686→        }),
   687→
   688→      addPlayerToMatch: (side, player) =>
   689→        set((state) => {
   690→          if (!state.activeMatch) return {};
   691→          const lineupKey = side === 'home' ? 'homeLineup' : 'awayLineup';
   692→          return {
   693→            activeMatch: {
   694→              ...state.activeMatch,
   695→              [lineupKey]: [...state.activeMatch[lineupKey], player],
   696→            },
   697→          };
   698→        }),
   699→
   700→      switchHalf: () =>
   701→        set((state) => {
   702→          if (!state.activeMatch) return {};
   703→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   704→          return {
   705→            activeMatch: {
   706→              ...state.activeMatch,
   707→              currentHalf: newHalf,
   708→              timer: state.activeMatch.halfDuration * 60,
   709→              isDoOrDie: false,
   710→              doOrDieTeamId: null,
   711→            },
   712→          };
   713→        }),
   714→
   715→      setTimer: (time) =>
   716→        set((state) => {
   717→          if (!state.activeMatch) return {};
   718→          return { activeMatch: { ...state.activeMatch, timer: time } };
   719→        }),
   720→
   721→      setDoOrDie: (value, teamId?) =>
   722→        set((state) => {
   723→          if (!state.activeMatch) return {};
   724→          return {
   725→            activeMatch: {
   726→              ...state.activeMatch,
   727→              isDoOrDie: value,
   728→              doOrDieTeamId: value ? (teamId || null) : null,
   729→            },
   730→          };
   731→        }),
   732→
   733→      switchRaidQueue: () =>
   734→        set((state) => {
   735→          if (!state.activeMatch) return {};
   736→          return {
   737→            activeMatch: {
   738→              ...state.activeMatch,
   739→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   740→            },
   741→          };
   742→        }),
   743→
   744→      callTimeout: (team) =>
   745→        set((state) => {
   746→          if (!state.activeMatch) return {};
   747→          return {
   748→            activeMatch: {
   749→              ...state.activeMatch,
   750→              homeTimeouts:
   751→                team === 'home'
   752→                  ? state.activeMatch.homeTimeouts + 1
   753→                  : state.activeMatch.homeTimeouts,
   754→              awayTimeouts:
   755→                team === 'away'
   756→                  ? state.activeMatch.awayTimeouts + 1
   757→                  : state.activeMatch.awayTimeouts,
   758→            },
   759→          };
   760→        }),
   761→
   762→      setOutPlayers: (team, count) =>
   763→        set((state) => {
   764→          if (!state.activeMatch) return {};
   765→          return {
   766→            activeMatch: {
   767→              ...state.activeMatch,
   768→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   769→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   770→            },
   771→          };
   772→        }),
   773→
   774→      fetchHomeData: async () => {
   775→        try {
   776→          const res = await fetch('/api/stats');
   777→          if (!res.ok) throw new Error('Failed to fetch stats');
   778→
   779→          const data = await res.json();
   780→
   781→          // Cache in store
   782→          set({ homeData: data });
   783→          return data;
   784→        } catch (err) {
   785→          console.error('fetchHomeData error:', err);
   786→          return null;
   787→        }
   788→      },
   789→
   790→      // Notification actions
   791→      addNotification: (notification) =>
   792→        set((state) => ({
   793→          notifications: [
   794→            {
   795→              ...notification,
   796→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   797→              timestamp: Date.now(),
   798→              read: false,
   799→            },
   800→            ...state.notifications,
   801→          ].slice(0, 50), // Keep max 50 notifications
   802→        })),
   803→
   804→      markNotificationRead: (id) =>
   805→        set((state) => ({
   806→          notifications: state.notifications.map((n) =>
   807→            n.id === id ? { ...n, read: true } : n
   808→          ),
   809→        })),
   810→
   811→      markAllRead: () =>
   812→        set((state) => ({
   813→          notifications: state.notifications.map((n) => ({ ...n, read: true })),
   814→        })),
   815→
   816→      clearNotifications: () => set({ notifications: [] }),
   817→
   818→      setLanguage: (lang) => set({ language: lang }),
   819→
   820→      // Onboarding actions
   821→      setHasCompletedOnboarding: (value) => set({ hasCompletedOnboarding: value }),
   822→
   823→      setOnboardingProfile: (profile) =>
   824→        set((state) => ({
   825→          onboardingProfile: { ...state.onboardingProfile, ...profile },
   826→        })),
   827→
   828→      completeOnboarding: () =>
   829→        set((state) => ({
   830→          hasCompletedOnboarding: true,
   831→          currentUser: state.currentUser
   832→            ? {
   833→                ...state.currentUser,
   834→                ...(state.onboardingProfile.position && { position: state.onboardingProfile.position }),
   835→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   836→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   837→                createdAt: state.currentUser.createdAt || Date.now(),
   838→              }
   839→            : null,
   840→        })),
   841→
   842→      // Team management actions
   843→      setTeamFilter: (filter) =>
   844→        set((state) => ({
   845→          teamManagement: { ...state.teamManagement, teamFilter: filter },
   846→        })),
   847→
   848→      setTeamSearch: (search) =>
   849→        set((state) => ({
   850→          teamManagement: { ...state.teamManagement, teamSearch: search },
   851→        })),
   852→
   853→      setSelectedTeamId: (teamId) =>
   854→        set((state) => ({
   855→          teamManagement: { ...state.teamManagement, selectedTeamId: teamId },
   856→        })),
   857→
   858→      setTeamDetailOpen: (open) =>
   859→        set((state) => ({
   860→          teamManagement: { ...state.teamManagement, teamDetailOpen: open },
   861→        })),
   862→
   863→      // Match transfer / handoff — load a match from a transfer
   864→      loadMatch: (match) =>
   865→        set({
   866→          activeMatch: match,
   867→          activeTab: 'quick-score' as TabId,
   868→          showToss: false,
   869→          tossMatchConfig: null,
   870→        }),
   871→    }),
   872→    {
   873→      name: 'kabaddi-pro-storage',
   874→      storage: createJSONStorage(() => localStorage),
   875→      // Only persist auth-related fields and notifications
   876→      partialize: (state) => ({
   877→        isAuthenticated: state.isAuthenticated,
   878→        isOnboarded: state.isOnboarded,
   879→        currentUser: state.currentUser,
   880→        hasSeenSplash: state.hasSeenSplash,
   881→        notifications: state.notifications,
   882→        language: state.language,
   883→        hasCompletedOnboarding: state.hasCompletedOnboarding,
   884→        onboardingProfile: state.onboardingProfile,
   885→        coachAcademies: state.coachAcademies,
   886→      }),
   887→    }
   888→  )
   889→);

   891→export function getEventPoints(eventType: EventType): number {
   892→  switch (eventType) {
   893→    case 'raid_point':
   894→      return 1;
   895→    case 'bonus_point':
   896→      return 1;
   897→    case 'tackle_point':
   898→      return 1;
   899→    case 'super_raid':
   900→      return 1;
   901→    case 'super_tackle':
   902→      return 1;
   903→    case 'do_or_die_raid':
   904→      return 1;
   905→    case 'all_out':
   906→      return 2;
   907→    case 'self_out':
   908→      return 1;
   909→    case 'empty_raid':
   910→      return 0;
   911→    case 'substitution':
   912→    case 'timeout':
   913→    case 'yellow_card':
   914→    case 'red_card':
   915→    case 'green_card':
   916→      return 0;
   917→    default:
   918→      return 0;
   919→  }
   920→}
