    81→// ─── Animated Value (count-up on first view) ────────────────────
    82→
    83→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    84→  const [display, setDisplay] = useState(0);
    85→  const [hasAnimated, setHasAnimated] = useState(false);
    86→  const ref = useRef<HTMLSpanElement>(null);
    87→  const valueRef = useRef(value);
    88→
    89→  useEffect(() => {
    90→    valueRef.current = value;
    91→  }, [value]);
    92→
    93→  useEffect(() => {
    94→    if (hasAnimated) return;
    95→    if (value === 0) return;
    96→
    97→    const observer = new IntersectionObserver(
    98→      ([entry]) => {
    99→        if (entry.isIntersecting && !hasAnimated) {
   100→          setHasAnimated(true);
   101→          const targetVal = valueRef.current;
   102→          const duration = 1200;
   103→          const startTime = performance.now();
   104→
   105→          function step(currentTime: number) {
   106→            const elapsed = currentTime - startTime;
   107→            const progress = Math.min(elapsed / duration, 1);
   108→            const eased = 1 - Math.pow(1 - progress, 3);
   109→            const current = eased * targetVal;
   110→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
   111→            if (progress < 1) {
   112→              requestAnimationFrame(step);
   113→            }
   114→          }
   115→
   116→          requestAnimationFrame(step);
   117→        }
   118→      },
   119→      { threshold: 0.1 }
   120→    );
   121→
   122→    if (ref.current) observer.observe(ref.current);
   123→    return () => observer.disconnect();
   124→  }, [value, decimals, hasAnimated]);
   125→
   126→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   127→}

   144→// ─── Time Ago helper ────────────────────
   145→
   146→function timeAgo(dateStr: string): string {
   147→  if (!dateStr || dateStr === '—') return '';
   148→  try {
   149→    const date = new Date(dateStr);
   150→    if (isNaN(date.getTime())) return dateStr;
   151→    const now = new Date();
   152→    const diff = now.getTime() - date.getTime();
   153→    const mins = Math.floor(diff / 60000);
   154→    if (mins < 60) return mins <= 1 ? 'just now' : `${mins}m ago`;
   155→    const hours = Math.floor(mins / 60);
   156→    if (hours < 24) return `${hours}h ago`;
   157→    const days = Math.floor(hours / 24);
   158→    if (days < 7) return `${days}d ago`;
   159→    const weeks = Math.floor(days / 7);
   160→    if (weeks < 4) return `${weeks}w ago`;
   161→    return dateStr;
   162→  } catch {
   163→    return dateStr;
   164→  }
   165→}

   169→function getPlayerLevel(totalMatches: number, totalPoints: number): { label: string; color: string; icon: string; progress: number } {
   170→  const score = totalMatches * 2 + totalPoints;
   171→  if (score >= 500) return { label: 'Legend', color: 'from-amber-400 to-yellow-600', icon: '👑', progress: 100 };
   172→  if (score >= 200) return { label: 'Pro', color: 'from-red-500 to-orange-500', icon: '🔥', progress: Math.min(((score - 200) / 300) * 100, 100) };
   173→  if (score >= 50) return { label: 'Intermediate', color: 'from-teal-500 to-emerald-500', icon: '⚡', progress: Math.min(((score - 50) / 150) * 100, 100) };
   174→  return { label: 'Beginner', color: 'from-slate-400 to-slate-500', icon: '🌱', progress: Math.min((score / 50) * 100, 100) };
   175→}
