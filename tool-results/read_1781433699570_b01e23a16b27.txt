   489→      const teamList: TeamOption[] = (data.teams || data || []).map((t: Record<string, unknown>) => ({
   490→        id: t.id as string,
   491→        name: (t.name as string) || 'Unnamed Team',
   492→        teamCode: (t.teamCode as string) || '',
   493→        memberCount: t.memberCount as number | undefined,
   494→      }));
   495→      setTeams(teamList);
   496→    } catch {
   497→      setTeams([]);
   498→    } finally {
   499→      setTeamsLoading(false);
   500→    }
   501→  }, []);
   502→
   503→  useEffect(() => {
   504→    if (currentStep === 2) {
   505→      void fetchTeams();
   506→    }
   507→  }, [currentStep, fetchTeams]);
   508→
   509→  const goToStep = useCallback((step: number) => {
   510→    setDirection(step > currentStep ? 1 : -1);
   511→    setCurrentStep(step);
   512→  }, [currentStep]);
   513→
   514→  const handleNext = useCallback(() => {
   515→    if (currentStep < TOTAL_STEPS - 1) {
   516→      // Save profile to store on each step transition
   517→      setOnboardingProfile(localProfile);
   518→      goToStep(currentStep + 1);
   519→    }
   520→  }, [currentStep, goToStep, localProfile, setOnboardingProfile]);
   521→
   522→  const handleBack = useCallback(() => {
   523→    if (currentStep > 0) {
   524→      goToStep(currentStep - 1);
   525→    }
   526→  }, [currentStep, goToStep]);
   527→
   528→  const handleSkip = useCallback(() => {
   529→    setOnboardingProfile(localProfile);
   530→    if (currentStep < TOTAL_STEPS - 1) {
   531→      goToStep(currentStep + 1);
   532→    }
   533→  }, [currentStep, goToStep, localProfile, setOnboardingProfile]);
   534→
   535→  const handleComplete = useCallback(() => {
   536→    setOnboardingProfile(localProfile);
   537→    completeOnboarding();
   538→  }, [completeOnboarding, localProfile, setOnboardingProfile]);
   539→
   540→  const handleLocalProfileChange = useCallback((data: { position?: OnboardingPosition | null; experience?: OnboardingExperience | null; weightCategory?: OnboardingWeightCategory | null }) => {
   541→    setLocalProfile((prev) => ({ ...prev, ...data }));
   542→  }, []);
   543→
   544→  const handleTeamSelect = useCallback((teamId: string | null) => {
   545→    setLocalProfile((prev) => ({ ...prev, selectedTeamId: teamId }));
   546→  }, []);
   547→
   548→  // Step titles for accessibility
   549→  const stepTitles = ['Welcome', 'Your Profile', 'Pick Your Team', 'All Set'];
   550→
   551→  return (
   552→    <motion.div
   553→      initial={{ opacity: 0 }}
   554→      animate={{ opacity: 1 }}
   555→      exit={{ opacity: 0 }}
   556→      className="fixed inset-0 z-50 flex flex-col bg-gradient-to-br from-red-800 via-red-900 to-red-950"
   557→    >
   558→      {/* Background pattern */}
   559→      <div
   560→        className="absolute inset-0 opacity-[0.04]"
   561→        style={{
   562→          backgroundImage: `repeating-linear-gradient(45deg, transparent, transparent 35px, rgba(255,255,255,0.1) 35px, rgba(255,255,255,0.1) 36px)`,
   563→        }}
   564→      />
   565→
   566→      {/* Skip button (steps 1-2) */}
   567→      {currentStep > 0 && currentStep < 3 && (
   568→        <motion.button
   569→          onClick={handleSkip}
   570→          className="absolute top-4 right-4 z-20 flex items-center gap-1 text-white/50 hover:text-white/80 text-xs font-medium transition-colors"
   571→          initial={{ opacity: 0 }}
   572→          animate={{ opacity: 1 }}
   573→          exit={{ opacity: 0 }}
   574→        >
   575→          <span>Skip</span>
   576→          <SkipForward className="w-3.5 h-3.5" />
   577→        </motion.button>
   578→      )}
   579→
   580→      {/* Back button (steps 1-3) */}
   581→      {currentStep > 0 && (
   582→        <motion.button
   583→          onClick={handleBack}
   584→          className="absolute top-4 left-4 z-20 w-8 h-8 rounded-full bg-white/10 flex items-center justify-center text-white/70 hover:bg-white/20 hover:text-white transition-colors"
   585→          initial={{ opacity: 0, x: -10 }}
   586→          animate={{ opacity: 1, x: 0 }}
   587→          exit={{ opacity: 0, x: -10 }}
   588→        >
   589→          <ChevronLeft className="w-4 h-4" />
   590→        </motion.button>
   591→      )}
   592→
   593→      {/* Main Content Area */}
   594→      <div className="flex-1 relative overflow-hidden flex items-start justify-center pt-16">
   595→        <AnimatePresence mode="wait" custom={direction}>
   596→          <motion.div
   597→            key={currentStep}
   598→            custom={direction}
   599→            initial={{ opacity: 0, x: direction * 60 }}
   600→            animate={{ opacity: 1, x: 0 }}
   601→            exit={{ opacity: 0, x: direction * -60 }}
   602→            transition={{ type: 'spring', damping: 24, stiffness: 200 }}
   603→            className="w-full max-w-md mx-auto"
   604→            role="region"
   605→            aria-label={stepTitles[currentStep]}
   606→          >
   607→            {currentStep === 0 && <WelcomeStep />}
   608→            {currentStep === 1 && (
   609→              <ProfileStep
   610→                profile={localProfile}
   611→                onProfileChange={handleLocalProfileChange}
   612→              />
   613→            )}
   614→            {currentStep === 2 && (
   615→              <TeamStep
   616→                selectedTeamId={localProfile.selectedTeamId}
   617→                onSelectTeam={handleTeamSelect}
   618→                teams={teams}
   619→                loading={teamsLoading}
   620→              />
   621→            )}
   622→            {currentStep === 3 && (
   623→              <CompleteStep profile={localProfile} />
   624→            )}
   625→          </motion.div>
   626→        </AnimatePresence>
   627→      </div>
   628→
   629→      {/* Bottom Navigation */}
   630→      <div className="relative z-10 px-6 pb-8 pt-4">
   631→        {/* Progress Dots */}
   632→        <div className="mb-5">
   633→          <ProgressDots currentStep={currentStep} />
   634→        </div>
   635→
   636→        {/* Action Buttons */}
   637→        <AnimatePresence mode="wait">
   638→          {currentStep === 0 && (
   639→            <motion.div
   640→              key="welcome-btn"
   641→              initial={{ opacity: 0, y: 20 }}
   642→              animate={{ opacity: 1, y: 0 }}
   643→              exit={{ opacity: 0, y: -20 }}
   644→            >
   645→              <Button
   646→                onClick={handleNext}
   647→                className="w-full h-12 rounded-xl bg-brand-gold hover:bg-brand-gold/90 text-red-950 font-black text-sm tracking-wide shadow-lg shadow-brand-gold/20"
   648→              >
   649→                Let&apos;s Get Started
   650→                <ChevronRight className="w-4 h-4 ml-1" />
   651→              </Button>
   652→            </motion.div>
   653→          )}
   654→
   655→          {currentStep === 1 && (
   656→            <motion.div
   657→              key="profile-btn"
   658→              initial={{ opacity: 0, y: 20 }}
   659→              animate={{ opacity: 1, y: 0 }}
   660→              exit={{ opacity: 0, y: -20 }}
   661→            >
   662→              <Button
   663→                onClick={handleNext}
   664→                className="w-full h-12 rounded-xl bg-brand-gold hover:bg-brand-gold/90 text-red-950 font-black text-sm tracking-wide shadow-lg shadow-brand-gold/20"
   665→              >
   666→                Continue
   667→                <ChevronRight className="w-4 h-4 ml-1" />
   668→              </Button>
   669→            </motion.div>
   670→          )}
   671→
   672→          {currentStep === 2 && (
   673→            <motion.div
   674→              key="team-btn"
   675→              initial={{ opacity: 0, y: 20 }}
   676→              animate={{ opacity: 1, y: 0 }}
   677→              exit={{ opacity: 0, y: -20 }}
   678→            >
   679→              <Button
   680→                onClick={handleNext}
   681→                className="w-full h-12 rounded-xl bg-brand-gold hover:bg-brand-gold/90 text-red-950 font-black text-sm tracking-wide shadow-lg shadow-brand-gold/20"
   682→              >
   683→                {localProfile.selectedTeamId ? 'Join & Continue' : 'Continue Without Team'}
   684→                <ChevronRight className="w-4 h-4 ml-1" />
   685→              </Button>
   686→            </motion.div>
   687→          )}
   688→
   689→          {currentStep === 3 && (
   690→            <motion.div
   691→              key="complete-btn"
   692→              initial={{ opacity: 0, y: 20 }}
   693→              animate={{ opacity: 1, y: 0 }}
   694→              exit={{ opacity: 0, y: -20 }}
   695→            >
   696→              <Button
   697→                onClick={handleComplete}
   698→                className="w-full h-12 rounded-xl bg-brand-gold hover:bg-brand-gold/90 text-red-950 font-black text-sm tracking-wide shadow-lg shadow-brand-gold/20"
   699→              >
   700→                Go to Home
   701→                <Sparkles className="w-4 h-4 ml-1" />
   702→              </Button>
   703→            </motion.div>
   704→          )}
   705→        </AnimatePresence>
   706→      </div>
   707→    </motion.div>
   708→  );
   709→}
