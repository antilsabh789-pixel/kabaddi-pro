    57→    57→/* ── Password Strength Helper ── */
    58→    58→function getPasswordStrength(pwd: string): { label: string; color: string; pct: number; barColor: string } {
    59→    59→  if (!pwd) return { label: '', color: '', pct: 0, barColor: '' };
    60→    60→  let score = 0;
    61→    61→  if (pwd.length >= 6) score++;
    62→    62→  if (pwd.length >= 8) score++;
    63→    63→  if (/[A-Z]/.test(pwd)) score++;
    64→    64→  if (/[0-9]/.test(pwd)) score++;
    65→    65→  if (/[^A-Za-z0-9]/.test(pwd)) score++;
    66→    66→
    67→    67→  if (score <= 1) return { label: 'Weak', color: 'text-red-500', pct: 25, barColor: 'bg-red-500' };
    68→    68→  if (score <= 2) return { label: 'Medium', color: 'text-amber-500', pct: 50, barColor: 'bg-amber-500' };
    69→    69→  if (score <= 3) return { label: 'Strong', color: 'text-emerald-500', pct: 75, barColor: 'bg-emerald-500' };
    70→    70→  return { label: 'Very Strong', color: 'text-teal-500', pct: 100, barColor: 'bg-teal-500' };
    71→    71→}

   160→   160→/* ── OTP Input Boxes ── */
   161→   161→function OTPInput({ value, onChange, length = 6 }: { value: string; onChange: (val: string) => void; length?: number }) {
   162→   162→  const inputRefs = useRef<(HTMLInputElement | null)[]>([]);
   163→   163→
   164→   164→  const handleChange = (index: number, digit: string) => {
   165→   165→    if (!/^\d*$/.test(digit)) return;
   166→   166→    const newVal = value.split('');
   167→   167→    newVal[index] = digit;
   168→   168→    const combined = newVal.join('');
   169→   169→    onChange(combined);
   170→   170→    if (digit && index < length - 1) {
   171→   171→      inputRefs.current[index + 1]?.focus();
   172→   172→    }
   173→   173→  };
   174→   174→
   175→   175→  const handleKeyDown = (index: number, e: React.KeyboardEvent<HTMLInputElement>) => {
   176→   176→    if (e.key === 'Backspace' && !value[index] && index > 0) {
   177→   177→      inputRefs.current[index - 1]?.focus();
   178→   178→    }
   179→   179→  };
   180→   180→
   181→   181→  const handlePaste = (e: React.ClipboardEvent) => {
   182→   182→    e.preventDefault();
   183→   183→    const pasted = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, length);
   184→   184→    onChange(pasted.padEnd(length, ''));
   185→   185→    const focusIdx = Math.min(pasted.length, length - 1);
   186→   186→    inputRefs.current[focusIdx]?.focus();
   187→   187→  };
   188→   188→
   189→   189→  return (
   190→   190→    <div className="flex gap-2 justify-center" onPaste={handlePaste}>
   191→   191→      {Array.from({ length }, (_, i) => (
   192→   192→        <motion.div
   193→   193→          key={i}
   194→   194→          initial={{ opacity: 0, y: 10 }}
   195→   195→          animate={{ opacity: 1, y: 0 }}
   196→   196→          transition={{ delay: i * 0.05 }}
   197→   197→        >
   198→   198→          <input
   199→   199→            ref={(el) => { inputRefs.current[i] = el; }}
   200→   200→            type="text"
   201→   201→            inputMode="numeric"
   202→   202→            maxLength={1}
   203→   203→            value={value[i] || ''}
   204→   204→            onChange={(e) => handleChange(i, e.target.value)}
   205→   205→            onKeyDown={(e) => handleKeyDown(i, e)}
   206→   206→            className="w-11 h-13 text-center text-lg font-bold rounded-xl border-2 border-warm-300/60 dark:border-warm-600/40 bg-white/60 dark:bg-white/5 text-warm-800 dark:text-warm-100 focus:border-brand-red focus:ring-2 focus:ring-brand-red/30 outline-none transition-all"
   207→   207→            style={{ height: 52 }}
   208→   208→          />
   209→   209→        </motion.div>
   210→   210→      ))}
   211→   211→    </div>
   212→   212→  );
   213→   213→}

   215→   215→/* ── Countdown Timer Component ── */
   216→   216→function CountdownTimer({ seconds, onComplete }: { seconds: number; onComplete: () => void }) {
   217→   217→  const [timeLeft, setTimeLeft] = useState(seconds);
   218→   218→
   219→   219→  useEffect(() => {
   220→   220→    if (timeLeft <= 0) {
   221→   221→      onComplete();
   222→   222→      return;
   223→   223→    }
   224→   224→    const timer = setTimeout(() => setTimeLeft(timeLeft - 1), 1000);
   225→   225→    return () => clearTimeout(timer);
   226→   226→  }, [timeLeft, onComplete]);
   227→   227→
   228→   228→  const minutes = Math.floor(timeLeft / 60);
   229→   229→  const secs = timeLeft % 60;
   230→   230→
   231→   231→  return (
   232→   232→    <div className="flex items-center gap-1.5 text-warm-500 dark:text-warm-400 text-xs">
   233→   233→      <Timer className="w-3.5 h-3.5" />
   234→   234→      <span>
   235→   235→        {minutes}:{secs.toString().padStart(2, '0')}
   236→   236→      </span>
   237→   237→    </div>
   238→   238→  );
   239→   239→}
