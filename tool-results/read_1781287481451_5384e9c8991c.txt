   118→function AnimatedCounter({ target, duration = 1000 }: { target: number; duration?: number }) {
   119→  const [count, setCount] = useState(0);
   120→  const prevTarget = useRef(target);
   121→  const hasAnimated = useRef(false);
   122→
   123→  useEffect(() => {
   124→    if (prevTarget.current === target && hasAnimated.current) return;
   125→    prevTarget.current = target;
   126→    hasAnimated.current = true;
   127→
   128→    let start = 0;
   129→    const startTime = performance.now();
   130→
   131→    function step(currentTime: number) {
   132→      const elapsed = currentTime - startTime;
   133→      const progress = Math.min(elapsed / duration, 1);
   134→      const eased = 1 - Math.pow(1 - progress, 3);
   135→      const current = Math.round(eased * target);
   136→      setCount(current);
   137→      if (progress < 1) {
   138→        requestAnimationFrame(step);
   139→      }
   140→    }
   141→
   142→    requestAnimationFrame(step);
   143→  }, [target, duration]);
   144→
   145→  return <span>{count}</span>;
   146→}

   150→function MiniProgressRing({ percent, size = 40, colorClass = 'stroke-emerald-500 dark:stroke-emerald-400' }: { percent: number; size?: number; colorClass?: string }) {
   151→  const strokeWidth = 4;
   152→  const radius = (size - strokeWidth) / 2;
   153→  const circumference = 2 * Math.PI * radius;
   154→  const offset = circumference - (percent / 100) * circumference;
   155→
   156→  return (
   157→    <svg width={size} height={size} className="transform -rotate-90">
   158→      <circle
   159→        cx={size / 2}
   160→        cy={size / 2}
   161→        r={radius}
   162→        strokeWidth={strokeWidth}
   163→        className="stroke-warm-200 dark:stroke-warm-700"
   164→        fill="none"
   165→      />
   166→      <motion.circle
   167→        cx={size / 2}
   168→        cy={size / 2}
   169→        r={radius}
   170→        strokeWidth={strokeWidth}
   171→        className={colorClass}
   172→        fill="none"
   173→        strokeLinecap="round"
   174→        strokeDasharray={circumference}
   175→        initial={{ strokeDashoffset: circumference }}
   176→        animate={{ strokeDashoffset: offset }}
   177→        transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   178→      />
   179→    </svg>
   180→  );
   181→}

   216→function getDateGroup(dateStr: string): string {
   217→  const date = new Date(dateStr);
   218→  const now = new Date();
   219→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   220→  const yesterday = new Date(today);
   221→  yesterday.setDate(yesterday.getDate() - 1);
   222→  const weekAgo = new Date(today);
   223→  weekAgo.setDate(weekAgo.getDate() - 7);
   224→
   225→  const matchDate = new Date(date.getFullYear(), date.getMonth(), date.getDate());
   226→
   227→  if (matchDate.getTime() === today.getTime()) return 'Today';
   228→  if (matchDate.getTime() === yesterday.getTime()) return 'Yesterday';
   229→  if (matchDate > weekAgo) return 'This Week';
   230→  return 'Earlier';
   231→}

   233→function formatMatchDate(dateStr: string): string {
   234→  const date = new Date(dateStr);
   235→  return date.toLocaleDateString('en-IN', {
   236→    day: 'numeric',
   237→    month: 'short',
   238→    year: date.getFullYear() !== new Date().getFullYear() ? 'numeric' : undefined,
   239→  });
   240→}
   241→
   242→function isWithinDateFilter(dateStr: string, filter: DateFilter): boolean {
   243→  if (filter === 'all') return true;
   244→  const date = new Date(dateStr);
   245→  const now = new Date();
   246→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   247→
   248→  if (filter === 'week') {
   249→    const weekAgo = new Date(today);
   250→    weekAgo.setDate(weekAgo.getDate() - 7);
   251→    return date >= weekAgo;
   252→  }
   253→  if (filter === 'month') {
   254→    const monthAgo = new Date(today);
   255→    monthAgo.setMonth(monthAgo.getMonth() - 1);
   256→    return date >= monthAgo;
   257→  }
   258→  if (filter === 'year') {
   259→    const yearAgo = new Date(today);
   260→    yearAgo.setFullYear(yearAgo.getFullYear() - 1);
   261→    return date >= yearAgo;
   262→  }
   263→  return true;
   264→}
   265→
   266→function getMatchResult(match: MatchItem, userId?: string): 'won' | 'lost' | 'draw' {
   267→  if (match.homeScore === match.awayScore) return 'draw';
   268→  const homeWon = match.homeScore > match.awayScore;
   269→
   270→  // If userId is a scorer, determine which side they were on
   271→  if (userId) {
   272→    const isScorer = match.scorers.some((s) => s.userId === userId);
   273→    if (isScorer) {
   274→      // Check if the user's team is home or away (based on scorer relation)
   275→      // Default: assume home team if we can't determine
   276→      return homeWon ? 'won' : 'lost';
   277→    }
   278→  }
   279→
   280→  // General: show from home team perspective
   281→  return homeWon ? 'won' : 'lost';
   282→}
   283→
   284→function getWinnerTeamId(match: MatchItem): string | null {
   285→  if (match.homeScore > match.awayScore) return match.homeTeam.id;
   286→  if (match.awayScore > match.homeScore) return match.awayTeam.id;
   287→  return null;
   288→}

   347→        const params = new URLSearchParams();
   348→        params.set('limit', PAGE_SIZE.toString());
   349→        params.set('offset', currentOffset.toString());

   353→          params.set('userId', userId);
   354→        }

   357→        if (!res.ok) throw new Error('Failed to fetch');
   358→
   359→        const data = await res.json();
   360→        const fetched: MatchItem[] = data.matches || [];
   361→
   362→        if (append) {
   363→          setMatches((prev) => [...prev, ...fetched]);
   364→        } else {
   365→          setMatches(fetched);
   366→        }
   367→
   368→        setTotalFromApi(fetched.length);
   369→        setHasMore(fetched.length >= PAGE_SIZE);
   370→      } catch (err) {
   371→        console.error('Match history fetch error:', err);
   372→      } finally {
   373→        setLoading(false);
   374→        setLoadingMore(false);
   375→      }
   376→    },
   377→    [userId]
   378→  );
   379→
   380→  useEffect(() => {
   381→    fetchMatches(0, false);
   382→  }, [fetchMatches]);
   383→
   384→  // Compute stats from all loaded matches (respecting date filter)
   385→  const stats = (() => {
   386→    let wins = 0;
   387→    let losses = 0;
   388→    let draws = 0;
   389→    let totalPoints = 0;
   390→    let highestScore = 0;
   391→    let highestScoringMatch: string | null = null;
   392→
   393→    const dateFiltered = matches.filter((m) =>
   394→      isWithinDateFilter(m.completedAt || m.createdAt, dateFilter)
   395→    );
   396→
   397→    for (const match of dateFiltered) {
   398→      const result = getMatchResult(match, userId);
   399→      if (result === 'won') wins++;
   400→      else if (result === 'lost') losses++;
   401→      else draws++;
   402→
   403→      const matchTotal = match.homeScore + match.awayScore;
   404→      totalPoints += matchTotal;
   405→
   406→      if (matchTotal > highestScore) {
   407→        highestScore = matchTotal;
   408→        highestScoringMatch = match.id;
   409→      }
   410→    }
   411→
   412→    // Current streak
   413→    const streak = (() => {
   414→      const sorted = [...dateFiltered].sort(
   415→        (a, b) =>
   416→          new Date(b.completedAt || b.createdAt).getTime() -
   417→          new Date(a.completedAt || a.createdAt).getTime()
   418→      );
   419→      if (sorted.length === 0) return { type: 'none', count: 0 };
   420→
   421→      const firstResult = getMatchResult(sorted[0], userId);
   422→      let streakCount = 0;
   423→      for (const m of sorted) {
   424→        const r = getMatchResult(m, userId);
   425→        if (r === firstResult) {
   426→          streakCount++;
   427→        } else if (r === 'draw' && firstResult !== 'draw') {
   428→          // draws don't break streaks but don't count either
   429→          continue;
   430→        } else {
   431→          break;
   432→        }
   433→      }
   434→      return { type: firstResult, count: streakCount };
   435→    })();
   436→
   437→    const avgScore = dateFiltered.length > 0
   438→      ? Math.round(totalPoints / dateFiltered.length)
   439→      : 0;
   440→
   441→    return {
   442→      total: dateFiltered.length,
   443→      wins,
   444→      losses,
   445→      draws,
   446→      winRate: dateFiltered.length > 0 ? Math.round((wins / dateFiltered.length) * 100) : 0,
   447→      totalPoints,
   448→      highestScore,
   449→      highestScoringMatch,
   450→      avgScore,
   451→      streak,
   452→    };

   459→    // Date filter
   460→    filtered = filtered.filter((m) =>
   461→      isWithinDateFilter(m.completedAt || m.createdAt, dateFilter)
   462→    );
   463→
   464→    // Result filter
   465→    if (resultFilter !== 'all') {
   466→      filtered = filtered.filter((m) => getMatchResult(m, userId) === resultFilter);
   467→    }

   478→      filtered = filtered.filter(
   479→        (m) => m.gender?.toLowerCase() === genderFilter
   480→      );
   481→    }
   482→
   483→    // Sort
   484→    if (sortFilter === 'newest') {
   485→      filtered.sort(
   486→        (a, b) =>
   487→          new Date(b.completedAt || b.createdAt).getTime() -
   488→          new Date(a.completedAt || a.createdAt).getTime()
   489→      );
   490→    } else if (sortFilter === 'oldest') {

   492→        (a, b) =>
   493→          new Date(a.completedAt || a.createdAt).getTime() -
   494→          new Date(b.completedAt || b.createdAt).getTime()

   496→    } else if (sortFilter === 'highest_score') {

   498→        (a, b) => (b.homeScore + b.awayScore) - (a.homeScore + a.awayScore)
