    91→// Map icon names to components for rendering
    92→function EventIcon({ iconName, className }: { iconName: string; className?: string }) {
    93→  const iconProps = { className };
    94→  switch (iconName) {
    95→    case 'Zap': return <Zap {...iconProps} />;
    96→    case 'Shield': return <Shield {...iconProps} />;
    97→    case 'Flame': return <Flame {...iconProps} />;
    98→    case 'Target': return <Target {...iconProps} />;
    99→    case 'Lock': return <Lock {...iconProps} />;
   100→    case 'Clock': return <Clock {...iconProps} />;
   101→    case 'AlertCircle': return <AlertCircle {...iconProps} />;
   102→    default: return <Zap {...iconProps} />;
   103→  }
   104→}

   108→function formatDateTime(iso: string | null): { date: string; time: string } {
   109→  if (!iso) return { date: '—', time: '' };
   110→  const d = new Date(iso);
   111→  return {
   112→    date: d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }),
   113→    time: d.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true }),
   114→  };
   115→}
   116→
   117→function formatDuration(isoStart: string | null, isoEnd: string | null): string {
   118→  if (!isoStart || !isoEnd) return '';
   119→  const ms = new Date(isoEnd).getTime() - new Date(isoStart).getTime();
   120→  const totalMins = Math.round(ms / 60000);
   121→  return `${totalMins} min`;
   122→}
   123→
   124→function getStatusConfig(status: string): { label: string; color: string; bgColor: string; pulse: boolean } {
   125→  switch (status) {
   126→    case 'LIVE':
   127→    case 'IN_PROGRESS':
   128→      return { label: 'LIVE', color: 'text-red-500', bgColor: 'bg-red-500/10 border-red-500/30', pulse: true };
   129→    case 'COMPLETED':
   130→    case 'FULL_TIME':
   131→      return { label: 'COMPLETED', color: 'text-brand-teal', bgColor: 'bg-brand-teal/10 border-brand-teal/30', pulse: false };
   132→    default:
   133→      return { label: 'UPCOMING', color: 'text-brand-gold', bgColor: 'bg-brand-gold/10 border-brand-gold/30', pulse: false };
   134→  }
   135→}

   139→export default function MatchDetailsScreen({ matchId, onClose }: MatchDetailsScreenProps) {
   140→  const [match, setMatch] = useState<MatchData | null>(null);
   141→  const [loading, setLoading] = useState(true);
   142→  const [error, setError] = useState<string | null>(null);
   143→  const [showShare, setShowShare] = useState(false);
   144→  const [showHighlights, setShowHighlights] = useState(false);
   145→  const [showReplay, setShowReplay] = useState(false);
   146→  const [prevHomeScore, setPrevHomeScore] = useState(0);
   147→  const [prevAwayScore, setPrevAwayScore] = useState(0);
   148→  const [homeScoreFlash, setHomeScoreFlash] = useState(false);
   149→  const [awayScoreFlash, setAwayScoreFlash] = useState(false);
   150→
   151→  const { toast } = useToast();
   152→  const activeMatch = useKabaddiStore((s) => s.activeMatch);
   153→  const timelineRef = useRef<HTMLDivElement>(null);
   154→
   155→  // ── Fetch match ────────────────────────────────────────────────────────────
   156→  const fetchMatch = useCallback(async () => {
   157→    setLoading(true);
   158→    setError(null);
   159→    try {
   160→      const res = await fetch(`/api/matches?id=${encodeURIComponent(matchId)}`);
   161→      if (!res.ok) throw new Error('Match not found');
   162→      const data = await res.json();
   163→      setMatch(data.match as MatchData);
   164→    } catch (err) {
   165→      console.error('Match details fetch error:', err);
   166→      setError('Failed to load match details');
   167→    } finally {
   168→      setLoading(false);
   169→    }
   170→  }, [matchId]);
   171→
   172→  useEffect(() => {
   173→    fetchMatch();
   174→  }, [fetchMatch]);
   175→
   176→  // ── Score change animation ────────────────────────────────────────────────
   177→  useEffect(() => {
   178→    if (!match) return;
   179→    if (match.homeScore !== prevHomeScore && prevHomeScore !== 0) {
   180→      setHomeScoreFlash(true);
   181→      const timer = setTimeout(() => setHomeScoreFlash(false), 800);
   182→      return () => clearTimeout(timer);
   183→    }
   184→    if (match.awayScore !== prevAwayScore && prevAwayScore !== 0) {
   185→      setAwayScoreFlash(true);
   186→      const timer = setTimeout(() => setAwayScoreFlash(false), 800);
   187→      return () => clearTimeout(timer);
   188→    }
   189→    setPrevHomeScore(match.homeScore);
   190→    setPrevAwayScore(match.awayScore);
   191→  }, [match?.homeScore, match?.awayScore, prevHomeScore, prevAwayScore]);
   192→
   193→  // ── Derived data ───────────────────────────────────────────────────────────
   194→  const homeColor = match?.homeTeam?.color || '#DC2626';
   195→  const awayColor = match?.awayTeam?.color || '#1E293B';
   196→  const isHomeWin = match ? match.homeScore > match.awayScore : false;
   197→  const isAwayWin = match ? match.awayScore > match.homeScore : false;
   198→  const isDraw = match ? match.homeScore === match.awayScore : false;
   199→  const winnerSide: 'home' | 'away' | 'draw' | null = match
   200→    ? isDraw ? 'draw' : isHomeWin ? 'home' : 'away'
   201→    : null;
   202→
   203→  const statusConfig = match ? getStatusConfig(match.status) : getStatusConfig('');
   204→
   205→  // ── Player contribution aggregation ────────────────────────────────────────
   206→  const aggregatePlayers = useCallback(() => {
   207→    if (!match) return { topRaiders: [], topDefenders: [] };
   208→
   209→    const raidMap: Record<string, { name: string; teamId: string; points: number; raidPoints: number; bonusPoints: number }> = {};
   210→    const tackleMap: Record<string, { name: string; teamId: string; points: number; tacklePoints: number; superTackles: number }> = {};
   211→
   212→    // Build player name lookup from scorers
   213→    const scorerNames: Record<string, string> = {};
   214→    for (const s of match.scorers) {
   215→      scorerNames[s.userId] = s.user.name;
   216→    }
   217→
   218→    for (const evt of match.events) {
   219→      if (!evt.playerId) continue;
   220→      const meta = EVENT_META[evt.eventType];
   221→      if (!meta) continue;
   222→
   223→      const name = scorerNames[evt.playerId] || evt.playerId.slice(0, 6);
   224→
   225→      if (meta.isRaid) {
   226→        if (!raidMap[evt.playerId]) raidMap[evt.playerId] = { name, teamId: evt.teamId, points: 0, raidPoints: 0, bonusPoints: 0 };
   227→        raidMap[evt.playerId].points += evt.value;
   228→        if (evt.eventType === 'bonus_point') raidMap[evt.playerId].bonusPoints += evt.value;
   229→        else raidMap[evt.playerId].raidPoints += evt.value;
   230→      }
   231→      if (meta.isTackle) {
   232→        if (!tackleMap[evt.playerId]) tackleMap[evt.playerId] = { name, teamId: evt.teamId, points: 0, tacklePoints: 0, superTackles: 0 };
   233→        tackleMap[evt.playerId].points += evt.value;
   234→        tackleMap[evt.playerId].tacklePoints += evt.value;
   235→        if (evt.eventType === 'super_tackle') tackleMap[evt.playerId].superTackles += 1;
   236→      }
   237→    }
   238→
   239→    const topRaiders = Object.values(raidMap).sort((a, b) => b.points - a.points).slice(0, 5);
   240→    const topDefenders = Object.values(tackleMap).sort((a, b) => b.points - a.points).slice(0, 5);
   241→
   242→    return { topRaiders, topDefenders };
   243→  }, [match]);
   244→
   245→  const { topRaiders, topDefenders } = aggregatePlayers();
   246→
   247→  // ── MOTM points ────────────────────────────────────────────────────────────
   248→  const motmPoints = useCallback(() => {
   249→    if (!match?.motmUser) return 0;
   250→    let total = 0;
   251→    for (const evt of match.events) {
   252→      if (evt.playerId === match.motmUser!.id) total += evt.value;
   253→    }
   254→    return total;
   255→  }, [match]);
   256→
   257→  // ── Team stats computation ─────────────────────────────────────────────────
   258→  const computeTeamStats = useCallback(() => {
   259→    if (!match) return null;
   260→    let homeRaidPts = 0, awayRaidPts = 0;
   261→    let homeTacklePts = 0, awayTacklePts = 0;
   262→    let homeBonusPts = 0, awayBonusPts = 0;
   263→    let homeAllOuts = 0, awayAllOuts = 0;
   264→
   265→    for (const evt of match.events) {
   266→      const isHome = evt.teamId === match.homeTeamId;
   267→      const meta = EVENT_META[evt.eventType];
   268→      if (!meta) continue;
   269→
   270→      if (meta.isRaid && evt.eventType !== 'bonus_point') {
   271→        if (isHome) homeRaidPts += evt.value; else awayRaidPts += evt.value;
   272→      }
   273→      if (meta.isTackle) {
   274→        if (isHome) homeTacklePts += evt.value; else awayTacklePts += evt.value;
   275→      }
   276→      if (evt.eventType === 'bonus_point') {
   277→        if (isHome) homeBonusPts += evt.value; else awayBonusPts += evt.value;
   278→      }
   279→      if (evt.eventType === 'all_out') {
   280→        if (isHome) homeAllOuts += 1; else awayAllOuts += 1;
   281→      }
   282→    }
   283→
   284→    return {
   285→      home: { total: match.homeScore, raidPoints: homeRaidPts, tacklePoints: homeTacklePts, bonusPoints: homeBonusPts, allOuts: homeAllOuts },
   286→      away: { total: match.awayScore, raidPoints: awayRaidPts, tacklePoints: awayTacklePts, bonusPoints: awayBonusPts, allOuts: awayAllOuts },
   287→    };
   288→  }, [match]);
   289→
   290→  const teamStats = computeTeamStats();
   291→
   292→  // ── Share data ─────────────────────────────────────────────────────────────
   293→  const buildShareData = useCallback(() => {
   294→    if (!match) return null;
   295→    const motm = match.motmUser ? { name: match.motmUser.name, points: motmPoints() } : null;
   296→    const tr = topRaiders[0] ? { name: topRaiders[0].name, points: topRaiders[0].points } : null;
   297→    const td = topDefenders[0] ? { name: topDefenders[0].name, points: topDefenders[0].points } : null;
   298→    return {
   299→      homeTeam: match.homeTeam.name,
   300→      awayTeam: match.awayTeam.name,
   301→      homeScore: match.homeScore,
   302→      awayScore: match.awayScore,
   303→      homeTeamColor: homeColor,
   304→      awayTeamColor: awayColor,
   305→      tournament: match.tournament?.name || null,
   306→      date: match.startedAt,
   307→      venue: match.venue || null,
   308→      gender: match.gender || null,
   309→      topRaider: tr,
   310→      topDefender: td,
   311→      motm,
   312→    };
   313→  }, [match, homeColor, awayColor, motmPoints, topRaiders, topDefenders]);
   314→
   315→  // ── Sorted events (chronological, oldest first) ────────────────────────────
   316→  const sortedEvents = match
   317→    ? [...match.events].sort(
   318→        (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   319→      )
   320→    : [];
   321→
   322→  // ── Half progress calculation ──────────────────────────────────────────────
   323→  const getHalfProgress = useCallback(() => {
   324→    if (!match?.startedAt) return 0;
   325→    const start = new Date(match.startedAt).getTime();
   326→    const now = match.completedAt
   327→      ? new Date(match.completedAt).getTime()
   328→      : Date.now();
   329→    const halfDurationMs = match.halfDuration * 60 * 1000;
   330→    const totalMatchMs = halfDurationMs * 2;
   331→    const elapsed = Math.min(now - start, totalMatchMs);
   332→    return Math.round((elapsed / totalMatchMs) * 100);
   333→  }, [match]);
   334→
   335→  // ── Render ─────────────────────────────────────────────────────────────────
   336→  return (
   337→    <AnimatePresence>
   338→      <motion.div
   339→        initial={{ opacity: 0 }}
   340→        animate={{ opacity: 1 }}
   341→        exit={{ opacity: 0 }}
   342→        className="fixed inset-0 z-50 bg-warm-50 dark:bg-warm-50 flex flex-col"
   343→      >
   344→        {/* ── Header bar ── */}
   345→        <header className="sticky top-0 z-10 bg-warm-50/90 dark:bg-warm-50/90 backdrop-blur-md border-b border-warm-200/60 dark:border-warm-200/60">
   346→          <div className="px-4 py-3 flex items-center justify-between">
   347→            <div className="flex items-center gap-2">
   348→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-brand-red to-brand-red-dark flex items-center justify-center">
   349→                <Swords className="w-4 h-4 text-white" />
   350→              </div>
   351→              <h1 className="text-sm font-black tracking-wider text-warm-800 dark:text-warm-800">MATCH DETAILS</h1>
   352→            </div>
   353→            <button
   354→              onClick={onClose}
   355→              className="w-8 h-8 rounded-full bg-warm-200 dark:bg-warm-200 flex items-center justify-center text-warm-600 dark:text-warm-600 hover:bg-warm-300 transition-colors"
   356→            >
   357→              <X className="w-4 h-4" />
   358→            </button>
   359→          </div>
   360→        </header>
   361→
   362→        {/* ── Scrollable content ── */}
   363→        <div className="flex-1 overflow-y-auto custom-scrollbar">
   364→          {loading ? (
   365→            <div className="px-4 py-8 space-y-4">
   366→              <div className="h-48 rounded-2xl bg-warm-100 dark:bg-warm-100 animate-pulse" />
   367→              <div className="h-24 rounded-2xl bg-warm-100 dark:bg-warm-100 animate-pulse" />
   368→              <div className="h-64 rounded-2xl bg-warm-100 dark:bg-warm-100 animate-pulse" />
   369→            </div>
   370→          ) : error || !match ? (
   371→            <div className="flex flex-col items-center justify-center py-20 px-6">
   372→              <Shield className="w-12 h-12 text-warm-300 dark:text-warm-300 mb-3" />
   373→              <p className="text-warm-600 dark:text-warm-600 text-sm font-medium">{error || 'Match not found'}</p>
   374→              <Button onClick={fetchMatch} variant="outline" className="mt-4 rounded-xl">
   375→                Retry
   376→              </Button>
   377→            </div>
   378→          ) : (
   379→            <div className="px-4 py-4 space-y-4 pb-28">
   380→
   381→              {/* ════════════════════════════════════════════════════════════════
   382→                  1. ENHANCED MATCH HEADER
   383→                  ════════════════════════════════════════════════════════════════ */}
   384→              <motion.div
   385→                initial={{ opacity: 0, y: 20 }}
   386→                animate={{ opacity: 1, y: 0 }}
   387→                transition={{ duration: 0.5 }}
   388→              >
   389→                <Card className="overflow-hidden border-0 shadow-lg text-white py-0 gap-0">
   390→                  {/* Team color gradient banner */}
   391→                  <div
   392→                    className="relative px-5 pt-4 pb-2"
   393→                    style={{
   394→                      background: `linear-gradient(135deg, ${homeColor} 0%, ${homeColor}cc 35%, ${awayColor}cc 65%, ${awayColor} 100%)`,
   395→                    }}
   396→                  >
   397→                    {/* Match status badge */}
   398→                    <div className="flex items-center justify-between mb-4">
   399→                      <Badge className={`text-[10px] font-black tracking-wider border px-2.5 py-1 ${statusConfig.bgColor} ${statusConfig.color}`}>
   400→                        {statusConfig.pulse && (
   401→                          <span className="relative flex h-2 w-2 mr-1.5">
   402→                            <span className="animate-ping absolute inline-flex h-full w-full rounded-full bg-red-500 opacity-75" />
   403→                            <span className="relative inline-flex rounded-full h-2 w-2 bg-red-500" />

   406→                        {statusConfig.label}
   407→                      </Badge>
   408→                      <div className="flex items-center gap-2">
   409→                        <Badge className="bg-white/20 text-white text-[10px] font-semibold border-0 px-2 py-0.5">

   416→                        </Badge>

   418→                    </div>
   419→
   420→                    {/* Teams & Score */}
   421→                    <div className="flex items-center justify-between pb-3">
   422→                      {/* Home Team */}
   423→                      <motion.div
   424→                        initial={{ opacity: 0, x: -30 }}
   425→                        animate={{ opacity: 1, x: 0 }}
   426→                        transition={{ duration: 0.5, delay: 0.1 }}
   427→                        className="flex-1 text-center"

   429→                        <div className="relative inline-block">
   430→                          <motion.div
   431→                            initial={{ scale: 0 }}
   432→                            animate={{ scale: 1 }}
   433→                            transition={{ type: 'spring', damping: 10, stiffness: 150, delay: 0.2 }}
   434→                            className="w-20 h-20 rounded-full flex items-center justify-center text-white font-bold text-3xl shadow-xl mx-auto ring-4 ring-white/20"
   435→                            style={{ backgroundColor: homeColor }}

   439→                          {winnerSide === 'home' && (
   440→                            <motion.div
   441→                              initial={{ scale: 0, rotate: -180 }}
   442→                              animate={{ scale: 1, rotate: 0 }}
   443→                              transition={{ type: 'spring', damping: 12, stiffness: 200, delay: 0.5 }}
