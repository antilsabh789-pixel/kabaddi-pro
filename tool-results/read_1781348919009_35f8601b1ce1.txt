    31→function ConfettiParticle({ delay, color, index }: { delay: number; color: string; index: number }) {
    32→  const leftPos = (index * 17 + 13) % 100;
    33→  const rotateEnd = (index * 47 + 23) % 720 - 360;
    34→  const xDrift = ((index * 31) % 80) - 40;
    35→
    36→  return (
    37→    <motion.div
    38→      initial={{ y: -20, x: 0, rotate: 0, opacity: 1 }}
    39→      animate={{ y: '100vh', x: xDrift, rotate: rotateEnd, opacity: 0 }}
    40→      transition={{ duration: 3 + (index % 3), delay, ease: 'easeIn' }}
    41→      className="absolute top-0 w-2 h-2 rounded-sm"
    42→      style={{ left: `${leftPos}%`, backgroundColor: color }}
    43→    />
    44→  );
    45→}

   388→function EventLogEntry({ event, matchInfo }: { event: MatchEvent; matchInfo: { homeTeamId: string; homeTeamColor: string; awayTeamColor: string } }) {
   389→  const isHome = event.teamId === matchInfo.homeTeamId;
   390→  const teamColor = isHome ? matchInfo.homeTeamColor : matchInfo.awayTeamColor;
   391→
   392→  const eventIcons: Record<string, string> = {
   393→    raid_point: '✅',
   394→    bonus_point: '🎯',
   395→    tackle_point: '🛡',
   396→    super_tackle: '⚡',
   397→    super_raid: '🔥',
   398→    do_or_die_raid: '🔥',
   399→    all_out: '💥',
   400→    empty_raid: '⏭',
   401→    substitution: '🔄',
   402→    timeout: '📋',
   403→    yellow_card: '🟨',
   404→    red_card: '🟥',
   405→    green_card: '🟩',
   406→  };
   407→
   408→  const eventLabels: Record<string, string> = {
   409→    raid_point: 'Raid Pt',
   410→    bonus_point: 'Bonus Pt',
   411→    tackle_point: 'Tackle Pt',
   412→    super_tackle: 'Super Tackle',
   413→    super_raid: 'Super Raid',
   414→    do_or_die_raid: 'Do-or-Die',
   415→    all_out: 'All Out',
   416→    empty_raid: 'Empty',
   417→    substitution: 'Sub',
   418→    timeout: 'Timeout',
   419→    yellow_card: 'Yellow Card',
   420→    red_card: 'Red Card',
   421→    green_card: 'Green Card',
   422→  };
   423→
   424→  const timeStr = new Date(event.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
   425→
   426→  return (
   427→    <motion.div
   428→      initial={{ opacity: 0, x: -8 }}
   429→      animate={{ opacity: 1, x: 0 }}
   430→      className="flex items-center gap-2 py-1 px-2 rounded-lg hover:bg-gray-800/30 dark:hover:bg-gray-700/30 transition-colors"

   436→      <span className="text-[10px] font-semibold text-gray-500 dark:text-gray-400 flex-1">{eventLabels[event.eventType] || event.eventType}</span>
   437→      {event.value > 0 && (
   438→        <span className="text-[10px] font-black px-1.5 py-0.5 rounded-md" style={{ backgroundColor: `${teamColor}20`, color: teamColor }}>
   439→          +{event.value}
   440→        </span>
   441→      )}
   442→    </motion.div>
   443→  );
   444→}
