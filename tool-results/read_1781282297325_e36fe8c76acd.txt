    56→function getInitial(name: string | null | undefined): string {
    57→  if (!name) return '?';
    58→  return name.trim().charAt(0).toUpperCase();
    59→}

    61→function getDisplayName(name: string | null | undefined): string {
    62→  return name?.trim() || 'Unnamed Player';
    63→}
    64→
    65→function formatTimeAgo(dateStr: string): string {
    66→  try {
    67→    const diff = Date.now() - new Date(dateStr).getTime();
    68→    const seconds = Math.floor(diff / 1000);
    69→    if (seconds < 60) return 'just now';
    70→    const minutes = Math.floor(seconds / 60);
    71→    if (minutes < 60) return `${minutes}m ago`;
    72→    const hours = Math.floor(minutes / 60);
    73→    if (hours < 24) return `${hours}h ago`;
    74→    const days = Math.floor(hours / 24);
    75→    if (days === 1) return 'Yesterday';
    76→    if (days < 7) return `${days}d ago`;
    77→    if (days < 30) return `${Math.floor(days / 7)}w ago`;
    78→    const months = Math.floor(days / 30);
    79→    return `${months}mo ago`;
    80→  } catch {
    81→    return '';
    82→  }
    83→}

   204→function ActivityCardContent({ activity }: { activity: Activity }) {
   205→  const config = getActivityConfig(activity.type);
   206→  const meta = parseMetadata(activity.metadata);
   207→
   208→  switch (activity.type) {
   209→    case 'match_completed': {
   210→      const homeTeam = (meta?.homeTeam as string) || 'Team A';
   211→      const awayTeam = (meta?.awayTeam as string) || 'Team B';
   212→      const homeScore = (meta?.homeScore as number) ?? 0;
   213→      const awayScore = (meta?.awayScore as number) ?? 0;
   214→      const winner = (meta?.winner as string) || null;
   215→
   216→      return (
   217→        <div className="mt-2.5 p-3 rounded-lg bg-white/60 border border-warm-200/60">
   218→          <div className="flex items-center justify-between gap-3">
   219→            <div className={`flex-1 text-center ${winner === 'home' ? 'font-bold text-brand-red' : 'text-warm-600'}`}>
   220→              <p className="text-xs truncate">{homeTeam}</p>
   221→              <p className="text-xl font-black">{homeScore}</p>
   222→            </div>
   223→            <div className="flex flex-col items-center gap-0.5">
   224→              <Swords className="w-4 h-4 text-warm-400" />
   225→              <span className="text-[9px] text-warm-400 font-semibold uppercase">vs</span>
   226→            </div>
   227→            <div className={`flex-1 text-center ${winner === 'away' ? 'font-bold text-brand-red' : 'text-warm-600'}`}>
   228→              <p className="text-xs truncate">{awayTeam}</p>
   229→              <p className="text-xl font-black">{awayScore}</p>
   230→            </div>
   231→          </div>
   232→        </div>
   233→      );
   234→    }
   235→
   236→    case 'tournament_joined': {
   237→      const tournamentName = (meta?.tournamentName as string) || activity.title || 'Tournament';
   238→      const teamCount = (meta?.teamCount as number) ?? null;
   239→
   240→      return (
   241→        <div className="mt-2.5 p-3 rounded-lg bg-white/60 border border-warm-200/60 flex items-center gap-2.5">
   242→          <div className={`w-9 h-9 rounded-lg ${config.badgeBg} flex items-center justify-center shrink-0`}>
   243→            <Flag className={`w-4 h-4 ${config.color}`} />
   244→          </div>
   245→          <div className="min-w-0">
   246→            <p className="text-sm font-semibold text-warm-800 truncate">{tournamentName}</p>
   247→            {teamCount && (
   248→              <p className="text-[11px] text-warm-400">{teamCount} teams participating</p>
   249→            )}
   250→          </div>
   251→        </div>
   252→      );
   253→    }
   254→
   255→    case 'achievement_unlocked': {
   256→      const achievementName = (meta?.achievementName as string) || activity.title || 'Achievement';
   257→      const achievementIcon = (meta?.achievementIcon as string) || '🏆';
   258→
   259→      return (
   260→        <div className="mt-2.5 p-3 rounded-lg bg-white/60 border border-warm-200/60 flex items-center gap-2.5">
   261→          <div className={`w-9 h-9 rounded-lg ${config.badgeBg} flex items-center justify-center shrink-0 text-lg`}>
   262→            {achievementIcon}
   263→          </div>
   264→          <div className="min-w-0">
   265→            <p className="text-sm font-semibold text-warm-800 truncate">{achievementName}</p>
   266→            <p className="text-[11px] text-warm-400">New achievement unlocked!</p>
   267→          </div>
   268→        </div>
   269→      );
   270→    }
   271→
   272→    case 'player_milestone': {
   273→      const milestoneType = (meta?.milestoneType as string) || activity.title || 'Milestone';
   274→      const milestoneValue = (meta?.milestoneValue as string | number) ?? null;
   275→
   276→      return (
   277→        <div className="mt-2.5 p-3 rounded-lg bg-white/60 border border-warm-200/60 flex items-center gap-2.5">
   278→          <div className={`w-9 h-9 rounded-lg ${config.badgeBg} flex items-center justify-center shrink-0`}>
   279→            <Milestone className={`w-4 h-4 ${config.color}`} />
   280→          </div>
   281→          <div className="min-w-0">
   282→            <p className="text-sm font-semibold text-warm-800 truncate">{milestoneType}</p>
   283→            {milestoneValue && (
   284→              <p className="text-[11px] text-warm-400">Reached {String(milestoneValue)}</p>
   285→            )}
   286→          </div>
   287→        </div>
   288→      );
   289→    }
   290→
   291→    default:
   292→      return activity.description ? (
   293→        <p className="mt-1.5 text-xs text-warm-500 leading-relaxed">{activity.description}</p>
   294→      ) : null;
   295→  }
   296→}

   371→// ─── Main Component ───────────────────────────────────────────────
   372→
   373→export default function SocialFeedScreen({ onClose }: SocialFeedScreenProps) {
   374→  const currentUser = useKabaddiStore((s) => s.currentUser);
   375→  const { toast } = useToast();
   376→
   377→  // Activity feed state
   378→  const [activities, setActivities] = useState<Activity[]>([]);
   379→  const [feedLoading, setFeedLoading] = useState(true);
   380→  const [refreshing, setRefreshing] = useState(false);
   381→  const [offset, setOffset] = useState(0);
   382→  const [hasMore, setHasMore] = useState(true);
   383→
   384→  // Suggested players state
   385→  const [suggestedPlayers, setSuggestedPlayers] = useState<SuggestedPlayer[]>([]);
   386→  const [suggestedLoading, setSuggestedLoading] = useState(false);
   387→  const [actionLoadingId, setActionLoadingId] = useState<string | null>(null);
   388→
   389→  // ─── Fetch Activities ───────────────────────────────────────
   390→
   391→  const fetchActivities = useCallback(
   392→    async (isRefresh = false) => {
   393→      if (!currentUser) return;
   394→
   395→      if (isRefresh) {
   396→        setRefreshing(true);
   397→      } else {
   398→        setFeedLoading(true);
   399→      }
   400→
   401→      try {
   402→        const params = new URLSearchParams({
   403→          userId: currentUser.id,
   404→          limit: '20',
   405→          offset: isRefresh ? '0' : String(offset),
   406→        });
   407→        const res = await fetch(`/api/activities?${params}`);
   408→        if (!res.ok) throw new Error('Failed to fetch');
   409→        const data = await res.json();
   410→        const newActivities: Activity[] = data.activities || [];
   411→
   412→        if (isRefresh) {
   413→          setActivities(newActivities);
   414→          setOffset(0);
   415→        } else {
   416→          setActivities((prev) => [...prev, ...newActivities]);
   417→        }
   418→        setHasMore(newActivities.length === 20);
   419→      } catch (err) {
   420→        console.error('Activities fetch error:', err);
   421→        if (isRefresh) setActivities([]);
   422→      } finally {
   423→        setFeedLoading(false);
   424→        setRefreshing(false);
   425→      }
   426→    },
   427→    [currentUser, offset]
   428→  );
   429→
   430→  // ─── Fetch Suggested Players ────────────────────────────────
   431→
   432→  const fetchSuggestedPlayers = useCallback(async () => {
   433→    if (!currentUser) return;
   434→    setSuggestedLoading(true);
   435→    try {
   436→      const params = new URLSearchParams({
   437→        userId: currentUser.id,
   438→        type: 'search',
   439→      });
   440→      const res = await fetch(`/api/follow?${params}`);
   441→      if (!res.ok) throw new Error('Failed to fetch');
   442→      const data = await res.json();
   443→      const players: SuggestedPlayer[] = (data.players || [])
   444→        .filter((p: SuggestedPlayer) => !p.isFollowing && p.id !== currentUser.id)
   445→        .slice(0, 5);
   446→      setSuggestedPlayers(players);
   447→    } catch (err) {
   448→      console.error('Suggested players fetch error:', err);
   449→      setSuggestedPlayers([]);
   450→    } finally {
   451→      setSuggestedLoading(false);
   452→    }
   453→  }, [currentUser]);
   454→
   455→  // ─── Follow / Unfollow action ───────────────────────────────
   456→
   457→  const handleFollowAction = async (
   458→    targetId: string,
   459→    isCurrentlyFollowing: boolean
   460→  ) => {
   461→    if (!currentUser || actionLoadingId) return;
   462→    setActionLoadingId(targetId);
   463→    try {
   464→      const res = await fetch('/api/follow', {
   465→        method: 'POST',
   466→        headers: { 'Content-Type': 'application/json' },
   467→        body: JSON.stringify({
   468→          followerId: currentUser.id,
   469→          followingId: targetId,
   470→          action: isCurrentlyFollowing ? 'unfollow' : 'follow',
   471→        }),
   472→      });
   473→      if (!res.ok) throw new Error('Action failed');
   474→
   475→      const nowFollowing = !isCurrentlyFollowing;
   476→
   477→      // Update suggested players
   478→      setSuggestedPlayers((prev) =>
   479→        prev.map((p) =>
   480→          p.id === targetId ? { ...p, isFollowing: nowFollowing } : p
   481→        )
   482→      );
   483→
   484→      toast({
   485→        title: nowFollowing ? 'Following!' : 'Unfollowed',
   486→        description: nowFollowing
   487→          ? 'You are now following this player'
   488→          : 'You unfollowed this player',
   489→      });
   490→
   491→      // Refresh feed to show new followed user's activities
   492→      if (nowFollowing) {
   493→        fetchActivities(true);
   494→      }
   495→    } catch (err) {
   496→      console.error('Follow action error:', err);
   497→      toast({
   498→        title: 'Error',
   499→        description: 'Something went wrong. Please try again.',
   500→        variant: 'destructive',
   501→      });
   502→    } finally {
   503→      setActionLoadingId(null);
   504→    }
   505→  };
   506→
   507→  // ─── Load more ──────────────────────────────────────────────
   508→
   509→  const loadMore = useCallback(() => {
   510→    if (!hasMore || feedLoading) return;
   511→    setOffset((prev) => prev + 20);
   512→  }, [hasMore, feedLoading]);
   513→
   514→  // ─── Effects ────────────────────────────────────────────────
   515→
   516→  useEffect(() => {
   517→    fetchActivities();
   518→  }, [fetchActivities]);
   519→
   520→  // Refetch when offset changes (load more)
   521→  useEffect(() => {
   522→    if (offset > 0) {
   523→      fetchActivities();
   524→    }
   525→  }, [offset]);
   526→
   527→  useEffect(() => {
   528→    fetchSuggestedPlayers();
   529→  }, [fetchSuggestedPlayers]);
   530→
   531→  // ─── Derived ────────────────────────────────────────────────
   532→
   533→  const feedIsEmpty = !feedLoading && activities.length === 0;
   534→
   535→  // ─── Render ─────────────────────────────────────────────────
   536→
   537→  return (
   538→    <motion.div
   539→      initial={{ y: '100%' }}
   540→      animate={{ y: 0 }}
   541→      exit={{ y: '100%' }}
   542→      transition={{ type: 'spring', damping: 28, stiffness: 300 }}
   543→      className="fixed inset-0 z-50 bg-warm-50 flex flex-col"
   544→    >
   545→      {/* ═══ Header ═══ */}
   546→      <header className="sticky top-0 z-10 bg-warm-50/90 backdrop-blur-md border-b border-warm-200/60">
   547→        <div className="px-4 py-3 flex items-center justify-between">
   548→          <div className="flex items-center gap-2">
   549→            <button
   550→              onClick={onClose}
   551→              className="w-8 h-8 rounded-full bg-warm-200 flex items-center justify-center text-warm-600 hover:bg-warm-300 transition-colors"
   552→            >
   553→              <ArrowLeft className="w-4 h-4" />
   554→            </button>
   555→            <div className="flex items-center gap-2">
   556→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-brand-teal to-brand-teal-dark flex items-center justify-center">
   557→                <Rss className="w-4 h-4 text-white" />
   558→              </div>
   559→              <h1 className="text-base font-black tracking-wider text-warm-800">
   560→                SOCIAL FEED
   561→              </h1>
   562→            </div>
   563→          </div>
   564→          <button
   565→            onClick={() => fetchActivities(true)}
   566→            disabled={refreshing}
   567→            className="w-8 h-8 rounded-full bg-warm-100 flex items-center justify-center text-warm-500 hover:bg-warm-200 hover:text-warm-700 transition-colors disabled:opacity-50"
   568→            aria-label="Refresh feed"
   569→          >
   570→            <RefreshCw
   571→              className={`w-4 h-4 ${refreshing ? 'animate-spin' : ''}`}
   572→            />
   573→          </button>
   574→        </div>
   575→      </header>
   576→
   577→      {/* ═══ Content ═══ */}
   578→      <div className="flex-1 overflow-y-auto custom-scrollbar">
   579→        <AnimatePresence mode="wait">
   580→          {feedLoading ? (
