    65→    65→function formatTimeAgo(dateStr: string): string {
    66→    66→  try {
    67→    67→    const diff = Date.now() - new Date(dateStr).getTime();
    68→    68→    const seconds = Math.floor(diff / 1000);
    69→    69→    if (seconds < 60) return 'just now';
    70→    70→    const minutes = Math.floor(seconds / 60);
    71→    71→    if (minutes < 60) return `${minutes}m ago`;
    72→    72→    const hours = Math.floor(minutes / 60);
    73→    73→    if (hours < 24) return `${hours}h ago`;
    74→    74→    const days = Math.floor(hours / 24);
    75→    75→    if (days === 1) return 'Yesterday';
    76→    76→    if (days < 7) return `${days}d ago`;
    77→    77→    if (days < 30) return `${Math.floor(days / 7)}w ago`;
    78→    78→    const months = Math.floor(days / 30);
    79→    79→    return `${months}mo ago`;
    80→    80→  } catch {
    81→    81→    return '';
    82→    82→  }
    83→    83→}

   373→   373→export default function SocialFeedScreen({ onClose }: SocialFeedScreenProps) {
   374→   374→  const currentUser = useKabaddiStore((s) => s.currentUser);
   375→   375→  const { toast } = useToast();
   376→   376→
   377→   377→  // Activity feed state
   378→   378→  const [activities, setActivities] = useState<Activity[]>([]);
   379→   379→  const [feedLoading, setFeedLoading] = useState(true);
   380→   380→  const [refreshing, setRefreshing] = useState(false);
   381→   381→  const [offset, setOffset] = useState(0);
   382→   382→  const [hasMore, setHasMore] = useState(true);
   383→   383→
   384→   384→  // Suggested players state
   385→   385→  const [suggestedPlayers, setSuggestedPlayers] = useState<SuggestedPlayer[]>([]);
   386→   386→  const [suggestedLoading, setSuggestedLoading] = useState(false);
   387→   387→  const [actionLoadingId, setActionLoadingId] = useState<string | null>(null);
   388→   388→
   389→   389→  // ─── Fetch Activities ───────────────────────────────────────
   390→   390→
   391→   391→  const fetchActivities = useCallback(
   392→   392→    async (isRefresh = false) => {
   393→   393→      if (!currentUser) return;
   394→   394→
   395→   395→      if (isRefresh) {
   396→   396→        setRefreshing(true);
   397→   397→      } else {
   398→   398→        setFeedLoading(true);
   399→   399→      }
   400→   400→
   401→   401→      try {
   402→   402→        const params = new URLSearchParams({
   403→   403→          userId: currentUser.id,
   404→   404→          limit: '20',
   405→   405→          offset: isRefresh ? '0' : String(offset),
   406→   406→        });
   407→   407→        const res = await fetch(`/api/activities?${params}`);
   408→   408→        if (!res.ok) throw new Error('Failed to fetch');
   409→   409→        const data = await res.json();
   410→   410→        const newActivities: Activity[] = data.activities || [];
   411→   411→
   412→   412→        if (isRefresh) {
   413→   413→          setActivities(newActivities);
   414→   414→          setOffset(0);
   415→   415→        } else {
   416→   416→          setActivities((prev) => [...prev, ...newActivities]);
   417→   417→        }
   418→   418→        setHasMore(newActivities.length === 20);
   419→   419→      } catch (err) {
   420→   420→        console.error('Activities fetch error:', err);
   421→   421→        if (isRefresh) setActivities([]);
   422→   422→      } finally {
   423→   423→        setFeedLoading(false);
   424→   424→        setRefreshing(false);
   425→   425→      }
   426→   426→    },
   427→   427→    [currentUser, offset]
   428→   428→  );
   429→   429→
   430→   430→  // ─── Fetch Suggested Players ────────────────────────────────
   431→   431→
   432→   432→  const fetchSuggestedPlayers = useCallback(async () => {
   433→   433→    if (!currentUser) return;
   434→   434→    setSuggestedLoading(true);
   435→   435→    try {
   436→   436→      const params = new URLSearchParams({
   437→   437→        userId: currentUser.id,
   438→   438→        type: 'search',
   439→   439→      });
   440→   440→      const res = await fetch(`/api/follow?${params}`);
   441→   441→      if (!res.ok) throw new Error('Failed to fetch');
   442→   442→      const data = await res.json();
   443→   443→      const players: SuggestedPlayer[] = (data.players || [])
   444→   444→        .filter((p: SuggestedPlayer) => !p.isFollowing && p.id !== currentUser.id)
   445→   445→        .slice(0, 5);
   446→   446→      setSuggestedPlayers(players);
   447→   447→    } catch (err) {
   448→   448→      console.error('Suggested players fetch error:', err);
   449→   449→      setSuggestedPlayers([]);
   450→   450→    } finally {
   451→   451→      setSuggestedLoading(false);
   452→   452→    }
   453→   453→  }, [currentUser]);
   454→   454→
   455→   455→  // ─── Follow / Unfollow action ───────────────────────────────
   456→   456→
   457→   457→  const handleFollowAction = async (
   458→   458→    targetId: string,
   459→   459→    isCurrentlyFollowing: boolean
   460→   460→  ) => {
   461→   461→    if (!currentUser || actionLoadingId) return;
   462→   462→    setActionLoadingId(targetId);
   463→   463→    try {
   464→   464→      const res = await fetch('/api/follow', {
   465→   465→        method: 'POST',
   466→   466→        headers: { 'Content-Type': 'application/json' },
   467→   467→        body: JSON.stringify({
   468→   468→          followerId: currentUser.id,
   469→   469→          followingId: targetId,
   470→   470→          action: isCurrentlyFollowing ? 'unfollow' : 'follow',
   471→   471→        }),
   472→   472→      });
   473→   473→      if (!res.ok) throw new Error('Action failed');
   474→   474→
   475→   475→      const nowFollowing = !isCurrentlyFollowing;
   476→   476→
