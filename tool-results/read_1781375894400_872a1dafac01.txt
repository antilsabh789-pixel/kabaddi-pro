   239→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   240→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   241→  let homeScore = 0;
   242→  let awayScore = 0;
   243→  let homeOutPlayers = 0;
   244→  let awayOutPlayers = 0;
   245→  let homeOutPlayerIds: string[] = [];
   246→  let awayOutPlayerIds: string[] = [];
   247→
   248→  for (const evt of events) {
   249→    const side = getTeamSide(match, evt.teamId);
   250→    const points = evt.value;
   251→    if (side === 'home') {
   252→      homeScore += points;
   253→    } else {
   254→      awayScore += points;
   255→    }
   256→
   257→    // Count-based out tracking (backward compat)
   258→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   259→      if (side === 'home') {
   260→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   261→      } else {
   262→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   263→      }
   264→    }
   265→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   266→      if (side === 'home') {
   267→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   268→      } else {
   269→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   270→      }
   271→    }
   272→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   273→    if (evt.eventType === 'self_out') {
   274→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   275→      if (side === 'home') {
   276→        // Home scored from self-out → away defender went out
   277→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   278→        // Home revives one player
   279→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   280→      } else {
   281→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   282→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   283→      }
   284→    }
   285→    if (evt.eventType === 'all_out') {
   286→      if (side === 'home') {
   287→        awayOutPlayers = 0;
   288→      } else {
   289→        homeOutPlayers = 0;
   290→      }
   291→    }
   292→
   293→    // ID-based out tracking
   294→    let details: Record<string, unknown> = {};
   295→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   296→
   297→    if (evt.eventType === 'raid_point') {
   298→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   299→      if (side === 'home') {
   300→        // Home team scored raid → away defenders go out
   301→        for (const tid of touchedIds) {
   302→          if (!awayOutPlayerIds.includes(tid)) {
   303→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   304→          }
   305→        }
   306→        // Home team revives players (first-out-first-in)
   307→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   308→        if (reviveCount > 0) {
   309→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   310→        }
   311→      } else {
   312→        // Away team scored raid → home defenders go out
   313→        for (const tid of touchedIds) {
   314→          if (!homeOutPlayerIds.includes(tid)) {
   315→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   316→          }
   317→        }
   318→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   319→        if (reviveCount > 0) {
   320→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   321→        }
   322→      }
   323→    }
   324→
   325→    if (evt.eventType === 'bonus_point') {
   326→      // One out player from scoring team revives
   327→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   328→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   329→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   330→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   331→      }
   332→    }
   333→
   334→    // Self-out: defender steps off mat during a live raid
   335→    if (evt.eventType === 'self_out') {
   336→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   337→      if (selfOutPlayerId) {
   338→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   339→        if (side === 'home') {
   340→          // Home scored → away defender self-out
   341→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   342→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   343→          }
   344→          // Home revives first-out player
   345→          if (homeOutPlayerIds.length > 0) {
   346→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   347→          }
   348→        } else {
   349→          // Away scored → home defender self-out
   350→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   351→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   352→          }
   353→          if (awayOutPlayerIds.length > 0) {
   354→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   355→          }
   356→        }
   357→      }
   358→    }
   359→
   360→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   361→      // Raider goes out. The tackle is scored by the DEFENDING team.
   362→      // So the raider is from the OPPOSING team.
   363→      const raiderId = (details.raiderId as string) || evt.playerId;
   364→      if (side === 'home' && raiderId) {
   365→        // Home team scored tackle → away raider was caught → away raider goes out
   366→        if (!awayOutPlayerIds.includes(raiderId)) {
   367→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   368→        }
   369→      } else if (side === 'away' && raiderId) {
   370→        // Away team scored tackle → home raider was caught → home raider goes out
   371→        if (!homeOutPlayerIds.includes(raiderId)) {
   372→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   373→        }
   374→      }
   375→    }
   376→
   377→    if (evt.eventType === 'all_out') {
   378→      if (side === 'home') {
   379→        awayOutPlayerIds = [];
   380→      } else {
   381→        homeOutPlayerIds = [];
   382→      }
   383→    }
   384→  }
   385→
   386→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   387→}

   389→/** Event types that constitute a complete raid and should flip the turn */
   390→const RAID_ENDING_EVENT_TYPES: string[] = [
   391→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   392→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   393→];
   394→
   395→/** Event types that do NOT flip the turn (non-raid events) */
   396→const NON_RAID_EVENT_TYPES: string[] = [
   397→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   398→];
   399→
   400→/**
   401→ * Event types where the SCORING team is the DEFENDING team.
   402→ * In these cases, the defending team scored (they made the tackle / raider failed do-or-die),
   403→ * so they should raid NEXT — the turn goes to the SAME side that scored.
   404→ *
   405→ * For all other raid-ending events (raid_point, bonus_point, super_raid, empty_raid, self_out),
   406→ * the scoring team is the RAIDING team, so the turn goes to the OPPOSITE side.
   407→ */
   408→const DEFENDING_TEAM_SCORES: Set<string> = new Set([
   409→  'tackle_point',
   410→  'super_tackle',
   411→  'do_or_die_raid',
   412→]);
   413→
   414→/** Helper: determine raid queue from last raid-ending event */
   415→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   416→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   417→  if (lastRaidEvent) {
   418→    const side = getTeamSide(match, lastRaidEvent.teamId);
   419→    // If the defending team scored (tackle/super_tackle/do_or_die_raid),
   420→    // THEY raid next — return the SAME side as the scoring team.
   421→    if (DEFENDING_TEAM_SCORES.has(lastRaidEvent.eventType)) {
   422→      return side;
   423→    }
   424→    // If the raiding team scored (touch points, bonus, super raid, etc.)
   425→    // or it was an empty raid, the DEFENDING team raids next — return the OPPOSITE side.
   426→    return side === 'home' ? 'away' : 'home';
   427→  }
   428→  return match.raidQueue;
   429→}

   591→            timestamp: Date.now(),

   593→          const updatedEvents = [...state.activeMatch.events, newEvent];
   594→
   595→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   596→
   597→          // Only flip raidQueue for raid-ending events.
   598→          // Non-raid events (timeout, substitution, cards, standalone all_out)
   599→          // should NOT change whose turn it is to raid.
   600→          const isRaidEndingEvent = RAID_ENDING_EVENT_TYPES.includes(event.eventType);
   601→          const raidQueue = isRaidEndingEvent
   602→            ? getRaidQueueFromEvents(state.activeMatch, updatedEvents)
   603→            : state.activeMatch.raidQueue;
   604→
   605→          return {
   606→            activeMatch: {
   607→              ...state.activeMatch,
   608→              events: updatedEvents,
   609→              ...calculated,
   610→              raidQueue,
   611→            },
   612→          };
   613→        }),
   614→
   615→      addBatchEvents: (events) =>
   616→        set((state) => {
   617→          if (!state.activeMatch || events.length === 0) return {};
   618→          const newEvents: MatchEvent[] = events.map((e, i) => ({
   619→            ...e,
   620→            id: `evt_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 7)}`,
   621→            timestamp: Date.now() + i,
   622→          }));
   623→          const updatedEvents = [...state.activeMatch.events, ...newEvents];
   624→
   625→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   626→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, updatedEvents);
   627→
   628→          return {
   629→            activeMatch: {
   630→              ...state.activeMatch,
   631→              events: updatedEvents,
   632→              ...calculated,
   633→              raidQueue,
   634→            },
   635→          };
   636→        }),
   637→
   638→      undoLastEvent: () =>
   639→        set((state) => {
   640→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   641→          const events = [...state.activeMatch.events];
   642→          events.pop();
   643→
   644→          const calculated = recalculateFromEvents(state.activeMatch, events);
   645→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   646→
   647→          return {
   648→            activeMatch: {
   649→              ...state.activeMatch,
   650→              events,
   651→              ...calculated,
   652→              raidQueue,
   653→            },
   654→          };
   655→        }),
   656→
   657→      undoLastRaid: () =>
   658→        set((state) => {
   659→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   660→
   661→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   662→          // A raid batch has events with the same second in timestamp
   663→          const events = [...state.activeMatch.events];
   664→          if (events.length === 0) return {};
   665→
   666→          // Find the timestamp of the last event
   667→          const lastTimestamp = events[events.length - 1].timestamp;
   668→          // Remove all events that have the same timestamp (within 5ms for batch events)
   669→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   670→            events.pop();
   671→          }
   672→
   673→          const calculated = recalculateFromEvents(state.activeMatch, events);
   674→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   675→
   676→          return {
   677→            activeMatch: {
   678→              ...state.activeMatch,
   679→              events,
   680→              ...calculated,
   681→              raidQueue,
   682→            },
   683→          };
   684→        }),
   685→
   686→      switchHalf: () =>
   687→        set((state) => {
   688→          if (!state.activeMatch) return {};
   689→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   690→          return {
   691→            activeMatch: {
   692→              ...state.activeMatch,
   693→              currentHalf: newHalf,
   694→              timer: state.activeMatch.halfDuration * 60,
   695→              isDoOrDie: false,
   696→              doOrDieTeamId: null,
   697→            },
   698→          };
   699→        }),
   700→
   701→      setTimer: (time) =>
   702→        set((state) => {
   703→          if (!state.activeMatch) return {};
   704→          return { activeMatch: { ...state.activeMatch, timer: time } };
   705→        }),
   706→
   707→      setDoOrDie: (value, teamId?) =>
   708→        set((state) => {
   709→          if (!state.activeMatch) return {};
   710→          return {
   711→            activeMatch: {
   712→              ...state.activeMatch,
   713→              isDoOrDie: value,
   714→              doOrDieTeamId: value ? (teamId || null) : null,
   715→            },
   716→          };
   717→        }),
   718→
   719→      switchRaidQueue: () =>
   720→        set((state) => {
   721→          if (!state.activeMatch) return {};
   722→          return {
   723→            activeMatch: {
   724→              ...state.activeMatch,
   725→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   726→            },
   727→          };
   728→        }),
   729→
   730→      callTimeout: (team) =>
   731→        set((state) => {
   732→          if (!state.activeMatch) return {};
   733→          return {
   734→            activeMatch: {
   735→              ...state.activeMatch,
   736→              homeTimeouts:
   737→                team === 'home'
   738→                  ? state.activeMatch.homeTimeouts + 1
   739→                  : state.activeMatch.homeTimeouts,
   740→              awayTimeouts:
   741→                team === 'away'
   742→                  ? state.activeMatch.awayTimeouts + 1
   743→                  : state.activeMatch.awayTimeouts,
   744→            },
   745→          };
   746→        }),
   747→
   748→      setOutPlayers: (team, count) =>
   749→        set((state) => {
   750→          if (!state.activeMatch) return {};
   751→          return {
   752→            activeMatch: {
   753→              ...state.activeMatch,
   754→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   755→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   756→            },
   757→          };
   758→        }),
   759→
   760→      fetchHomeData: async () => {
   761→        try {
   762→          const res = await fetch('/api/stats');
   763→          if (!res.ok) throw new Error('Failed to fetch stats');
   764→
   765→          const data = await res.json();
   766→
   767→          // Cache in store
   768→          set({ homeData: data });
   769→          return data;
   770→        } catch (err) {
   771→          console.error('fetchHomeData error:', err);
   772→          return null;
   773→        }
   774→      },
   775→
   776→      // Notification actions
   777→      addNotification: (notification) =>
   778→        set((state) => ({
   779→          notifications: [
   780→            {
   781→              ...notification,
   782→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   783→              timestamp: Date.now(),
   784→              read: false,
   785→            },
   786→            ...state.notifications,
   787→          ].slice(0, 50), // Keep max 50 notifications
   788→        })),
   789→
   790→      markNotificationRead: (id) =>
   791→        set((state) => ({
   792→          notifications: state.notifications.map((n) =>
   793→            n.id === id ? { ...n, read: true } : n
   794→          ),
   795→        })),
   796→
   797→      markAllRead: () =>
   798→        set((state) => ({
   799→          notifications: state.notifications.map((n) => ({ ...n, read: true })),
   800→        })),
   801→
   802→      clearNotifications: () => set({ notifications: [] }),
   803→
   804→      setLanguage: (lang) => set({ language: lang }),
   805→
   806→      // Onboarding actions
   807→      setHasCompletedOnboarding: (value) => set({ hasCompletedOnboarding: value }),
   808→
   809→      setOnboardingProfile: (profile) =>
   810→        set((state) => ({
   811→          onboardingProfile: { ...state.onboardingProfile, ...profile },
   812→        })),
   813→
   814→      completeOnboarding: () =>
   815→        set((state) => ({
   816→          hasCompletedOnboarding: true,
   817→          currentUser: state.currentUser
   818→            ? {
   819→                ...state.currentUser,
   820→                ...(state.onboardingProfile.position && { role: state.onboardingProfile.position }),
   821→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   822→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   823→                createdAt: state.currentUser.createdAt || Date.now(),
   824→              }
   825→            : null,
   826→        })),
   827→
   828→      // Team management actions
   829→      setTeamFilter: (filter) =>
   830→        set((state) => ({
   831→          teamManagement: { ...state.teamManagement, teamFilter: filter },
   832→        })),
   833→
   834→      setTeamSearch: (search) =>
   835→        set((state) => ({
   836→          teamManagement: { ...state.teamManagement, teamSearch: search },
   837→        })),
   838→
   839→      setSelectedTeamId: (teamId) =>
   840→        set((state) => ({
   841→          teamManagement: { ...state.teamManagement, selectedTeamId: teamId },
   842→        })),
   843→
   844→      setTeamDetailOpen: (open) =>
   845→        set((state) => ({
   846→          teamManagement: { ...state.teamManagement, teamDetailOpen: open },
   847→        })),
   848→
   849→      // Match transfer / handoff — load a match from a transfer
   850→      loadMatch: (match) =>
   851→        set({
   852→          activeMatch: match,
   853→          activeTab: 'quick-score' as TabId,
   854→          showToss: false,
   855→          tossMatchConfig: null,
   856→        }),
   857→    }),
   858→    {
   859→      name: 'kabaddi-pro-storage',
   860→      storage: createJSONStorage(() => localStorage),
   861→      // Only persist auth-related fields and notifications
   862→      partialize: (state) => ({
   863→        isAuthenticated: state.isAuthenticated,
   864→        isOnboarded: state.isOnboarded,
   865→        currentUser: state.currentUser,
   866→        hasSeenSplash: state.hasSeenSplash,
   867→        notifications: state.notifications,
   868→        language: state.language,
   869→        hasCompletedOnboarding: state.hasCompletedOnboarding,
   870→        onboardingProfile: state.onboardingProfile,
   871→        coachAcademies: state.coachAcademies,
   872→      }),
   873→    }
   874→  )
   875→);

   877→export function getEventPoints(eventType: EventType): number {
   878→  switch (eventType) {
   879→    case 'raid_point':
   880→      return 1;
   881→    case 'bonus_point':
   882→      return 1;
   883→    case 'tackle_point':
   884→      return 1;
   885→    case 'super_raid':
   886→      return 1;
   887→    case 'super_tackle':
   888→      return 1;
   889→    case 'do_or_die_raid':
   890→      return 1;
   891→    case 'all_out':
   892→      return 2;
   893→    case 'self_out':
   894→      return 1;
   895→    case 'empty_raid':
   896→      return 0;
   897→    case 'substitution':
   898→    case 'timeout':
   899→    case 'yellow_card':
   900→    case 'red_card':
   901→    case 'green_card':
   902→      return 0;
   903→    default:
   904→      return 0;
   905→  }
   906→}
