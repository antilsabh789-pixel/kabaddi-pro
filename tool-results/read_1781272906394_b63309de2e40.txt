    88→// ─── Helpers ──────────────────────────────────────────────────────────────────
    89→
    90→function formatDateTime(iso: string | null): { date: string; time: string } {
    91→  if (!iso) return { date: '—', time: '' };
    92→  const d = new Date(iso);
    93→  return {
    94→    date: d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' }),
    95→    time: d.toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit', hour12: true }),
    96→  };
    97→}
    98→
    99→function formatDuration(isoStart: string | null, isoEnd: string | null): string {
   100→  if (!isoStart || !isoEnd) return '';
   101→  const ms = new Date(isoEnd).getTime() - new Date(isoStart).getTime();
   102→  const totalMins = Math.round(ms / 60000);
   103→  return `${totalMins} min`;
   104→}
