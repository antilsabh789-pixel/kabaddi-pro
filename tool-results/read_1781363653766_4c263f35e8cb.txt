   236→   236→   236→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   237→   237→   237→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   238→   238→   238→  let homeScore = 0;
   239→   239→   239→  let awayScore = 0;
   240→   240→   240→  let homeOutPlayers = 0;
   241→   241→   241→  let awayOutPlayers = 0;
   242→   242→   242→  let homeOutPlayerIds: string[] = [];
   243→   243→   243→  let awayOutPlayerIds: string[] = [];
   244→   244→   244→
   245→   245→   245→  for (const evt of events) {
   246→   246→   246→    const side = getTeamSide(match, evt.teamId);
   247→   247→   247→    const points = evt.value;
   248→   248→   248→    if (side === 'home') {
   249→   249→   249→      homeScore += points;
   250→   250→   250→    } else {
   251→   251→   251→      awayScore += points;
   252→   252→   252→    }
   253→   253→   253→
   254→   254→   254→    // Count-based out tracking (backward compat)
   255→   255→   255→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   256→   256→   256→      if (side === 'home') {
   257→   257→   257→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   258→   258→   258→      } else {
   259→   259→   259→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   260→   260→   260→      }
   261→   261→   261→    }
   262→   262→   262→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   263→   263→   263→      if (side === 'home') {
   264→   264→   264→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   265→   265→   265→      } else {
   266→   266→   266→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   267→   267→   267→      }
   268→   268→   268→    }
   269→   269→   269→    if (evt.eventType === 'all_out') {
   270→   270→   270→      if (side === 'home') {
   271→   271→   271→        awayOutPlayers = 0;
   272→   272→   272→      } else {
   273→   273→   273→        homeOutPlayers = 0;
   274→   274→   274→      }
   275→   275→   275→    }
   276→   276→   276→
   277→   277→   277→    // ID-based out tracking
   278→   278→   278→    let details: Record<string, unknown> = {};
   279→   279→   279→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   280→   280→   280→
   281→   281→   281→    if (evt.eventType === 'raid_point') {
   282→   282→   282→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   283→   283→   283→      if (side === 'home') {
   284→   284→   284→        // Home team scored raid → away defenders go out
   285→   285→   285→        for (const tid of touchedIds) {
   286→   286→   286→          if (!awayOutPlayerIds.includes(tid)) {
   287→   287→   287→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   288→   288→   288→          }
   289→   289→   289→        }
   290→   290→   290→        // Home team revives players (first-out-first-in)
   291→   291→   291→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   292→   292→   292→        if (reviveCount > 0) {
   293→   293→   293→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   294→   294→   294→        }
   295→   295→   295→      } else {
   296→   296→   296→        // Away team scored raid → home defenders go out
   297→   297→   297→        for (const tid of touchedIds) {
   298→   298→   298→          if (!homeOutPlayerIds.includes(tid)) {
   299→   299→   299→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   300→   300→   300→          }
   301→   301→   301→        }
   302→   302→   302→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   303→   303→   303→        if (reviveCount > 0) {
   304→   304→   304→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   305→   305→   305→        }
   306→   306→   306→      }
   307→   307→   307→    }
   308→   308→   308→
   309→   309→   309→    if (evt.eventType === 'bonus_point') {
   310→   310→   310→      // One out player from scoring team revives
   311→   311→   311→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   312→   312→   312→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   313→   313→   313→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   314→   314→   314→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   315→   315→   315→      }
   316→   316→   316→    }
   317→   317→   317→
   318→   318→   318→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   319→   319→   319→      // Raider goes out. The tackle is scored by the DEFENDING team.
   320→   320→   320→      // So the raider is from the OPPOSING team.
   321→   321→   321→      const raiderId = (details.raiderId as string) || evt.playerId;
   322→   322→   322→      if (side === 'home' && raiderId) {
   323→   323→   323→        // Home team scored tackle → away raider was caught → away raider goes out
   324→   324→   324→        if (!awayOutPlayerIds.includes(raiderId)) {
   325→   325→   325→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   326→   326→   326→        }
   327→   327→   327→      } else if (side === 'away' && raiderId) {
   328→   328→   328→        // Away team scored tackle → home raider was caught → home raider goes out
   329→   329→   329→        if (!homeOutPlayerIds.includes(raiderId)) {
   330→   330→   330→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   331→   331→   331→        }
   332→   332→   332→      }
   333→   333→   333→    }
   334→   334→   334→
   335→   335→   335→    if (evt.eventType === 'all_out') {
   336→   336→   336→      if (side === 'home') {
   337→   337→   337→        awayOutPlayerIds = [];
   338→   338→   338→      } else {
   339→   339→   339→        homeOutPlayerIds = [];
   340→   340→   340→      }
   341→   341→   341→    }
   342→   342→   342→  }
   343→   343→   343→
   344→   344→   344→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   345→   345→   345→}

   347→   347→   347→/** Helper: determine raid queue from last raid event */
   348→   348→   348→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   349→   349→   349→  const raidEventTypes = ['raid_point', 'tackle_point', 'super_tackle', 'empty_raid', 'bonus_point', 'do_or_die_raid'];
   350→   350→   350→  const lastRaidEvent = [...events].reverse().find(e => raidEventTypes.includes(e.eventType));
   351→   351→   351→  if (lastRaidEvent) {
   352→   352→   352→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   353→   353→   353→  }
   354→   354→   354→  return match.raidQueue;
   355→   355→   355→}
