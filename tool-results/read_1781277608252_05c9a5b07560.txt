   240→   240→// ─── Animated Counter ───────────────────────────────────────────────

   242→   242→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   243→   243→  const [count, setCount] = useState(0);
   244→   244→  const prevTarget = useRef(target);
   245→   245→
   246→   246→  useEffect(() => {
   247→   247→    if (prevTarget.current === target && count !== 0) return;
   248→   248→    prevTarget.current = target;
   249→   249→
   250→   250→    let start = 0;
   251→   251→    const startTime = performance.now();
   252→   252→
   253→   253→    function step(currentTime: number) {
   254→   254→      const elapsed = currentTime - startTime;
   255→   255→      const progress = Math.min(elapsed / duration, 1);
   256→   256→      // ease-out cubic
   257→   257→      const eased = 1 - Math.pow(1 - progress, 3);
   258→   258→      const current = Math.round(eased * target);
   259→   259→      setCount(current);
   260→   260→      if (progress < 1) {
   261→   261→        requestAnimationFrame(step);
   262→   262→      }
   263→   263→    }
   264→   264→
   265→   265→    requestAnimationFrame(step);
   266→   266→  }, [target, duration]);
   267→   267→
   268→   268→  return <span>{count}</span>;
   269→   269→}

   383→   383→// ─── Countdown Timer ────────────────────────────────────────────────
   384→   384→
   385→   385→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   386→   386→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   387→   387→  const [expired, setExpired] = useState(!targetDate);
   388→   388→
   389→   389→  useEffect(() => {
   390→   390→    if (!targetDate) {
   391→   391→      return;
   392→   392→    }
   393→   393→
   394→   394→    function calculate() {
   395→   395→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   396→   396→      if (diff === 0) {
   397→   397→        setExpired(true);
   398→   398→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   399→   399→        return true;
   400→   400→      }
   401→   401→      setExpired(false);
   402→   402→      setTimeLeft({
   403→   403→        d: Math.floor(diff / 86400000),
   404→   404→        h: Math.floor((diff % 86400000) / 3600000),
   405→   405→        m: Math.floor((diff % 3600000) / 60000),
   406→   406→        s: Math.floor((diff % 60000) / 1000),
   407→   407→      });
   408→   408→      return false;
   409→   409→    }
   410→   410→
   411→   411→    const done = calculate();
   412→   412→    if (done) return;
   413→   413→
   414→   414→    const id = setInterval(() => {
   415→   415→      const finished = calculate();
   416→   416→      if (finished) clearInterval(id);
   417→   417→    }, 1000);
   418→   418→
   419→   419→    return () => clearInterval(id);
   420→   420→  }, [targetDate]);
   421→   421→
   422→   422→  if (expired) {
   423→   423→    return (
   424→   424→      <motion.div
   425→   425→        className="flex items-center justify-center gap-1.5 py-1"
   426→   426→        animate={{ opacity: [1, 0.5, 1] }}
   427→   427→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   428→   428→      >
   429→   429→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   430→   430→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   431→   431→      </motion.div>
   432→   432→    );
   433→   433→  }
   434→   434→
   435→   435→  if (!timeLeft) return null;
   436→   436→
   437→   437→  const units = [
   438→   438→    { value: timeLeft.d, label: 'D' },
   439→   439→    { value: timeLeft.h, label: 'H' },
   440→   440→    { value: timeLeft.m, label: 'M' },
   441→   441→    { value: timeLeft.s, label: 'S' },
   442→   442→  ];
   443→   443→
   444→   444→  return (
   445→   445→    <div className="flex items-center justify-center gap-1.5">
   446→   446→      {units.map((unit, idx) => (
   447→   447→        <div key={unit.label} className="flex items-center gap-1.5">
   448→   448→          <div className="flex flex-col items-center">
   449→   449→            <motion.div
   450→   450→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   451→   451→              key={`${unit.label}-${unit.value}`}
   452→   452→              initial={{ rotateX: -90, opacity: 0 }}
   453→   453→              animate={{ rotateX: 0, opacity: 1 }}
   454→   454→              transition={{ duration: 0.3, ease: 'easeOut' }}
   455→   455→            >
   456→   456→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums">
   457→   457→                {String(unit.value).padStart(2, '0')}
   458→   458→              </span>
   459→   459→            </motion.div>
   460→   460→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   461→   461→              {unit.label}
   462→   462→            </span>
   463→   463→          </div>
   464→   464→          {idx < units.length - 1 && (
   465→   465→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   466→   466→          )}
   467→   467→        </div>
   468→   468→      ))}
   469→   469→    </div>
   470→   470→  );
   471→   471→}
