   231→/** Helper: determine which side a teamId belongs to in a match */
   232→function getTeamSide(match: ActiveMatch, teamId: string): 'home' | 'away' {
   233→  if (teamId === match.homeTeamId) return 'home';
   234→  return 'away';

   237→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   238→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   239→  let homeScore = 0;
   240→  let awayScore = 0;
   241→  let homeOutPlayers = 0;
   242→  let awayOutPlayers = 0;
   243→  let homeOutPlayerIds: string[] = [];
   244→  let awayOutPlayerIds: string[] = [];
   245→
   246→  for (const evt of events) {
   247→    const side = getTeamSide(match, evt.teamId);
   248→    const points = evt.value;
   249→    if (side === 'home') {
   250→      homeScore += points;
   251→    } else {
   252→      awayScore += points;
   253→    }
   254→
   255→    // Count-based out tracking (backward compat)
   256→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   257→      if (side === 'home') {
   258→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   259→      } else {
   260→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   261→      }
   262→    }
   263→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   264→      if (side === 'home') {
   265→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   266→      } else {
   267→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   268→      }
   269→    }
   270→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   271→    if (evt.eventType === 'self_out') {
   272→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   273→      if (side === 'home') {
   274→        // Home scored from self-out → away defender went out
   275→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   276→        // Home revives one player
   277→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   278→      } else {
   279→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   280→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   281→      }
   282→    }
   283→    if (evt.eventType === 'all_out') {
   284→      if (side === 'home') {
   285→        awayOutPlayers = 0;
   286→      } else {
   287→        homeOutPlayers = 0;
   288→      }
   289→    }
   290→
   291→    // ID-based out tracking
   292→    let details: Record<string, unknown> = {};
   293→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   294→
   295→    if (evt.eventType === 'raid_point') {
   296→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   297→      if (side === 'home') {
   298→        // Home team scored raid → away defenders go out
   299→        for (const tid of touchedIds) {
   300→          if (!awayOutPlayerIds.includes(tid)) {
   301→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   302→          }
   303→        }
   304→        // Home team revives players (first-out-first-in)
   305→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   306→        if (reviveCount > 0) {
   307→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   308→        }
   309→      } else {
   310→        // Away team scored raid → home defenders go out
   311→        for (const tid of touchedIds) {
   312→          if (!homeOutPlayerIds.includes(tid)) {
   313→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   314→          }
   315→        }
   316→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   317→        if (reviveCount > 0) {
   318→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   319→        }
   320→      }
   321→    }
   322→
   323→    if (evt.eventType === 'bonus_point') {
   324→      // One out player from scoring team revives
   325→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   326→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   327→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   328→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   329→      }
   330→    }
   331→
   332→    // Self-out: defender steps off mat during a live raid
   333→    if (evt.eventType === 'self_out') {
   334→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   335→      if (selfOutPlayerId) {
   336→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   337→        if (side === 'home') {
   338→          // Home scored → away defender self-out
   339→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   340→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   341→          }
   342→          // Home revives first-out player
   343→          if (homeOutPlayerIds.length > 0) {
   344→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   345→          }
   346→        } else {
   347→          // Away scored → home defender self-out
   348→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   349→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   350→          }
   351→          if (awayOutPlayerIds.length > 0) {
   352→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   353→          }
   354→        }
   355→      }
   356→    }
   357→
   358→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   359→      // Raider goes out. The tackle is scored by the DEFENDING team.
   360→      // So the raider is from the OPPOSING team.
   361→      const raiderId = (details.raiderId as string) || evt.playerId;
   362→      if (side === 'home' && raiderId) {
   363→        // Home team scored tackle → away raider was caught → away raider goes out
   364→        if (!awayOutPlayerIds.includes(raiderId)) {
   365→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   366→        }
   367→      } else if (side === 'away' && raiderId) {
   368→        // Away team scored tackle → home raider was caught → home raider goes out
   369→        if (!homeOutPlayerIds.includes(raiderId)) {
   370→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   371→        }
   372→      }
   373→    }
   374→
   375→    if (evt.eventType === 'all_out') {
   376→      if (side === 'home') {
   377→        awayOutPlayerIds = [];
   378→      } else {
   379→        homeOutPlayerIds = [];
   380→      }
   381→    }
   382→  }
   383→
   384→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   385→}

   387→/** Helper: determine raid queue from last raid event */
   388→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   389→  const raidEventTypes = ['raid_point', 'tackle_point', 'super_tackle', 'empty_raid', 'bonus_point', 'do_or_die_raid'];
   390→  const lastRaidEvent = [...events].reverse().find(e => raidEventTypes.includes(e.eventType));
   391→  if (lastRaidEvent) {
   392→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   393→  }
   394→  return match.raidQueue;
   395→}

   606→            activeMatch: {
   607→              ...state.activeMatch,
   608→              events,
   609→              ...calculated,
   610→              raidQueue,
   611→            },
   612→          };
   613→        }),
   614→
   615→      undoLastRaid: () =>
   616→        set((state) => {
   617→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   618→
   619→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   620→          // A raid batch has events with the same second in timestamp
   621→          const events = [...state.activeMatch.events];
   622→          if (events.length === 0) return {};
   623→
   624→          // Find the timestamp of the last event
   625→          const lastTimestamp = events[events.length - 1].timestamp;
   626→          // Remove all events that have the same timestamp (within 5ms for batch events)
   627→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   628→            events.pop();
   629→          }
   630→
   631→          const calculated = recalculateFromEvents(state.activeMatch, events);
   632→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   633→
   634→          return {
   635→            activeMatch: {
   636→              ...state.activeMatch,
   637→              events,
   638→              ...calculated,
   639→              raidQueue,
   640→            },
   641→          };
   642→        }),
   643→
   644→      switchHalf: () =>
   645→        set((state) => {
   646→          if (!state.activeMatch) return {};
   647→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   648→          return {
   649→            activeMatch: {
   650→              ...state.activeMatch,
   651→              currentHalf: newHalf,
   652→              timer: state.activeMatch.halfDuration * 60,
   653→              isDoOrDie: false,
   654→            },
   655→          };
   656→        }),
   657→
   658→      setTimer: (time) =>
   659→        set((state) => {
   660→          if (!state.activeMatch) return {};
   661→          return { activeMatch: { ...state.activeMatch, timer: time } };
   662→        }),
   663→
   664→      setDoOrDie: (value) =>
   665→        set((state) => {
   666→          if (!state.activeMatch) return {};
   667→          return { activeMatch: { ...state.activeMatch, isDoOrDie: value } };
   668→        }),
   669→
   670→      switchRaidQueue: () =>
   671→        set((state) => {
   672→          if (!state.activeMatch) return {};
   673→          return {
   674→            activeMatch: {
   675→              ...state.activeMatch,
   676→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   677→            },
   678→          };
   679→        }),
   680→
   681→      callTimeout: (team) =>
   682→        set((state) => {
   683→          if (!state.activeMatch) return {};
   684→          return {
   685→            activeMatch: {
   686→              ...state.activeMatch,
   687→              homeTimeouts:
   688→                team === 'home'
   689→                  ? state.activeMatch.homeTimeouts + 1
   690→                  : state.activeMatch.homeTimeouts,
   691→              awayTimeouts:
   692→                team === 'away'
   693→                  ? state.activeMatch.awayTimeouts + 1
   694→                  : state.activeMatch.awayTimeouts,
   695→            },
   696→          };
   697→        }),
   698→
   699→      setOutPlayers: (team, count) =>
   700→        set((state) => {
   701→          if (!state.activeMatch) return {};
   702→          return {
   703→            activeMatch: {
   704→              ...state.activeMatch,
   705→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   706→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   707→            },
   708→          };
   709→        }),
   710→
   711→      fetchHomeData: async () => {
   712→        try {
   713→          const res = await fetch('/api/stats');
   714→          if (!res.ok) throw new Error('Failed to fetch stats');
   715→
   716→          const data = await res.json();
   717→
   718→          // Cache in store
   719→          set({ homeData: data });
   720→          return data;
   721→        } catch (err) {
   722→          console.error('fetchHomeData error:', err);
   723→          return null;
   724→        }
   725→      },
   726→
   727→      // Notification actions
   728→      addNotification: (notification) =>
   729→        set((state) => ({
   730→          notifications: [
   731→            {
   732→              ...notification,
   733→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   734→              timestamp: Date.now(),
   735→              read: false,
   736→            },
   737→            ...state.notifications,
   738→          ].slice(0, 50), // Keep max 50 notifications
   739→        })),
   740→
   741→      markNotificationRead: (id) =>
   742→        set((state) => ({
   743→          notifications: state.notifications.map((n) =>
   744→            n.id === id ? { ...n, read: true } : n
   745→          ),
   746→        })),
   747→
   748→      markAllRead: () =>
   749→        set((state) => ({
   750→          notifications: state.notifications.map((n) => ({ ...n, read: true })),
   751→        })),
   752→
   753→      clearNotifications: () => set({ notifications: [] }),
   754→
   755→      setLanguage: (lang) => set({ language: lang }),
   756→
   757→      // Onboarding actions
   758→      setHasCompletedOnboarding: (value) => set({ hasCompletedOnboarding: value }),
   759→
   760→      setOnboardingProfile: (profile) =>
   761→        set((state) => ({
   762→          onboardingProfile: { ...state.onboardingProfile, ...profile },
   763→        })),
   764→
   765→      completeOnboarding: () =>
   766→        set((state) => ({
   767→          hasCompletedOnboarding: true,
   768→          currentUser: state.currentUser
   769→            ? {
   770→                ...state.currentUser,
   771→                ...(state.onboardingProfile.position && { role: state.onboardingProfile.position }),
   772→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   773→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   774→                createdAt: state.currentUser.createdAt || Date.now(),
   775→              }
   776→            : null,
   777→        })),
   778→
   779→      // Team management actions
   780→      setTeamFilter: (filter) =>
   781→        set((state) => ({
   782→          teamManagement: { ...state.teamManagement, teamFilter: filter },
   783→        })),
   784→
   785→      setTeamSearch: (search) =>
   786→        set((state) => ({
   787→          teamManagement: { ...state.teamManagement, teamSearch: search },
   788→        })),
   789→
   790→      setSelectedTeamId: (teamId) =>
   791→        set((state) => ({
   792→          teamManagement: { ...state.teamManagement, selectedTeamId: teamId },
   793→        })),
   794→
   795→      setTeamDetailOpen: (open) =>
   796→        set((state) => ({
   797→          teamManagement: { ...state.teamManagement, teamDetailOpen: open },
   798→        })),
   799→
   800→      // Match transfer / handoff — load a match from a transfer
   801→      loadMatch: (match) =>
   802→        set({
   803→          activeMatch: match,
   804→          activeTab: 'quick-score' as TabId,
   805→          showToss: false,
   806→          tossMatchConfig: null,
   807→        }),
   808→    }),
   809→    {
   810→      name: 'kabaddi-pro-storage',
   811→      storage: createJSONStorage(() => localStorage),
   812→      // Only persist auth-related fields and notifications
   813→      partialize: (state) => ({
   814→        isAuthenticated: state.isAuthenticated,
   815→        isOnboarded: state.isOnboarded,
   816→        currentUser: state.currentUser,
   817→        hasSeenSplash: state.hasSeenSplash,
   818→        notifications: state.notifications,
   819→        language: state.language,
   820→        hasCompletedOnboarding: state.hasCompletedOnboarding,
   821→        onboardingProfile: state.onboardingProfile,
   822→        coachAcademies: state.coachAcademies,
   823→      }),
   824→    }
   825→  )
   826→);

   829→  switch (eventType) {
