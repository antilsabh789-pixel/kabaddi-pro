    54→/* ── Floating particles for background ── */
    55→function FloatingParticles() {
    56→  const particles = useMemo(
    57→    () =>
    58→      Array.from({ length: 18 }, (_, i) => ({
    59→        id: i,
    60→        left: `${Math.random() * 100}%`,
    61→        top: `${Math.random() * 100}%`,
    62→        size: 2 + Math.random() * 4,
    63→        delay: Math.random() * 5,
    64→        duration: 6 + Math.random() * 8,
    65→        opacity: 0.15 + Math.random() * 0.25,
    66→      })),
    67→    [],
    68→  );
    69→
    70→  return (
    71→    <>
    72→      {particles.map((p) => (
    73→        <motion.div
    74→          key={p.id}
    75→          className="absolute rounded-full bg-brand-red/60 dark:bg-brand-red-light/40"
    76→          style={{
    77→            left: p.left,
    78→            top: p.top,
    79→            width: p.size,
    80→            height: p.size,
    81→            opacity: p.opacity,
    82→          }}
    83→          animate={{
    84→            y: [0, -20, 0],
    85→            x: [0, Math.random() > 0.5 ? 8 : -8, 0],
    86→            opacity: [p.opacity, p.opacity * 0.5, p.opacity],
    87→          }}
    88→          transition={{
    89→            duration: p.duration,
    90→            delay: p.delay,
    91→            repeat: Infinity,
    92→            ease: 'easeInOut',
    93→          }}
    94→        />
    95→      ))}
    96→    </>
    97→  );
    98→}
