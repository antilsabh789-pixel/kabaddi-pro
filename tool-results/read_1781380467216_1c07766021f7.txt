   137→// ─── Animated Counter Hook ────────────────────────────────────────
   138→
   139→function useAnimatedCounter(target: number, duration = 1200, delay = 0) {
   140→  const [count, setCount] = useState(0);
   141→  const startTime = useRef<number | null>(null);
   142→  const rafRef = useRef<number | null>(null);
   143→
   144→  useEffect(() => {
   145→    const timeout = setTimeout(() => {
   146→      startTime.current = performance.now();
   147→
   148→      const animate = (now: number) => {
   149→        if (!startTime.current) return;
   150→        const elapsed = now - startTime.current;
   151→        const progress = Math.min(elapsed / duration, 1);
   152→        // easeOutExpo
   153→        const eased = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   154→        setCount(Math.round(eased * target));
   155→
   156→        if (progress < 1) {
   157→          rafRef.current = requestAnimationFrame(animate);
   158→        }
   159→      };
   160→
   161→      rafRef.current = requestAnimationFrame(animate);
   162→    }, delay);
   163→
   164→    return () => {
   165→      clearTimeout(timeout);
   166→      if (rafRef.current) cancelAnimationFrame(rafRef.current);
   167→    };
   168→  }, [target, duration, delay]);
   169→
   170→  return count;
   171→}

   173→// ─── Circular Progress Ring ───────────────────────────────────────
   174→
   175→function CircularProgressRing({
   176→  value,
   177→  maxValue = 100,
   178→  size = 180,
   179→  strokeWidth = 12,
   180→}: {
   181→  value: number;
   182→  maxValue?: number;
   183→  size?: number;
   184→  strokeWidth?: number;
   185→}) {
   186→  const { theme } = useTheme();
   187→  const isDark = theme === 'dark';
   188→  const radius = (size - strokeWidth) / 2;
   189→  const circumference = 2 * Math.PI * radius;
   190→  const percentage = Math.min(value / maxValue, 1);
   191→  const offset = circumference - percentage * circumference;
   192→  const animatedValue = useAnimatedCounter(Math.round(value), 1500, 400);
   193→
   194→  // Color based on rating
   195→  const getColor = (val: number) => {
   196→    if (val < 40) return { ring: '#EF4444', bg: 'from-red-500/10 to-red-600/5', text: 'text-red-500' };
   197→    if (val <= 70) return { ring: '#EAB308', bg: 'from-yellow-500/10 to-yellow-600/5', text: 'text-yellow-500' };
   198→    return { ring: '#22C55E', bg: 'from-green-500/10 to-green-600/5', text: 'text-green-500' };
   199→  };
   200→
   201→  const colorInfo = getColor(value);
   202→
   203→  return (
   204→    <div className="flex flex-col items-center gap-2">
   205→      <div className="relative" style={{ width: size, height: size }}>
   206→        <svg width={size} height={size} className="-rotate-90">
   207→          {/* Background circle */}
   208→          <circle
   209→            cx={size / 2}
   210→            cy={size / 2}
   211→            r={radius}
   212→            fill="none"
   213→            stroke={isDark ? '#3B3428' : '#F0EBE1'}
   214→            strokeWidth={strokeWidth}
   215→          />
   216→          {/* Glow effect */}
   217→          <motion.circle
   218→            cx={size / 2}
   219→            cy={size / 2}
   220→            r={radius}
   221→            fill="none"
   222→            stroke={colorInfo.ring}
   223→            strokeWidth={strokeWidth + 6}
   224→            strokeLinecap="round"
   225→            strokeDasharray={circumference}
   226→            initial={{ strokeDashoffset: circumference, opacity: 0.15 }}
   227→            animate={{ strokeDashoffset: offset, opacity: 0.15 }}
   228→            transition={{ duration: 1.5, ease: 'easeOut', delay: 0.3 }}
   229→            style={{ filter: 'blur(6px)' }}
   230→          />
   231→          {/* Progress circle */}
   232→          <motion.circle
   233→            cx={size / 2}
   234→            cy={size / 2}
   235→            r={radius}
   236→            fill="none"
   237→            stroke={colorInfo.ring}
   238→            strokeWidth={strokeWidth}
   239→            strokeLinecap="round"
   240→            strokeDasharray={circumference}
   241→            initial={{ strokeDashoffset: circumference }}
   242→            animate={{ strokeDashoffset: offset }}
   243→            transition={{ duration: 1.5, ease: 'easeOut', delay: 0.3 }}
   244→          />
   245→        </svg>
   246→        {/* Center content */}
   247→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   248→          <motion.span
   249→            className="text-4xl font-black"
   250→            style={{ color: colorInfo.ring }}
   251→            initial={{ opacity: 0, scale: 0.5 }}
   252→            animate={{ opacity: 1, scale: 1 }}
   253→            transition={{ delay: 0.8, type: 'spring', damping: 12 }}
   254→          >
   255→            {animatedValue}
   256→          </motion.span>
   257→          <span className="text-[10px] font-semibold text-warm-500 dark:text-warm-400 uppercase tracking-wider mt-0.5">
   258→            Overall
   259→          </span>
   260→        </div>
   261→      </div>
   262→    </div>
   263→  );
   264→}

   317→// ─── Performance Bar Component ────────────────────────────────────
   318→
   319→function PerformanceBar({
   320→  label,
   321→  value,
   322→  maxValue,
   323→  color,
   324→  delay = 0,
   325→}: {
   326→  label: string;
   327→  value: number;
   328→  maxValue: number;
   329→  color: string;
   330→  delay?: number;
   331→}) {
   332→  const percentage = maxValue > 0 ? Math.round((value / maxValue) * 100) : 0;
   333→  const animatedWidth = useAnimatedCounter(percentage, 1000, 300 + delay * 100);
   334→
   335→  return (
   336→    <motion.div
   337→      initial={{ opacity: 0, x: -20 }}
   338→      animate={{ opacity: 1, x: 0 }}
   339→      transition={{ delay: 0.3 + delay * 0.15, duration: 0.4 }}
   340→      className="space-y-1.5"
   341→    >
   342→      <div className="flex items-center justify-between">
   343→        <span className="text-xs font-semibold text-warm-700 dark:text-warm-300">{label}</span>
   344→        <span className="text-xs font-bold text-warm-800 dark:text-warm-200">
   345→          {value} <span className="text-warm-400 font-normal text-[10px]">({animatedWidth}%)</span>
   346→        </span>
   347→      </div>
   348→      <div className="h-3 rounded-full bg-warm-200 dark:bg-warm-700 overflow-hidden">
   349→        <motion.div
   350→          className={`h-full rounded-full bg-gradient-to-r ${color}`}
   351→          initial={{ width: 0 }}
   352→          animate={{ width: `${percentage}%` }}
   353→          transition={{ duration: 1, ease: 'easeOut', delay: 0.5 + delay * 0.15 }}
   354→        />
   355→      </div>
   356→    </motion.div>
   357→  );
   358→}

   407→export default function PlayerStatsScreen({ userId, onClose }: PlayerStatsScreenProps) {
   408→  const { theme } = useTheme();
   409→  const isDark = theme === 'dark';
   410→  const currentUser = useKabaddiStore((s) => s.currentUser);
   411→
   412→  const [player, setPlayer] = useState<PlayerData | null>(null);
   413→  const [profile, setProfile] = useState<ProfileData | null>(null);
   414→  const [loading, setLoading] = useState(true);
   415→  const [error, setError] = useState<string | null>(null);
   416→  const [matchType, setMatchType] = useState<'tournament' | 'practice'>('tournament');
   417→  const [recentForm, setRecentForm] = useState<RecentMatchResult[]>([]);
   418→  const [positionRank, setPositionRank] = useState<{ rank: number; total: number } | null>(null);
   419→
   420→  const isPremium = currentUser?.isPremium || false;
   421→
   422→  // Fetch player data
   423→  const fetchPlayerData = useCallback(async () => {
   424→    setLoading(true);
   425→    setError(null);
   426→    try {
   427→      const res = await fetch(`/api/players/${userId}`);
   428→      if (!res.ok) throw new Error('Failed to fetch player');
   429→      const data = await res.json();
   430→      setPlayer(data.player);
   431→      setProfile(data.profile || null);
   432→    } catch (err) {
   433→      console.error('Fetch error:', err);
   434→      setError('Failed to load player stats');
   435→    } finally {
   436→      setLoading(false);
   437→    }
   438→  }, [userId]);
   439→
   440→  // Fetch recent form and rank
   441→  const fetchAdditionalData = useCallback(async () => {
   442→    try {
   443→      // Fetch recent match results for this player
   444→      const matchRes = await fetch(`/api/matches?userId=${userId}&limit=5`);
   445→      if (matchRes.ok) {
   446→        const matchData = await matchRes.json();
   447→        if (matchData.matches && Array.isArray(matchData.matches)) {
   448→          const formResults: RecentMatchResult[] = matchData.matches
   449→            .slice(0, 5)
   450→            .map((m: Record<string, unknown>) => {
   451→              const homeScore = m.homeScore as number;
   452→              const awayScore = m.awayScore as number;
   453→              const homeTeam = (m.homeTeam as Record<string, string>)?.name || 'Home';
   454→              const awayTeam = (m.awayTeam as Record<string, string>)?.name || 'Away';
   455→              const completedAt = (m.completedAt as string) || '';
   456→
   457→              // Determine user's team
   458→              const isHome = Array.isArray(m.scorers) && m.scorers.some((s: Record<string, unknown>) => s.userId === userId && s.teamId === m.homeTeamId);
   459→              const userScore = isHome ? homeScore : awayScore;
   460→              const oppScore = isHome ? awayScore : homeScore;
   461→              const opponent = isHome ? awayTeam : homeTeam;
   462→
   463→              let result: 'W' | 'L' | 'D' = 'D';
   464→              if (userScore > oppScore) result = 'W';
   465→              else if (userScore < oppScore) result = 'L';
   466→
   467→              return {
   468→                id: m.id as string,
   469→                opponent,
   470→                result,
   471→                score: `${homeScore}-${awayScore}`,
   472→                date: completedAt ? new Date(completedAt).toLocaleDateString('en-IN', { day: 'numeric', month: 'short' }) : '',
   473→              };
   474→            });
   475→          setRecentForm(formResults);
   476→        }
   477→      }
   478→
   479→      // Fetch position rank
   480→      if (profile?.position) {
   481→        const rankRes = await fetch(`/api/player-stats?leaderboard=true&position=${profile.position}&limit=1000`);
   482→        if (rankRes.ok) {
   483→          const rankData = await rankRes.json();
   484→          if (rankData.leaderboard && Array.isArray(rankData.leaderboard)) {
   485→            const idx = rankData.leaderboard.findIndex((p: Record<string, unknown>) => p.userId === userId);
   486→            if (idx >= 0) {
   487→              setPositionRank({ rank: idx + 1, total: rankData.leaderboard.length });
   488→            }
   489→          }
   490→        }
   491→      }
   492→    } catch {
   493→      // Non-critical, just don't show
   494→    }
   495→  }, [userId, profile?.position]);
   496→
   497→  useEffect(() => {
   498→    fetchPlayerData();
   499→  }, [fetchPlayerData]);
   500→
   501→  useEffect(() => {
   502→    if (profile) {
   503→      fetchAdditionalData();
   504→    }
   505→  }, [profile, fetchAdditionalData]);
   506→
   507→  // Derived stats based on match type
   508→  const getStats = useCallback(() => {
   509→    if (!profile) {
   510→      return {
   511→        raidPoints: 0,
   512→        tacklePoints: 0,
   513→        totalMatches: 0,
   514→        successRate: 0,
   515→        superTackles: 0,
   516→        bonusPoints: 0,
   517→        totalRaids: 0,
   518→        successfulRaids: 0,
   519→        totalTackles: 0,
   520→        successfulTackles: 0,
   521→        totalPoints: 0,
   522→        raidTrend: null as 'up' | 'down' | 'neutral' | null,
   523→        tackleTrend: null as 'up' | 'down' | 'neutral' | null,
   524→      };
   525→    }
   526→
   527→    if (matchType === 'tournament') {
   528→      const successRate = profile.tournamentTotalRaids > 0
   529→        ? Math.round((profile.tournamentSuccessfulRaids / profile.tournamentTotalRaids) * 100)
   530→        : 0;
   531→      return {
   532→        raidPoints: profile.tournamentRaidPoints,
   533→        tacklePoints: profile.tournamentTacklePoints,
   534→        totalMatches: profile.tournamentMatches,
   535→        successRate,
   536→        superTackles: profile.tournamentSuperTackles,
   537→        bonusPoints: profile.tournamentBonusPoints,
   538→        totalRaids: profile.tournamentTotalRaids,
   539→        successfulRaids: profile.tournamentSuccessfulRaids,
   540→        totalTackles: profile.tournamentTotalTackles,
   541→        successfulTackles: profile.tournamentSuccessfulTackles,
   542→        totalPoints: profile.tournamentTotalPoints,
   543→        raidTrend: profile.tournamentRaidPoints > 0 ? 'up' as const : null,
   544→        tackleTrend: profile.tournamentTacklePoints > 0 ? 'up' as const : null,
   545→      };
   546→    }
   547→
   548→    const successRate = profile.practiceTotalRaids > 0
   549→      ? Math.round((profile.practiceSuccessfulRaids / profile.practiceTotalRaids) * 100)
   550→      : 0;
   551→    return {
   552→      raidPoints: profile.practiceRaidPoints,
   553→      tacklePoints: profile.practiceTacklePoints,
   554→      totalMatches: profile.practiceMatches,
   555→      successRate,
   556→      superTackles: profile.practiceSuperTackles,
   557→      bonusPoints: profile.practiceBonusPoints,
   558→      totalRaids: profile.practiceTotalRaids,
   559→      successfulRaids: profile.practiceSuccessfulRaids,
   560→      totalTackles: profile.practiceTotalTackles,
   561→      successfulTackles: profile.practiceSuccessfulTackles,
   562→      totalPoints: profile.practiceTotalPoints,
   563→      raidTrend: profile.practiceRaidPoints > 0 ? 'up' as const : null,
   564→      tackleTrend: profile.practiceTacklePoints > 0 ? 'up' as const : null,
