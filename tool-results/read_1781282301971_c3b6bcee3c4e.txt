    94→    94→function getInitials(name: string | null | undefined): string {
    95→    95→  if (!name) return '?';
    96→    96→  const parts = name.trim().split(/\s+/);
    97→    97→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
    98→    98→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
    99→    99→}
   100→   100→
   101→   101→function getDisplayName(name: string | null | undefined): string {
   102→   102→  return name?.trim() || 'Unnamed Player';
