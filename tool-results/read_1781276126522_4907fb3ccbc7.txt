   240→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   241→  const [count, setCount] = useState(0);
   242→  const prevTarget = useRef(target);
   243→
   244→  useEffect(() => {
   245→    if (prevTarget.current === target && count !== 0) return;
   246→    prevTarget.current = target;
   247→
   248→    let start = 0;
   249→    const startTime = performance.now();
   250→
   251→    function step(currentTime: number) {
   252→      const elapsed = currentTime - startTime;
   253→      const progress = Math.min(elapsed / duration, 1);
   254→      // ease-out cubic
   255→      const eased = 1 - Math.pow(1 - progress, 3);
   256→      const current = Math.round(eased * target);
   257→      setCount(current);
   258→      if (progress < 1) {
   259→        requestAnimationFrame(step);
   260→      }
   261→    }
   262→
   263→    requestAnimationFrame(step);
   264→  }, [target, duration]);
   265→
   266→  return <span>{count}</span>;
   267→}

   321→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   322→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   323→  const [expired, setExpired] = useState(!targetDate);
   324→
   325→  useEffect(() => {
   326→    if (!targetDate) {
   327→      return;
   328→    }
   329→
   330→    function calculate() {
   331→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   332→      if (diff === 0) {
   333→        setExpired(true);
   334→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   335→        return true;
   336→      }
   337→      setExpired(false);
   338→      setTimeLeft({
   339→        d: Math.floor(diff / 86400000),
   340→        h: Math.floor((diff % 86400000) / 3600000),
   341→        m: Math.floor((diff % 3600000) / 60000),
   342→        s: Math.floor((diff % 60000) / 1000),
   343→      });
   344→      return false;
   345→    }
   346→
   347→    const done = calculate();
   348→    if (done) return;
   349→
   350→    const id = setInterval(() => {
   351→      const finished = calculate();
   352→      if (finished) clearInterval(id);
   353→    }, 1000);
   354→
   355→    return () => clearInterval(id);
   356→  }, [targetDate]);
   357→
   358→  if (expired) {
   359→    return (
   360→      <motion.div
   361→        className="flex items-center justify-center gap-1.5 py-1"
   362→        animate={{ opacity: [1, 0.5, 1] }}
   363→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   364→      >
   365→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   366→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   367→      </motion.div>
   368→    );
   369→  }
   370→
   371→  if (!timeLeft) return null;
   372→
   373→  const units = [
   374→    { value: timeLeft.d, label: 'D' },
   375→    { value: timeLeft.h, label: 'H' },
   376→    { value: timeLeft.m, label: 'M' },
   377→    { value: timeLeft.s, label: 'S' },
   378→  ];
   379→
   380→  return (
   381→    <div className="flex items-center justify-center gap-1.5">
   382→      {units.map((unit, idx) => (
   383→        <div key={unit.label} className="flex items-center gap-1.5">
   384→          <div className="flex flex-col items-center">
   385→            <motion.div
   386→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   387→              key={`${unit.label}-${unit.value}`}
   388→              initial={{ rotateX: -90, opacity: 0 }}
   389→              animate={{ rotateX: 0, opacity: 1 }}
   390→              transition={{ duration: 0.3, ease: 'easeOut' }}
   391→            >
   392→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums">
   393→                {String(unit.value).padStart(2, '0')}
   394→              </span>
   395→            </motion.div>
   396→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   397→              {unit.label}
   398→            </span>
   399→          </div>
   400→          {idx < units.length - 1 && (
   401→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   402→          )}
   403→        </div>
   404→      ))}
   405→    </div>
   406→  );
   407→}
