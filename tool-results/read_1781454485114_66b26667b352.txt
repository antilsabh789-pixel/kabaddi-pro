     1→'use client';
     2→
     3→import { useState, useEffect, useCallback, useRef } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import {
     6→  X, Share2, Download, Copy, Check, Shield, Clock,
     7→  Smartphone, AlertTriangle, RefreshCw, QrCode,
     8→  ArrowRightLeft, Users, Zap,
     9→} from 'lucide-react';
    10→import { Card, CardContent } from '@/components/ui/card';
    11→import { Button } from '@/components/ui/button';
    12→import { Input } from '@/components/ui/input';
    13→import { Badge } from '@/components/ui/badge';
    14→import { useKabaddiStore, type ActiveMatch } from '@/lib/store';
    15→import { useToast } from '@/hooks/use-toast';
    16→
    17→type TransferTab = 'share' | 'receive';
    18→
    19→interface ScorerTransferScreenProps {
    20→  onClose: () => void;
    21→  /** If provided, we're sharing an active match. If not, we're in receive mode. */
    22→  activeMatch?: ActiveMatch | null;
    23→}
    24→
    25→export default function ScorerTransferScreen({ onClose, activeMatch }: ScorerTransferScreenProps) {
    26→  const currentUser = useKabaddiStore((s) => s.currentUser);
    27→  const loadMatch = useKabaddiStore((s) => s.loadMatch);
    28→  const { toast } = useToast();
    29→
    30→  // Default to 'share' if we have an active match, otherwise 'receive'
    31→  const [tab, setTab] = useState<TransferTab>(activeMatch ? 'share' : 'receive');
    32→
    33→  // Share state
    34→  const [transferCode, setTransferCode] = useState<string | null>(null);
    35→  const [codeExpiry, setCodeExpiry] = useState<Date | null>(null);
    36→  const [isGenerating, setIsGenerating] = useState(false);
    37→  const [copied, setCopied] = useState(false);
    38→  const [codeTimer, setCodeTimer] = useState<string>('');
    39→
    40→  // Receive state
    41→  const [inputCode, setInputCode] = useState('');
    42→  const [isValidating, setIsValidating] = useState(false);
    43→  const [receiveError, setReceiveError] = useState<string | null>(null);
    44→  const [previewMatch, setPreviewMatch] = useState<ActiveMatch | null>(null);
    45→  const [previewScorer, setPreviewScorer] = useState<string | null>(null);
    46→  const [isClaiming, setIsClaiming] = useState(false);
    47→
    48→  // Timer for code expiry
    49→  const timerRef = useRef<ReturnType<typeof setInterval> | null>(null);
    50→
    51→  useEffect(() => {
    52→    if (transferCode && codeExpiry) {
    53→      const updateTimer = () => {
    54→        const now = Date.now();
    55→        const expiry = new Date(codeExpiry).getTime();
    56→        const remaining = expiry - now;
    57→        if (remaining <= 0) {
    58→          setCodeTimer('Expired');
    59→          if (timerRef.current) clearInterval(timerRef.current);
    60→          return;
    61→        }
    62→        const mins = Math.floor(remaining / 60000);
    63→        const secs = Math.floor((remaining % 60000) / 1000);
    64→        setCodeTimer(`${mins}:${secs.toString().padStart(2, '0')}`);
    65→      };
    66→      updateTimer();
    67→      timerRef.current = setInterval(updateTimer, 1000);
    68→      return () => {
    69→        if (timerRef.current) clearInterval(timerRef.current);
    70→      };
    71→    }
    72→  }, [transferCode, codeExpiry]);
    73→
    74→  const handleGenerateCode = useCallback(async () => {
    75→    if (!activeMatch) return;
    76→    setIsGenerating(true);
    77→    try {
    78→      const res = await fetch('/api/match-transfer', {
    79→        method: 'POST',
    80→        headers: { 'Content-Type': 'application/json' },
    81→        body: JSON.stringify({
    82→          matchState: activeMatch,
    83→          scorerUserId: currentUser?.id,
    84→          scorerName: currentUser?.name,
    85→          matchId: activeMatch.id,
    86→        }),
    87→      });
    88→      const data = await res.json();
    89→      if (!res.ok) {
    90→        toast({ title: 'Error', description: data.error || 'Failed to generate code', variant: 'destructive' });
    91→        return;
    92→      }
    93→      setTransferCode(data.transferCode);
    94→      setCodeExpiry(data.expiresAt);
    95→    } catch (err) {
    96→      console.error('Generate code error:', err);
    97→      toast({ title: 'Error', description: 'Failed to generate transfer code', variant: 'destructive' });
    98→    } finally {
    99→      setIsGenerating(false);
   100→    }
   101→  }, [activeMatch, currentUser, toast]);
   102→
   103→  const handleCopyCode = useCallback(() => {
   104→    if (!transferCode) return;
   105→    navigator.clipboard.writeText(transferCode).then(() => {
   106→      setCopied(true);
   107→      toast({ title: 'Copied!', description: 'Transfer code copied to clipboard' });
   108→      setTimeout(() => setCopied(false), 2000);
   109→    }).catch(() => {
   110→      // Fallback
   111→      const textArea = document.createElement('textarea');
   112→      textArea.value = transferCode;
   113→      document.body.appendChild(textArea);
   114→      textArea.select();
   115→      document.execCommand('copy');
   116→      document.body.removeChild(textArea);
   117→      setCopied(true);
   118→      setTimeout(() => setCopied(false), 2000);
   119→    });
   120→  }, [transferCode, toast]);
   121→
   122→  const handleValidateCode = useCallback(async () => {
   123→    if (!inputCode.trim()) return;
   124→    setIsValidating(true);
   125→    setReceiveError(null);
   126→    setPreviewMatch(null);
   127→    setPreviewScorer(null);
   128→
   129→    try {
   130→      const res = await fetch(`/api/match-transfer?code=${inputCode.trim().toUpperCase()}`);
   131→      const data = await res.json();
   132→
   133→      if (!res.ok) {
   134→        setReceiveError(data.error || 'Invalid code');
   135→        return;
   136→      }
   137→
   138→      setPreviewMatch(data.matchState);
   139→      setPreviewScorer(data.scorerName);
   140→    } catch (err) {
   141→      console.error('Validate code error:', err);
   142→      setReceiveError('Failed to validate code. Please try again.');
   143→    } finally {
   144→      setIsValidating(false);
   145→    }
   146→  }, [inputCode]);
   147→
   148→  const handleClaimTransfer = useCallback(async () => {
   149→    if (!previewMatch) return;
   150→    setIsClaiming(true);
   151→
   152→    try {
   153→      const res = await fetch('/api/match-transfer', {
   154→        method: 'PATCH',
   155→        headers: { 'Content-Type': 'application/json' },
   156→        body: JSON.stringify({
   157→          code: inputCode.trim().toUpperCase(),
   158→          receiverUserId: currentUser?.id,
   159→        }),
   160→      });
   161→      const data = await res.json();
   162→
   163→      if (!res.ok) {
   164→        setReceiveError(data.error || 'Failed to claim transfer');
   165→        return;
   166→      }
   167→
   168→      // Load the match into the store
   169→      loadMatch(previewMatch);
   170→
   171→      toast({
   172→        title: 'Match Transferred! 🎉',
   173→        description: `You can now continue scoring ${previewMatch.homeTeam} vs ${previewMatch.awayTeam}`,
   174→      });
   175→
   176→      onClose();
   177→    } catch (err) {
   178→      console.error('Claim transfer error:', err);
   179→      setReceiveError('Failed to claim transfer. Please try again.');
   180→    } finally {
   181→      setIsClaiming(false);
   182→    }
   183→  }, [previewMatch, inputCode, currentUser, loadMatch, toast, onClose]);
   184→
   185→  return (
   186→    <motion.div
   187→      initial={{ opacity: 0 }}
   188→      animate={{ opacity: 1 }}
   189→      exit={{ opacity: 0 }}
   190→      className="fixed inset-0 z-50 bg-warm-50 dark:bg-warm-800 flex flex-col"
   191→    >
   192→      {/* Header */}
   193→      <header className="sticky top-0 z-10 bg-warm-50/90 dark:bg-warm-800/90 backdrop-blur-md border-b border-warm-200/60 dark:border-warm-600/30">
   194→        <div className="px-4 py-3 flex items-center justify-between">
   195→          <div className="flex items-center gap-2">
   196→            <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-teal-500 to-teal-600 flex items-center justify-center">
   197→              <ArrowRightLeft className="w-4 h-4 text-white" />
   198→            </div>
   199→            <div>
   200→              <h1 className="text-base font-black tracking-wider text-warm-800 dark:text-warm-100">
   201→                HANDOFF SCORER
   202→              </h1>
   203→              <p className="text-[9px] text-warm-400 dark:text-warm-500 font-medium">
   204→                Transfer live scoring to another device
   205→              </p>
   206→            </div>
   207→          </div>
   208→          <button
   209→            onClick={onClose}
   210→            className="w-8 h-8 rounded-full bg-warm-200 dark:bg-warm-600 flex items-center justify-center text-warm-600 dark:text-warm-300 hover:bg-warm-300 dark:hover:bg-warm-500 transition-colors"
   211→          >
   212→            <X className="w-4 h-4" />
   213→          </button>
   214→        </div>
   215→
   216→        {/* Tab Toggle */}
   217→        <div className="flex gap-2 px-4 pb-3">
   218→          <button
   219→            onClick={() => setTab('share')}
   220→            className={`flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-semibold transition-all ${
   221→              tab === 'share'
   222→                ? 'bg-teal-600 text-white shadow-sm'
   223→                : 'bg-warm-100 dark:bg-warm-700 text-warm-600 dark:text-warm-300 hover:bg-warm-200 dark:hover:bg-warm-600'
   224→            }`}
   225→          >
   226→            <Share2 className="w-3 h-3" />
   227→            Share Match
   228→          </button>
   229→          <button
   230→            onClick={() => setTab('receive')}
   231→            className={`flex items-center gap-1.5 px-3 py-1.5 rounded-full text-xs font-semibold transition-all ${
   232→              tab === 'receive'
   233→                ? 'bg-teal-600 text-white shadow-sm'
   234→                : 'bg-warm-100 dark:bg-warm-700 text-warm-600 dark:text-warm-300 hover:bg-warm-200 dark:hover:bg-warm-600'
   235→            }`}
   236→          >
   237→            <Download className="w-3 h-3" />
   238→            Receive Match
   239→          </button>
   240→        </div>
   241→      </header>
   242→
   243→      {/* Content */}
   244→      <div className="flex-1 overflow-y-auto px-4 py-4 custom-scrollbar">
   245→        <AnimatePresence mode="wait">
   246→          {tab === 'share' ? (
   247→            <motion.div
   248→              key="share"
   249→              initial={{ opacity: 0, x: -20 }}
   250→              animate={{ opacity: 1, x: 0 }}
   251→              exit={{ opacity: 0, x: 20 }}
   252→              className="flex flex-col gap-4"
   253→            >
   254→              {/* Info Card */}
   255→              <Card className="bg-gradient-to-br from-teal-50/80 to-warm-50 dark:from-teal-900/20 dark:to-warm-800 border-teal-200/50 dark:border-teal-700/30">
   256→                <CardContent className="p-4">
   257→                  <div className="flex items-start gap-3">
   258→                    <div className="w-10 h-10 rounded-xl bg-teal-100 dark:bg-teal-800/40 flex items-center justify-center shrink-0">
   259→                      <Smartphone className="w-5 h-5 text-teal-600 dark:text-teal-400" />
   260→                    </div>
   261→                    <div>
   262→                      <h3 className="text-sm font-bold text-warm-800 dark:text-warm-100">How it works</h3>
   263→                      <p className="text-xs text-warm-500 dark:text-warm-400 mt-1 leading-relaxed">
   264→                        Generate a transfer code and share it with the new scorer. They enter the code on their phone to continue scoring from exactly where you left off. Works for both tournament and practice matches.
   265→                      </p>
   266→                    </div>
   267→                  </div>
   268→                  <div className="grid grid-cols-3 gap-2 mt-3">
   269→                    <div className="bg-white/60 dark:bg-warm-700/40 rounded-lg p-2 text-center">
   270→                      <p className="text-lg">1️⃣</p>
   271→                      <p className="text-[9px] text-warm-500 dark:text-warm-400 font-medium">Generate code</p>
   272→                    </div>
   273→                    <div className="bg-white/60 dark:bg-warm-700/40 rounded-lg p-2 text-center">
   274→                      <p className="text-lg">2️⃣</p>
   275→                      <p className="text-[9px] text-warm-500 dark:text-warm-400 font-medium">Share code</p>
   276→                    </div>
   277→                    <div className="bg-white/60 dark:bg-warm-700/40 rounded-lg p-2 text-center">
   278→                      <p className="text-lg">3️⃣</p>
   279→                      <p className="text-[9px] text-warm-500 dark:text-warm-400 font-medium">Continue scoring</p>
   280→                    </div>
   281→                  </div>
   282→                </CardContent>
   283→              </Card>
   284→
   285→              {/* Current Match Info */}
   286→              {activeMatch && (
   287→                <Card className="bg-warm-100/60 dark:bg-warm-700/40 border-warm-200 dark:border-warm-600/30">
   288→                  <CardContent className="p-4">
   289→                    <div className="flex items-center gap-2 mb-3">
   290→                      <Zap className="w-4 h-4 text-brand-red" />
   291→                      <span className="text-xs font-bold text-warm-700 dark:text-warm-200">Current Live Match</span>
   292→                      <Badge className="bg-red-500 text-white text-[8px] border-0 animate-pulse">LIVE</Badge>
   293→                    </div>
   294→                    <div className="flex items-center justify-between">
   295→                      <div className="flex items-center gap-2">
   296→                        <div
   297→                          className="w-8 h-8 rounded-full flex items-center justify-center text-white text-[10px] font-bold"
   298→                          style={{ backgroundColor: activeMatch.homeTeamColor || '#DC2626' }}
   299→                        >
   300→                          {(activeMatch.homeTeam || 'H').slice(0, 2).toUpperCase()}
   301→                        </div>
   302→                        <span className="text-sm font-bold text-warm-800 dark:text-warm-100">{activeMatch.homeTeam}</span>
   303→                      </div>
   304→                      <div className="flex items-center gap-2">
   305→                        <span className="text-lg font-black text-warm-800 dark:text-warm-100">
   306→                          {activeMatch.homeScore} - {activeMatch.awayScore}
   307→                        </span>
   308→                      </div>
   309→                      <div className="flex items-center gap-2">
   310→                        <span className="text-sm font-bold text-warm-800 dark:text-warm-100">{activeMatch.awayTeam}</span>
   311→                        <div
   312→                          className="w-8 h-8 rounded-full flex items-center justify-center text-white text-[10px] font-bold"
   313→                          style={{ backgroundColor: activeMatch.awayTeamColor || '#1E293B' }}
   314→                        >
   315→                          {(activeMatch.awayTeam || 'A').slice(0, 2).toUpperCase()}
   316→                        </div>
   317→                      </div>
   318→                    </div>
   319→                    <div className="flex items-center justify-center gap-3 mt-2">
   320→                      <Badge variant="outline" className="text-[9px]">
   321→                        H{activeMatch.currentHalf}
   322→                      </Badge>
   323→                      <Badge variant="outline" className="text-[9px]">
   324→                        {activeMatch.isPractice ? 'Practice' : 'Tournament'}
   325→                      </Badge>
   326→                      <Badge variant="outline" className="text-[9px]">
   327→                        {activeMatch.events.length} events
   328→                      </Badge>
   329→                    </div>
   330→                  </CardContent>
   331→                </Card>
   332→              )}
   333→
   334→              {/* Generate Code Section */}
   335→              {!transferCode ? (
   336→                <Button
   337→                  onClick={handleGenerateCode}
   338→                  disabled={isGenerating || !activeMatch}
   339→                  className="w-full bg-gradient-to-r from-teal-600 to-teal-500 hover:from-teal-700 hover:to-teal-600 text-white font-bold rounded-xl py-6 text-sm shadow-lg shadow-teal-500/20"
   340→                >
   341→                  {isGenerating ? (
   342→                    <RefreshCw className="w-4 h-4 mr-2 animate-spin" />
   343→                  ) : (
   344→                    <Share2 className="w-4 h-4 mr-2" />
   345→                  )}
   346→                  {isGenerating ? 'Generating...' : 'Generate Transfer Code'}
   347→                </Button>
   348→              ) : (
   349→                <motion.div
   350→                  initial={{ opacity: 0, scale: 0.95 }}
   351→                  animate={{ opacity: 1, scale: 1 }}
   352→                  transition={{ type: 'spring', damping: 15, stiffness: 200 }}
   353→                >
   354→                  <Card className="bg-gradient-to-br from-teal-50 to-white dark:from-teal-900/30 dark:to-warm-800 border-2 border-teal-300 dark:border-teal-600/50 shadow-lg">
   355→                    <CardContent className="p-5">
   356→                      {/* Success indicator */}
   357→                      <div className="flex items-center gap-2 mb-4">
   358→                        <div className="w-8 h-8 rounded-full bg-teal-100 dark:bg-teal-800/40 flex items-center justify-center">
   359→                          <Check className="w-4 h-4 text-teal-600 dark:text-teal-400" />
   360→                        </div>
   361→                        <div>
   362→                          <p className="text-sm font-bold text-teal-700 dark:text-teal-300">Code Generated!</p>
   363→                          <p className="text-[10px] text-warm-500 dark:text-warm-400">Share this code with the new scorer</p>
   364→                        </div>
   365→                      </div>
   366→
   367→                      {/* The Code */}
   368→                      <div className="bg-white dark:bg-warm-700 rounded-2xl p-4 text-center border border-teal-200/50 dark:border-teal-600/30 shadow-inner">
   369→                        <p className="text-[10px] text-warm-400 dark:text-warm-500 font-medium mb-1">TRANSFER CODE</p>
   370→                        <div className="flex items-center justify-center gap-1.5">
   371→                          {transferCode.split('').map((char, i) => (
   372→                            <motion.span
   373→                              key={i}
   374→                              initial={{ opacity: 0, y: 8 }}
   375→                              animate={{ opacity: 1, y: 0 }}
   376→                              transition={{ delay: i * 0.08 }}
   377→                              className="w-9 h-12 rounded-lg bg-teal-50 dark:bg-teal-900/30 border border-teal-200 dark:border-teal-700/50 flex items-center justify-center text-xl font-black text-teal-700 dark:text-teal-300 tracking-wider"
   378→                            >
   379→                              {char}
   380→                            </motion.span>
   381→                          ))}
   382→                        </div>
   383→
   384→                        {/* Timer */}
   385→                        {codeTimer && (
   386→                          <div className="flex items-center justify-center gap-1.5 mt-3">
   387→                            <Clock className="w-3 h-3 text-warm-400 dark:text-warm-500" />
   388→                            <span className={`text-[10px] font-semibold ${
   389→                              codeTimer === 'Expired'
   390→                                ? 'text-red-500'
   391→                                : 'text-warm-500 dark:text-warm-400'
   392→                            }`}>
   393→                              {codeTimer === 'Expired' ? 'Code expired' : `Expires in ${codeTimer}`}
   394→                            </span>
   395→                          </div>
   396→                        )}
   397→                      </div>
   398→
   399→                      {/* Copy Button */}
   400→                      <div className="flex gap-2 mt-4">
   401→                        <Button
   402→                          onClick={handleCopyCode}
   403→                          className="flex-1 bg-teal-600 hover:bg-teal-700 text-white font-bold rounded-xl"
   404→                        >
   405→                          {copied ? <Check className="w-4 h-4 mr-1.5" /> : <Copy className="w-4 h-4 mr-1.5" />}
   406→                          {copied ? 'Copied!' : 'Copy Code'}
   407→                        </Button>
   408→                        <Button
   409→                          onClick={handleGenerateCode}
   410→                          variant="outline"
   411→                          className="flex-1 border-teal-300 dark:border-teal-600 text-teal-700 dark:text-teal-300 font-bold rounded-xl"
   412→                        >
   413→                          <RefreshCw className="w-4 h-4 mr-1.5" />
   414→                          New Code
   415→                        </Button>
   416→                      </div>
   417→                    </CardContent>
   418→                  </Card>
   419→                </motion.div>
   420→              )}
   421→
   422→              {/* Warning */}
   423→              <Card className="bg-amber-50/80 dark:bg-amber-900/15 border-amber-200/50 dark:border-amber-700/20">
   424→                <CardContent className="p-3">
   425→                  <div className="flex items-start gap-2">
   426→                    <AlertTriangle className="w-4 h-4 text-amber-500 shrink-0 mt-0.5" />
   427→                    <div>
   428→                      <p className="text-xs font-semibold text-amber-700 dark:text-amber-400">Important</p>
   429→                      <ul className="text-[10px] text-amber-600 dark:text-amber-400/80 mt-1 space-y-0.5">
   430→                        <li>• Code expires in 30 minutes</li>
   431→                        <li>• Keep scoring until the new scorer confirms receipt</li>
   432→                        <li>• Both scorers will see the same match state</li>
   433→                        <li>• Only one person should score at a time after transfer</li>
   434→                      </ul>
   435→                    </div>
   436→                  </div>
   437→                </CardContent>
   438→              </Card>
   439→            </motion.div>
   440→          ) : (
   441→            <motion.div
   442→              key="receive"
   443→              initial={{ opacity: 0, x: 20 }}
   444→              animate={{ opacity: 1, x: 0 }}
   445→              exit={{ opacity: 0, x: -20 }}
   446→              className="flex flex-col gap-4"
   447→            >
   448→              {/* Info Card */}
   449→              <Card className="bg-gradient-to-br from-teal-50/80 to-warm-50 dark:from-teal-900/20 dark:to-warm-800 border-teal-200/50 dark:border-teal-700/30">
   450→                <CardContent className="p-4">
   451→                  <div className="flex items-start gap-3">
   452→                    <div className="w-10 h-10 rounded-xl bg-teal-100 dark:bg-teal-800/40 flex items-center justify-center shrink-0">
   453→                      <Download className="w-5 h-5 text-teal-600 dark:text-teal-400" />
   454→                    </div>
   455→                    <div>
   456→                      <h3 className="text-sm font-bold text-warm-800 dark:text-warm-100">Receive a Match</h3>
   457→                      <p className="text-xs text-warm-500 dark:text-warm-400 mt-1 leading-relaxed">
   458→                        Enter the transfer code shared by the current scorer. You&apos;ll take over the live scoring from exactly where they left off.
   459→                      </p>
   460→                    </div>
   461→                  </div>
   462→                </CardContent>
   463→              </Card>
   464→
   465→              {/* Code Input */}
   466→              <Card className="bg-warm-100/60 dark:bg-warm-700/40 border-warm-200 dark:border-warm-600/30">
   467→                <CardContent className="p-4">
   468→                  <label className="text-xs font-bold text-warm-700 dark:text-warm-200 block mb-2">
   469→                    Enter Transfer Code
   470→                  </label>
   471→                  <div className="flex gap-2">
   472→                    <Input
   473→                      value={inputCode}
   474→                      onChange={(e) => {
   475→                        setInputCode(e.target.value.toUpperCase().replace(/[^A-Z0-9]/g, '').slice(0, 6));
   476→                        setReceiveError(null);
   477→                        setPreviewMatch(null);
   478→                      }}
   479→                      placeholder="e.g. ABC123"
   480→                      className="text-center text-lg font-black tracking-[0.3em] uppercase bg-white dark:bg-warm-800 border-2 border-teal-200 dark:border-teal-700/50 focus:border-teal-500 rounded-xl h-12"
   481→                      maxLength={6}
   482→                    />
   483→                    <Button
