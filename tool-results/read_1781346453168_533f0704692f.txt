   130→   130→function getInitials(name: string | null | undefined): string {
   131→   131→  if (!name) return '?';
   132→   132→  const parts = name.trim().split(/\s+/);
   133→   133→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
   134→   134→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
   135→   135→}
   136→   136→
   137→   137→function getDisplayName(name: string | null | undefined): string {
   138→   138→  return name?.trim() || 'Unnamed Player';
   139→   139→}
   140→   140→
   141→   141→function generateShortName(name: string): string {
   142→   142→  const words = name.trim().split(/\s+/);
   143→   143→  if (words.length >= 3) {
   144→   144→    return (words[0].charAt(0) + words[1].charAt(0) + words[2].charAt(0)).toUpperCase();
   145→   145→  }
   146→   146→  if (words.length === 2) {
   147→   147→    return (words[0].charAt(0) + words[1].charAt(0)).toUpperCase();
   148→   148→  }
   149→   149→  return name.slice(0, 3).toUpperCase();
   150→   150→}
