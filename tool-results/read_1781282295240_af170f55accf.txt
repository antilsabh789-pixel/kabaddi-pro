   123→   123→function formatDuration(isoStart: string | null, isoEnd: string | null): string {
   124→   124→  if (!isoStart || !isoEnd) return '';
   125→   125→  const ms = new Date(isoEnd).getTime() - new Date(isoStart).getTime();
   126→   126→  const totalMins = Math.round(ms / 60000);
   127→   127→  return `${totalMins} min`;
   128→   128→}
   129→   129→
   130→   130→function formatDateTime(iso: string | null): string {
   131→   131→  if (!iso) return '';
   132→   132→  const d = new Date(iso);
   133→   133→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' });
   134→   134→}

   177→   177→    setLoading(true);
   178→   178→    setError(null);
   179→   179→    try {
   180→   180→      const res = await fetch(`/api/matches?id=${encodeURIComponent(matchId)}`);
   181→   181→      if (!res.ok) throw new Error('Match not found');
   182→   182→      const data = await res.json();
   183→   183→      setMatch(data.match as MatchData);
   184→   184→    } catch (err) {
   185→   185→      console.error('Match highlights fetch error:', err);
   186→   186→      setError('Failed to load match highlights');
   187→   187→    } finally {
   188→   188→      setLoading(false);
   189→   189→    }
   190→   190→  }, [matchId]);
   191→   191→
   192→   192→  useEffect(() => {

   195→   195→
   196→   196→  // ── Derived data ───────────────────────────────────────────────────────────

   198→   198→  const awayColor = match?.awayTeam?.color || '#1E293B';
   199→   199→  const isHomeWin = match ? match.homeScore > match.awayScore : false;
   200→   200→  const isAwayWin = match ? match.awayScore > match.homeScore : false;

   203→   203→    ? isDraw ? 'draw' : isHomeWin ? 'home' : 'away'

   205→   205→
   206→   206→  // Build scorer name lookup
   207→   207→  const scorerNames = useMemo(() => {
   208→   208→    if (!match) return {} as Record<string, string>;
   209→   209→    const map: Record<string, string> = {};
   210→   210→    for (const s of match.scorers) {
   211→   211→      map[s.userId] = s.user.name;
   212→   212→    }
   213→   213→    return map;
   214→   214→  }, [match]);
   215→   215→
   216→   216→  // ── Sorted events (chronological) ──────────────────────────────────────────
   217→   217→  const sortedEvents = useMemo(() => {
   218→   218→    if (!match) return [];
   219→   219→    return [...match.events].sort(
   220→   220→      (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()

   224→   224→  // ── Key Moments ────────────────────────────────────────────────────────────
   225→   225→  const keyMoments = useMemo((): KeyMoment[] => {
   226→   226→    if (!match) return [];

   228→   228→    const moments: KeyMoment[] = [];

   230→   230→    let awayScore = 0;

   232→   232→

   234→   234→      const isHome = evt.teamId === match.homeTeamId;
   235→   235→      const teamName = isHome ? match.homeTeam.name : match.awayTeam.name;
   236→   236→      const teamColor = isHome ? homeColor : awayColor;
   237→   237→      const playerName = evt.playerId ? scorerNames[evt.playerId] : undefined;
   238→   238→
   239→   239→      // Track score for comeback detection
   240→   240→      if (isHome) {
   241→   241→        homeScore += evt.value;
   242→   242→      } else {
   243→   243→        awayScore += evt.value;
   244→   244→      }
   245→   245→
   246→   246→      // Detect lead change (comeback)
   247→   247→      const currentLeader = homeScore > awayScore
   248→   248→        ? match.homeTeamId
   249→   249→        : awayScore > homeScore
   250→   250→          ? match.awayTeamId
   251→   251→          : null;
   252→   252→
   253→   253→      const isComeback = currentLeader !== null
   254→   254→        && leadingTeamId !== null
   255→   255→        && currentLeader !== leadingTeamId

   258→   258→      leadingTeamId = currentLeader;

   263→   263→        let description = '';
   264→   264→
   265→   265→        switch (evt.eventType) {

   269→   269→          case 'super_tackle':
   270→   270→            description = `${playerName || teamName} executes a super tackle with 3 or fewer defenders!`;
   271→   271→            break;
   272→   272→          case 'all_out':
   273→   273→            description = `${teamName} inflicts an All Out — entire opposition sent off the mat!`;
   274→   274→            break;
   275→   275→          case 'do_or_die_raid':
   276→   276→            description = `${playerName || teamName} delivers in the crucial Do-or-Die raid!`;
   277→   277→            break;
   278→   278→        }
   279→   279→
   280→   280→        moments.push({
   281→   281→          id: evt.id,
   282→   282→          eventType: evt.eventType,
   283→   283→          label: meta.label,
   284→   284→          icon: meta.icon,
   285→   285→          teamId: evt.teamId,
   286→   286→          teamName,
   287→   287→          teamColor,
   288→   288→          playerName,
   289→   289→          value: evt.value,
   290→   290→          half: evt.half,
   291→   291→          timestamp: evt.timestamp,
   292→   292→          description,
   293→   293→        });
   294→   294→      }
   295→   295→
   296→   296→      // Add comeback moment
   297→   297→      if (isComeback && currentLeader !== null) {
   298→   298→        const comebackTeam = currentLeader === match.homeTeamId ? match.homeTeam : match.awayTeam;
   299→   299→        const comebackColor = currentLeader === match.homeTeamId ? homeColor : awayColor;
   300→   300→        moments.push({
   301→   301→          id: `comeback_${evt.id}`,
   302→   302→          eventType: 'comeback',
   303→   303→          label: 'Comeback!',
   304→   304→          icon: '🔄',
   305→   305→          teamId: currentLeader,
   306→   306→          teamName: comebackTeam.name,
   307→   307→          teamColor: comebackColor,
   308→   308→          value: 0,
   309→   309→          half: evt.half,
   310→   310→          timestamp: evt.timestamp,
   311→   311→          description: `${comebackTeam.name} takes the lead! The momentum shifts!`,
   312→   312→        });
   313→   313→      }
   314→   314→    }
   315→   315→
   316→   316→    return moments;
   317→   317→  }, [match, sortedEvents, homeColor, awayColor, scorerNames]);
   318→   318→
   319→   319→  // ── MVP Breakdown ──────────────────────────────────────────────────────────
   320→   320→  const mvpBreakdown = useMemo(() => {
   321→   321→    if (!match?.motmUser) return null;
   322→   322→
   323→   323→    let raidPoints = 0;
   324→   324→    let tacklePoints = 0;
   325→   325→    let bonusPoints = 0;
   326→   326→
   327→   327→    for (const evt of match.events) {
   328→   328→      if (evt.playerId !== match.motmUser.id) continue;
   329→   329→      switch (evt.eventType) {
   330→   330→        case 'raid_point':
   331→   331→        case 'super_raid':
   332→   332→        case 'do_or_die_raid':
   333→   333→          raidPoints += evt.value;
   334→   334→          break;
   335→   335→        case 'tackle_point':
   336→   336→        case 'super_tackle':
   337→   337→          tacklePoints += evt.value;
   338→   338→          break;
   339→   339→        case 'bonus_point':
   340→   340→          bonusPoints += evt.value;
   341→   341→          break;
   342→   342→      }
   343→   343→    }
   344→   344→
   345→   345→    const totalPoints = raidPoints + tacklePoints + bonusPoints;
   346→   346→    const motmTeamId = match.events.find(e => e.playerId === match.motmUser!.id)?.teamId;
   347→   347→    const teamName = motmTeamId === match.homeTeamId
   348→   348→      ? match.homeTeam.name
   349→   349→      : motmTeamId === match.awayTeamId
   350→   350→        ? match.awayTeam.name
   351→   351→        : '';
   352→   352→
   353→   353→    return {
   354→   354→      raidPoints,
   355→   355→      tacklePoints,
   356→   356→      bonusPoints,
   357→   357→      totalPoints,
   358→   358→      teamName,
   359→   359→    };
   360→   360→  }, [match]);
   361→   361→
   362→   362→  // ── Score Progression (for momentum graph) ─────────────────────────────────

   364→   364→    if (!match) return [];
   365→   365→
   366→   366→    let homeScore = 0;
   367→   367→    let awayScore = 0;

   377→   377→      } else {
   378→   378→        awayScore += evt.value;

   380→   380→      progression.push({
   381→   381→        index: idx + 1,
   382→   382→        homeScore,
   383→   383→        awayScore,
   384→   384→        gap: homeScore - awayScore,
   385→   385→        half: evt.half,
   386→   386→      });
   387→   387→    });
   388→   388→
   389→   389→    return progression;
   390→   390→  }, [match, sortedEvents]);
   391→   391→
   392→   392→  // ── Turning Points ─────────────────────────────────────────────────────────
   393→   393→  const turningPoints = useMemo((): TurningPoint[] => {
   394→   394→    if (!match) return [];
   395→   395→
   396→   396→    const points: TurningPoint[] = [];
   397→   397→    let homeScore = 0;
   398→   398→    let awayScore = 0;
   399→   399→    let prevLeader: 'home' | 'away' | 'draw' = 'draw';
   400→   400→
   401→   401→    for (const evt of sortedEvents) {
   402→   402→      const isHome = evt.teamId === match.homeTeamId;

   404→   404→        homeScore += evt.value;
   405→   405→      } else {
   406→   406→        awayScore += evt.value;
   407→   407→      }
   408→   408→
   409→   409→      const currentLeader: 'home' | 'away' | 'draw' =
   410→   410→        homeScore > awayScore ? 'home' : awayScore > homeScore ? 'away' : 'draw';
   411→   411→
   412→   412→      // All Out that changed lead
   413→   413→      if (evt.eventType === 'all_out' && currentLeader !== prevLeader && currentLeader !== 'draw') {
   414→   414→        const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;

   418→   418→          type: 'all_out_lead_change',
   419→   419→          label: 'All Out Lead Change',

   424→   424→          timestamp: evt.timestamp,
   425→   425→          half: evt.half,
   426→   426→        });
   427→   427→      }
   428→   428→
   429→   429→      // Do-or-Die raid that shifted momentum (lead change)
   430→   430→      if (evt.eventType === 'do_or_die_raid' && currentLeader !== prevLeader && currentLeader !== 'draw') {
   431→   431→        const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;
   432→   432→        const color = currentLeader === 'home' ? homeColor : awayColor;
   433→   433→        points.push({
   434→   434→          id: `tp_dod_${evt.id}`,
   435→   435→          type: 'do_or_die_shift',
   436→   436→          label: 'Do-or-Die Shift',
   437→   437→          icon: '⚡',
   438→   438→          description: `${team.name} seized the lead in a Do-or-Die raid! ${homeScore}-${awayScore}`,
   439→   439→          teamName: team.name,
   440→   440→          teamColor: color,
   441→   441→          timestamp: evt.timestamp,
   442→   442→          half: evt.half,
   443→   443→        });
   444→   444→      }
   445→   445→
   446→   446→      // Comeback: team was losing by 3+ and takes lead
   447→   447→      if (currentLeader !== 'draw' && currentLeader !== prevLeader) {
   448→   448→        const prevGap = Math.abs(homeScore - awayScore);
   449→   449→        if (prevGap >= 3) {
   450→   450→          const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;
   451→   451→          const color = currentLeader === 'home' ? homeColor : awayColor;
   452→   452→          points.push({
   453→   453→            id: `tp_cb_${evt.id}`,
   454→   454→            type: 'comeback',
   455→   455→            label: 'Comeback Moment',
   456→   456→            icon: '🔄',
   457→   457→            description: `${team.name} fights back from ${prevGap} points down to take the lead!`,
   458→   458→            teamName: team.name,
   459→   459→            teamColor: color,
   460→   460→            timestamp: evt.timestamp,
   461→   461→            half: evt.half,
   462→   462→          });
   463→   463→        }
   464→   464→      }
   465→   465→
   466→   466→      prevLeader = currentLeader;
   467→   467→    }
   468→   468→
   469→   469→    return points;
   470→   470→  }, [match, sortedEvents, homeColor, awayColor]);
   471→   471→
   472→   472→  // ── Share data ─────────────────────────────────────────────────────────────
   473→   473→  const buildShareData = useCallback(() => {
   474→   474→    if (!match) return null;
   475→   475→
   476→   476→    // Top raider/defender from events
   477→   477→    const raidMap: Record<string, { name: string; points: number }> = {};
   478→   478→    const tackleMap: Record<string, { name: string; points: number }> = {};
