   234→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   235→  const [count, setCount] = useState(0);
   236→  const prevTarget = useRef(target);
   237→
   238→  useEffect(() => {
   239→    if (prevTarget.current === target && count !== 0) return;
   240→    prevTarget.current = target;
   241→
   242→    let start = 0;
   243→    const startTime = performance.now();
   244→
   245→    function step(currentTime: number) {
   246→      const elapsed = currentTime - startTime;
   247→      const progress = Math.min(elapsed / duration, 1);
   248→      // ease-out cubic
   249→      const eased = 1 - Math.pow(1 - progress, 3);
   250→      const current = Math.round(eased * target);
   251→      setCount(current);
   252→      if (progress < 1) {
   253→        requestAnimationFrame(step);
   254→      }
   255→    }
   256→
   257→    requestAnimationFrame(step);
   258→  }, [target, duration]);
   259→
   260→  return <span>{count}</span>;
   261→}

   315→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   316→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   317→  const [expired, setExpired] = useState(!targetDate);
   318→
   319→  useEffect(() => {
   320→    if (!targetDate) {
   321→      return;
   322→    }
   323→
   324→    function calculate() {
   325→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   326→      if (diff === 0) {
   327→        setExpired(true);
   328→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   329→        return true;
   330→      }
   331→      setExpired(false);
   332→      setTimeLeft({
   333→        d: Math.floor(diff / 86400000),
   334→        h: Math.floor((diff % 86400000) / 3600000),
   335→        m: Math.floor((diff % 3600000) / 60000),
   336→        s: Math.floor((diff % 60000) / 1000),
   337→      });
   338→      return false;
   339→    }
   340→
   341→    const done = calculate();
   342→    if (done) return;
   343→
   344→    const id = setInterval(() => {
   345→      const finished = calculate();
   346→      if (finished) clearInterval(id);
   347→    }, 1000);
   348→
   349→    return () => clearInterval(id);
   350→  }, [targetDate]);
   351→
   352→  if (expired) {
   353→    return (
   354→      <motion.div
   355→        className="flex items-center justify-center gap-1.5 py-1"
   356→        animate={{ opacity: [1, 0.5, 1] }}
   357→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   358→      >
   359→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   360→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   361→      </motion.div>
   362→    );
   363→  }
   364→
   365→  if (!timeLeft) return null;
   366→
   367→  const units = [
   368→    { value: timeLeft.d, label: 'D' },
   369→    { value: timeLeft.h, label: 'H' },
   370→    { value: timeLeft.m, label: 'M' },
   371→    { value: timeLeft.s, label: 'S' },
   372→  ];
   373→
   374→  return (
   375→    <div className="flex items-center justify-center gap-1.5">
   376→      {units.map((unit, idx) => (
   377→        <div key={unit.label} className="flex items-center gap-1.5">
   378→          <div className="flex flex-col items-center">
   379→            <motion.div
   380→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   381→              key={`${unit.label}-${unit.value}`}
   382→              initial={{ rotateX: -90, opacity: 0 }}
   383→              animate={{ rotateX: 0, opacity: 1 }}
   384→              transition={{ duration: 0.3, ease: 'easeOut' }}
   385→            >
   386→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums">
   387→                {String(unit.value).padStart(2, '0')}
   388→              </span>
   389→            </motion.div>
   390→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   391→              {unit.label}
   392→            </span>
   393→          </div>
   394→          {idx < units.length - 1 && (
   395→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   396→          )}
   397→        </div>
   398→      ))}
   399→    </div>
   400→  );
   401→}
