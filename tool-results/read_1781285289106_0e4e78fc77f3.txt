    69→function getPositionCategory(position: string | null): 'raider' | 'defender' | 'all-rounder' | 'unknown' {
    70→  if (!position) return 'unknown';
    71→  const p = position.toLowerCase();
    72→  if (p.includes('raider')) return 'raider';
    73→  if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return 'defender';
    74→  if (p.includes('all')) return 'all-rounder';
    75→  return 'unknown';
    76→}

    80→function LineupValidation({ lineup, playersPerSide, teamName }: {
    81→  lineup: MatchPlayer[];
    82→  playersPerSide: number;
    83→  teamName: string;
    84→}) {
    85→  const warnings: { message: string; severity: 'error' | 'warning' | 'info' }[] = [];
    86→
    87→  if (lineup.length === 0) {
    88→    warnings.push({ message: 'No players added', severity: 'error' });
    89→  } else if (lineup.length < playersPerSide) {
    90→    warnings.push({
    91→      message: `${lineup.length}/${playersPerSide} players — need ${playersPerSide - lineup.length} more`,
    92→      severity: 'warning',
    93→    });
    94→  } else if (lineup.length > playersPerSide) {
    95→    warnings.push({
    96→      message: `${lineup.length} players — ${lineup.length - playersPerSide} extra`,
    97→      severity: 'info',
    98→    });
    99→  }
   100→
   101→  // Position balance check (only if we have position data from DB)
   102→  const balance = POSITION_BALANCE[playersPerSide];
   103→  // Note: position data might not be available for quick-added players
   104→  // We check this based on the allPlayers data which is in parent scope
   105→
   106→  if (warnings.length === 0) return null;
   107→
   108→  return (
   109→    <motion.div
   110→      initial={{ opacity: 0, y: -5 }}
   111→      animate={{ opacity: 1, y: 0 }}
   112→      className="space-y-1.5"
   113→    >
   114→      {warnings.map((w, i) => (
   115→        <div
   116→          key={i}
   117→          className={`flex items-center gap-2 px-3 py-2 rounded-lg text-xs font-medium ${
   118→            w.severity === 'error'
   119→              ? 'bg-brand-red/10 text-brand-red dark:bg-brand-red/15'
   120→              : w.severity === 'warning'
   121→                ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400 dark:bg-amber-500/15'
   122→                : 'bg-blue-500/10 text-blue-600 dark:text-blue-400 dark:bg-blue-500/15'
   123→          }`}
   124→        >
   125→          <AlertTriangle className={`w-3.5 h-3.5 shrink-0 ${w.severity === 'info' ? 'hidden' : ''}`} />
   126→          <Info className={`w-3.5 h-3.5 shrink-0 ${w.severity !== 'info' ? 'hidden' : ''}`} />
   127→          <span>{teamName}: {w.message}</span>
   128→        </div>
   129→      ))}
   130→    </motion.div>
   131→  );
   132→}

   136→function PlayerStatsTooltip({ player, allPlayers }: { player: DbPlayer; allPlayers: DbPlayer[] }) {
   137→  const dbPlayer = allPlayers.find(p => p.id === player.id);
   138→  const profile = dbPlayer?.profile;
   139→
   140→  if (!profile || !profile.totalPoints) return null;
   141→
   142→  const raidSuccessRate = profile.totalRaids && profile.totalRaids > 0
   143→    ? Math.round(((profile.successfulRaids || 0) / profile.totalRaids) * 100)
   144→    : 0;
   145→
   146→  return (
   147→    <div className="absolute z-30 bottom-full left-0 mb-1 p-2 bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-600 rounded-lg shadow-lg text-[10px] min-w-[140px]">
   148→      <div className="space-y-1">
   149→        <div className="flex justify-between">
   150→          <span className="text-warm-500 dark:text-warm-400">Total Pts</span>
   151→          <span className="font-bold text-brand-navy dark:text-brand-navy-light">{profile.totalPoints}</span>
   152→        </div>
   153→        <div className="flex justify-between">
   154→          <span className="text-warm-500 dark:text-warm-400">Raid Pts</span>
   155→          <span className="font-bold text-brand-red">{profile.raidPoints || 0}</span>
   156→        </div>
   157→        <div className="flex justify-between">
   158→          <span className="text-warm-500 dark:text-warm-400">Tackle Pts</span>
   159→          <span className="font-bold text-brand-teal">{profile.tacklePoints || 0}</span>
   160→        </div>
   161→        <div className="flex justify-between">
   162→          <span className="text-warm-500 dark:text-warm-400">Raid Success</span>
   163→          <span className="font-bold text-brand-gold">{raidSuccessRate}%</span>
   164→        </div>
   165→      </div>
   166→      {/* Tooltip arrow */}
   167→      <div className="absolute -bottom-1 left-4 w-2 h-2 bg-white dark:bg-warm-800 border-r border-b border-warm-200 dark:border-warm-600 rotate-45" />
   168→    </div>
   169→  );
   170→}

   174→function FormationVisualization({ lineup, teamColor, teamName, side }: {
   175→  lineup: MatchPlayer[];
   176→  teamColor: string;
   177→  teamName: string;
   178→  side: 'left' | 'right';
   179→}) {
   180→  const maxDisplay = Math.min(lineup.length, 7);
   181→  const displayPlayers = lineup.slice(0, maxDisplay);
   182→
   183→  return (
   184→    <div className="flex-1 relative">
   185→      {/* Team header */}
   186→      <div className="text-center mb-2">
   187→        <div
   188→          className="w-10 h-10 rounded-xl mx-auto flex items-center justify-center text-white font-black text-sm shadow-md"
   189→          style={{ backgroundColor: teamColor }}
   190→        >
   191→          {teamName.charAt(0).toUpperCase()}
   192→        </div>
   193→        <p className="text-[10px] font-bold text-warm-600 dark:text-warm-300 mt-1 truncate max-w-[80px] mx-auto">
   194→          {teamName}
   195→        </p>
   196→      </div>
   197→
   198→      {/* Formation layout */}
   199→      <div className="relative bg-warm-100/40 dark:bg-warm-700/20 rounded-xl p-2 min-h-[90px]">
   200→        {/* Court half line */}
   201→        <div className={`absolute top-1/2 left-0 right-0 h-px bg-warm-200 dark:bg-warm-600 ${
   202→          side === 'left' ? 'bg-gradient-to-r from-transparent to-warm-200 dark:to-warm-600' : 'bg-gradient-to-l from-transparent to-warm-200 dark:to-warm-600'
   203→        }`} />
   204→
   205→        {/* Players in formation */}
   206→        <div className="grid grid-cols-3 gap-1 place-items-center">
   207→          {displayPlayers.map((player, i) => {
   208→            // Simple formation: top row = defenders, middle = all-rounders, bottom = raiders
   209→            return (
   210→              <motion.div
   211→                key={player.id}
   212→                initial={{ scale: 0, opacity: 0 }}
   213→                animate={{ scale: 1, opacity: 1 }}
   214→                transition={{ delay: i * 0.05, type: 'spring', stiffness: 300, damping: 20 }}
   215→                className="flex flex-col items-center"
   216→              >
   217→                <div
   218→                  className="w-7 h-7 rounded-full flex items-center justify-center text-white text-[8px] font-bold shadow-sm"
   219→                  style={{ backgroundColor: teamColor }}
   220→                >
   221→                  {player.jerseyNumber || i + 1}
   222→                </div>
   223→                <span className="text-[7px] text-warm-400 dark:text-warm-500 mt-0.5 truncate max-w-[32px]">
   224→                  {player.name.split(' ')[0]}
   225→                </span>
   226→              </motion.div>
   227→            );
   228→          })}
   229→          {/* Empty slots */}
   230→          {Array.from({ length: Math.max(0, 7 - displayPlayers.length) }).map((_, i) => (
   231→            <div key={`empty-${i}`} className="flex flex-col items-center">
   232→              <div className="w-7 h-7 rounded-full border-2 border-dashed border-warm-200 dark:border-warm-600 flex items-center justify-center">
   233→                <span className="text-[8px] text-warm-300 dark:text-warm-600">?</span>
   234→              </div>
   235→              <span className="text-[7px] text-warm-300 dark:text-warm-600 mt-0.5">Empty</span>
   236→            </div>
   237→          ))}
   238→        </div>
   239→      </div>
   240→    </div>
   241→  );
   242→}

   562→    const sortedUnknowns = sortByRating(unknowns);

   568→      const picked = pool.slice(0, count);

   570→        suggested.push({
   571→          id: p.id,
   572→          name: p.name || 'Unknown',
   573→          jerseyNumber: p.profile?.jerseyNumber || suggested.length + 1,
   574→          playerCode: p.playerCode || undefined,
   575→          team: activeLineupTeam,
   576→        });
   577→      });

   579→    };

   582→    let needDefenders = balance.defenders;
   583→    let needAllRounders = balance.allRounders;
   584→

   586→    const pickedRaiders = pickFrom(sortedRaiders, needRaiders);
   587→    needRaiders -= pickedRaiders;
   588→    const pickedDefenders = pickFrom(sortedDefenders, needDefenders);
   589→    needDefenders -= pickedDefenders;
   590→    const pickedAllRounders = pickFrom(sortedAllRounders, needAllRounders);
