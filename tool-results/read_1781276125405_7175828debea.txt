   173→    // Reset 5-min warning flag when half changes

   234→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);

   265→    if (raidGapTimer <= 0) {
   266→      if (raidGapRef.current) clearInterval(raidGapRef.current);

   281→          if (raidGapRef.current) clearInterval(raidGapRef.current);
   282→          return 0;
   283→        }
   284→        return prev - 1;
   285→      });
   286→    }, 1000);
   287→
   288→    return () => {
   289→      if (raidGapRef.current) clearInterval(raidGapRef.current);
   290→    };

   293→  // Calculate MOTM

   303→      }
   304→    }
   305→    let topPlayer: { name: string; points: number } | null = null;
   306→    for (const p of Object.values(playerPoints)) {
   307→      if (!topPlayer || p.points > topPlayer.points) topPlayer = p;
   308→    }
   309→    return topPlayer;
   310→  }, [match]);
   311→
   312→  const calculateTopRaiderDefender = useCallback(() => {
   313→    if (!match) return { topRaider: null, topDefender: null };
   314→    const playerRaidPoints: Record<string, { name: string; points: number }> = {};
   315→    const playerTacklePoints: Record<string, { name: string; points: number }> = {};
   316→
   317→    for (const event of match.events) {
   318→      if (!event.playerId || !event.playerName) continue;
   319→      if (['raid_point', 'bonus_point', 'super_raid', 'do_or_die_raid'].includes(event.eventType)) {
   320→        if (!playerRaidPoints[event.playerId]) playerRaidPoints[event.playerId] = { name: event.playerName, points: 0 };
   321→        playerRaidPoints[event.playerId].points += event.value;
   322→      }
   323→      if (['tackle_point', 'super_tackle'].includes(event.eventType)) {
   324→        if (!playerTacklePoints[event.playerId]) playerTacklePoints[event.playerId] = { name: event.playerName, points: 0 };
   325→        playerTacklePoints[event.playerId].points += event.value;
   326→      }
   327→    }
   328→
   329→    let topRaider: { name: string; points: number } | null = null;
   330→    for (const p of Object.values(playerRaidPoints)) {
   331→      if (!topRaider || p.points > topRaider.points) topRaider = p;
   332→    }
   333→    let topDefender: { name: string; points: number } | null = null;
   334→    for (const p of Object.values(playerTacklePoints)) {
   335→      if (!topDefender || p.points > topDefender.points) topDefender = p;
   336→    }
   337→    return { topRaider, topDefender };
   338→  }, [match]);
   339→
   340→  if (!match) return null;
   341→
   342→  // ═══ DERIVED STATE ═══
   343→  const raidingTeam = match.raidQueue;
   344→  const defendingTeam: 'home' | 'away' = raidingTeam === 'home' ? 'away' : 'home';
   345→
   346→  const fullRaidingLineup = raidingTeam === 'home' ? match.homeLineup : match.awayLineup;
   347→  const fullDefendingLineup = defendingTeam === 'home' ? match.homeLineup : match.awayLineup;
   348→
   349→  const raidingOutIds = raidingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   350→  const defendingOutIds = defendingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   351→
   352→  const raidingTeamColor = raidingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   353→  const defendingTeamColor = defendingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   354→
   355→  const raidingTeamName = raidingTeam === 'home' ? match.homeTeam : match.awayTeam;

   358→  // Split lineup: first 7 = on court, rest = substitutes
   359→  const splitLineup = (lineup: MatchPlayer[], outIds: string[]) => {
   360→    const onCourt = lineup.slice(0, ON_COURT_MAX);
   361→    const substitutes = lineup.slice(ON_COURT_MAX);
   362→    const onCourtActive = onCourt.filter(p => !outIds.includes(p.id));
   363→    const onCourtOut = onCourt.filter(p => outIds.includes(p.id));
   364→    return { onCourt, substitutes, onCourtActive, onCourtOut };
   365→  };
   366→
   367→  // Handle raider selection → starts raid timer AND match timer
   368→  const handleSelectRaider = (player: MatchPlayer) => {
   369→    if (raidingOutIds.includes(player.id)) return;
   370→    if (raidPhase !== 'idle') return;
   371→

   374→

   377→
   378→    // Clear raid gap timer

   383→    setRaidResult(null);
   384→    setSelectedDefenders(new Set());

   386→
   387→    // Start 30s raid timer

   389→

   392→      triggerFeedback(SoundType.DO_OR_DIE);
   393→    } else {
   394→      triggerFeedback(SoundType.WHISTLE);
   395→    }
   396→  };
   397→
   398→  // Handle raid result selection
   399→  const handleSelectResult = (result: RaidResult) => {
   400→    if (result === 'empty') {
   401→      processRaidResult(result, new Set(), false);
   402→      return;
   403→    }
   404→    setRaidResult(result);
   405→    setRaidPhase('defenders');
   406→    setSelectedDefenders(new Set());
   407→  };
   408→
   409→  // Toggle defender selection

   413→      if (next.has(playerId)) next.delete(playerId);
   414→      else next.add(playerId);

   416→    });
   417→  };
   418→
   419→  // Process raid result
   420→  const processRaidResult = (result: RaidResult, touchedDefenders: Set<string>, hasBonus: boolean) => {
   421→    if (!match || !raider) return;
   422→
   423→    if (!hasStartedRaiding) setHasStartedRaiding(true);
   424→
   425→    // Stop raid timer
   426→    setRaidTimer(null);
   427→    if (raidTimerRef.current) clearInterval(raidTimerRef.current);

   429→    const raidingTeamId = raidingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   430→    const defendingTeamId = defendingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   431→
   432→    const events: Omit<MatchEvent, 'id' | 'timestamp'>[] = [];
   433→
   434→    if (result === 'success') {
   435→      const touchCount = touchedDefenders.size;
   436→      if (touchCount > 0) {
   437→        events.push({
   438→          matchId: match.id, eventType: 'raid_point', teamId: raidingTeamId,
   439→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   440→          value: touchCount,
   441→          details: JSON.stringify({ touchedPlayerIds: Array.from(touchedDefenders), raiderId: raider.id }),
   442→        });
   443→      }
   444→      if (hasBonus) {
   445→        events.push({
   446→          matchId: match.id, eventType: 'bonus_point', teamId: raidingTeamId,
   447→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   448→          value: 1, details: JSON.stringify({ raiderId: raider.id }),
   449→        });
   450→      }
   451→
   452→      // All out check
   453→      const { onCourt } = splitLineup(fullDefendingLineup, defendingOutIds);
   454→      const defendingOnCourtOut = onCourt.filter(p => defendingOutIds.includes(p.id)).length;
   455→      const newDefendingOutCount = defendingOnCourtOut + touchCount;
   456→      if (newDefendingOutCount >= onCourt.length) {
   457→        events.push({
   458→          matchId: match.id, eventType: 'all_out', teamId: raidingTeamId,
   459→          half: match.currentHalf, value: 2,
   460→        });
   461→      }
   462→    } else if (result === 'caught') {
   463→      const caughtByIds = Array.from(touchedDefenders);
   464→      const primaryCatcher = caughtByIds.length > 0
   465→        ? fullDefendingLineup.find(p => caughtByIds.includes(p.id))
   466→        : null;
   467→
   468→      events.push({
   469→        matchId: match.id, eventType: 'tackle_point', teamId: defendingTeamId,
   470→        half: match.currentHalf,
   471→        playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   472→        value: 1,
   473→        details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   474→      });
   475→
   476→      const { onCourtActive } = splitLineup(fullDefendingLineup, defendingOutIds);
   477→      if (onCourtActive.length <= 3) {
   478→        events.push({
   479→          matchId: match.id, eventType: 'super_tackle', teamId: defendingTeamId,
   480→          half: match.currentHalf,
   481→          playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   482→          value: 1,
   483→          details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   484→        });
   485→      }
   486→    } else if (result === 'empty') {
   487→      events.push({
   488→        matchId: match.id, eventType: 'empty_raid', teamId: raidingTeamId,
   489→        half: match.currentHalf, playerId: raider.id, playerName: raider.name, value: 0,
   490→      });
   491→
   492→      // Track consecutive empty raids for do-or-die
   493→      consecutiveEmptyRaidsRef.current += 1;
   494→    }
   495→
   496→    // Reset consecutive empty raids on any successful raid (points scored)
   497→    if (result === 'success') {
   498→      consecutiveEmptyRaidsRef.current = 0;
   499→    }
   500→    // Raider caught (tackle) also resets because the defending team scored
   501→    if (result === 'caught') {
   502→      consecutiveEmptyRaidsRef.current = 0;
   503→    }
   504→
   505→    // ═══ DO-OR-DIE LOGIC ═══
   506→    // In kabaddi, after 2 consecutive empty raids, the next raid becomes do-or-die
   507→    if (consecutiveEmptyRaidsRef.current >= 2) {
   508→      setDoOrDie(true);
   509→      // 🔊 Do-or-Die activated!
   510→      triggerFeedback(SoundType.DO_OR_DIE);
   511→      toast({
   512→        title: '🔥 DO OR DIE RAID!',
   513→        description: 'Next raider must score — or raider is out!',
   514→        duration: 3000,
   515→      });
   516→    } else {
   517→      setDoOrDie(false);
   518→    }
   519→
   520→    if (events.length > 0) {
   521→      addBatchEvents(events);
   522→    }
   523→
   524→    // Reset raid state
   525→    setRaidPhase('idle');
   526→    setRaider(null);
   527→    setRaidResult(null);
   528→    setSelectedDefenders(new Set());
   529→    setBonusPoint(false);
