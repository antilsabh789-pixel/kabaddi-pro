    83→// ─── Animated Value (count-up on first view) ────────────────────
    84→
    85→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    86→  const [display, setDisplay] = useState(0);
    87→  const [hasAnimated, setHasAnimated] = useState(false);
    88→  const ref = useRef<HTMLSpanElement>(null);
    89→  const valueRef = useRef(value);
    90→
    91→  useEffect(() => {
    92→    valueRef.current = value;
    93→  }, [value]);
    94→
    95→  useEffect(() => {
    96→    if (hasAnimated) return;
    97→    if (value === 0) return;
    98→
    99→    const observer = new IntersectionObserver(
   100→      ([entry]) => {
   101→        if (entry.isIntersecting && !hasAnimated) {
   102→          setHasAnimated(true);
   103→          const targetVal = valueRef.current;
   104→          const duration = 1200;
   105→          const startTime = performance.now();
   106→
   107→          function step(currentTime: number) {
   108→            const elapsed = currentTime - startTime;
   109→            const progress = Math.min(elapsed / duration, 1);
   110→            const eased = 1 - Math.pow(1 - progress, 3);
   111→            const current = eased * targetVal;
   112→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
   113→            if (progress < 1) {
   114→              requestAnimationFrame(step);
   115→            }
   116→          }
   117→
   118→          requestAnimationFrame(step);
   119→        }
   120→      },
   121→      { threshold: 0.1 }
   122→    );
   123→
   124→    if (ref.current) observer.observe(ref.current);
   125→    return () => observer.disconnect();
   126→  }, [value, decimals, hasAnimated]);
   127→
   128→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   129→}

   146→// ─── Time Ago helper ────────────────────
   147→
   148→function timeAgo(dateStr: string): string {
   149→  if (!dateStr || dateStr === '—') return '';
   150→  try {
   151→    const date = new Date(dateStr);
   152→    if (isNaN(date.getTime())) return dateStr;
   153→    const now = new Date();
   154→    const diff = now.getTime() - date.getTime();
   155→    const mins = Math.floor(diff / 60000);
   156→    if (mins < 60) return mins <= 1 ? 'just now' : `${mins}m ago`;
   157→    const hours = Math.floor(mins / 60);
   158→    if (hours < 24) return `${hours}h ago`;
   159→    const days = Math.floor(hours / 24);
   160→    if (days < 7) return `${days}d ago`;
   161→    const weeks = Math.floor(days / 7);
   162→    if (weeks < 4) return `${weeks}w ago`;
   163→    return dateStr;
   164→  } catch {
   165→    return dateStr;
   166→  }
   167→}

   171→function getPlayerLevel(totalMatches: number, totalPoints: number): { label: string; color: string; icon: string; progress: number } {
   172→  const score = totalMatches * 2 + totalPoints;
   173→  if (score >= 500) return { label: 'Legend', color: 'from-amber-400 to-yellow-600', icon: '👑', progress: 100 };
   174→  if (score >= 200) return { label: 'Pro', color: 'from-red-500 to-orange-500', icon: '🔥', progress: Math.min(((score - 200) / 300) * 100, 100) };
   175→  if (score >= 50) return { label: 'Intermediate', color: 'from-teal-500 to-emerald-500', icon: '⚡', progress: Math.min(((score - 50) / 150) * 100, 100) };
   176→  return { label: 'Beginner', color: 'from-slate-400 to-slate-500', icon: '🌱', progress: Math.min((score / 50) * 100, 100) };
   177→}
