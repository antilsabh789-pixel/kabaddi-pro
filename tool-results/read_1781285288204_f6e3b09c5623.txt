   128→function SearchableTeamSelect({
   129→  teams,
   130→  selectedTeamId,
   131→  onSelect,
   132→  label,
   133→  placeholder,
   134→  excludeTeamId,
   135→}: {
   136→  teams: TeamBasic[];
   137→  selectedTeamId: string;
   138→  onSelect: (id: string) => void;
   139→  label: string;
   140→  placeholder: string;
   141→  excludeTeamId?: string;
   142→}) {
   143→  const [open, setOpen] = useState(false);
   144→  const [search, setSearch] = useState('');
   145→  const containerRef = useRef<HTMLDivElement>(null);
   146→
   147→  const filteredTeams = useMemo(
   148→    () =>
   149→      teams
   150→        .filter((t) => !excludeTeamId || t.id !== excludeTeamId)
   151→        .filter(
   152→          (t) =>
   153→            t.name.toLowerCase().includes(search.toLowerCase()) ||
   154→            (t.shortName && t.shortName.toLowerCase().includes(search.toLowerCase()))
   155→        ),
   156→    [teams, excludeTeamId, search]
   157→  );
   158→
   159→  const selectedTeam = teams.find((t) => t.id === selectedTeamId);
   160→
   161→  useEffect(() => {
   162→    function handleClickOutside(e: MouseEvent) {
   163→      if (containerRef.current && !containerRef.current.contains(e.target as Node)) {
   164→        setOpen(false);
   165→      }
   166→    }
   167→    document.addEventListener('mousedown', handleClickOutside);
   168→    return () => document.removeEventListener('mousedown', handleClickOutside);
   169→  }, []);
   170→
   171→  return (
   172→    <div ref={containerRef}>
   173→      <label className="text-xs font-semibold text-warm-600 dark:text-warm-400 mb-1.5 block">
   174→        {label}
   175→      </label>
   176→      <div className="relative">
   177→        <button
   178→          type="button"
   179→          onClick={() => setOpen(!open)}
   180→          className="w-full h-10 rounded-xl border border-warm-300 dark:border-warm-700 bg-white/80 dark:bg-warm-800/80 backdrop-blur-sm px-3 text-sm text-warm-800 dark:text-warm-200 flex items-center justify-between hover:border-brand-red/40 focus:outline-none focus:ring-2 focus:ring-brand-red/20 transition-all"
   181→        >
   182→          <div className="flex items-center gap-2 min-w-0">
   183→            {selectedTeam ? (
   184→              <>
   185→                <div
   186→                  className="w-6 h-6 rounded-md flex items-center justify-center text-[9px] font-bold text-white shrink-0"
   187→                  style={{ backgroundColor: selectedTeam.color || '#64748B' }}
   188→                >
   189→                  {(selectedTeam.shortName || selectedTeam.name.slice(0, 2)).toUpperCase()}
   190→                </div>
   191→                <span className="truncate">{selectedTeam.name}</span>
   192→              </>
   193→            ) : (
   194→              <span className="text-warm-400">{placeholder}</span>
   195→            )}
   196→          </div>
   197→          <ChevronDown className={`w-4 h-4 text-warm-400 transition-transform ${open ? 'rotate-180' : ''}`} />
   198→        </button>
   199→
   200→        <AnimatePresence>
   201→          {open && (
   202→            <motion.div
   203→              initial={{ opacity: 0, y: -4, scale: 0.98 }}
   204→              animate={{ opacity: 1, y: 0, scale: 1 }}
   205→              exit={{ opacity: 0, y: -4, scale: 0.98 }}
   206→              transition={{ duration: 0.15 }}
   207→              className="absolute z-20 top-full mt-1 w-full bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-700 rounded-xl shadow-lg overflow-hidden"
   208→            >
   209→              <div className="p-2 border-b border-warm-100 dark:border-warm-700">
   210→                <div className="relative">
   211→                  <Search className="w-3.5 h-3.5 absolute left-2.5 top-1/2 -translate-y-1/2 text-warm-400" />
   212→                  <input
   213→                    type="text"
   214→                    value={search}
   215→                    onChange={(e) => setSearch(e.target.value)}
   216→                    placeholder="Search teams..."
   217→                    className="w-full h-8 pl-8 pr-3 rounded-lg bg-warm-50 dark:bg-warm-900 text-xs text-warm-800 dark:text-warm-200 placeholder:text-warm-400 focus:outline-none"
   218→                    autoFocus
   219→                  />
   220→                </div>
   221→              </div>
   222→              <div className="max-h-48 overflow-y-auto custom-scrollbar">
   223→                {filteredTeams.length === 0 ? (
   224→                  <p className="p-3 text-xs text-warm-400 text-center">No teams found</p>
   225→                ) : (
   226→                  filteredTeams.map((team) => (
   227→                    <button
   228→                      key={team.id}
   229→                      type="button"
   230→                      onClick={() => {
   231→                        onSelect(team.id);
   232→                        setOpen(false);
   233→                        setSearch('');
   234→                      }}
   235→                      className={`w-full flex items-center gap-2.5 px-3 py-2.5 text-sm hover:bg-warm-50 dark:hover:bg-warm-700/50 transition-colors ${
   236→                        team.id === selectedTeamId
   237→                          ? 'bg-brand-red/5 dark:bg-brand-red/10 text-brand-red'
   238→                          : 'text-warm-800 dark:text-warm-200'
   239→                      }`}
   240→                    >
   241→                      <div
   242→                        className="w-7 h-7 rounded-lg flex items-center justify-center text-[10px] font-bold text-white shrink-0"
   243→                        style={{ backgroundColor: team.color || '#64748B' }}
   244→                      >
   245→                        {(team.shortName || team.name.slice(0, 2)).toUpperCase()}
   246→                      </div>
   247→                      <div className="text-left min-w-0">
   248→                        <p className="font-semibold text-xs truncate">{team.name}</p>
   249→                        {team.shortName && (
   250→                          <p className="text-[10px] text-warm-400">{team.shortName}</p>
   251→                        )}
   252→                      </div>
   253→                    </button>
   254→                  ))
   255→                )}
   256→              </div>
   257→            </motion.div>
   258→          )}
   259→        </AnimatePresence>
   260→      </div>
   261→    </div>
   262→  );
   263→}
