   238→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   239→  const [count, setCount] = useState(0);
   240→  const prevTarget = useRef(target);
   241→
   242→  useEffect(() => {
   243→    if (prevTarget.current === target && count !== 0) return;
   244→    prevTarget.current = target;
   245→
   246→    let start = 0;
   247→    const startTime = performance.now();
   248→
   249→    function step(currentTime: number) {
   250→      const elapsed = currentTime - startTime;
   251→      const progress = Math.min(elapsed / duration, 1);
   252→      // ease-out cubic
   253→      const eased = 1 - Math.pow(1 - progress, 3);
   254→      const current = Math.round(eased * target);
   255→      setCount(current);
   256→      if (progress < 1) {
   257→        requestAnimationFrame(step);
   258→      }
   259→    }
   260→
   261→    requestAnimationFrame(step);
   262→  }, [target, duration]);
   263→
   264→  return <span>{count}</span>;
   265→}

   317→// ─── Countdown Timer ────────────────────────────────────────────────

   319→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   320→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   321→  const [expired, setExpired] = useState(!targetDate);
   322→
   323→  useEffect(() => {
   324→    if (!targetDate) {
   325→      return;
   326→    }
   327→
   328→    function calculate() {
   329→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   330→      if (diff === 0) {
   331→        setExpired(true);
   332→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   333→        return true;
   334→      }
   335→      setExpired(false);
   336→      setTimeLeft({
   337→        d: Math.floor(diff / 86400000),
   338→        h: Math.floor((diff % 86400000) / 3600000),
   339→        m: Math.floor((diff % 3600000) / 60000),
   340→        s: Math.floor((diff % 60000) / 1000),
   341→      });
   342→      return false;
   343→    }
   344→
   345→    const done = calculate();
   346→    if (done) return;
   347→
   348→    const id = setInterval(() => {
   349→      const finished = calculate();
   350→      if (finished) clearInterval(id);
   351→    }, 1000);
   352→
   353→    return () => clearInterval(id);
   354→  }, [targetDate]);
   355→
   356→  if (expired) {
   357→    return (
   358→      <motion.div
   359→        className="flex items-center justify-center gap-1.5 py-1"
   360→        animate={{ opacity: [1, 0.5, 1] }}
   361→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   362→      >
   363→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   364→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   365→      </motion.div>
   366→    );
   367→  }
   368→
   369→  if (!timeLeft) return null;
   370→
   371→  const units = [
   372→    { value: timeLeft.d, label: 'D' },
   373→    { value: timeLeft.h, label: 'H' },
   374→    { value: timeLeft.m, label: 'M' },
   375→    { value: timeLeft.s, label: 'S' },
   376→  ];
   377→
   378→  return (
   379→    <div className="flex items-center justify-center gap-1.5">
   380→      {units.map((unit, idx) => (
   381→        <div key={unit.label} className="flex items-center gap-1.5">
   382→          <div className="flex flex-col items-center">
   383→            <motion.div
   384→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   385→              key={`${unit.label}-${unit.value}`}
   386→              initial={{ rotateX: -90, opacity: 0 }}
   387→              animate={{ rotateX: 0, opacity: 1 }}
   388→              transition={{ duration: 0.3, ease: 'easeOut' }}
   389→            >
   390→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums">
   391→                {String(unit.value).padStart(2, '0')}
   392→              </span>
   393→            </motion.div>
   394→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   395→              {unit.label}
   396→            </span>
   397→          </div>
   398→          {idx < units.length - 1 && (
   399→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   400→          )}
   401→        </div>
   402→      ))}
   403→    </div>
   404→  );
   405→}
