    79→/** Animated status indicator dot */
    80→function StatusIndicator({ status }: { status: string }) {
    81→  const config = {
    82→    ongoing: { color: 'bg-emerald-500', ring: 'ring-emerald-500/30', pulse: true },
    83→    upcoming: { color: 'bg-amber-500', ring: 'ring-amber-500/30', pulse: false },
    84→    past: { color: 'bg-warm-400', ring: 'ring-warm-400/30', pulse: false },
    85→  }[status] || { color: 'bg-warm-400', ring: 'ring-warm-400/30', pulse: false };
    86→
    87→  return (
    88→    <span className="relative flex h-3 w-3">
    89→      {config.pulse && (
    90→        <span className={`animate-ping absolute inline-flex h-full w-full rounded-full ${config.color} opacity-75`} />
    91→      )}
    92→      <span className={`relative inline-flex rounded-full h-3 w-3 ${config.color} ring-2 ${config.ring}`} />
    93→    </span>
    94→  );
    95→}

   132→/** Knockout bracket view */
   133→function BracketView({ tournament }: { tournament: Tournament }) {
   134→  const [bracketMatches, setBracketMatches] = useState<BracketMatch[]>([]);
   135→  const [loadingBracket, setLoadingBracket] = useState(false);
   136→
   137→  useEffect(() => {
   138→    const fetchBracket = async () => {
   139→      setLoadingBracket(true);
   140→      try {
   141→        const res = await fetch(`/api/tournaments/${tournament.id}/matches`);
   142→        if (res.ok) {
   143→          const data = await res.json();
   144→          const matches: BracketMatch[] = (data.matches || []).map((m: Record<string, unknown>) => ({
   145→            id: m.id as string,
   146→            round: (m.round as number) || 1,
   147→            position: (m.position as number) || 0,
   148→            team1Id: (m.team1Id as string) || null,
   149→            team2Id: (m.team2Id as string) || null,
   150→            team1Score: (m.team1Score as number) ?? null,
   151→            team2Score: (m.team2Score as number) ?? null,
   152→            winnerId: (m.winnerId as string) || null,
   153→            status: (m.status as string) || 'scheduled',
   154→          }));
   155→          setBracketMatches(matches);
   156→        }
   157→      } catch {
   158→        // silently fail
   159→      } finally {
   160→        setLoadingBracket(false);
   161→      }
   162→    };
   163→    fetchBracket();
   164→  }, [tournament.id]);
   165→
   166→  const teamsMap = new Map(tournament.teams.map(t => [t.id, t]));
   167→
   168→  // Group matches by round
   169→  const rounds = new Map<number, BracketMatch[]>();
   170→  bracketMatches.forEach(m => {
   171→    const list = rounds.get(m.round) || [];
   172→    list.push(m);
   173→    rounds.set(m.round, list);
   174→  });
   175→  const sortedRounds = Array.from(rounds.entries()).sort((a, b) => a[0] - b[0]);
   176→
   177→  // If knockout with no matches yet, show placeholder bracket
   178→  const showPlaceholder = bracketMatches.length === 0 && !loadingBracket;
   179→
   180→  // Generate placeholder bracket based on team count
   181→  const generatePlaceholderRounds = () => {
   182→    const teamCount = tournament.teams.length;
   183→    if (teamCount < 2) return [];
   184→    const roundsCount = Math.ceil(Math.log2(teamCount));
   185→    const result: { round: number; matches: { team1: string | null; team2: string | null }[] }[] = [];
   186→    let currentMatches = Math.floor(teamCount / 2);
   187→
   188→    for (let r = 1; r <= roundsCount; r++) {
   189→      const matches = [];
   190→      for (let m = 0; m < currentMatches; m++) {
   191→        const team1Idx = m * 2;
   192→        const team2Idx = m * 2 + 1;
   193→        matches.push({
   194→          team1: tournament.teams[team1Idx]?.id || null,
   195→          team2: team2Idx < tournament.teams.length ? tournament.teams[team2Idx]?.id || null : null,
   196→        });
   197→      }
   198→      result.push({ round: r, matches });
   199→      currentMatches = Math.ceil(currentMatches / 2);
   200→    }
   201→    return result;
   202→  };
   203→
   204→  const placeholderRounds = showPlaceholder ? generatePlaceholderRounds() : [];
   205→
   206→  const getRoundName = (round: number, totalRounds: number) => {
   207→    if (round === totalRounds) return 'Final';
   208→    if (round === totalRounds - 1) return 'Semi-Final';
   209→    if (round === totalRounds - 2) return 'Quarter-Final';
   210→    return `Round ${round}`;
   211→  };
   212→
   213→  return (
   214→    <motion.div
   215→      initial={{ opacity: 0, y: 20 }}
   216→      animate={{ opacity: 1, y: 0 }}
   217→      transition={{ duration: 0.5, ease: 'easeOut' }}
   218→    >
   219→      <div className="flex items-center gap-2 mb-3">
   220→        <Swords className="w-4 h-4 text-brand-red" />
   221→        <h4 className="text-xs font-bold text-warm-500 dark:text-warm-400 uppercase tracking-wider">
   222→          {tournament.type === 'knockout' ? 'Knockout Bracket' : tournament.type === 'league' ? 'League Fixtures' : 'Tournament Bracket'}
   223→        </h4>
   224→      </div>
   225→
   226→      {loadingBracket ? (
   227→        <div className="flex items-center justify-center py-8 gap-2">
   228→          <Loader2 className="w-5 h-5 animate-spin text-brand-red" />
   229→          <span className="text-sm text-warm-400">Loading bracket...</span>
   230→        </div>
   231→      ) : (
   232→        <div className="overflow-x-auto pb-2 -mx-1">
   233→          <div className="flex gap-6 min-w-max px-1">
   234→            {/* Real bracket from API data */}
   235→            {sortedRounds.length > 0 ? sortedRounds.map(([roundNum, matches], rIdx) => {
   236→              const totalRounds = sortedRounds.length;
   237→              return (
   238→                <motion.div
   239→                  key={roundNum}
   240→                  className="flex flex-col gap-3 min-w-[180px]"
   241→                  initial={{ opacity: 0, x: 30 }}
   242→                  animate={{ opacity: 1, x: 0 }}
   243→                  transition={{ delay: rIdx * 0.15, duration: 0.4 }}
   244→                >
   245→                  <div className="text-center">
   246→                    <Badge className="bg-brand-red/10 text-brand-red text-[10px] border-0 font-bold">
   247→                      {getRoundName(roundNum, totalRounds)}
   248→                    </Badge>
   249→                  </div>
   250→                  <div className="flex flex-col justify-around flex-1 gap-2">
   251→                    {matches.sort((a, b) => a.position - b.position).map((match) => {
   252→                      const team1 = match.team1Id ? teamsMap.get(match.team1Id) : null;
   253→                      const team2 = match.team2Id ? teamsMap.get(match.team2Id) : null;
   254→                      const isLive = match.status === 'live' || match.status === 'ongoing';
   255→                      const isCompleted = match.status === 'completed';
   256→                      return (
   257→                        <motion.div
   258→                          key={match.id}
   259→                          className={`rounded-xl border p-2.5 text-xs transition-all ${
   260→                            isLive
   261→                              ? 'border-brand-red bg-brand-red/5 dark:bg-brand-red/10 ring-1 ring-brand-red/30'
   262→                              : isCompleted
   263→                                ? 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/50'
   264→                                : 'border-warm-200 dark:border-warm-700 bg-white dark:bg-warm-800/30'
   265→                          }`}
   266→                          initial={{ opacity: 0, scale: 0.9 }}
   267→                          animate={{ opacity: 1, scale: 1 }}
   268→                          transition={{ delay: rIdx * 0.1 + 0.1 }}
   269→                        >
   270→                          {isLive && (
   271→                            <div className="flex items-center gap-1 mb-1">
   272→                              <Radio className="w-2.5 h-2.5 text-brand-red animate-pulse" />
   273→                              <span className="text-[9px] font-bold text-brand-red uppercase">Live</span>
   274→                            </div>
   275→                          )}
   276→                          {/* Team 1 */}
   277→                          <div className={`flex items-center gap-2 py-1 ${match.winnerId === match.team1Id ? 'font-bold' : ''}`}>
   278→                            {team1 ? (
   279→                              <>
   280→                                <div className="w-5 h-5 rounded-md flex items-center justify-center text-[8px] font-bold text-white shrink-0" style={{ backgroundColor: team1.color }}>
   281→                                  {team1.shortName?.charAt(0) || team1.name.charAt(0)}
   282→                                </div>
   283→                                <span className={`truncate flex-1 ${match.winnerId === match.team1Id ? 'text-warm-800 dark:text-warm-100' : 'text-warm-600 dark:text-warm-300'}`}>
   284→                                  {team1.name}
   285→                                </span>
   286→                              </>
   287→                            ) : (
   288→                              <span className="text-warm-400 italic">TBD</span>
   289→                            )}
   290→                            {match.team1Score !== null && (
   291→                              <span className={`font-mono font-bold ${match.winnerId === match.team1Id ? 'text-brand-green' : 'text-warm-500'}`}>
   292→                                {match.team1Score}
   293→                              </span>
   294→                            )}
   295→                          </div>
   296→                          <div className="border-t border-warm-100 dark:border-warm-700/50 my-0.5" />
   297→                          {/* Team 2 */}
   298→                          <div className={`flex items-center gap-2 py-1 ${match.winnerId === match.team2Id ? 'font-bold' : ''}`}>
   299→                            {team2 ? (
   300→                              <>
   301→                                <div className="w-5 h-5 rounded-md flex items-center justify-center text-[8px] font-bold text-white shrink-0" style={{ backgroundColor: team2.color }}>
   302→                                  {team2.shortName?.charAt(0) || team2.name.charAt(0)}
   303→                                </div>
   304→                                <span className={`truncate flex-1 ${match.winnerId === match.team2Id ? 'text-warm-800 dark:text-warm-100' : 'text-warm-600 dark:text-warm-300'}`}>
   305→                                  {team2.name}
   306→                                </span>
   307→                              </>
   308→                            ) : (
   309→                              <span className="text-warm-400 italic">TBD</span>
   310→                            )}
   311→                            {match.team2Score !== null && (
   312→                              <span className={`font-mono font-bold ${match.winnerId === match.team2Id ? 'text-brand-green' : 'text-warm-500'}`}>
   313→                                {match.team2Score}
   314→                              </span>
   315→                            )}
   316→                          </div>
   317→                          {match.winnerId && (
   318→                            <div className="flex items-center gap-1 mt-1 pt-1 border-t border-warm-100 dark:border-warm-700/30">
   319→                              <Trophy className="w-2.5 h-2.5 text-brand-gold" />
   320→                              <span className="text-[9px] text-brand-gold font-bold">
   321→                                {teamsMap.get(match.winnerId)?.name || 'Winner'}
   322→                              </span>
   323→                            </div>
   324→                          )}
   325→                        </motion.div>
   326→                      );
   327→                    })}
   328→                  </div>
   329→                </motion.div>
   330→              );
   331→            }) : null}
   332→
   333→            {/* Placeholder bracket */}
   334→            {placeholderRounds.length > 0 ? placeholderRounds.map((round, rIdx) => {
   335→              const totalRounds = placeholderRounds.length;
   336→              return (
   337→                <motion.div
   338→                  key={`placeholder-${round.round}`}
   339→                  className="flex flex-col gap-3 min-w-[160px]"
   340→                  initial={{ opacity: 0, x: 30 }}
   341→                  animate={{ opacity: 1, x: 0 }}
   342→                  transition={{ delay: rIdx * 0.15, duration: 0.4 }}
   343→                >
   344→                  <div className="text-center">
   345→                    <Badge className="bg-warm-200 dark:bg-warm-700 text-warm-500 dark:text-warm-400 text-[10px] border-0 font-bold">
   346→                      {getRoundName(round.round, totalRounds)}
   347→                    </Badge>
   348→                  </div>
   349→                  <div className="flex flex-col justify-around flex-1 gap-2">
   350→                    {round.matches.map((match, mIdx) => {
   351→                      const team1 = match.team1 ? teamsMap.get(match.team1) : null;
   352→                      const team2 = match.team2 ? teamsMap.get(match.team2) : null;
   353→                      return (
   354→                        <div key={mIdx} className="rounded-xl border border-dashed border-warm-300 dark:border-warm-600 p-2.5 text-xs bg-warm-50 dark:bg-warm-800/20">
   355→                          <div className="flex items-center gap-2 py-0.5">
   356→                            {team1 ? (
   357→                              <>
   358→                                <div className="w-5 h-5 rounded-md flex items-center justify-center text-[8px] font-bold text-white shrink-0" style={{ backgroundColor: team1.color }}>
   359→                                  {team1.shortName?.charAt(0) || team1.name.charAt(0)}
   360→                                </div>
   361→                                <span className="truncate text-warm-500 dark:text-warm-400">{team1.name}</span>
   362→                              </>
   363→                            ) : (
   364→                              <span className="text-warm-400 italic">TBD</span>
   365→                            )}
   366→                          </div>
   367→                          <div className="text-warm-300 dark:text-warm-600 text-center py-0.5 text-[10px]">VS</div>
   368→                          <div className="flex items-center gap-2 py-0.5">
   369→                            {team2 ? (
   370→                              <>
   371→                                <div className="w-5 h-5 rounded-md flex items-center justify-center text-[8px] font-bold text-white shrink-0" style={{ backgroundColor: team2.color }}>
   372→                                  {team2.shortName?.charAt(0) || team2.name.charAt(0)}
   373→                                </div>
   374→                                <span className="truncate text-warm-500 dark:text-warm-400">{team2.name}</span>
   375→                              </>
   376→                            ) : (
   377→                              <span className="text-warm-400 italic">BYE</span>
   378→                            )}
   379→                          </div>
   380→                        </div>
   381→                      );
   382→                    })}
   383→                  </div>
   384→                </motion.div>
   385→              );
   386→            }) : null}
   387→
   388→            {/* No data at all */}
   389→            {sortedRounds.length === 0 && placeholderRounds.length === 0 && !loadingBracket && (
   390→              <div className="text-center py-6 w-full">
   391→                <Target className="w-8 h-8 text-warm-300 dark:text-warm-600 mx-auto mb-2" />
   392→                <p className="text-sm text-warm-400 dark:text-warm-500">No bracket generated yet</p>
   393→                <p className="text-[10px] text-warm-400 dark:text-warm-500 mt-1">Add teams and generate a bracket to see the tournament tree</p>
   394→              </div>
   395→            )}
   396→          </div>
   397→        </div>
   398→      )}
   399→    </motion.div>
   400→  );
   401→}
