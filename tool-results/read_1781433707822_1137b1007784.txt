   240→   240→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   241→   241→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   242→   242→  let homeScore = 0;
   243→   243→  let awayScore = 0;
   244→   244→  let homeOutPlayers = 0;
   245→   245→  let awayOutPlayers = 0;
   246→   246→  let homeOutPlayerIds: string[] = [];
   247→   247→  let awayOutPlayerIds: string[] = [];
   248→   248→
   249→   249→  for (const evt of events) {
   250→   250→    const side = getTeamSide(match, evt.teamId);
   251→   251→    const points = evt.value;
   252→   252→    if (side === 'home') {
   253→   253→      homeScore += points;
   254→   254→    } else {
   255→   255→      awayScore += points;
   256→   256→    }
   257→   257→
   258→   258→    // Count-based out tracking (backward compat)
   259→   259→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   260→   260→      if (side === 'home') {
   261→   261→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   262→   262→      } else {
   263→   263→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   264→   264→      }
   265→   265→    }
   266→   266→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   267→   267→      if (side === 'home') {
   268→   268→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   269→   269→      } else {
   270→   270→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   271→   271→      }
   272→   272→    }
   273→   273→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   274→   274→    if (evt.eventType === 'self_out') {
   275→   275→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   276→   276→      if (side === 'home') {
   277→   277→        // Home scored from self-out → away defender went out
   278→   278→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   279→   279→        // Home revives one player
   280→   280→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   281→   281→      } else {
   282→   282→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   283→   283→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   284→   284→      }
   285→   285→    }
   286→   286→    if (evt.eventType === 'all_out') {
   287→   287→      if (side === 'home') {
   288→   288→        awayOutPlayers = 0;
   289→   289→      } else {
   290→   290→        homeOutPlayers = 0;
   291→   291→      }
   292→   292→    }
   293→   293→
   294→   294→    // ID-based out tracking
   295→   295→    let details: Record<string, unknown> = {};
   296→   296→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   297→   297→
   298→   298→    if (evt.eventType === 'raid_point') {
   299→   299→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   300→   300→      if (side === 'home') {
   301→   301→        // Home team scored raid → away defenders go out
   302→   302→        for (const tid of touchedIds) {
   303→   303→          if (!awayOutPlayerIds.includes(tid)) {
   304→   304→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   305→   305→          }
   306→   306→        }
   307→   307→        // Home team revives players (first-out-first-in)
   308→   308→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   309→   309→        if (reviveCount > 0) {
   310→   310→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   311→   311→        }
   312→   312→      } else {
   313→   313→        // Away team scored raid → home defenders go out
   314→   314→        for (const tid of touchedIds) {
   315→   315→          if (!homeOutPlayerIds.includes(tid)) {
   316→   316→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   317→   317→          }
   318→   318→        }
   319→   319→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   320→   320→        if (reviveCount > 0) {
   321→   321→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   322→   322→        }
   323→   323→      }
   324→   324→    }
   325→   325→
   326→   326→    if (evt.eventType === 'bonus_point') {
   327→   327→      // One out player from scoring team revives
   328→   328→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   329→   329→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   330→   330→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   331→   331→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   332→   332→      }
   333→   333→    }
   334→   334→
   335→   335→    // Self-out: defender steps off mat during a live raid
   336→   336→    if (evt.eventType === 'self_out') {
   337→   337→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   338→   338→      if (selfOutPlayerId) {
   339→   339→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   340→   340→        if (side === 'home') {
   341→   341→          // Home scored → away defender self-out
   342→   342→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   343→   343→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   344→   344→          }
   345→   345→          // Home revives first-out player
   346→   346→          if (homeOutPlayerIds.length > 0) {
   347→   347→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   348→   348→          }
   349→   349→        } else {
   350→   350→          // Away scored → home defender self-out
   351→   351→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   352→   352→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   353→   353→          }
   354→   354→          if (awayOutPlayerIds.length > 0) {
   355→   355→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   356→   356→          }
   357→   357→        }
   358→   358→      }
   359→   359→    }
   360→   360→
   361→   361→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   362→   362→      // Raider goes out. The tackle is scored by the DEFENDING team.
   363→   363→      // So the raider is from the OPPOSING team.
   364→   364→      const raiderId = (details.raiderId as string) || evt.playerId;
   365→   365→      if (side === 'home' && raiderId) {
   366→   366→        // Home team scored tackle → away raider was caught → away raider goes out
   367→   367→        if (!awayOutPlayerIds.includes(raiderId)) {
   368→   368→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   369→   369→        }
   370→   370→      } else if (side === 'away' && raiderId) {
   371→   371→        // Away team scored tackle → home raider was caught → home raider goes out
   372→   372→        if (!homeOutPlayerIds.includes(raiderId)) {
   373→   373→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   374→   374→        }
   375→   375→      }
   376→   376→    }
   377→   377→
   378→   378→    if (evt.eventType === 'all_out') {
   379→   379→      if (side === 'home') {
   380→   380→        awayOutPlayerIds = [];
   381→   381→      } else {
   382→   382→        homeOutPlayerIds = [];
   383→   383→      }
   384→   384→    }
   385→   385→  }
   386→   386→
   387→   387→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   388→   388→}

   390→   390→/** Event types that constitute a complete raid and should flip the turn */
   391→   391→const RAID_ENDING_EVENT_TYPES: string[] = [
   392→   392→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   393→   393→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   394→   394→];
   395→   395→
   396→   396→/** Event types that do NOT flip the turn (non-raid events) */
   397→   397→const NON_RAID_EVENT_TYPES: string[] = [
   398→   398→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   399→   399→];
   400→   400→
   401→   401→/**
   402→   402→ * Event types where the SCORING team is the DEFENDING team.
   403→   403→ * In these cases, the defending team scored (they made the tackle / raider failed do-or-die),
   404→   404→ * so they should raid NEXT — the turn goes to the SAME side that scored.
   405→   405→ *
   406→   406→ * For all other raid-ending events (raid_point, bonus_point, super_raid, empty_raid, self_out),
   407→   407→ * the scoring team is the RAIDING team, so the turn goes to the OPPOSITE side.
   408→   408→ */
   409→   409→const DEFENDING_TEAM_SCORES: Set<string> = new Set([
   410→   410→  'tackle_point',
   411→   411→  'super_tackle',
   412→   412→  'do_or_die_raid',
   413→   413→]);
   414→   414→
   415→   415→/** Helper: determine raid queue from last raid-ending event */
   416→   416→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   417→   417→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   418→   418→  if (lastRaidEvent) {
   419→   419→    const side = getTeamSide(match, lastRaidEvent.teamId);
   420→   420→    // If the defending team scored (tackle/super_tackle/do_or_die_raid),
   421→   421→    // THEY raid next — return the SAME side as the scoring team.
   422→   422→    if (DEFENDING_TEAM_SCORES.has(lastRaidEvent.eventType)) {
   423→   423→      return side;
   424→   424→    }
   425→   425→    // If the raiding team scored (touch points, bonus, super raid, etc.)
   426→   426→    // or it was an empty raid, the DEFENDING team raids next — return the OPPOSITE side.
   427→   427→    return side === 'home' ? 'away' : 'home';
   428→   428→  }
   429→   429→  return match.raidQueue;
   430→   430→}
