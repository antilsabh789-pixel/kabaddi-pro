    87→    // ignore storage errors
    88→  }

   196→// ─── Confetti Component ───
   197→
   198→function ConfettiEffect() {
   199→  const particles = useMemo(() =>
   200→    Array.from({ length: 30 }, (_, i) => ({
   201→      id: i,
   202→      left: Math.random() * 100,
   203→      delay: Math.random() * 0.5,
   204→      duration: 1.5 + Math.random() * 1.5,
   205→      color: ['#DC2626', '#F59E0B', '#14B8A6', '#EC4899', '#8B5CF6', '#22C55E'][Math.floor(Math.random() * 6)],
   206→      size: 4 + Math.random() * 6,
   207→      rotation: Math.random() * 360,
   208→    })),
   209→  []);
   210→
   211→  return (
   212→    <div className="fixed inset-0 pointer-events-none z-50">
   213→      {particles.map((p) => (
   214→        <motion.div
   215→          key={p.id}
   216→          initial={{ opacity: 1, y: -20, x: `${p.left}vw`, rotate: 0 }}
   217→          animate={{
   218→            opacity: 0,
   219→            y: '100vh',
   220→            rotate: p.rotation + 720,
   221→            x: `${p.left + (Math.random() - 0.5) * 20}vw`,
   222→          }}
   223→          transition={{ duration: p.duration, delay: p.delay, ease: 'easeOut' }}
   224→          className="absolute rounded-sm"
   225→          style={{
   226→            width: p.size,
   227→            height: p.size,
   228→            backgroundColor: p.color,
   229→            left: `${p.left}%`,
   230→          }}
   231→        />
   232→      ))}
   233→    </div>
   234→  );
   235→}

   244→  const [predictions, setPredictions] = useState<UserPrediction[]>(() => loadPredictions());
   245→  const [localPicks, setLocalPicks] = useState<Record<string, 'teamA' | 'draw' | 'teamB'>>({});
   246→  const [activeTab, setActiveTab] = useState('predict');
   247→  const [historyFilter, setHistoryFilter] = useState<'all' | 'correct' | 'incorrect'>('all');
   248→  const [showConfetti, setShowConfetti] = useState(false);
   249→  const [revealingMatch, setRevealingMatch] = useState<string | null>(null);
   250→
   251→  // Calculate stats
   252→  const stats = useMemo(() => {
   253→    const completedPreds = predictions.filter((p) => {
   254→      const match = COMPLETED_MATCHES.find((m) => m.id === p.matchId);
   255→      return match?.status === 'completed';
   256→    });
   257→
   258→    const correctCount = completedPreds.filter((p) => {
   259→      const match = COMPLETED_MATCHES.find((m) => m.id === p.matchId);
   260→      return match?.result === p.pick;
   261→    }).length;
   262→
   263→    const totalPoints = completedPreds.reduce((sum, p) => {
   264→      const match = COMPLETED_MATCHES.find((m) => m.id === p.matchId);
   265→      if (!match?.result) return sum;
   266→      const correct = match.result === p.pick;
   267→      if (!correct) return sum;
   268→
   269→      let points = 10; // Base points for correct winner
   270→
   271→      // Bonus for draw prediction
   272→      if (p.pick === 'draw') points = 25;
   273→
   274→      // Streak bonus
   275→      return sum + points;
   276→    }, 0);
   277→
   278→    // Calculate streak
   279→    let streak = 0;
   280→    const sortedCompleted = [...completedPreds].sort((a, b) => b.timestamp - a.timestamp);
   281→    for (const p of sortedCompleted) {
   282→      const match = COMPLETED_MATCHES.find((m) => m.id === p.matchId);
   283→      if (match?.result === p.pick) {
   284→        streak++;
   285→      } else {
   286→        break;
   287→      }
   288→    }
   289→
   290→    // Add streak bonus
   291→    const streakBonus = streak >= 3 ? (streak - 2) * 5 : 0;
   292→
   293→    return {
   294→      totalPoints: totalPoints + streakBonus,
   295→      totalPredictions: completedPreds.length,
   296→      correctCount,
   297→      accuracy: completedPreds.length > 0 ? Math.round((correctCount / completedPreds.length) * 100) : 0,
   298→      streak,
   299→    };
   300→  }, [predictions]);
   301→
   302→  // Build prediction history
   303→  const predictionHistory = useMemo((): PredictionHistoryItem[] => {
   304→    const history: PredictionHistoryItem[] = [];
   305→    for (const pred of predictions) {
   306→      const allMatches = [...UPCOMPING_MATCHES, ...COMPLETED_MATCHES];
   307→      const match = allMatches.find((m) => m.id === pred.matchId);
   308→      if (!match) continue;
   309→
   310→      const isCompleted = match.status === 'completed';
   311→      const result = isCompleted ? match.result ?? null : null;
   312→      const correct = result !== null ? result === pred.pick : null;
   313→      const points = correct ? (pred.pick === 'draw' ? 25 : 10) : 0;
   314→
   315→      history.push({
   316→        matchId: pred.matchId,
   317→        teamA: match.teamA.short,
   318→        teamB: match.teamB.short,
   319→        pick: pred.pick,
   320→        result,
   321→        points,
   322→        correct,
   323→        timestamp: pred.timestamp,
   324→      });
   325→    }
   326→    return history.sort((a, b) => b.timestamp - a.timestamp);
   327→  }, [predictions]);
   328→
   329→  const filteredHistory = useMemo(() => {
   330→    if (historyFilter === 'all') return predictionHistory;
   331→    if (historyFilter === 'correct') return predictionHistory.filter((h) => h.correct === true);
   332→    return predictionHistory.filter((h) => h.correct === false);
   333→  }, [predictionHistory, historyFilter]);
   334→
   335→  // Updated leaderboard with user stats
   336→  const updatedLeaderboard = useMemo(() => {
   337→    const lb = LEADERBOARD.map((entry) => {
   338→      if (entry.isUser) {
   339→        return {
   340→          ...entry,
   341→          points: stats.totalPoints,
   342→          accuracy: stats.accuracy,
   343→          predictions: stats.totalPredictions,
   344→        };
   345→      }
   346→      return entry;
   347→    });
   348→    return lb.sort((a, b) => b.points - a.points);
   349→  }, [stats]);
   350→
   351→  const userRank = updatedLeaderboard.findIndex((e) => e.isUser) + 1;
   352→
   353→  // Submit prediction
   354→  const handleSubmitPrediction = useCallback(
   355→    (matchId: string) => {
   356→      const pick = localPicks[matchId];
   357→      if (!pick) return;
   358→
   359→      const newPrediction: UserPrediction = {
   360→        matchId,
   361→        pick,
   362→        timestamp: Date.now(),
   363→      };
   364→
   365→      const updated = [...predictions.filter((p) => p.matchId !== matchId), newPrediction];
   366→      setPredictions(updated);
   367→      savePredictions(updated);
   368→      setLocalPicks((prev) => {
   369→        const next = { ...prev };
   370→        delete next[matchId];
   371→        return next;
   372→      });
   373→    },
   374→    [localPicks, predictions],
   375→  );
   376→
   377→  // Reveal result animation
   378→  const handleReveal = useCallback(
   379→    (matchId: string) => {
   380→      setRevealingMatch(matchId);
   381→      setTimeout(() => {
   382→        const pred = predictions.find((p) => p.matchId === matchId);
   383→        const match = COMPLETED_MATCHES.find((m) => m.id === matchId);
   384→        if (pred && match?.result === pred.pick) {
   385→          setShowConfetti(true);
   386→          setTimeout(() => setShowConfetti(false), 3000);
   387→        }
   388→        setRevealingMatch(null);
   389→      }, 800);
   390→    },
   391→    [predictions],
   392→  );
   393→
   394→  // Get prediction for match
   395→  const getPrediction = useCallback(
   396→    (matchId: string) => predictions.find((p) => p.matchId === matchId),
   397→    [predictions],
   398→  );
   399→
   400→  return (
   401→    <div className="fixed inset-0 z-40 bg-warm-50 dark:bg-warm-900 overflow-y-auto custom-scrollbar">

   412→          >
   413→            <ArrowLeft className="w-4 h-4" />
   414→          </Button>
