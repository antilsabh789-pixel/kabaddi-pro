    33→function ConfettiParticle({ delay, color, index }: { delay: number; color: string; index: number }) {
    34→  const leftPos = (index * 17 + 13) % 100;
    35→  const rotateEnd = (index * 47 + 23) % 720 - 360;
    36→  const xDrift = ((index * 31) % 80) - 40;
    37→
    38→  return (
    39→    <motion.div
    40→      initial={{ y: -20, x: 0, rotate: 0, opacity: 1 }}
    41→      animate={{ y: '100vh', x: xDrift, rotate: rotateEnd, opacity: 0 }}
    42→      transition={{ duration: 3 + (index % 3), delay, ease: 'easeIn' }}
    43→      className="absolute top-0 w-2 h-2 rounded-sm"
    44→      style={{ left: `${leftPos}%`, backgroundColor: color }}
    45→    />
    46→  );
    47→}

   404→function EventLogEntry({ event, matchInfo }: { event: MatchEvent; matchInfo: { homeTeamId: string; homeTeamColor: string; awayTeamColor: string } }) {
   405→  const isHome = event.teamId === matchInfo.homeTeamId;
   406→  const teamColor = isHome ? matchInfo.homeTeamColor : matchInfo.awayTeamColor;
   407→
   408→  const eventIcons: Record<string, string> = {
   409→    raid_point: '✅',
   410→    bonus_point: '🎯',
   411→    tackle_point: '🛡',
   412→    super_tackle: '⚡',
   413→    super_raid: '🔥',
   414→    do_or_die_raid: '🔥',
   415→    all_out: '💥',
   416→    empty_raid: '⏭',
   417→    substitution: '🔄',
   418→    timeout: '📋',
   419→    yellow_card: '🟨',
   420→    red_card: '🟥',
   421→    green_card: '🟩',
   422→  };
   423→
   424→  const eventLabels: Record<string, string> = {
   425→    raid_point: 'Raid Pt',
   426→    bonus_point: 'Bonus Pt',
   427→    tackle_point: 'Tackle Pt',
   428→    super_tackle: 'Super Tackle',
   429→    super_raid: 'Super Raid',
   430→    do_or_die_raid: 'Do-or-Die',
   431→    all_out: 'All Out',
   432→    empty_raid: 'Empty',
   433→    substitution: 'Sub',
   434→    timeout: 'Timeout',
   435→    yellow_card: 'Yellow Card',
   436→    red_card: 'Red Card',
   437→    green_card: 'Green Card',
   438→  };
   439→
   440→  const timeStr = new Date(event.timestamp).toLocaleTimeString([], { hour: '2-digit', minute: '2-digit', second: '2-digit' });
   441→
   442→  return (
   443→    <motion.div
   444→      initial={{ opacity: 0, x: -8 }}
   445→      animate={{ opacity: 1, x: 0 }}
   446→      className="flex items-center gap-2 py-1 px-2 rounded-lg hover:bg-gray-800/30 dark:hover:bg-gray-700/30 transition-colors"

   451→      <span className="text-xs">{eventIcons[event.eventType] || '📌'}</span>
   452→      <span className="text-[10px] font-semibold text-gray-500 dark:text-gray-400 flex-1">{eventLabels[event.eventType] || event.eventType}</span>
   453→      {event.value > 0 && (
   454→        <span className="text-[10px] font-black px-1.5 py-0.5 rounded-md" style={{ backgroundColor: `${teamColor}20`, color: teamColor }}>
   455→          +{event.value}
   456→        </span>
   457→      )}
   458→    </motion.div>
   459→  );
   460→}
