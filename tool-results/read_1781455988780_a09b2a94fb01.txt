   241→/** Helper: recalculate scores, out counts, and out player IDs from a list of events
   242→ *  Implements proper Pro Kabaddi rules:
   243→ *  - Revival: 1 point = 1 player revived from front of out queue (FIFO)
   244→ *  - All-Out: When all 7 defenders are out, +2 points to raiding team, all defenders revive
   245→ *  - Tackle: Raider goes out, defending team revives 1 player per tackle point
   246→ *  - Do-or-Die: 2 consecutive empty raids → next raid for that team is do-or-die
   247→ */
   248→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   249→  let homeScore = 0;
   250→  let awayScore = 0;
   251→  let homeOutPlayers = 0;
   252→  let awayOutPlayers = 0;
   253→  let homeOutPlayerIds: string[] = [];
   254→  let awayOutPlayerIds: string[] = [];
   255→
   256→  // Track consecutive empty raids per team for do-or-die
   257→  const emptyRaidCount: Record<string, number> = {};
   258→
   259→  for (const evt of events) {
   260→    const side = getTeamSide(match, evt.teamId);
   261→    const points = evt.value;
   262→
   263→    // Add points to score
   264→    if (side === 'home') {
   265→      homeScore += points;
   266→    } else {
   267→      awayScore += points;
   268→    }
   269→
   270→    let details: Record<string, unknown> = {};
   271→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   272→
   273→    // ─── RAID POINT: Raider scores touch points ───
   274→    if (evt.eventType === 'raid_point') {
   275→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   276→      const scoringSide = side;
   277→
   278→      // Defenders go to out queue
   279→      const defendingOutIds = scoringSide === 'home' ? awayOutPlayerIds : homeOutPlayerIds;
   280→      const defendingLineup = scoringSide === 'home' ? match.awayLineup : match.homeLineup;
   281→
   282→      for (const tid of touchedIds) {
   283→        if (!defendingOutIds.includes(tid)) {
   284→          defendingOutIds.push(tid);
   285→        }
   286→      }
   287→
   288→      if (scoringSide === 'home') {
   289→        awayOutPlayerIds = defendingOutIds;
   290→      } else {
   291→        homeOutPlayerIds = defendingOutIds;
   292→      }
   293→
   294→      // Revival: each point revives 1 player from front of scoring team's out queue
   295→      const scoringOutIds = scoringSide === 'home' ? homeOutPlayerIds : awayOutPlayerIds;
   296→      const reviveCount = Math.min(points, scoringOutIds.length);
   297→      if (reviveCount > 0) {
   298→        const remaining = scoringOutIds.slice(reviveCount);
   299→        if (scoringSide === 'home') {
   300→          homeOutPlayerIds = remaining;
   301→        } else {
   302→          awayOutPlayerIds = remaining;
   303→        }
   304→      }
   305→
   306→      // Reset empty raid counter for scoring team
   307→      emptyRaidCount[evt.teamId] = 0;
   308→
   309→      // Auto All-Out check: if all 7 defenders are out
   310→      const defendingActive = defendingLineup.slice(0, match.playersPerSide).length - defendingOutIds.length;
   311→      if (defendingActive <= 0 && defendingOutIds.length > 0) {
   312→        // Note: all_out event should follow in the event list, but if it doesn't,
   313→        // we handle it here by clearing the out queue
   314→      }
   315→    }
   316→
   317→    // ─── BONUS POINT: Raider crosses bonus line ───
   318→    if (evt.eventType === 'bonus_point') {
   319→      // Revival: 1 player from scoring team's out queue
   320→      const scoringOutIds = side === 'home' ? homeOutPlayerIds : awayOutPlayerIds;
   321→      if (scoringOutIds.length > 0) {
   322→        const remaining = scoringOutIds.slice(1);
   323→        if (side === 'home') {
   324→          homeOutPlayerIds = remaining;
   325→        } else {
   326→          awayOutPlayerIds = remaining;
   327→        }
   328→      }
   329→
   330→      // Reset empty raid counter
   331→      emptyRaidCount[evt.teamId] = 0;
   332→    }
   333→
   334→    // ─── TACKLE / SUPER TACKLE: Raider is caught ───
   335→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   336→      const raiderId = (details.raiderId as string) || evt.playerId;
   337→
   338→      // Raider goes to out queue (raider is from the OPPOSING team)
   339→      if (side === 'home' && raiderId) {
   340→        // Home scored tackle → away raider goes out
   341→        if (!awayOutPlayerIds.includes(raiderId)) {
   342→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   343→        }
   344→      } else if (side === 'away' && raiderId) {
   345→        // Away scored tackle → home raider goes out
   346→        if (!homeOutPlayerIds.includes(raiderId)) {
   347→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   348→        }
   349→      }
   350→
   351→      // Revival: defending team revives 1 player per tackle point
   352→      const tacklePoints = evt.eventType === 'super_tackle' ? 2 : 1;
   353→      const defendingOutIds = side === 'home' ? homeOutPlayerIds : awayOutPlayerIds;
   354→      const reviveCount = Math.min(tacklePoints, defendingOutIds.length);
   355→      if (reviveCount > 0) {
   356→        const remaining = defendingOutIds.slice(reviveCount);
   357→        if (side === 'home') {
   358→          homeOutPlayerIds = remaining;
   359→        } else {
   360→          awayOutPlayerIds = remaining;
   361→        }
   362→      }
   363→
   364→      // Reset empty raid counter for the defending team (they scored)
   365→      emptyRaidCount[evt.teamId] = 0;
   366→    }
   367→
   368→    // ─── SELF-OUT: Defender steps off mat during raid ───
   369→    if (evt.eventType === 'self_out') {
   370→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   371→      if (selfOutPlayerId) {
   372→        // Self-out player is from DEFENDING team (opposite of scoring team)
   373→        if (side === 'home') {
   374→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   375→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   376→          }
   377→          // Home revives 1 player
   378→          if (homeOutPlayerIds.length > 0) {
   379→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   380→          }
   381→        } else {
   382→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   383→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   384→          }
   385→          if (awayOutPlayerIds.length > 0) {
   386→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   387→          }
   388→        }
   389→      }
   390→
   391→      // Reset empty raid counter
   392→      emptyRaidCount[evt.teamId] = 0;
   393→    }
   394→
   395→    // ─── ALL OUT: All defenders eliminated ───
   396→    if (evt.eventType === 'all_out') {
   397→      // Clear the eliminated team's out queue - all players revive
   398→      if (side === 'home') {
   399→        awayOutPlayerIds = [];
   400→      } else {
   401→        homeOutPlayerIds = [];
   402→      }
   403→    }
   404→
   405→    // ─── DO-OR-DIE RAID: Failed do-or-die ───
   406→    if (evt.eventType === 'do_or_die_raid') {
   407→      // Raider goes out (failed do-or-die)
   408→      const raiderId = (details.raiderId as string) || evt.playerId;
   409→      if (side === 'home') {
   410→        // Home scored from do-or-die → away raider is out
   411→        if (raiderId && !awayOutPlayerIds.includes(raiderId)) {
   412→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   413→        }
   414→        // Home revives 1 player
   415→        if (homeOutPlayerIds.length > 0) {
   416→          homeOutPlayerIds = homeOutPlayerIds.slice(1);
   417→        }
   418→      } else {
   419→        if (raiderId && !homeOutPlayerIds.includes(raiderId)) {
   420→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   421→        }
   422→        if (awayOutPlayerIds.length > 0) {
   423→          awayOutPlayerIds = awayOutPlayerIds.slice(1);
   424→        }
   425→      }
   426→
   427→      // Reset empty raid counter for the team that had do-or-die
   428→      const failedTeamId = side === 'home' ? match.awayTeamId : match.homeTeamId;
   429→      emptyRaidCount[failedTeamId] = 0;
   430→    }
   431→
   432→    // ─── EMPTY RAID: No points scored ───
   433→    if (evt.eventType === 'empty_raid') {
   434→      emptyRaidCount[evt.teamId] = (emptyRaidCount[evt.teamId] || 0) + 1;
   435→    }
   436→
   437→    // ─── SUPER RAID: 3+ points in single raid (just a marker, points already counted via raid_point) ───
   438→    if (evt.eventType === 'super_raid') {
   439→      // Points already handled by raid_point events, no additional state change needed
   440→    }
   441→  }
   442→
   443→  // Count-based out tracking (backward compat)
   444→  homeOutPlayers = homeOutPlayerIds.length;
   445→  awayOutPlayers = awayOutPlayerIds.length;
   446→
   447→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   448→}

   450→/** Event types that constitute a complete raid and should flip the turn */
   451→const RAID_ENDING_EVENT_TYPES: string[] = [
   452→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   453→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   454→];
   455→
   456→/** Event types that do NOT flip the turn (non-raid events) */
   457→const NON_RAID_EVENT_TYPES: string[] = [
   458→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   459→];
   460→
   461→/**
   462→ * Event types where the SCORING team is the DEFENDING team.
   463→ * In these cases, the defending team scored (they made the tackle / raider failed do-or-die),
   464→ * so they should raid NEXT — the turn goes to the SAME side that scored.
   465→ *
   466→ * For all other raid-ending events (raid_point, bonus_point, super_raid, empty_raid, self_out),
   467→ * the scoring team is the RAIDING team, so the turn goes to the OPPOSITE side.
   468→ */
   469→const DEFENDING_TEAM_SCORES: Set<string> = new Set([
   470→  'tackle_point',
   471→  'super_tackle',
   472→  'do_or_die_raid',
   473→]);
   474→
   475→/** Helper: determine raid queue from last raid-ending event */
   476→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   477→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   478→  if (lastRaidEvent) {
   479→    const side = getTeamSide(match, lastRaidEvent.teamId);
   480→    // If the defending team scored (tackle/super_tackle/do_or_die_raid),
   481→    // THEY raid next — return the SAME side as the scoring team.
   482→    if (DEFENDING_TEAM_SCORES.has(lastRaidEvent.eventType)) {
   483→      return side;
   484→    }
   485→    // If the raiding team scored (touch points, bonus, super raid, etc.)
   486→    // or it was an empty raid, the DEFENDING team raids next — return the OPPOSITE side.
   487→    return side === 'home' ? 'away' : 'home';
   488→  }
   489→  return match.raidQueue;
   490→}

   585→      setActiveTab: (tab) => set({ activeTab: tab }),

   601→      startMatch: (match, firstRaidTeam) =>

   623→          tossMatchConfig: null,
   624→        }),
   625→
   626→      endMatch: () => set({ activeMatch: null }),
   627→
   628→      updateScore: (team, delta) =>
   629→        set((state) => {
   630→          if (!state.activeMatch) return {};
   631→          return {
   632→            activeMatch: {
   633→              ...state.activeMatch,
   634→              homeScore:
   635→                team === 'home'
   636→                  ? Math.max(0, state.activeMatch.homeScore + delta)
   637→                  : state.activeMatch.homeScore,
   638→              awayScore:
   639→                team === 'away'
   640→                  ? Math.max(0, state.activeMatch.awayScore + delta)
   641→                  : state.activeMatch.awayScore,
   642→            },
   643→          };
   644→        }),
   645→
   646→      addEvent: (event) =>
   647→        set((state) => {
   648→          if (!state.activeMatch) return {};
   649→          const newEvent: MatchEvent = {
   650→            ...event,
   651→            id: `evt_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   652→            timestamp: Date.now(),
   653→          };
   654→          const updatedEvents = [...state.activeMatch.events, newEvent];
   655→
   656→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   657→
   658→          // Only flip raidQueue for raid-ending events.
   659→          // Non-raid events (timeout, substitution, cards, standalone all_out)
   660→          // should NOT change whose turn it is to raid.
   661→          const isRaidEndingEvent = RAID_ENDING_EVENT_TYPES.includes(event.eventType);
   662→          const raidQueue = isRaidEndingEvent
   663→            ? getRaidQueueFromEvents(state.activeMatch, updatedEvents)
   664→            : state.activeMatch.raidQueue;
   665→
   666→          return {
   667→            activeMatch: {
   668→              ...state.activeMatch,
   669→              events: updatedEvents,
   670→              ...calculated,
   671→              raidQueue,
   672→            },
   673→          };
   674→        }),
   675→
   676→      addBatchEvents: (events) =>
   677→        set((state) => {
   678→          if (!state.activeMatch || events.length === 0) return {};
   679→          const newEvents: MatchEvent[] = events.map((e, i) => ({
   680→            ...e,
   681→            id: `evt_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 7)}`,
   682→            timestamp: Date.now() + i,
   683→          }));
   684→          const updatedEvents = [...state.activeMatch.events, ...newEvents];
   685→
   686→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   687→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, updatedEvents);
   688→
   689→          return {
   690→            activeMatch: {
   691→              ...state.activeMatch,
   692→              events: updatedEvents,
   693→              ...calculated,
   694→              raidQueue,
   695→            },
   696→          };
   697→        }),
   698→
   699→      undoLastEvent: () =>
   700→        set((state) => {
   701→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   702→          const events = [...state.activeMatch.events];
   703→          events.pop();
   704→
   705→          const calculated = recalculateFromEvents(state.activeMatch, events);
   706→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   707→
   708→          return {
   709→            activeMatch: {
   710→              ...state.activeMatch,
   711→              events,
   712→              ...calculated,
   713→              raidQueue,
   714→            },
   715→          };
   716→        }),
   717→
   718→      undoLastRaid: () =>
   719→        set((state) => {
   720→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   721→
   722→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   723→          // A raid batch has events with the same second in timestamp
   724→          const events = [...state.activeMatch.events];
   725→          if (events.length === 0) return {};
   726→
   727→          // Find the timestamp of the last event
   728→          const lastTimestamp = events[events.length - 1].timestamp;
   729→          // Remove all events that have the same timestamp (within 5ms for batch events)
   730→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   731→            events.pop();
   732→          }
   733→
   734→          const calculated = recalculateFromEvents(state.activeMatch, events);
   735→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   736→
   737→          return {
   738→            activeMatch: {
   739→              ...state.activeMatch,
   740→              events,
   741→              ...calculated,
   742→              raidQueue,
   743→            },
   744→          };
   745→        }),
   746→
   747→      addPlayerToMatch: (side, player) =>
   748→        set((state) => {
   749→          if (!state.activeMatch) return {};
   750→          const lineupKey = side === 'home' ? 'homeLineup' : 'awayLineup';
   751→          return {
   752→            activeMatch: {
   753→              ...state.activeMatch,
   754→              [lineupKey]: [...state.activeMatch[lineupKey], player],
   755→            },
   756→          };
   757→        }),
   758→
   759→      switchHalf: () =>
   760→        set((state) => {
   761→          if (!state.activeMatch) return {};
   762→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   763→          return {
   764→            activeMatch: {
   765→              ...state.activeMatch,
   766→              currentHalf: newHalf,
   767→              timer: state.activeMatch.halfDuration * 60,
   768→              isDoOrDie: false,
   769→              doOrDieTeamId: null,
   770→            },
   771→          };
   772→        }),
   773→
   774→      setTimer: (time) =>
   775→        set((state) => {
   776→          if (!state.activeMatch) return {};
   777→          return { activeMatch: { ...state.activeMatch, timer: time } };
   778→        }),
   779→
   780→      setDoOrDie: (value, teamId?) =>
   781→        set((state) => {
   782→          if (!state.activeMatch) return {};
   783→          return {
   784→            activeMatch: {
   785→              ...state.activeMatch,
   786→              isDoOrDie: value,
   787→              doOrDieTeamId: value ? (teamId || null) : null,
   788→            },
   789→          };
   790→        }),
   791→
   792→      switchRaidQueue: () =>
   793→        set((state) => {
   794→          if (!state.activeMatch) return {};
   795→          return {
   796→            activeMatch: {
   797→              ...state.activeMatch,
   798→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   799→            },
   800→          };
   801→        }),
   802→
   803→      callTimeout: (team) =>
   804→        set((state) => {
   805→          if (!state.activeMatch) return {};
   806→          return {
   807→            activeMatch: {
   808→              ...state.activeMatch,
   809→              homeTimeouts:
   810→                team === 'home'
   811→                  ? state.activeMatch.homeTimeouts + 1
   812→                  : state.activeMatch.homeTimeouts,
   813→              awayTimeouts:
   814→                team === 'away'
   815→                  ? state.activeMatch.awayTimeouts + 1
   816→                  : state.activeMatch.awayTimeouts,
   817→            },
   818→          };
   819→        }),
   820→
   821→      setOutPlayers: (team, count) =>
   822→        set((state) => {
   823→          if (!state.activeMatch) return {};
   824→          return {
   825→            activeMatch: {
   826→              ...state.activeMatch,
   827→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   828→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   829→            },
   830→          };
   831→        }),
   832→
   833→      fetchHomeData: async () => {
   834→        try {
   835→          const res = await fetch('/api/stats');
   836→          if (!res.ok) throw new Error('Failed to fetch stats');
   837→
   838→          const data = await res.json();
   839→
   840→          // Cache in store
   841→          set({ homeData: data });
   842→          return data;
   843→        } catch (err) {
   844→          console.error('fetchHomeData error:', err);
   845→          return null;
   846→        }
   847→      },
   848→
   849→      // Notification actions
   850→      addNotification: (notification) =>
   851→        set((state) => ({
   852→          notifications: [
   853→            {
   854→              ...notification,
   855→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,

   874→
   875→      clearNotifications: () => set({ notifications: [] }),
   876→
   877→      setLanguage: (lang) => set({ language: lang }),

   894→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),

   900→
   901→      // Team management actions
   902→      setTeamFilter: (filter) =>

   950→export function getEventPoints(eventType: EventType): number {
   951→  switch (eventType) {
   952→    case 'raid_point':
   953→      return 1;
   954→    case 'bonus_point':
   955→      return 1;
   956→    case 'tackle_point':
   957→      return 1;
   958→    case 'super_raid':
   959→      return 1;
   960→    case 'super_tackle':
   961→      return 1;
   962→    case 'do_or_die_raid':
   963→      return 1;
   964→    case 'all_out':
   965→      return 2;
   966→    case 'self_out':
   967→      return 1;
   968→    case 'empty_raid':
   969→      return 0;
   970→    case 'substitution':
   971→    case 'timeout':
   972→    case 'yellow_card':
   973→    case 'red_card':
   974→    case 'green_card':
   975→      return 0;
   976→    default:
   977→      return 0;
   978→  }
   979→}
