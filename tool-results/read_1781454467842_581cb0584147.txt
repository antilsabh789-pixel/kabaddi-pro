    72→function getPositionCategory(position: string | null): 'raider' | 'defender' | 'all-rounder' | 'unknown' {
    73→  if (!position) return 'unknown';
    74→  const p = position.toLowerCase();
    75→  if (p.includes('raider')) return 'raider';
    76→  if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return 'defender';
    77→  if (p.includes('all')) return 'all-rounder';
    78→  return 'unknown';
    79→}

    83→function LineupValidation({ lineup, playersPerSide, teamName }: {
    84→  lineup: MatchPlayer[];
    85→  playersPerSide: number;
    86→  teamName: string;
    87→}) {
    88→  const warnings: { message: string; severity: 'error' | 'warning' | 'info' }[] = [];
    89→
    90→  if (lineup.length === 0) {
    91→    warnings.push({ message: 'No players added', severity: 'error' });
    92→  } else if (lineup.length < playersPerSide) {
    93→    warnings.push({
    94→      message: `${lineup.length}/${playersPerSide} players — need ${playersPerSide - lineup.length} more`,
    95→      severity: 'warning',
    96→    });
    97→  } else if (lineup.length > playersPerSide) {
    98→    warnings.push({
    99→      message: `${lineup.length} players — ${lineup.length - playersPerSide} extra`,
   100→      severity: 'info',
   101→    });
   102→  }
   103→
   104→  // Position balance check (only if we have position data from DB)
   105→  const balance = POSITION_BALANCE[playersPerSide];
   106→  // Note: position data might not be available for quick-added players
   107→  // We check this based on the allPlayers data which is in parent scope
   108→
   109→  if (warnings.length === 0) return null;
   110→
   111→  return (
   112→    <motion.div
   113→      initial={{ opacity: 0, y: -5 }}
   114→      animate={{ opacity: 1, y: 0 }}
   115→      className="space-y-1.5"
   116→    >
   117→      {warnings.map((w, i) => (
   118→        <div
   119→          key={i}
   120→          className={`flex items-center gap-2 px-3 py-2 rounded-lg text-xs font-medium ${
   121→            w.severity === 'error'
   122→              ? 'bg-brand-red/10 text-brand-red dark:bg-brand-red/15'
   123→              : w.severity === 'warning'
   124→                ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400 dark:bg-amber-500/15'
   125→                : 'bg-blue-500/10 text-blue-600 dark:text-blue-400 dark:bg-blue-500/15'
   126→          }`}
   127→        >
   128→          <AlertTriangle className={`w-3.5 h-3.5 shrink-0 ${w.severity === 'info' ? 'hidden' : ''}`} />
   129→          <Info className={`w-3.5 h-3.5 shrink-0 ${w.severity !== 'info' ? 'hidden' : ''}`} />
   130→          <span>{teamName}: {w.message}</span>
   131→        </div>
   132→      ))}
   133→    </motion.div>
   134→  );
   135→}

   139→function PlayerStatsTooltip({ player, allPlayers }: { player: DbPlayer; allPlayers: DbPlayer[] }) {
   140→  const dbPlayer = allPlayers.find(p => p.id === player.id);
   141→  const profile = dbPlayer?.profile;
   142→
   143→  if (!profile || !profile.totalPoints) return null;
   144→
   145→  const raidSuccessRate = profile.totalRaids && profile.totalRaids > 0
   146→    ? Math.round(((profile.successfulRaids || 0) / profile.totalRaids) * 100)
   147→    : 0;
   148→
   149→  return (
   150→    <div className="absolute z-30 bottom-full left-0 mb-1 p-2 bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-600 rounded-lg shadow-lg text-[10px] min-w-[140px]">
   151→      <div className="space-y-1">
   152→        <div className="flex justify-between">
   153→          <span className="text-warm-500 dark:text-warm-400">Total Pts</span>
   154→          <span className="font-bold text-brand-navy dark:text-brand-navy-light">{profile.totalPoints}</span>
   155→        </div>
   156→        <div className="flex justify-between">
   157→          <span className="text-warm-500 dark:text-warm-400">Raid Pts</span>
   158→          <span className="font-bold text-brand-red">{profile.raidPoints || 0}</span>
   159→        </div>
   160→        <div className="flex justify-between">
   161→          <span className="text-warm-500 dark:text-warm-400">Tackle Pts</span>
   162→          <span className="font-bold text-brand-teal">{profile.tacklePoints || 0}</span>
   163→        </div>
   164→        <div className="flex justify-between">
   165→          <span className="text-warm-500 dark:text-warm-400">Raid Success</span>
   166→          <span className="font-bold text-brand-gold">{raidSuccessRate}%</span>
   167→        </div>
   168→      </div>
   169→      {/* Tooltip arrow */}
   170→      <div className="absolute -bottom-1 left-4 w-2 h-2 bg-white dark:bg-warm-800 border-r border-b border-warm-200 dark:border-warm-600 rotate-45" />
   171→    </div>
   172→  );
   173→}

   179→  teamColor: string;
   180→  teamName: string;
   181→  side: 'left' | 'right';

   191→          className="w-10 h-10 rounded-xl mx-auto flex items-center justify-center text-white font-black text-sm shadow-md"

   193→        >
   194→          {teamName.charAt(0).toUpperCase()}
   195→        </div>
   196→        <p className="text-[10px] font-bold text-warm-600 dark:text-warm-300 mt-1 truncate max-w-[80px] mx-auto">
   197→          {teamName}
   198→        </p>
   199→      </div>
   200→
   201→      {/* Formation layout */}
   202→      <div className="relative bg-warm-100/40 dark:bg-warm-700/20 rounded-xl p-2 min-h-[90px]">
   203→        {/* Court half line */}
   204→        <div className={`absolute top-1/2 left-0 right-0 h-px bg-warm-200 dark:bg-warm-600 ${
   205→          side === 'left' ? 'bg-gradient-to-r from-transparent to-warm-200 dark:to-warm-600' : 'bg-gradient-to-l from-transparent to-warm-200 dark:to-warm-600'
   206→        }`} />
   207→
   208→        {/* Players in formation */}
   209→        <div className="grid grid-cols-3 gap-1 place-items-center">
   210→          {displayPlayers.map((player, i) => {
   211→            // Simple formation: top row = defenders, middle = all-rounders, bottom = raiders
   212→            return (
   213→              <motion.div
   214→                key={player.id}
   215→                initial={{ scale: 0, opacity: 0 }}
   216→                animate={{ scale: 1, opacity: 1 }}
   217→                transition={{ delay: i * 0.05, type: 'spring', stiffness: 300, damping: 20 }}
   218→                className="flex flex-col items-center"
   219→              >
   220→                <div
   221→                  className="w-7 h-7 rounded-full flex items-center justify-center text-white text-[8px] font-bold shadow-sm"
   222→                  style={{ backgroundColor: teamColor }}
   223→                >
   224→                  {player.jerseyNumber || i + 1}
   225→                </div>
   226→                <span className="text-[7px] text-warm-400 dark:text-warm-500 mt-0.5 truncate max-w-[32px]">
   227→                  {player.name.split(' ')[0]}
   228→                </span>
   229→              </motion.div>
   230→            );
   231→          })}
   232→          {/* Empty slots */}
   233→          {Array.from({ length: Math.max(0, 7 - displayPlayers.length) }).map((_, i) => (
   234→            <div key={`empty-${i}`} className="flex flex-col items-center">
   235→              <div className="w-7 h-7 rounded-full border-2 border-dashed border-warm-200 dark:border-warm-600 flex items-center justify-center">
   236→                <span className="text-[8px] text-warm-300 dark:text-warm-600">?</span>
   237→              </div>
   238→              <span className="text-[7px] text-warm-300 dark:text-warm-600 mt-0.5">Empty</span>
   239→            </div>
   240→          ))}
   241→        </div>
   242→      </div>
   243→    </div>
   244→  );
   245→}
