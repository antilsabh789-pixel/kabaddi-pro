   237→   237→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   238→   238→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   239→   239→  let homeScore = 0;
   240→   240→  let awayScore = 0;
   241→   241→  let homeOutPlayers = 0;
   242→   242→  let awayOutPlayers = 0;
   243→   243→  let homeOutPlayerIds: string[] = [];
   244→   244→  let awayOutPlayerIds: string[] = [];
   245→   245→
   246→   246→  for (const evt of events) {
   247→   247→    const side = getTeamSide(match, evt.teamId);
   248→   248→    const points = evt.value;
   249→   249→    if (side === 'home') {
   250→   250→      homeScore += points;
   251→   251→    } else {
   252→   252→      awayScore += points;
   253→   253→    }
   254→   254→
   255→   255→    // Count-based out tracking (backward compat)
   256→   256→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   257→   257→      if (side === 'home') {
   258→   258→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   259→   259→      } else {
   260→   260→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   261→   261→      }
   262→   262→    }
   263→   263→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   264→   264→      if (side === 'home') {
   265→   265→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   266→   266→      } else {
   267→   267→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   268→   268→      }
   269→   269→    }
   270→   270→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   271→   271→    if (evt.eventType === 'self_out') {
   272→   272→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   273→   273→      if (side === 'home') {
   274→   274→        // Home scored from self-out → away defender went out
   275→   275→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   276→   276→        // Home revives one player
   277→   277→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   278→   278→      } else {
   279→   279→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   280→   280→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   281→   281→      }
   282→   282→    }
   283→   283→    if (evt.eventType === 'all_out') {
   284→   284→      if (side === 'home') {
   285→   285→        awayOutPlayers = 0;
   286→   286→      } else {
   287→   287→        homeOutPlayers = 0;
   288→   288→      }
   289→   289→    }
   290→   290→
   291→   291→    // ID-based out tracking
   292→   292→    let details: Record<string, unknown> = {};
   293→   293→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   294→   294→
   295→   295→    if (evt.eventType === 'raid_point') {
   296→   296→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   297→   297→      if (side === 'home') {
   298→   298→        // Home team scored raid → away defenders go out
   299→   299→        for (const tid of touchedIds) {
   300→   300→          if (!awayOutPlayerIds.includes(tid)) {
   301→   301→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   302→   302→          }
   303→   303→        }
   304→   304→        // Home team revives players (first-out-first-in)
   305→   305→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   306→   306→        if (reviveCount > 0) {
   307→   307→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   308→   308→        }
   309→   309→      } else {
   310→   310→        // Away team scored raid → home defenders go out
   311→   311→        for (const tid of touchedIds) {
   312→   312→          if (!homeOutPlayerIds.includes(tid)) {
   313→   313→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   314→   314→          }
   315→   315→        }
   316→   316→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   317→   317→        if (reviveCount > 0) {
   318→   318→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   319→   319→        }
   320→   320→      }
   321→   321→    }
   322→   322→
   323→   323→    if (evt.eventType === 'bonus_point') {
   324→   324→      // One out player from scoring team revives
   325→   325→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   326→   326→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   327→   327→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   328→   328→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   329→   329→      }
   330→   330→    }
   331→   331→
   332→   332→    // Self-out: defender steps off mat during a live raid
   333→   333→    if (evt.eventType === 'self_out') {
   334→   334→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   335→   335→      if (selfOutPlayerId) {
   336→   336→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   337→   337→        if (side === 'home') {
   338→   338→          // Home scored → away defender self-out
   339→   339→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   340→   340→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   341→   341→          }
   342→   342→          // Home revives first-out player
   343→   343→          if (homeOutPlayerIds.length > 0) {
   344→   344→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   345→   345→          }
   346→   346→        } else {
   347→   347→          // Away scored → home defender self-out
   348→   348→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   349→   349→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   350→   350→          }
   351→   351→          if (awayOutPlayerIds.length > 0) {
   352→   352→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   353→   353→          }
   354→   354→        }
   355→   355→      }
   356→   356→    }
   357→   357→
   358→   358→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   359→   359→      // Raider goes out. The tackle is scored by the DEFENDING team.
   360→   360→      // So the raider is from the OPPOSING team.
   361→   361→      const raiderId = (details.raiderId as string) || evt.playerId;
   362→   362→      if (side === 'home' && raiderId) {
   363→   363→        // Home team scored tackle → away raider was caught → away raider goes out
   364→   364→        if (!awayOutPlayerIds.includes(raiderId)) {
   365→   365→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   366→   366→        }
   367→   367→      } else if (side === 'away' && raiderId) {
   368→   368→        // Away team scored tackle → home raider was caught → home raider goes out
   369→   369→        if (!homeOutPlayerIds.includes(raiderId)) {
   370→   370→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   371→   371→        }
   372→   372→      }
   373→   373→    }
   374→   374→
   375→   375→    if (evt.eventType === 'all_out') {
   376→   376→      if (side === 'home') {
   377→   377→        awayOutPlayerIds = [];
   378→   378→      } else {
   379→   379→        homeOutPlayerIds = [];
   380→   380→      }
   381→   381→    }
   382→   382→  }
   383→   383→
   384→   384→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   385→   385→}

   387→   387→/** Helper: determine raid queue from last raid event */
   388→   388→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   389→   389→  const raidEventTypes = ['raid_point', 'tackle_point', 'super_tackle', 'empty_raid', 'bonus_point', 'do_or_die_raid'];
   390→   390→  const lastRaidEvent = [...events].reverse().find(e => raidEventTypes.includes(e.eventType));
   391→   391→  if (lastRaidEvent) {
   392→   392→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   393→   393→  }
   394→   394→  return match.raidQueue;
   395→   395→}
