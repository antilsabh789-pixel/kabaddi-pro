    79→function getInitial(name: string | null | undefined): string {

    88→function formatTimeAgo(dateStr: string): string {
    89→  try {
    90→    const diff = Date.now() - new Date(dateStr).getTime();
    91→    const seconds = Math.floor(diff / 1000);
    92→    if (seconds < 60) return 'just now';
    93→    const minutes = Math.floor(seconds / 60);
    94→    if (minutes < 60) return `${minutes}m ago`;
    95→    const hours = Math.floor(minutes / 60);
    96→    if (hours < 24) return `${hours}h ago`;
    97→    const days = Math.floor(hours / 24);
    98→    if (days === 1) return 'Yesterday';
    99→    if (days < 7) return `${days}d ago`;
   100→    if (days < 30) return `${Math.floor(days / 7)}w ago`;
   101→    const months = Math.floor(days / 30);
   102→    return `${months}mo ago`;
   103→  } catch {
   104→    return '';
   105→  }
   106→}
   107→
   108→function formatTimestampAgo(timestamp: number): string {
   109→  const diff = Date.now() - timestamp;
   110→  const seconds = Math.floor(diff / 1000);
   111→  if (seconds < 60) return 'just now';
   112→  const minutes = Math.floor(seconds / 60);
   113→  if (minutes < 60) return `${minutes}m ago`;
   114→  const hours = Math.floor(minutes / 60);
   115→  if (hours < 24) return `${hours}h ago`;
   116→  const days = Math.floor(hours / 24);
   117→  if (days === 1) return 'Yesterday';
   118→  if (days < 7) return `${days}d ago`;
   119→  return `${Math.floor(days / 7)}w ago`;
   120→}

   904→export default function SocialFeedScreen({ onClose }: SocialFeedScreenProps) {

   906→  const { toast } = useToast();
   907→
   908→  // Activity feed state
   909→  const [activities, setActivities] = useState<Activity[]>([]);
   910→  const [feedLoading, setFeedLoading] = useState(true);
   911→  const [refreshing, setRefreshing] = useState(false);
   912→  const [offset, setOffset] = useState(0);
   913→  const [hasMore, setHasMore] = useState(true);
   914→
   915→  // Suggested players state
   916→  const [suggestedPlayers, setSuggestedPlayers] = useState<SuggestedPlayer[]>([]);
   917→  const [suggestedLoading, setSuggestedLoading] = useState(false);
   918→  const [actionLoadingId, setActionLoadingId] = useState<string | null>(null);

   927→  const [pulling, setPulling] = useState(false);
   928→
   929→  // ─── Fetch Activities ───────────────────────────────────────
   930→
   931→  const fetchActivities = useCallback(
   932→    async (isRefresh = false) => {
   933→      if (!currentUser) return;
   934→
   935→      if (isRefresh) {
   936→        setRefreshing(true);
   937→      } else {
   938→        setFeedLoading(true);
   939→      }
   940→
   941→      try {
   942→        const params = new URLSearchParams({
   943→          userId: currentUser.id,
   944→          limit: '20',
   945→          offset: isRefresh ? '0' : String(offset),
   946→        });
   947→        const res = await fetch(`/api/activities?${params}`);
   948→        if (!res.ok) throw new Error('Failed to fetch');
   949→        const data = await res.json();
   950→        const newActivities: Activity[] = data.activities || [];
   951→
   952→        if (isRefresh) {
   953→          setActivities(newActivities);
   954→          setOffset(0);
   955→        } else {
   956→          setActivities((prev) => [...prev, ...newActivities]);
   957→        }
   958→        setHasMore(newActivities.length === 20);
   959→      } catch (err) {
   960→        console.error('Activities fetch error:', err);
   961→        if (isRefresh) setActivities([]);
   962→      } finally {
   963→        setFeedLoading(false);
   964→        setRefreshing(false);
   965→        setPulling(false);
   966→      }
   967→    },
   968→    [currentUser, offset]
   969→  );
   970→
   971→  // ─── Fetch Suggested Players ────────────────────────────────
   972→
   973→  const fetchSuggestedPlayers = useCallback(async () => {
   974→    if (!currentUser) return;
   975→    setSuggestedLoading(true);
   976→    try {
   977→      const params = new URLSearchParams({
   978→        userId: currentUser.id,
   979→        type: 'search',
   980→      });
   981→      const res = await fetch(`/api/follow?${params}`);
   982→      if (!res.ok) throw new Error('Failed to fetch');
   983→      const data = await res.json();
   984→      const players: SuggestedPlayer[] = (data.players || [])
   985→        .filter((p: SuggestedPlayer) => !p.isFollowing && p.id !== currentUser.id)
   986→        .slice(0, 5);
   987→      setSuggestedPlayers(players);
   988→    } catch (err) {
   989→      console.error('Suggested players fetch error:', err);
   990→      setSuggestedPlayers([]);
   991→    } finally {
   992→      setSuggestedLoading(false);
   993→    }
   994→  }, [currentUser]);
   995→
   996→  // ─── Follow / Unfollow action ───────────────────────────────
   997→
   998→  const handleFollowAction = async (
   999→    targetId: string,
  1000→    isCurrentlyFollowing: boolean
  1001→  ) => {
  1002→    if (!currentUser || actionLoadingId) return;
  1003→    setActionLoadingId(targetId);
  1004→    try {
  1005→      const res = await fetch('/api/follow', {
  1006→        method: 'POST',
  1007→        headers: { 'Content-Type': 'application/json' },
  1008→        body: JSON.stringify({
  1009→          followerId: currentUser.id,
  1010→          followingId: targetId,
  1011→          action: isCurrentlyFollowing ? 'unfollow' : 'follow',
  1012→        }),
  1013→      });
  1014→      if (!res.ok) throw new Error('Action failed');
  1015→
  1016→      const nowFollowing = !isCurrentlyFollowing;
  1017→
  1018→      // Update suggested players
  1019→      setSuggestedPlayers((prev) =>
  1020→        prev.map((p) =>
  1021→          p.id === targetId ? { ...p, isFollowing: nowFollowing } : p
  1022→        )
  1023→      );
  1024→
  1025→      toast({
  1026→        title: nowFollowing ? 'Following!' : 'Unfollowed',
  1027→        description: nowFollowing
  1028→          ? 'You are now following this player'
  1029→          : 'You unfollowed this player',

  1031→
  1032→      // Refresh feed to show new followed user's activities
  1033→      if (nowFollowing) {
  1034→        fetchActivities(true);
  1035→      }
  1036→    } catch (err) {
  1037→      console.error('Follow action error:', err);
  1038→      toast({
  1039→        title: 'Error',
  1040→        description: 'Something went wrong. Please try again.',
  1041→        variant: 'destructive',
  1042→      });
  1043→    } finally {
  1044→      setActionLoadingId(null);
  1045→    }
  1046→  };
  1047→
  1048→  // ─── Load more ──────────────────────────────────────────────
  1049→
  1050→  const loadMore = useCallback(() => {
  1051→    if (!hasMore || feedLoading) return;
  1052→    setOffset((prev) => prev + 20);
  1053→  }, [hasMore, feedLoading]);
  1054→
  1055→  // ─── Effects ────────────────────────────────────────────────
  1056→
  1057→  useEffect(() => {
  1058→    fetchActivities();
  1059→  }, [fetchActivities]);
  1060→
  1061→  // Refetch when offset changes (load more)
  1062→  useEffect(() => {
  1063→    if (offset > 0) {
  1064→      fetchActivities();
  1065→    }
  1066→  }, [offset]);
  1067→
  1068→  useEffect(() => {
  1069→    fetchSuggestedPlayers();
  1070→  }, [fetchSuggestedPlayers]);
  1071→
  1072→  // ─── Community Post Actions ─────────────────────────────────
  1073→
  1074→  const handleCreatePost = useCallback((content: string, type: FeedType) => {

  1076→      id: `post_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
  1077→      userId: currentUser?.id || 'anonymous',
  1078→      userName: currentUser?.name || 'You',
  1079→      content,
  1080→      createdAt: Date.now(),
  1081→      likes: 0,
  1082→      isLiked: false,
  1083→      type,
  1084→    };
  1085→    setCommunityPosts((prev) => [newPost, ...prev]);
