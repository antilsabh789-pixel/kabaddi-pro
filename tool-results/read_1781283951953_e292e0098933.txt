    29→function ConfettiParticle({ delay, color, index }: { delay: number; color: string; index: number }) {
    30→  const leftPos = (index * 17 + 13) % 100;
    31→  const rotateEnd = (index * 47 + 23) % 720 - 360;
    32→  const xDrift = ((index * 31) % 80) - 40;
    33→
    34→  return (
    35→    <motion.div
    36→      initial={{ y: -20, x: 0, rotate: 0, opacity: 1 }}
    37→      animate={{ y: '100vh', x: xDrift, rotate: rotateEnd, opacity: 0 }}
    38→      transition={{ duration: 3 + (index % 3), delay, ease: 'easeIn' }}
    39→      className="absolute top-0 w-2 h-2 rounded-sm"
    40→      style={{ left: `${leftPos}%`, backgroundColor: color }}
    41→    />
    42→  );
    43→}

   527→  // ═══ 5-MINUTE WARNING & HALF/END TIME DETECTION ═══
   528→  useEffect(() => {

   539→        description: 'Time is running out',
   540→        duration: 3000,
   541→      });
   542→    }
   543→
   544→    if (timer === halfDuration) {
   545→      fiveMinWarningFiredRef.current = false;
   546→    }
   547→
   548→    if (timer === 0 && hasStartedRaiding) {
   549→      if (match.currentHalf === 1) {
   550→        triggerFeedback(SoundType.HALF_END);
   551→        setShowHalfTimeTransition(true);
   552→      } else {
   553→        triggerFeedback(SoundType.HALF_END);
   554→        setTimeout(() => {
   555→          triggerFeedback(SoundType.MATCH_END);
   556→        }, 800);
   557→      }
   558→    }
   559→  }, [match?.timer, match?.isLive, match?.currentHalf, match?.halfDuration, hasStartedRaiding, toast]);
   560→
   561→  // ═══ RAID TIMER LOGIC ═══
   562→  useEffect(() => {
   563→    if (raidTimer === null) {
   564→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   565→      return;
   566→    }
   567→
   568→    if (raidTimer <= 0) {
   569→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   570→      setRaidTimer(null);
   571→      if (raidPhase !== 'idle' && raider) {
   572→        triggerFeedback(SoundType.RAID_TIME_EXPIRED);

   575→      }
   576→      return;
   577→    }
   578→
   579→    raidTimerRef.current = setInterval(() => {
   580→      setRaidTimer(prev => {
   581→        if (prev === null || prev <= 1) {
   582→          if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   583→          return 0;
   584→        }
   585→        return prev - 1;
   586→      });
   587→    }, 1000);
   588→
   589→    return () => {
   590→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   591→    };
   592→  }, [raidTimer !== null, raidPhase, raider, toast]);

   596→    if (raidTimer === null || raidPhase === 'idle') return;

   599→      vibrate([40]);

   606→    }
   607→  }, [raidTimer, raidPhase]);

   612→      if (raidGapRef.current) clearInterval(raidGapRef.current);
   613→      return;
   614→    }
   615→
   616→    if (raidGapTimer <= 0) {
   617→      if (raidGapRef.current) clearInterval(raidGapRef.current);

   619→      if (raidPhase === 'idle' && hasStartedRaiding && !isPaused) {
   620→        triggerFeedback(SoundType.RAID_GAP_WARNING);
   621→        setIsPaused(true);

   629→        if (prev === null || prev <= 1) {
   630→          if (raidGapRef.current) clearInterval(raidGapRef.current);
   631→          return 0;
   632→        }
   633→        return prev - 1;
   634→      });
   635→    }, 1000);
   636→
   637→    return () => {
   638→      if (raidGapRef.current) clearInterval(raidGapRef.current);
   639→    };

   663→  // Calculate MOTM
   664→  const calculateMotm = useCallback(() => {
   665→    if (!match) return null;
   666→    const playerPoints: Record<string, { name: string; points: number }> = {};
   667→    for (const event of match.events) {
   668→      if (event.playerId && event.playerName) {
   669→        if (!playerPoints[event.playerId]) {
   670→          playerPoints[event.playerId] = { name: event.playerName, points: 0 };
   671→        }
   672→        playerPoints[event.playerId].points += event.value;
   673→      }
   674→    }
   675→    let topPlayer: { name: string; points: number } | null = null;
   676→    for (const p of Object.values(playerPoints)) {
   677→      if (!topPlayer || p.points > topPlayer.points) topPlayer = p;
   678→    }
   679→    return topPlayer;
   680→  }, [match]);
   681→
   682→  const calculateTopRaiderDefender = useCallback(() => {
   683→    if (!match) return { topRaider: null, topDefender: null };
   684→    const playerRaidPoints: Record<string, { name: string; points: number }> = {};
   685→    const playerTacklePoints: Record<string, { name: string; points: number }> = {};
   686→
   687→    for (const event of match.events) {
   688→      if (!event.playerId || !event.playerName) continue;
   689→      if (['raid_point', 'bonus_point', 'super_raid', 'do_or_die_raid'].includes(event.eventType)) {
   690→        if (!playerRaidPoints[event.playerId]) playerRaidPoints[event.playerId] = { name: event.playerName, points: 0 };
   691→        playerRaidPoints[event.playerId].points += event.value;
   692→      }
   693→      if (['tackle_point', 'super_tackle'].includes(event.eventType)) {
   694→        if (!playerTacklePoints[event.playerId]) playerTacklePoints[event.playerId] = { name: event.playerName, points: 0 };
   695→        playerTacklePoints[event.playerId].points += event.value;
   696→      }
   697→    }
   698→
   699→    let topRaider: { name: string; points: number } | null = null;
   700→    for (const p of Object.values(playerRaidPoints)) {
   701→      if (!topRaider || p.points > topRaider.points) topRaider = p;
   702→    }
   703→    let topDefender: { name: string; points: number } | null = null;
   704→    for (const p of Object.values(playerTacklePoints)) {
   705→      if (!topDefender || p.points > topDefender.points) topDefender = p;
   706→    }
   707→    return { topRaider, topDefender };
   708→  }, [match]);
   709→
   710→  // Get player points from match events (before early return for hooks rule)
   711→  const getPlayerPoints = useCallback((playerId: string): { raid: number; tackle: number } => {
   712→    if (!match) return { raid: 0, tackle: 0 };
   713→    let raid = 0;
   714→    let tackle = 0;
   715→    for (const event of match.events) {
   716→      if (event.playerId !== playerId) continue;
   717→      if (['raid_point', 'super_raid', 'do_or_die_raid', 'bonus_point'].includes(event.eventType)) {
   718→        raid += event.value;
   719→      }
   720→      if (['tackle_point', 'super_tackle'].includes(event.eventType)) {
   721→        tackle += event.value;
   722→      }
   723→    }
   724→    return { raid, tackle };
   725→  }, [match?.events]);
   726→
   727→  if (!match) return null;
   728→
   729→  // ═══ DERIVED STATE ═══
   730→  const raidingTeam = match.raidQueue;
   731→  const defendingTeam: 'home' | 'away' = raidingTeam === 'home' ? 'away' : 'home';
   732→
   733→  const fullRaidingLineup = raidingTeam === 'home' ? match.homeLineup : match.awayLineup;
   734→  const fullDefendingLineup = defendingTeam === 'home' ? match.homeLineup : match.awayLineup;
   735→
   736→  const raidingOutIds = raidingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   737→  const defendingOutIds = defendingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   738→
   739→  const raidingTeamColor = raidingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   740→  const defendingTeamColor = defendingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   741→
   742→  const raidingTeamName = raidingTeam === 'home' ? match.homeTeam : match.awayTeam;
   743→  const defendingTeamName = defendingTeam === 'home' ? match.homeTeam : match.awayTeam;
   744→
   745→  // Split lineup: first 7 = on court, rest = substitutes
   746→  const splitLineup = (lineup: MatchPlayer[], outIds: string[]) => {
   747→    const onCourt = lineup.slice(0, ON_COURT_MAX);
   748→    const substitutes = lineup.slice(ON_COURT_MAX);
   749→    const onCourtActive = onCourt.filter(p => !outIds.includes(p.id));
   750→    const onCourtOut = onCourt.filter(p => outIds.includes(p.id));
   751→    return { onCourt, substitutes, onCourtActive, onCourtOut };
   752→  };
   753→  // Handle raider selection → starts raid timer AND match timer
   754→  const handleSelectRaider = (player: MatchPlayer) => {
   755→    if (raidingOutIds.includes(player.id)) return;
   756→    if (raidPhase !== 'idle') return;
   757→
   758→    if (!hasStartedRaiding) setHasStartedRaiding(true);
   759→    if (isPaused) setIsPaused(false);
   760→    clearRaidGap();
   761→
   762→    setRaider(player);
   763→    setRaidPhase('result');
   764→    setRaidResult(null);
   765→    setSelectedDefenders(new Set());
   766→    setBonusPoint(false);
   767→
   768→    setRaidTimer(RAID_TIME_LIMIT);
   769→
   770→    if (match.isDoOrDie) {
   771→      triggerFeedback(SoundType.DO_OR_DIE);
   772→    } else {
   773→      triggerFeedback(SoundType.WHISTLE);
   774→    }
   775→  };
   776→
   777→  // Handle raid result selection
   778→  const handleSelectResult = (result: RaidResult) => {
   779→    if (result === 'empty') {
   780→      processRaidResult(result, new Set(), false);
   781→      return;
   782→    }
   783→    setRaidResult(result);
   784→    setRaidPhase('defenders');
   785→    setSelectedDefenders(new Set());
   786→  };
   787→
   788→  // Toggle defender selection
   789→  const toggleDefender = (playerId: string) => {
   790→    setSelectedDefenders(prev => {
   791→      const next = new Set(prev);
   792→      if (next.has(playerId)) next.delete(playerId);
   793→      else next.add(playerId);

   795→    });
   796→  };
   797→
   798→  // Process raid result
   799→  const processRaidResult = (result: RaidResult, touchedDefenders: Set<string>, hasBonus: boolean) => {
   800→    if (!match || !raider) return;
   801→
   802→    if (!hasStartedRaiding) setHasStartedRaiding(true);
   803→
   804→    // Stop raid timer
   805→    setRaidTimer(null);
   806→    if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   807→
   808→    const raidingTeamId = raidingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   809→    const defendingTeamId = defendingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   810→
   811→    const events: Omit<MatchEvent, 'id' | 'timestamp'>[] = [];
   812→
   813→    if (result === 'success') {
   814→      const touchCount = touchedDefenders.size;
   815→      if (touchCount > 0) {
   816→        events.push({
   817→          matchId: match.id, eventType: 'raid_point', teamId: raidingTeamId,
   818→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   819→          value: touchCount,
   820→          details: JSON.stringify({ touchedPlayerIds: Array.from(touchedDefenders), raiderId: raider.id }),
   821→        });
   822→      }
   823→      if (hasBonus) {
   824→        events.push({
   825→          matchId: match.id, eventType: 'bonus_point', teamId: raidingTeamId,
   826→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   827→          value: 1, details: JSON.stringify({ raiderId: raider.id }),
   828→        });
   829→      }
   830→
   831→      // All out check
   832→      const { onCourt } = splitLineup(fullDefendingLineup, defendingOutIds);
   833→      const defendingOnCourtOut = onCourt.filter(p => defendingOutIds.includes(p.id)).length;
   834→      const newDefendingOutCount = defendingOnCourtOut + touchCount;
   835→      if (newDefendingOutCount >= onCourt.length) {
   836→        events.push({
   837→          matchId: match.id, eventType: 'all_out', teamId: raidingTeamId,
   838→          half: match.currentHalf, value: 2,
   839→        });
   840→        // Trigger all-out celebration
   841→        setAllOutCelebration({ teamName: raidingTeamName, teamColor: raidingTeamColor });
   842→      }
   843→    } else if (result === 'caught') {
   844→      const caughtByIds = Array.from(touchedDefenders);
   845→      const primaryCatcher = caughtByIds.length > 0
   846→        ? fullDefendingLineup.find(p => caughtByIds.includes(p.id))
   847→        : null;
   848→
   849→      events.push({
   850→        matchId: match.id, eventType: 'tackle_point', teamId: defendingTeamId,
   851→        half: match.currentHalf,
   852→        playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   853→        value: 1,
   854→        details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   855→      });
   856→
   857→      const { onCourtActive } = splitLineup(fullDefendingLineup, defendingOutIds);
   858→      if (onCourtActive.length <= 3) {
   859→        events.push({
   860→          matchId: match.id, eventType: 'super_tackle', teamId: defendingTeamId,
   861→          half: match.currentHalf,
   862→          playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   863→          value: 1,
   864→          details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   865→        });
   866→      }
   867→    } else if (result === 'empty') {
   868→      events.push({
   869→        matchId: match.id, eventType: 'empty_raid', teamId: raidingTeamId,
   870→        half: match.currentHalf, playerId: raider.id, playerName: raider.name, value: 0,
   871→      });
   872→
   873→      consecutiveEmptyRaidsRef.current += 1;
   874→    }
   875→
   876→    // Reset consecutive empty raids on any successful raid (points scored)
   877→    if (result === 'success') {
   878→      consecutiveEmptyRaidsRef.current = 0;
   879→    }
   880→    if (result === 'caught') {
   881→      consecutiveEmptyRaidsRef.current = 0;
   882→    }
   883→
   884→    // ═══ DO-OR-DIE LOGIC ═══
   885→    if (consecutiveEmptyRaidsRef.current >= 2) {
   886→      setDoOrDie(true);
   887→      triggerFeedback(SoundType.DO_OR_DIE);
   888→      toast({
   889→        title: '🔥 DO OR DIE RAID!',
   890→        description: 'Next raider must score — or raider is out!',
   891→        duration: 3000,
   892→      });
   893→    } else {
   894→      setDoOrDie(false);
   895→    }
   896→
   897→    if (events.length > 0) {
   898→      addBatchEvents(events);
   899→    }
   900→
   901→    // Show event confirmation
   902→    const pointValue = result === 'success' ? (touchedDefenders.size + (hasBonus ? 1 : 0)) : result === 'caught' ? 1 : 0;
   903→    const confirmMsg = result === 'success'
   904→      ? `${raider.name} +${pointValue} raid point${pointValue > 1 ? 's' : ''}`
   905→      : result === 'caught'
   906→        ? `${defendingTeamName} +1 tackle`
   907→        : 'Empty raid recorded';
   908→    setEventConfirm({ message: confirmMsg, teamColor: result === 'caught' ? defendingTeamColor : raidingTeamColor });
   909→
   910→    // Reset raid state
   911→    setRaidPhase('idle');
   912→    setRaider(null);
   913→    setRaidResult(null);
   914→    setSelectedDefenders(new Set());
   915→    setBonusPoint(false);
   916→
   917→    // Start 5-second raid gap timer
   918→    setRaidGapTimer(RAID_GAP_TIMEOUT);
   919→  };
   920→
   921→  // Keep ref in sync with latest processRaidResult
   922→  // This is a standard React pattern for accessing latest closure from effects

   924→  processRaidResultRef.current = processRaidResult;
   925→
   926→  const cancelRaid = () => {
   927→    setRaidTimer(null);
   928→    if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   929→    setRaidPhase('idle');
   930→    setRaider(null);
   931→    setRaidResult(null);
   932→    setSelectedDefenders(new Set());
   933→    setBonusPoint(false);
   934→
   935→    if (hasStartedRaiding) setRaidGapTimer(RAID_GAP_TIMEOUT);
   936→  };
   937→
   938→  const handleEndMatch = async () => {

   944→        homeTeam: match.homeTeam, awayTeam: match.awayTeam,
   945→        homeScore: match.homeScore, awayScore: match.awayScore,

   950→      try {
   951→        await fetch('/api/matches', {
   952→          method: 'POST',
   953→          headers: { 'Content-Type': 'application/json' },
   954→          body: JSON.stringify({
   955→            homeTeamName: match.homeTeam, awayTeamName: match.awayTeam,
   956→            homeTeamColor: match.homeTeamColor, awayTeamColor: match.awayTeamColor,
   957→            homeScore: match.homeScore, awayScore: match.awayScore,
   958→            gender: match.gender, isPractice: match.isPractice,
   959→            halfDuration: match.halfDuration, playersPerSide: match.playersPerSide,
   960→            events: match.events.map(e => ({
   961→              eventType: e.eventType, teamId: e.teamId, half: e.half,
   962→              playerId: e.playerId, value: e.value, details: e.details,
   963→            })),
   964→          }),
   965→        });
   966→      } catch (err) {
   967→        console.error('Failed to save match:', err);
   968→      }
   969→
   970→      addNotification(matchNotification(match.homeTeam, match.awayTeam, match.homeScore, match.awayScore));
   971→
   972→      if (motm) {

   978→      setShowEndMatchConfirm(false);
   979→      setShowMatchEndCelebration(true);
   980→    } else {
   981→      setShowEndMatchConfirm(true);
   982→      setTimeout(() => setShowEndMatchConfirm(false), 3000);
   983→    }
   984→  };
   985→
   986→  const handleEndHalf = () => {
   987→    if (showEndHalfConfirm) {
   988→      triggerFeedback(SoundType.HALF_END);
   989→      switchHalf();
   990→      setHasStartedRaiding(false);
   991→      setRaidTimer(null);
   992→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);

   995→      setRaider(null);
   996→      setDoOrDie(false);

  1000→      setShowHalfTimeTransition(true);

  1002→      setShowEndHalfConfirm(true);

  1007→  const handlePause = () => {
  1008→    const newPaused = !isPaused;

  1010→
  1011→    if (newPaused) {

  1019→              if (raidTimerRef.current) clearInterval(raidTimerRef.current);
  1020→              return 0;
  1021→            }
  1022→            return prev - 1;

  1027→  };
  1028→

  1035→
  1036→  const handleTimeout = () => {

  1038→    const teamId = raidingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
  1039→    callTimeout(raidingTeam);
  1040→    setIsPaused(true);
  1041→    if (raidTimerRef.current) clearInterval(raidTimerRef.current);

  1044→      matchId: match.id, eventType: 'timeout', teamId,
  1045→      half: match.currentHalf, value: 0,

  1048→  };
  1049→
  1050→  // Substitute handler
  1051→  const handleSub = (outPlayer: MatchPlayer, inPlayer: MatchPlayer) => {
  1052→    const teamId = outPlayer.team === 'home' ? match.homeTeamId : match.awayTeamId;
  1053→    addEvent({
  1054→      matchId: match.id,
  1055→      eventType: 'substitution' as const,
  1056→      teamId,
  1057→      half: match.currentHalf,
  1058→      playerId: inPlayer.id,
