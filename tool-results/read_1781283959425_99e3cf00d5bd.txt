    74→function CircularGauge({
    75→  value,
    76→  max = 100,
    77→  size = 100,
    78→  strokeWidth = 8,
    79→  color = '#14B8A6',
    80→  label,
    81→  sublabel,
    82→  children,
    83→}: {
    84→  value: number;
    85→  max?: number;
    86→  size?: number;
    87→  strokeWidth?: number;
    88→  color?: string;
    89→  label?: string;
    90→  sublabel?: string;
    91→  children?: React.ReactNode;
    92→}) {
    93→  const radius = (size - strokeWidth) / 2;
    94→  const circumference = 2 * Math.PI * radius;
    95→  const percentage = Math.min(value / max, 1);
    96→  const offset = circumference - percentage * circumference;
    97→
    98→  return (
    99→    <div className="flex flex-col items-center gap-1">
   100→      <div className="relative" style={{ width: size, height: size }}>
   101→        <svg width={size} height={size} className="-rotate-90">
   102→          {/* Background circle */}
   103→          <circle
   104→            cx={size / 2}
   105→            cy={size / 2}
   106→            r={radius}
   107→            fill="none"
   108→            stroke="currentColor"
   109→            strokeWidth={strokeWidth}
   110→            className="text-warm-200"
   111→          />
   112→          {/* Progress circle */}
   113→          <motion.circle
   114→            cx={size / 2}
   115→            cy={size / 2}
   116→            r={radius}
   117→            fill="none"
   118→            stroke={color}
   119→            strokeWidth={strokeWidth}
   120→            strokeLinecap="round"
   121→            strokeDasharray={circumference}
   122→            initial={{ strokeDashoffset: circumference }}
   123→            animate={{ strokeDashoffset: offset }}
   124→            transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   125→          />
   126→        </svg>
   127→        {/* Center content */}
   128→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   129→          {children || (
   130→            <>
   131→              <motion.span
   132→                className="text-lg font-black"
   133→                style={{ color }}
   134→                initial={{ opacity: 0, scale: 0.5 }}
   135→                animate={{ opacity: 1, scale: 1 }}
   136→                transition={{ delay: 0.6, type: 'spring', damping: 12 }}
   137→              >
   138→                {value.toFixed(value % 1 === 0 ? 0 : 1)}
   139→              </motion.span>
   140→              {label && (
   141→                <span className="text-[9px] font-semibold text-warm-500 uppercase tracking-wide">
   142→                  {label}
   143→                </span>
   144→              )}
   145→            </>
   146→          )}
   147→        </div>
   148→      </div>
   149→      {sublabel && (
   150→        <span className="text-[10px] text-warm-500 font-medium">{sublabel}</span>
   151→      )}
   152→    </div>
   153→  );
   154→}

   158→function SkillRadar({ dimensions }: { dimensions: { label: string; value: number }[] }) {
   159→  const size = 220;
   160→  const center = size / 2;
   161→  const radius = 80;
   162→  const levels = 4;
   163→  const count = dimensions.length;
   164→  const angleStep = (2 * Math.PI) / count;
   165→
   166→  const getPoint = (index: number, value: number) => {
   167→    const angle = angleStep * index - Math.PI / 2;
   168→    const r = (value / 100) * radius;
   169→    return {
   170→      x: center + r * Math.cos(angle),
   171→      y: center + r * Math.sin(angle),
   172→    };
   173→  };
   174→
   175→  const radarPoints = dimensions.map((d, i) => getPoint(i, d.value));
   176→  const radarPath = radarPoints.map((p) => `${p.x},${p.y}`).join(' ');
   177→
   178→  return (
   179→    <div className="flex flex-col items-center">
   180→      <svg width={size} height={size} viewBox={`0 0 ${size} ${size}`}>
   181→        {/* Grid levels */}
   182→        {Array.from({ length: levels }).map((_, level) => {
   183→          const r = radius * ((level + 1) / levels);
   184→          const points = Array.from({ length: count }).map((_, i) => {
   185→            const angle = angleStep * i - Math.PI / 2;
   186→            return `${center + r * Math.cos(angle)},${center + r * Math.sin(angle)}`;
   187→          });
   188→          return (
   189→            <polygon
   190→              key={level}
   191→              points={points.join(' ')}
   192→              fill="none"
   193→              stroke="#E2E8F0"
   194→              strokeWidth={1}
   195→            />
   196→          );
   197→        })}
   198→
   199→        {/* Axis lines */}
   200→        {dimensions.map((_, i) => {
   201→          const angle = angleStep * i - Math.PI / 2;
   202→          return (
   203→            <line
   204→              key={i}
   205→              x1={center}
   206→              y1={center}
   207→              x2={center + radius * Math.cos(angle)}
   208→              y2={center + radius * Math.sin(angle)}
   209→              stroke="#E2E8F0"
   210→              strokeWidth={1}
   211→            />
   212→          );
   213→        })}
   214→
   215→        {/* Radar fill */}
   216→        <motion.polygon
   217→          points={radarPath}
   218→          fill="rgba(20, 184, 166, 0.2)"
   219→          stroke="#14B8A6"
   220→          strokeWidth={2}
   221→          initial={{ opacity: 0, scale: 0.3 }}
   222→          animate={{ opacity: 1, scale: 1 }}
   223→          transition={{ duration: 0.8, ease: 'easeOut', delay: 0.4 }}
   224→          style={{ transformOrigin: `${center}px ${center}px` }}
   225→        />
   226→
   227→        {/* Data points */}
   228→        {radarPoints.map((p, i) => (
   229→          <motion.circle
   230→            key={i}
   231→            cx={p.x}
   232→            cy={p.y}
   233→            r={4}
   234→            fill="#14B8A6"
   235→            stroke="white"
   236→            strokeWidth={2}
   237→            initial={{ r: 0 }}
   238→            animate={{ r: 4 }}
   239→            transition={{ delay: 0.8 + i * 0.1, type: 'spring', damping: 12 }}
   240→          />
   241→        ))}
   242→
   243→        {/* Labels */}
   244→        {dimensions.map((d, i) => {
   245→          const angle = angleStep * i - Math.PI / 2;
   246→          const labelR = radius + 22;
   247→          const x = center + labelR * Math.cos(angle);
   248→          const y = center + labelR * Math.sin(angle);
   249→          const anchor = Math.abs(Math.cos(angle)) < 0.1 ? 'middle' : Math.cos(angle) > 0 ? 'start' : 'end';
   250→          const dy = Math.sin(angle) < -0.5 ? -4 : Math.sin(angle) > 0.5 ? 8 : 2;
   251→          return (
   252→            <text
   253→              key={i}
   254→              x={x}
   255→              y={y + dy}
   256→              textAnchor={anchor}
   257→              className="fill-warm-700"
   258→              fontSize="9"
   259→              fontWeight="700"
   260→            >
   261→              {d.label}
   262→            </text>
   263→          );
   264→        })}
   265→      </svg>
   266→    </div>
   267→  );
   268→}

   367→// ─── Simulated match history ─────────────────────────────────────
   368→
   369→function generateMatchHistory(
   370→  profile: PlayerProfileData,
   371→  user: PlayerUser,
   372→): MatchHistoryEntry[] {
   373→  const totalPoints = profile.successfulRaids + profile.bonusPoints + profile.successfulTackles;
   374→  const matchCount = Math.max(1, Math.round(totalPoints / 6));
   375→  const opponents = [
   376→    'Warriors FC', 'Royal Titans', 'Storm Riders', 'Thunder Hawks',
   377→    'Phoenix United', 'Dark Panthers', 'Golden Eagles', 'Silver Sharks',
   378→  ];
   379→  const matches: MatchHistoryEntry[] = [];
   380→
   381→  for (let i = 0; i < Math.min(7, matchCount); i++) {
   382→    const pts = Math.floor(Math.random() * 8) + 2;
   383→    const teamScore = pts + Math.floor(Math.random() * 15) + 10;
   384→    const oppScore = Math.floor(Math.random() * 25) + 8;
   385→    const result: 'W' | 'L' | 'D' = teamScore > oppScore ? 'W' : teamScore < oppScore ? 'L' : 'D';
   386→
   387→    matches.push({
   388→      id: `match_${i}`,
   389→      opponent: opponents[i % opponents.length],
   390→      score: `${teamScore}-${oppScore}`,
   391→      playerPoints: pts,
   392→      result,
   393→    });
   394→  }
   395→
   396→  return matches;
   397→}

   399→// ─── Simulated raid trend ────────────────────────────────────────
   400→
   401→function generateRaidTrend(profile: PlayerProfileData): number[] {
   402→  if (profile.totalRaids === 0) return [0, 0, 0, 0, 0];
   403→  const avg = profile.successfulRaids / Math.max(1, profile.totalRaids) * 100;
   404→  return Array.from({ length: 5 }, () =>
   405→    Math.max(0, Math.min(100, avg + (Math.random() - 0.5) * 30))
   406→  );
   407→}
