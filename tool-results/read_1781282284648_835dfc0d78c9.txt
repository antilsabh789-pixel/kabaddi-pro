    79→// ─── Animated Value (count-up on first view) ────────────────────

    81→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    82→  const [display, setDisplay] = useState(0);
    83→  const [hasAnimated, setHasAnimated] = useState(false);
    84→  const ref = useRef<HTMLSpanElement>(null);
    85→  const valueRef = useRef(value);
    86→
    87→  useEffect(() => {
    88→    valueRef.current = value;
    89→  }, [value]);
    90→
    91→  useEffect(() => {
    92→    if (hasAnimated) return;
    93→    if (value === 0) return;
    94→
    95→    const observer = new IntersectionObserver(
    96→      ([entry]) => {
    97→        if (entry.isIntersecting && !hasAnimated) {
    98→          setHasAnimated(true);
    99→          const targetVal = valueRef.current;
   100→          const duration = 1200;
   101→          const startTime = performance.now();
   102→
   103→          function step(currentTime: number) {
   104→            const elapsed = currentTime - startTime;
   105→            const progress = Math.min(elapsed / duration, 1);
   106→            const eased = 1 - Math.pow(1 - progress, 3);
   107→            const current = eased * targetVal;
   108→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
   109→            if (progress < 1) {
   110→              requestAnimationFrame(step);
   111→            }
   112→          }
   113→
   114→          requestAnimationFrame(step);
   115→        }
   116→      },
   117→      { threshold: 0.1 }
   118→    );
   119→
   120→    if (ref.current) observer.observe(ref.current);
   121→    return () => observer.disconnect();
   122→  }, [value, decimals, hasAnimated]);
   123→
   124→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   125→}

   142→// ─── Time Ago helper ────────────────────
   143→
   144→function timeAgo(dateStr: string): string {
   145→  if (!dateStr || dateStr === '—') return '';
   146→  try {
   147→    const date = new Date(dateStr);
   148→    if (isNaN(date.getTime())) return dateStr;
   149→    const now = new Date();
   150→    const diff = now.getTime() - date.getTime();
   151→    const mins = Math.floor(diff / 60000);
   152→    if (mins < 60) return mins <= 1 ? 'just now' : `${mins}m ago`;
   153→    const hours = Math.floor(mins / 60);
   154→    if (hours < 24) return `${hours}h ago`;
   155→    const days = Math.floor(hours / 24);
   156→    if (days < 7) return `${days}d ago`;
   157→    const weeks = Math.floor(days / 7);
   158→    if (weeks < 4) return `${weeks}w ago`;
   159→    return dateStr;
   160→  } catch {
   161→    return dateStr;
   162→  }
   163→}

   167→function getPlayerLevel(totalMatches: number, totalPoints: number): { label: string; color: string; icon: string; progress: number } {
   168→  const score = totalMatches * 2 + totalPoints;
   169→  if (score >= 500) return { label: 'Legend', color: 'from-amber-400 to-yellow-600', icon: '👑', progress: 100 };
   170→  if (score >= 200) return { label: 'Pro', color: 'from-red-500 to-orange-500', icon: '🔥', progress: Math.min(((score - 200) / 300) * 100, 100) };
   171→  if (score >= 50) return { label: 'Intermediate', color: 'from-teal-500 to-emerald-500', icon: '⚡', progress: Math.min(((score - 50) / 150) * 100, 100) };
   172→  return { label: 'Beginner', color: 'from-slate-400 to-slate-500', icon: '🌱', progress: Math.min((score / 50) * 100, 100) };
   173→}
