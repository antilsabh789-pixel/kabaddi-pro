    53→// ─── Position Helpers ────────────────────────────────────────────────
    54→
    55→type PositionType = 'raider' | 'defender' | 'all-rounder';
    56→
    57→function getPositionType(position: string | null): PositionType {
    58→  if (!position) return 'all-rounder';
    59→  const p = position.toLowerCase();
    60→  if (p.includes('raider')) return 'raider';
    61→  if (p.includes('corner') || p.includes('cover') || p.includes('defend')) return 'defender';
    62→  return 'all-rounder';
    63→}
    64→
    65→function getPositionColor(position: string | null) {
    66→  const type = getPositionType(position);
    67→  switch (type) {
    68→    case 'raider': return { ring: 'ring-red-500', bg: 'bg-red-500', text: 'text-red-500', gradient: 'from-red-500 to-orange-500', border: 'border-red-500/30' };
    69→    case 'defender': return { ring: 'ring-blue-500', bg: 'bg-blue-500', text: 'text-blue-500', gradient: 'from-blue-500 to-cyan-500', border: 'border-blue-500/30' };
    70→    case 'all-rounder': return { ring: 'ring-amber-500', bg: 'bg-amber-500', text: 'text-amber-500', gradient: 'from-amber-500 to-yellow-400', border: 'border-amber-500/30' };
    71→  }
    72→}

    99→// ─── Animated Counter ────────────────────────────────────────────────
   100→
   101→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   102→  const [display, setDisplay] = useState(value);
   103→  const prevValueRef = useRef(value);
   104→
   105→  useEffect(() => {
   106→    const end = value;
   107→    if (end === 0) {
   108→      prevValueRef.current = 0;
   109→      return;
   110→    }
   111→    const start = prevValueRef.current;
   112→    if (start === end) return;
   113→    const increment = (end - start) / (duration / 16);
   114→    let current = start;
   115→    const timer = setInterval(() => {
   116→      current += increment;
   117→      if ((increment > 0 && current >= end) || (increment < 0 && current <= end)) {
   118→        setDisplay(end);
   119→        prevValueRef.current = end;
   120→        clearInterval(timer);
   121→      } else {
   122→        setDisplay(Math.floor(current));
   123→      }
   124→    }, 16);
   125→    return () => clearInterval(timer);
   126→  }, [value, duration]);
   127→
   128→  return <>{display}</>;
   129→}

   131→// ─── Circular Rating Indicator ───────────────────────────────────────
   132→
   133→function CircularRating({ rating, size = 48 }: { rating: number; size?: number }) {
   134→  const radius = (size - 8) / 2;
   135→  const circumference = 2 * Math.PI * radius;
   136→  const progress = Math.min(rating / 10, 1);
   137→  const offset = circumference - progress * circumference;
   138→
   139→  const getColor = (r: number) => {
   140→    if (r >= 8) return '#22c55e';
   141→    if (r >= 6) return '#eab308';
   142→    if (r >= 4) return '#f97316';
   143→    return '#ef4444';
   144→  };
   145→
   146→  return (
   147→    <div className="relative" style={{ width: size, height: size }}>
   148→      <svg width={size} height={size} className="-rotate-90">
   149→        <circle
   150→          cx={size / 2}
   151→          cy={size / 2}
   152→          r={radius}
   153→          stroke="rgba(255,255,255,0.15)"
   154→          strokeWidth="4"
   155→          fill="none"
   156→        />
   157→        <circle
   158→          cx={size / 2}
   159→          cy={size / 2}
   160→          r={radius}
   161→          stroke={getColor(rating)}
   162→          strokeWidth="4"
   163→          fill="none"
   164→          strokeLinecap="round"
   165→          strokeDasharray={circumference}
   166→          strokeDashoffset={offset}
   167→          className="transition-all duration-1000 ease-out"
   168→        />
   169→      </svg>
   170→      <div className="absolute inset-0 flex items-center justify-center">
   171→        <span className="text-xs font-black text-white">{rating.toFixed(1)}</span>
   172→      </div>
   173→    </div>
   174→  );
   175→}
