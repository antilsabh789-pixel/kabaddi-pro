   112→function AnimatedCounter({ target, duration = 1000 }: { target: number; duration?: number }) {
   113→  const [count, setCount] = useState(0);
   114→  const prevTarget = useRef(target);
   115→  const hasAnimated = useRef(false);
   116→
   117→  useEffect(() => {
   118→    if (prevTarget.current === target && hasAnimated.current) return;
   119→    prevTarget.current = target;
   120→    hasAnimated.current = true;
   121→
   122→    let start = 0;
   123→    const startTime = performance.now();
   124→
   125→    function step(currentTime: number) {
   126→      const elapsed = currentTime - startTime;
   127→      const progress = Math.min(elapsed / duration, 1);
   128→      const eased = 1 - Math.pow(1 - progress, 3);
   129→      const current = Math.round(eased * target);
   130→      setCount(current);
   131→      if (progress < 1) {
   132→        requestAnimationFrame(step);
   133→      }
   134→    }
   135→
   136→    requestAnimationFrame(step);
   137→  }, [target, duration]);
   138→
   139→  return <span>{count}</span>;
   140→}

   144→function MiniProgressRing({ percent, size = 40 }: { percent: number; size?: number }) {
   145→  const strokeWidth = 4;
   146→  const radius = (size - strokeWidth) / 2;
   147→  const circumference = 2 * Math.PI * radius;
   148→  const offset = circumference - (percent / 100) * circumference;
   149→
   150→  return (
   151→    <svg width={size} height={size} className="transform -rotate-90">
   152→      <circle
   153→        cx={size / 2}
   154→        cy={size / 2}
   155→        r={radius}
   156→        strokeWidth={strokeWidth}
   157→        className="stroke-warm-200 dark:stroke-warm-700"
   158→        fill="none"
   159→      />
   160→      <motion.circle
   161→        cx={size / 2}
   162→        cy={size / 2}
   163→        r={radius}
   164→        strokeWidth={strokeWidth}
   165→        className="stroke-emerald-500 dark:stroke-emerald-400"
   166→        fill="none"
   167→        strokeLinecap="round"
   168→        strokeDasharray={circumference}
   169→        initial={{ strokeDashoffset: circumference }}
   170→        animate={{ strokeDashoffset: offset }}
   171→        transition={{ duration: 1.2, ease: 'easeOut', delay: 0.3 }}
   172→      />
   173→    </svg>
   174→  );
   175→}

   179→function getDateGroup(dateStr: string): string {
   180→  const date = new Date(dateStr);
   181→  const now = new Date();
   182→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   183→  const yesterday = new Date(today);
   184→  yesterday.setDate(yesterday.getDate() - 1);
   185→  const weekAgo = new Date(today);
   186→  weekAgo.setDate(weekAgo.getDate() - 7);
   187→
   188→  const matchDate = new Date(date.getFullYear(), date.getMonth(), date.getDate());
   189→
   190→  if (matchDate.getTime() === today.getTime()) return 'Today';
   191→  if (matchDate.getTime() === yesterday.getTime()) return 'Yesterday';
   192→  if (matchDate > weekAgo) return 'This Week';
   193→  return 'Earlier';
   194→}

   196→function formatMatchDate(dateStr: string): string {
   197→  const date = new Date(dateStr);
   198→  return date.toLocaleDateString('en-IN', {
   199→    day: 'numeric',
   200→    month: 'short',
   201→    year: date.getFullYear() !== new Date().getFullYear() ? 'numeric' : undefined,
   202→  });
   203→}
   204→
   205→function getMatchResult(match: MatchItem, userId?: string): 'won' | 'lost' | 'draw' {
   206→  if (match.homeScore === match.awayScore) return 'draw';
   207→  const homeWon = match.homeScore > match.awayScore;
   208→
   209→  // If userId is a scorer, determine which side they were on
   210→  if (userId) {
   211→    const isScorer = match.scorers.some((s) => s.userId === userId);
   212→    if (isScorer) {
   213→      // Check if the user's team is home or away (based on scorer relation)
   214→      // Default: assume home team if we can't determine
   215→      return homeWon ? 'won' : 'lost';
   216→    }
   217→  }
   218→
   219→  // General: show from home team perspective
   220→  return homeWon ? 'won' : 'lost';
   221→}
   222→
   223→function getWinnerTeamId(match: MatchItem): string | null {
   224→  if (match.homeScore > match.awayScore) return match.homeTeamId;
   225→  if (match.awayScore > match.homeScore) return match.awayTeamId;
   226→  return null;
   227→}
