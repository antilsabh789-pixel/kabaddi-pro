    63→// ─── Animated Value (count-up on first view) ────────────────────
    64→
    65→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    66→  const [display, setDisplay] = useState(0);
    67→  const [hasAnimated, setHasAnimated] = useState(false);
    68→  const ref = useRef<HTMLSpanElement>(null);
    69→  const valueRef = useRef(value);
    70→
    71→  useEffect(() => {
    72→    valueRef.current = value;
    73→  }, [value]);
    74→
    75→  useEffect(() => {
    76→    if (hasAnimated) return;
    77→    if (value === 0) return;
    78→
    79→    const observer = new IntersectionObserver(
    80→      ([entry]) => {
    81→        if (entry.isIntersecting && !hasAnimated) {
    82→          setHasAnimated(true);
    83→          const targetVal = valueRef.current;
    84→          const duration = 800;
    85→          const startTime = performance.now();
    86→
    87→          function step(currentTime: number) {
    88→            const elapsed = currentTime - startTime;
    89→            const progress = Math.min(elapsed / duration, 1);
    90→            const eased = 1 - Math.pow(1 - progress, 3);
    91→            const current = eased * targetVal;
    92→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
    93→            if (progress < 1) {
    94→              requestAnimationFrame(step);
    95→            }
    96→          }
    97→
    98→          requestAnimationFrame(step);
    99→        }
   100→      },
   101→      { threshold: 0.1 }
   102→    );
   103→
   104→    if (ref.current) observer.observe(ref.current);
   105→    return () => observer.disconnect();
   106→  }, [value, decimals, hasAnimated]);
   107→
   108→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   109→}

   495→  const getMatchResult = (match: RecentMatch): 'W' | 'L' | 'D' => {
   496→    if (match.userTeamSide === 'unknown') {
   497→      return match.homeScore === match.awayScore ? 'D' : 'D';
   498→    }
   499→    const myScore = match.userTeamSide === 'home' ? match.homeScore : match.awayScore;
   500→    const oppScore = match.userTeamSide === 'home' ? match.awayScore : match.homeScore;
   501→    if (myScore > oppScore) return 'W';
