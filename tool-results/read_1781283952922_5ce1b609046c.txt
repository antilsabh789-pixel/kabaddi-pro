   130→function SeasonComparisonChart({ seasons }: { seasons: Season[] }) {
   131→  const maxTeams = Math.max(...seasons.map((s) => s.teamCount), 1);
   132→  const maxMatches = Math.max(...seasons.map((s) => s.matchCount), 1);
   133→  const barWidth = 28;
   134→  const gap = 12;
   135→  const chartWidth = seasons.length * (barWidth + gap) + 60;
   136→  const chartHeight = 140;
   137→  const barAreaHeight = 100;
   138→  const barAreaY = 20;
   139→
   140→  return (
   141→    <div className="w-full overflow-x-auto">
   142→      <svg
   143→        width={Math.max(chartWidth, 260)}
   144→        height={chartHeight}
   145→        viewBox={`0 0 ${Math.max(chartWidth, 260)} ${chartHeight}`}
   146→        className="mx-auto"
   147→      >
   148→        {/* Y-axis labels */}
   149→        <text x="8" y={barAreaY + 5} className="fill-warm-400 dark:fill-warm-500" fontSize="8" textAnchor="start">
   150→          {maxTeams}T
   151→        </text>
   152→        <text x="8" y={barAreaY + barAreaHeight / 2} className="fill-warm-400 dark:fill-warm-500" fontSize="8" textAnchor="start">
   153→          {Math.round(maxTeams / 2)}T
   154→        </text>
   155→        <text x="8" y={barAreaY + barAreaHeight} className="fill-warm-400 dark:fill-warm-500" fontSize="8" textAnchor="start">
   156→          0
   157→        </text>
   158→
   159→        {/* Grid lines */}
   160→        <line
   161→          x1="30" y1={barAreaY} x2={chartWidth - 10} y2={barAreaY}
   162→          stroke="currentColor" strokeWidth="0.5" className="text-warm-200 dark:text-warm-700"
   163→        />
   164→        <line
   165→          x1="30" y1={barAreaY + barAreaHeight / 2} x2={chartWidth - 10} y2={barAreaY + barAreaHeight / 2}
   166→          stroke="currentColor" strokeWidth="0.5" strokeDasharray="4 2" className="text-warm-200 dark:text-warm-700"
   167→        />
   168→        <line
   169→          x1="30" y1={barAreaY + barAreaHeight} x2={chartWidth - 10} y2={barAreaY + barAreaHeight}
   170→          stroke="currentColor" strokeWidth="0.5" className="text-warm-200 dark:text-warm-700"
   171→        />
   172→
   173→        {/* Bars */}
   174→        {seasons.map((season, i) => {
   175→          const x = 40 + i * (barWidth + gap);
   176→          const teamHeight = maxTeams > 0 ? (season.teamCount / maxTeams) * barAreaHeight : 0;
   177→          const matchHeight = maxMatches > 0 ? (season.matchCount / maxMatches) * barAreaHeight : 0;
   178→
   179→          return (
   180→            <g key={season.id}>
   181→              {/* Teams bar */}
   182→              <motion.rect
   183→                x={x}
   184→                y={barAreaY + barAreaHeight - teamHeight}
   185→                width={barWidth / 2 - 2}
   186→                height={teamHeight}
   187→                rx={3}
   188→                className="fill-brand-teal"
   189→                initial={{ height: 0, y: barAreaY + barAreaHeight }}
   190→                animate={{ height: teamHeight, y: barAreaY + barAreaHeight - teamHeight }}
   191→                transition={{ delay: i * 0.1, duration: 0.5, ease: 'easeOut' }}
   192→              />
   193→              {/* Matches bar */}
   194→              <motion.rect
   195→                x={x + barWidth / 2 + 1}
   196→                y={barAreaY + barAreaHeight - matchHeight}
   197→                width={barWidth / 2 - 2}

   206→              <text

   220→        {/* Legend */}

   230→// ─── Progress Tracker Component ───────────────────────────────────
   231→
   232→function SeasonProgressTracker({ season }: { season: Season }) {
   233→  const completedMatches = season.matches.filter((m) => m.status === 'completed').length;
   234→  const totalMatches = season.matchCount || 1;
   235→  const matchProgress = (completedMatches / totalMatches) * 100;
   236→  const totalTeams = season.teamCount;
   237→
   238→  // Estimate total expected matches (round robin)
   239→  const expectedMatches = totalTeams > 1 ? (totalTeams * (totalTeams - 1)) / 2 : totalMatches;
   240→  const seasonProgress = Math.min(100, (completedMatches / Math.max(expectedMatches, 1)) * 100);
   241→
   242→  return (
   243→    <motion.div
   244→      initial={{ opacity: 0, y: 10 }}
   245→      animate={{ opacity: 1, y: 0 }}
   246→      transition={{ delay: 0.2 }}
   247→      className="bg-white dark:bg-warm-800/50 rounded-2xl border border-warm-200/60 dark:border-warm-700/60 p-4 space-y-3"
   248→    >
   249→      <div className="flex items-center gap-2">
   250→        <div className="w-6 h-6 rounded-md bg-brand-teal/10 flex items-center justify-center">
   251→          <Activity className="w-3.5 h-3.5 text-brand-teal" />
   252→        </div>
   253→        <h3 className="text-xs font-black tracking-wider text-warm-800 dark:text-warm-200">SEASON PROGRESS</h3>
   254→      </div>
   255→
   256→      {/* Main Progress Bar */}
   257→      <div>
   258→        <div className="flex justify-between text-[10px] mb-1.5">
   259→          <span className="text-warm-500 dark:text-warm-400">Season Completion</span>
   260→          <span className="font-bold text-brand-teal">{Math.round(seasonProgress)}%</span>
   261→        </div>
   262→        <div className="h-2.5 bg-warm-100 dark:bg-warm-700 rounded-full overflow-hidden">
   263→          <motion.div
   264→            className="h-full rounded-full bg-gradient-to-r from-brand-teal to-emerald-400"
   265→            initial={{ width: 0 }}
   266→            animate={{ width: `${seasonProgress}%` }}
   267→            transition={{ duration: 1, ease: 'easeOut', delay: 0.3 }}
   268→          />
   269→        </div>
   270→      </div>
   271→
   272→      {/* Stat Chips */}
   273→      <div className="grid grid-cols-3 gap-2">
   274→        <div className="text-center p-2 bg-warm-50 dark:bg-warm-700/30 rounded-xl">
   275→          <p className="text-lg font-black text-brand-navy dark:text-brand-navy-light">{completedMatches}</p>
   276→          <p className="text-[9px] text-warm-400 dark:text-warm-500 font-medium">Played</p>
   277→        </div>
   278→        <div className="text-center p-2 bg-warm-50 dark:bg-warm-700/30 rounded-xl">
   279→          <p className="text-lg font-black text-brand-teal">{totalMatches}</p>
   280→          <p className="text-[9px] text-warm-400 dark:text-warm-500 font-medium">Total</p>
   281→        </div>
   282→        <div className="text-center p-2 bg-warm-50 dark:bg-warm-700/30 rounded-xl">
   283→          <p className="text-lg font-black text-brand-gold">{totalTeams}</p>
   284→          <p className="text-[9px] text-warm-400 dark:text-warm-500 font-medium">Teams</p>
   285→        </div>
   286→      </div>
   287→
   288→      {/* Match Progress Sub-bar */}
   289→      {matchProgress > 0 && (
   290→        <div>
   291→          <div className="flex justify-between text-[9px] mb-1">
   292→            <span className="text-warm-400 dark:text-warm-500">Matches Played</span>
   293→            <span className="text-warm-500 dark:text-warm-400 font-semibold">{completedMatches}/{totalMatches}</span>
   294→          </div>
   295→          <div className="h-1.5 bg-warm-100 dark:bg-warm-700 rounded-full overflow-hidden">
   296→            <motion.div
   297→              className="h-full rounded-full bg-gradient-to-r from-brand-navy to-brand-navy-light"
   298→              initial={{ width: 0 }}
   299→              animate={{ width: `${matchProgress}%` }}
   300→              transition={{ duration: 0.8, ease: 'easeOut', delay: 0.5 }}
   301→            />
   302→          </div>
   303→        </div>
   304→      )}
   305→    </motion.div>
   306→  );
   307→}
