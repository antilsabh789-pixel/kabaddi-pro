   174→    // Reset 5-min warning flag when half changes

   176→      fiveMinWarningFiredRef.current = false;

   178→

   181→      if (match.currentHalf === 1) {
   182→        // Half 1 complete
   183→        triggerFeedback(SoundType.HALF_END);
   184→        toast({

   227→          if (raidTimerRef.current) clearInterval(raidTimerRef.current);

   233→
   234→    return () => {
   235→      if (raidTimerRef.current) clearInterval(raidTimerRef.current);
   236→    };

   246→      vibrate([40]);

   248→    // At 5 seconds: urgent double vibration

   252→    // At 3, 2, 1 seconds: countdown ticks
   253→    if (raidTimer <= 3 && raidTimer > 0) {

   266→    if (raidGapTimer <= 0) {
   267→      if (raidGapRef.current) clearInterval(raidGapRef.current);

   282→          if (raidGapRef.current) clearInterval(raidGapRef.current);
   283→          return 0;
   284→        }
   285→        return prev - 1;
   286→      });

   288→

   290→      if (raidGapRef.current) clearInterval(raidGapRef.current);
   291→    };

   294→  // Calculate MOTM

   301→          playerPoints[event.playerId] = { name: event.playerName, points: 0 };
   302→        }
   303→        playerPoints[event.playerId].points += event.value;
   304→      }
   305→    }
   306→    let topPlayer: { name: string; points: number } | null = null;
   307→    for (const p of Object.values(playerPoints)) {
   308→      if (!topPlayer || p.points > topPlayer.points) topPlayer = p;
   309→    }
   310→    return topPlayer;
   311→  }, [match]);
   312→
   313→  const calculateTopRaiderDefender = useCallback(() => {
   314→    if (!match) return { topRaider: null, topDefender: null };
   315→    const playerRaidPoints: Record<string, { name: string; points: number }> = {};
   316→    const playerTacklePoints: Record<string, { name: string; points: number }> = {};
   317→
   318→    for (const event of match.events) {
   319→      if (!event.playerId || !event.playerName) continue;
   320→      if (['raid_point', 'bonus_point', 'super_raid', 'do_or_die_raid'].includes(event.eventType)) {
   321→        if (!playerRaidPoints[event.playerId]) playerRaidPoints[event.playerId] = { name: event.playerName, points: 0 };
   322→        playerRaidPoints[event.playerId].points += event.value;
   323→      }
   324→      if (['tackle_point', 'super_tackle'].includes(event.eventType)) {
   325→        if (!playerTacklePoints[event.playerId]) playerTacklePoints[event.playerId] = { name: event.playerName, points: 0 };
   326→        playerTacklePoints[event.playerId].points += event.value;
   327→      }
   328→    }
   329→
   330→    let topRaider: { name: string; points: number } | null = null;
   331→    for (const p of Object.values(playerRaidPoints)) {
   332→      if (!topRaider || p.points > topRaider.points) topRaider = p;
   333→    }
   334→    let topDefender: { name: string; points: number } | null = null;
   335→    for (const p of Object.values(playerTacklePoints)) {
   336→      if (!topDefender || p.points > topDefender.points) topDefender = p;
   337→    }
   338→    return { topRaider, topDefender };
   339→  }, [match]);
   340→
   341→  if (!match) return null;
   342→
   343→  // ═══ DERIVED STATE ═══
   344→  const raidingTeam = match.raidQueue;
   345→  const defendingTeam: 'home' | 'away' = raidingTeam === 'home' ? 'away' : 'home';
   346→
   347→  const fullRaidingLineup = raidingTeam === 'home' ? match.homeLineup : match.awayLineup;
   348→  const fullDefendingLineup = defendingTeam === 'home' ? match.homeLineup : match.awayLineup;
   349→
   350→  const raidingOutIds = raidingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   351→  const defendingOutIds = defendingTeam === 'home' ? match.homeOutPlayerIds : match.awayOutPlayerIds;
   352→
   353→  const raidingTeamColor = raidingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   354→  const defendingTeamColor = defendingTeam === 'home' ? match.homeTeamColor : match.awayTeamColor;
   355→

   357→  const defendingTeamName = defendingTeam === 'home' ? match.homeTeam : match.awayTeam;

   359→  // Split lineup: first 7 = on court, rest = substitutes
   360→  const splitLineup = (lineup: MatchPlayer[], outIds: string[]) => {

   362→    const substitutes = lineup.slice(ON_COURT_MAX);
   363→    const onCourtActive = onCourt.filter(p => !outIds.includes(p.id));
   364→    const onCourtOut = onCourt.filter(p => outIds.includes(p.id));
   365→    return { onCourt, substitutes, onCourtActive, onCourtOut };
   366→  };
   367→
   368→  // Handle raider selection → starts raid timer AND match timer
   369→  const handleSelectRaider = (player: MatchPlayer) => {
   370→    if (raidingOutIds.includes(player.id)) return;
   371→    if (raidPhase !== 'idle') return;
   372→

   375→

   378→
   379→    // Clear raid gap timer

   383→    setRaidPhase('result');
   384→    setRaidResult(null);
   385→    setSelectedDefenders(new Set());

   394→    } else {

   396→    }

   398→
   399→  // Handle raid result selection
   400→  const handleSelectResult = (result: RaidResult) => {
   401→    if (result === 'empty') {
   402→      processRaidResult(result, new Set(), false);
   403→      return;
   404→    }
   405→    setRaidResult(result);
   406→    setRaidPhase('defenders');

   415→      else next.add(playerId);

   417→    });

   420→  // Process raid result
   421→  const processRaidResult = (result: RaidResult, touchedDefenders: Set<string>, hasBonus: boolean) => {
   422→    if (!match || !raider) return;
   423→
   424→    if (!hasStartedRaiding) setHasStartedRaiding(true);
   425→
   426→    // Stop raid timer
   427→    setRaidTimer(null);
   428→    if (raidTimerRef.current) clearInterval(raidTimerRef.current);

   430→    const raidingTeamId = raidingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   431→    const defendingTeamId = defendingTeam === 'home' ? match.homeTeamId : match.awayTeamId;
   432→
   433→    const events: Omit<MatchEvent, 'id' | 'timestamp'>[] = [];
   434→
   435→    if (result === 'success') {
   436→      const touchCount = touchedDefenders.size;
   437→      if (touchCount > 0) {
   438→        events.push({
   439→          matchId: match.id, eventType: 'raid_point', teamId: raidingTeamId,
   440→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   441→          value: touchCount,
   442→          details: JSON.stringify({ touchedPlayerIds: Array.from(touchedDefenders), raiderId: raider.id }),
   443→        });
   444→      }
   445→      if (hasBonus) {
   446→        events.push({
   447→          matchId: match.id, eventType: 'bonus_point', teamId: raidingTeamId,
   448→          half: match.currentHalf, playerId: raider.id, playerName: raider.name,
   449→          value: 1, details: JSON.stringify({ raiderId: raider.id }),
   450→        });
   451→      }
   452→
   453→      // All out check
   454→      const { onCourt } = splitLineup(fullDefendingLineup, defendingOutIds);
   455→      const defendingOnCourtOut = onCourt.filter(p => defendingOutIds.includes(p.id)).length;
   456→      const newDefendingOutCount = defendingOnCourtOut + touchCount;
   457→      if (newDefendingOutCount >= onCourt.length) {
   458→        events.push({
   459→          matchId: match.id, eventType: 'all_out', teamId: raidingTeamId,
   460→          half: match.currentHalf, value: 2,
   461→        });
   462→      }
   463→    } else if (result === 'caught') {
   464→      const caughtByIds = Array.from(touchedDefenders);
   465→      const primaryCatcher = caughtByIds.length > 0
   466→        ? fullDefendingLineup.find(p => caughtByIds.includes(p.id))
   467→        : null;
   468→
   469→      events.push({
   470→        matchId: match.id, eventType: 'tackle_point', teamId: defendingTeamId,
   471→        half: match.currentHalf,
   472→        playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   473→        value: 1,
   474→        details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   475→      });
   476→
   477→      const { onCourtActive } = splitLineup(fullDefendingLineup, defendingOutIds);
   478→      if (onCourtActive.length <= 3) {
   479→        events.push({
   480→          matchId: match.id, eventType: 'super_tackle', teamId: defendingTeamId,
   481→          half: match.currentHalf,
   482→          playerId: primaryCatcher?.id || raider.id, playerName: primaryCatcher?.name || raider.name,
   483→          value: 1,
   484→          details: JSON.stringify({ caughtByIds, raiderId: raider.id }),
   485→        });
   486→      }
   487→    } else if (result === 'empty') {
   488→      events.push({
   489→        matchId: match.id, eventType: 'empty_raid', teamId: raidingTeamId,
   490→        half: match.currentHalf, playerId: raider.id, playerName: raider.name, value: 0,
   491→      });
   492→
   493→      // Track consecutive empty raids for do-or-die
   494→      consecutiveEmptyRaidsRef.current += 1;
   495→    }
   496→
   497→    // Reset consecutive empty raids on any successful raid (points scored)
   498→    if (result === 'success') {
   499→      consecutiveEmptyRaidsRef.current = 0;
   500→    }
   501→    // Raider caught (tackle) also resets because the defending team scored
   502→    if (result === 'caught') {
   503→      consecutiveEmptyRaidsRef.current = 0;
   504→    }
   505→
   506→    // ═══ DO-OR-DIE LOGIC ═══
   507→    // In kabaddi, after 2 consecutive empty raids, the next raid becomes do-or-die
   508→    if (consecutiveEmptyRaidsRef.current >= 2) {
   509→      setDoOrDie(true);
   510→      // 🔊 Do-or-Die activated!
   511→      triggerFeedback(SoundType.DO_OR_DIE);
   512→      toast({
   513→        title: '🔥 DO OR DIE RAID!',
   514→        description: 'Next raider must score — or raider is out!',
   515→        duration: 3000,
   516→      });
   517→    } else {
   518→      setDoOrDie(false);
   519→    }
   520→
   521→    if (events.length > 0) {
   522→      addBatchEvents(events);
   523→    }
   524→
   525→    // Reset raid state
   526→    setRaidPhase('idle');
   527→    setRaider(null);
   528→    setRaidResult(null);
   529→    setSelectedDefenders(new Set());
