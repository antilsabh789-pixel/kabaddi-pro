   113→function getInitials(name: string | null | undefined): string {
   114→  if (!name) return '?';
   115→  const parts = name.trim().split(/\s+/);
   116→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
   117→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
   118→}
   119→
   120→function getDisplayName(name: string | null | undefined): string {
   121→  return name?.trim() || 'Unnamed Player';
   122→}
   123→
   124→function hexToRgb(hex: string | null): string {
   125→  if (!hex) return '220, 38, 38';
   126→  const result = /^#?([a-f\d]{2})([a-f\d]{2})([a-f\d]{2})$/i.exec(hex);
   127→  if (!result) return '220, 38, 38';
   128→  return `${parseInt(result[1], 16)}, ${parseInt(result[2], 16)}, ${parseInt(result[3], 16)}`;
   129→}
