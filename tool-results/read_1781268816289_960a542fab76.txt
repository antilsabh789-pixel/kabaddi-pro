     1→'use client';
     2→
     3→import { useState, useCallback } from 'react';
     4→import { motion, AnimatePresence } from 'framer-motion';
     5→import { Trophy, Shield, Megaphone, ChevronRight, ArrowLeft, Phone, Eye, EyeOff } from 'lucide-react';
     6→import { Button } from '@/components/ui/button';
     7→import { Input } from '@/components/ui/input';
     8→import { useKabaddiStore } from '@/lib/store';
     9→
    10→type Stage = 'auth' | 'details' | 'role';
    11→
    12→const roles = [
    13→  {
    14→    id: 'player',
    15→    title: 'Player',
    16→    description: 'Track your raids, tackles, and performance stats',
    17→    icon: Shield,
    18→    color: 'from-brand-red to-brand-red-dark',
    19→    borderColor: 'border-brand-red/40',
    20→    glowColor: 'shadow-brand-red/20',
    21→  },
    22→  {
    23→    id: 'coach',
    24→    title: 'Coach',
    25→    description: 'Manage teams, strategies, and match tactics',
    26→    icon: Megaphone,
    27→    color: 'from-brand-green to-brand-green-dark',
    28→    borderColor: 'border-brand-green/40',
    29→    glowColor: 'shadow-brand-green/20',
    30→  },
    31→] as const;
    32→
    33→const slideVariants = {
    34→  enter: (direction: number) => ({
    35→    x: direction > 0 ? 300 : -300,
    36→    opacity: 0,
    37→  }),
    38→  center: {
    39→    x: 0,
    40→    opacity: 1,
    41→  },
    42→  exit: (direction: number) => ({
    43→    x: direction < 0 ? 300 : -300,
    44→    opacity: 0,
    45→  }),
    46→};
    47→
    48→export default function AuthScreen() {
    49→  const [stage, setStage] = useState<Stage>('auth');
    50→  const [isSignUp, setIsSignUp] = useState(false);
    51→  const [phone, setPhone] = useState('');
    52→  const [name, setName] = useState('');
    53→  const [password, setPassword] = useState('');
    54→  const [confirmPassword, setConfirmPassword] = useState('');
    55→  const [showPassword, setShowPassword] = useState(false);
    56→  const [showConfirmPassword, setShowConfirmPassword] = useState(false);
    57→  const [selectedRoles, setSelectedRoles] = useState<Set<string>>(new Set());
    58→  const [isSubmitting, setIsSubmitting] = useState(false);
    59→  const [direction, setDirection] = useState(1);
    60→  const [countryCode] = useState('+91');
    61→  const [error, setError] = useState('');
    62→
    63→  // New fields
    64→  const [gender, setGender] = useState<'boy' | 'girl' | ''>('');
    65→  const [weight, setWeight] = useState('');
    66→  const [practiceGround, setPracticeGround] = useState('');
    67→  const [position, setPosition] = useState('');
    68→
    69→  const login = useKabaddiStore((s) => s.login);
    70→  const setOnboarded = useKabaddiStore((s) => s.setOnboarded);
    71→
    72→  const goNext = useCallback((next: Stage) => {
    73→    setDirection(1);
    74→    setStage(next);
    75→  }, []);
    76→
    77→  const goBack = useCallback(() => {
    78→    setDirection(-1);
    79→    if (stage === 'role') setStage('details');
    80→    else if (stage === 'details') setStage('auth');
    81→  }, [stage]);
    82→
    83→  const toggleRole = useCallback((roleId: string) => {
    84→    setSelectedRoles((prev) => {
    85→      const next = new Set(prev);
    86→      if (next.has(roleId)) {
    87→        next.delete(roleId);
    88→      } else {
    89→        next.add(roleId);
    90→      }
    91→      return next;
    92→    });
    93→  }, []);
    94→
    95→  const handleAuth = useCallback(async () => {
    96→    setError('');
    97→
    98→    if (phone.length !== 10) {
    99→      setError('Please enter a valid 10-digit phone number');
   100→      return;
   101→    }
   102→    if (password.length < 6) {
   103→      setError('Password must be at least 6 characters');
   104→      return;
   105→    }
   106→    if (isSignUp && !name.trim()) {
   107→      setError('Please enter your name');
   108→      return;
   109→    }
   110→    if (isSignUp && password !== confirmPassword) {
   111→      setError('Passwords do not match');
   112→      return;
   113→    }
   114→
   115→    setIsSubmitting(true);
   116→    try {
   117→      const res = await fetch('/api/auth', {
   118→        method: 'POST',
   119→        headers: { 'Content-Type': 'application/json' },
   120→        body: JSON.stringify({
   121→          phone: `${countryCode}${phone}`,
   122→          name: name.trim(),
   123→          password,
   124→          role: selectedRoles.values().next().value ?? 'player',
   125→          action: isSignUp ? 'register' : 'login',
   126→          gender: gender === 'boy' ? 'male' : gender === 'girl' ? 'female' : undefined,
   127→          weight: weight ? `${weight}kg` : undefined,
   128→          practiceGround: practiceGround || undefined,
   129→          position: position || undefined,
   130→        }),
   131→      });
   132→
   133→      const data = await res.json();
   134→
   135→      if (!res.ok) {
   136→        setError(data.error || 'Authentication failed');
   137→        return;
   138→      }
   139→
   140→      const user = data.user;
   141→      // Set user state directly
   142→      login({
   143→        id: user.id,
   144→        phone: user.phone,
   145→        playerCode: user.playerCode || undefined,
   146→        name: user.name || name.trim(),
   147→        role: user.role || 'player',
   148→        avatar: user.avatar || '',
   149→        isPremium: user.isPremium || false,
   150→        isAdmin: user.isAdmin || false,
   151→        gender: user.gender || (gender === 'boy' ? 'male' : gender === 'girl' ? 'female' : undefined),
   152→        weight: user.weight || (weight ? `${weight}kg` : undefined),
   153→        practiceGround: user.practiceGround || practiceGround || undefined,
   154→      });
   155→
   156→      if (isSignUp) {
   157→        // New user — go to details page first, then role selection
   158→        goNext('details');
   159→      } else {
   160→        // Returning user — go straight in
   161→        setOnboarded(true);
   162→      }
   163→    } catch (err) {
   164→      setError('Something went wrong. Please try again.');
   165→      console.error('Auth failed:', err);
   166→    } finally {
   167→      setIsSubmitting(false);
   168→    }
   169→  }, [phone, name, password, confirmPassword, isSignUp, selectedRoles, countryCode, login, setOnboarded, goNext, gender, weight, practiceGround]);
   170→
   171→  const handleDetailsContinue = useCallback(() => {
   172→    setError('');
   173→    if (!gender) {
   174→      setError('Please select your gender');
   175→      return;
   176→    }
   177→
   178→    // Update user details via API
   179→    const currentUser = useKabaddiStore.getState().currentUser;
   180→    if (currentUser?.id) {
   181→      fetch('/api/auth', {
   182→        method: 'POST',
   183→        headers: { 'Content-Type': 'application/json' },
   184→        body: JSON.stringify({
   185→          userId: currentUser.id,
   186→          action: 'update-details',
   187→          gender: gender === 'boy' ? 'male' : gender === 'girl' ? 'female' : gender,
   188→          weight: weight ? `${weight}kg` : undefined,
   189→          practiceGround: practiceGround || undefined,
   190→        }),
   191→      }).catch(() => {});
   192→
   193→      // Update position on player profile
   194→      if (position) {
   195→        fetch(`/api/players/${currentUser.id}`, {
   196→          method: 'PUT',
   197→          headers: { 'Content-Type': 'application/json' },
   198→          body: JSON.stringify({ position }),
   199→        }).catch(() => {});
   200→      }
   201→    }
   202→
   203→    // Update local state
   204→    useKabaddiStore.getState().updateUser({
   205→      gender: gender === 'boy' ? 'male' : gender === 'girl' ? 'female' : gender,
   206→      weight: weight ? `${weight}kg` : undefined,
   207→      practiceGround: practiceGround || undefined,
   208→    });
   209→
   210→    goNext('role');
   211→  }, [gender, weight, practiceGround, position, goNext]);
   212→
   213→  const handleGetStarted = useCallback(async () => {
   214→    setIsSubmitting(true);
   215→    try {
   216→      // Update user role via API
   217→      const currentUser = useKabaddiStore.getState().currentUser;
   218→      if (currentUser?.id) {
   219→        const roleValue = Array.from(selectedRoles).join(',') || 'player';
   220→        await fetch('/api/auth', {
   221→          method: 'POST',
   222→          headers: { 'Content-Type': 'application/json' },
   223→          body: JSON.stringify({
   224→            userId: currentUser.id,
   225→            action: 'update-details',
   226→            role: roleValue,
   227→          }),
   228→        });
   229→
   230→        // Update local state
   231→        useKabaddiStore.getState().updateUser({ role: roleValue });
   232→      }
   233→      setOnboarded(true);
   234→    } finally {
   235→      setIsSubmitting(false);
   236→    }
   237→  }, [setOnboarded, selectedRoles]);
   238→
   239→  const stageIndex: Record<Stage, number> = { auth: 0, details: 1, role: 2 };
   240→
   241→  // Gender-based accent color
   242→  const genderAccent = gender === 'boy' ? '#1E293B' : gender === 'girl' ? '#DC2626' : '';
   243→
   244→  return (
   245→    <div className="min-h-screen bg-warm-50 flex flex-col items-center justify-center px-4 relative overflow-hidden">
   246→      {/* Background decoration */}
   247→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   248→        <div className="absolute -top-40 -right-40 w-80 h-80 bg-brand-red/10 rounded-full blur-3xl" />
   249→        <div className="absolute -bottom-40 -left-40 w-80 h-80 bg-brand-red-dark/10 rounded-full blur-3xl" />
   250→        <div className="absolute top-1/4 left-1/2 -translate-x-1/2 w-96 h-96 bg-brand-red/20 rounded-full blur-3xl" />
   251→      </div>
   252→
   253→      {/* Logo - always visible */}
   254→      <motion.div
   255→        className="flex flex-col items-center mb-8 relative z-10"
   256→        initial={{ opacity: 0, y: -20 }}
   257→        animate={{ opacity: 1, y: 0 }}
   258→        transition={{ duration: 0.5 }}
   259→      >
   260→        <div className="relative mb-3">
   261→          <div className="w-16 h-16 rounded-2xl bg-gradient-to-br from-brand-red to-brand-red-dark flex items-center justify-center shadow-lg shadow-brand-red/30">
   262→            <Trophy className="w-8 h-8 text-white" />
   263→          </div>
   264→          <div className="absolute -inset-1 rounded-2xl bg-brand-red/30 blur-md -z-10" />
   265→        </div>
   266→        <h1 className="text-2xl font-black tracking-wider text-warm-800">
   267→          KABADDI <span className="text-brand-red">PRO</span>
   268→        </h1>
   269→      </motion.div>
   270→
   271→      {/* Stage content */}
   272→      <div className="w-full max-w-sm relative z-10">
   273→        <AnimatePresence mode="wait" custom={direction}>
   274→          {/* ─── Stage 1: Login / Sign Up ─── */}
   275→          {stage === 'auth' && (
   276→            <motion.div
   277→              key="auth"
   278→              custom={direction}
   279→              variants={slideVariants}
   280→              initial="enter"
   281→              animate="center"
   282→              exit="exit"
   283→              transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   284→              className="flex flex-col items-center"
   285→            >
   286→              <h2 className="text-xl font-bold text-warm-800 mb-1">
   287→                {isSignUp ? 'Create Account' : 'Welcome Back!'}
   288→              </h2>
   289→              <p className="text-warm-600 text-sm mb-6 text-center">
   290→                {isSignUp
   291→                  ? 'Sign up to get started with Kabaddi Pro'
   292→                  : 'Login to your Kabaddi Pro account'}
   293→              </p>
   294→
   295→              {/* Phone Input */}
   296→              <div className="w-full flex gap-2 mb-3">
   297→                <div className="flex items-center justify-center h-12 px-3 rounded-lg bg-warm-100 border border-warm-300 text-warm-800 font-semibold text-sm shrink-0">
   298→                  <Phone className="w-3.5 h-3.5 mr-1.5 text-warm-600" />
   299→                  {countryCode}
   300→                </div>
   301→                <Input
   302→                  type="tel"
   303→                  inputMode="numeric"
   304→                  maxLength={10}
   305→                  placeholder="Enter 10-digit number"
   306→                  value={phone}
   307→                  onChange={(e) => {
   308→                    const val = e.target.value.replace(/\D/g, '');
   309→                    setPhone(val);
   310→                    if (error) setError('');
   311→                  }}
   312→                  className="h-12 bg-warm-100 border-warm-300 text-warm-800 placeholder:text-warm-500 text-base"
   313→                />
   314→              </div>
   315→
   316→              {/* Name Input (Sign Up only) */}
   317→              {isSignUp && (
   318→                <motion.div
   319→                  initial={{ opacity: 0, height: 0 }}
   320→                  animate={{ opacity: 1, height: 'auto' }}
   321→                  exit={{ opacity: 0, height: 0 }}
   322→                  className="w-full mb-3"
   323→                >
   324→                  <Input
   325→                    type="text"
   326→                    placeholder="Enter your name"
   327→                    value={name}
   328→                    onChange={(e) => {
   329→                      setName(e.target.value);
   330→                      if (error) setError('');
   331→                    }}
   332→                    className="h-12 bg-warm-100 border-warm-300 text-warm-800 placeholder:text-warm-500 text-base"
   333→                  />
   334→                </motion.div>
   335→              )}
   336→
   337→              {/* Password Input */}
   338→              <div className="w-full relative mb-3">
   339→                <div className="relative">
   340→                  <Input
   341→                    type={showPassword ? 'text' : 'password'}
   342→                    placeholder={isSignUp ? 'Create a password' : 'Enter password'}
   343→                    value={password}
   344→                    onChange={(e) => {
   345→                      setPassword(e.target.value);
   346→                      if (error) setError('');
   347→                    }}
   348→                    onKeyDown={(e) => {
   349→                      if (e.key === 'Enter') handleAuth();
   350→                    }}
   351→                    className="h-12 bg-warm-100 border-warm-300 text-warm-800 placeholder:text-warm-500 text-base pr-12"
   352→                  />
   353→                  <button
   354→                    type="button"
   355→                    onClick={() => setShowPassword(!showPassword)}
   356→                    className="absolute right-3 top-1/2 -translate-y-1/2 text-warm-500 hover:text-warm-700 transition-colors"
   357→                  >
   358→                    {showPassword ? (
   359→                      <EyeOff className="w-5 h-5" />
   360→                    ) : (
   361→                      <Eye className="w-5 h-5" />
   362→                    )}
   363→                  </button>
   364→                </div>
   365→                {isSignUp && password && password.length < 6 && (
   366→                  <p className="text-xs text-warm-500 mt-1 ml-1">
   367→                    Password must be at least 6 characters
   368→                  </p>
   369→                )}
   370→              </div>
   371→
   372→              {/* Confirm Password (Sign Up only) */}
   373→              {isSignUp && (
   374→                <motion.div
   375→                  initial={{ opacity: 0, height: 0 }}
   376→                  animate={{ opacity: 1, height: 'auto' }}
   377→                  exit={{ opacity: 0, height: 0 }}
   378→                  className="w-full mb-4"
   379→                >
   380→                  <div className="relative">
   381→                    <Input
   382→                      type={showConfirmPassword ? 'text' : 'password'}
   383→                      placeholder="Confirm password"
   384→                      value={confirmPassword}
   385→                      onChange={(e) => {
   386→                        setConfirmPassword(e.target.value);
   387→                        if (error) setError('');
   388→                      }}
   389→                      onKeyDown={(e) => {
   390→                        if (e.key === 'Enter') handleAuth();
   391→                      }}
   392→                      className="h-12 bg-warm-100 border-warm-300 text-warm-800 placeholder:text-warm-500 text-base pr-12"
   393→                    />
   394→                    <button
   395→                      type="button"
   396→                      onClick={() => setShowConfirmPassword(!showConfirmPassword)}
   397→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-warm-500 hover:text-warm-700 transition-colors"
   398→                    >
   399→                      {showConfirmPassword ? (
   400→                        <EyeOff className="w-5 h-5" />
   401→                      ) : (
   402→                        <Eye className="w-5 h-5" />
   403→                      )}
   404→                    </button>
   405→                  </div>
   406→                  {confirmPassword && password !== confirmPassword && (
   407→                    <p className="text-xs text-red-500 mt-1 ml-1">
   408→                      Passwords do not match
   409→                    </p>
   410→                  )}
   411→                </motion.div>
   412→              )}
   413→
   414→              {/* Error Message */}
   415→              {error && (
   416→                <motion.p
   417→                  initial={{ opacity: 0, y: -4 }}
   418→                  animate={{ opacity: 1, y: 0 }}
   419→                  className="text-brand-red text-sm mb-3 text-center w-full"
   420→                >
   421→                  {error}
   422→                </motion.p>
   423→              )}
   424→
   425→              {/* Submit Button */}
   426→              <Button
   427→                onClick={handleAuth}
   428→                disabled={
   429→                  isSubmitting ||
   430→                  phone.length !== 10 ||
   431→                  password.length < 6 ||
   432→                  (isSignUp && (!confirmPassword || password !== confirmPassword))
   433→                }
   434→                className="w-full h-12 bg-brand-red-dark hover:bg-brand-red-dark/90 text-white font-semibold text-base rounded-lg disabled:opacity-40"
   435→              >
   436→                {isSubmitting ? (
   437→                  <div className="flex items-center gap-2">
   438→                    <div className="w-4 h-4 border-2 border-white/30 border-t-white rounded-full animate-spin" />
   439→                    Please wait...
   440→                  </div>
   441→                ) : isSignUp ? (
   442→                  <>
   443→                    Create Account
   444→                    <ChevronRight className="w-4 h-4 ml-1" />
   445→                  </>
   446→                ) : (
   447→                  <>
   448→                    Login
   449→                    <ChevronRight className="w-4 h-4 ml-1" />
   450→                  </>
   451→                )}
   452→              </Button>
   453→
   454→              {/* Toggle Login/Signup */}
   455→              <button
   456→                className="text-brand-red hover:text-brand-red-light text-sm font-medium mt-4 transition-colors"
   457→                onClick={() => {
   458→                  setIsSignUp(!isSignUp);
   459→                  setError('');
   460→                  setConfirmPassword('');
   461→                }}
   462→              >
   463→                {isSignUp
   464→                  ? 'Already have an account? Login'
   465→                  : "Don't have an account? Sign Up"}
   466→              </button>
   467→            </motion.div>
   468→          )}
   469→
   470→          {/* ─── Stage 2: Details (Gender, Weight, Practice Ground) ─── */}
   471→          {stage === 'details' && (
   472→            <motion.div
   473→              key="details"
   474→              custom={direction}
   475→              variants={slideVariants}
   476→              initial="enter"
   477→              animate="center"
   478→              exit="exit"
   479→              transition={{ type: 'spring', stiffness: 300, damping: 30 }}
   480→              className="flex flex-col items-center"
   481→            >
   482→              <button
   483→                onClick={goBack}
   484→                className="self-start flex items-center text-warm-600 hover:text-warm-800 transition-colors mb-4 text-sm"
   485→              >
   486→                <ArrowLeft className="w-4 h-4 mr-1" />
   487→                Back
   488→              </button>
   489→
   490→              <h2 className="text-xl font-bold text-warm-800 mb-1">Tell us about yourself</h2>
   491→              <p className="text-warm-600 text-sm mb-6 text-center">
   492→                Help us personalize your experience
   493→              </p>
   494→
   495→              {/* Gender Selection */}
   496→              <div className="w-full mb-4">
   497→                <label className="text-sm font-semibold text-warm-700 mb-2 block">
   498→                  Gender <span className="text-brand-red">*</span>
   499→                </label>
   500→                <div className="grid grid-cols-2 gap-3">
   501→                  {/* Boy Button */}
   502→                  <motion.button
   503→                    type="button"
   504→                    whileTap={{ scale: 0.97 }}
   505→                    onClick={() => {
   506→                      setGender('boy');
   507→                      if (error) setError('');
   508→                    }}
   509→                    className={`flex flex-col items-center justify-center gap-2 p-5 rounded-xl border-2 transition-all ${
   510→                      gender === 'boy'
   511→                        ? 'border-brand-blue bg-brand-blue shadow-lg shadow-brand-blue/20'
   512→                        : 'border-warm-300 bg-warm-100 hover:border-brand-blue/40'
   513→                    }`}
   514→                  >
   515→                    <span
   516→                      className={`text-3xl leading-none ${
   517→                        gender === 'boy' ? 'text-white' : 'text-brand-blue'
   518→                      }`}
   519→                    >
   520→                      ♂
   521→                    </span>
   522→                    <span
   523→                      className={`text-sm font-bold ${
   524→                        gender === 'boy' ? 'text-white' : 'text-warm-700'
   525→                      }`}
   526→                    >
   527→                      Boy
   528→                    </span>
   529→                  </motion.button>
   530→
   531→                  {/* Girl Button */}
   532→                  <motion.button
   533→                    type="button"
   534→                    whileTap={{ scale: 0.97 }}
   535→                    onClick={() => {
   536→                      setGender('girl');
   537→                      if (error) setError('');
   538→                    }}
   539→                    className={`flex flex-col items-center justify-center gap-2 p-5 rounded-xl border-2 transition-all ${
   540→                      gender === 'girl'
   541→                        ? 'border-brand-red bg-brand-red shadow-lg shadow-brand-red/20'
   542→                        : 'border-warm-300 bg-warm-100 hover:border-brand-red/40'
   543→                    }`}
   544→                  >
   545→                    <span
   546→                      className={`text-3xl leading-none ${
   547→                        gender === 'girl' ? 'text-white' : 'text-brand-red'
   548→                      }`}
   549→                    >
   550→                      ♀
   551→                    </span>
   552→                    <span
   553→                      className={`text-sm font-bold ${
   554→                        gender === 'girl' ? 'text-white' : 'text-warm-700'
   555→                      }`}
   556→                    >
   557→                      Girl
   558→                    </span>
   559→                  </motion.button>
   560→                </div>
   561→              </div>
   562→
   563→              {/* Weight Input */}
   564→              <div className="w-full mb-3">
   565→                <label className="text-sm font-semibold text-warm-700 mb-2 block">
   566→                  Weight
   567→                </label>
   568→                <div className="relative">
   569→                  <Input
   570→                    type="number"
   571→                    inputMode="numeric"
   572→                    placeholder="Enter weight"
   573→                    value={weight}
   574→                    onChange={(e) => {
   575→                      setWeight(e.target.value);
   576→                    }}
   577→                    className="h-12 bg-warm-100 border-warm-300 text-warm-800 placeholder:text-warm-500 text-base pr-12"
   578→                  />
   579→                  <span className="absolute right-3 top-1/2 -translate-y-1/2 text-warm-500 text-sm font-medium pointer-events-none">
   580→                    kg
   581→                  </span>
   582→                </div>
   583→              </div>
   584→
   585→              {/* Practice Ground Input */}
