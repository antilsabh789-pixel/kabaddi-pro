    51→// ─── Position Helpers ────────────────────────────────────────────────
    52→
    53→type PositionType = 'raider' | 'defender' | 'all-rounder';
    54→
    55→function getPositionType(position: string | null): PositionType {
    56→  if (!position) return 'all-rounder';
    57→  const p = position.toLowerCase();
    58→  if (p.includes('raider')) return 'raider';
    59→  if (p.includes('corner') || p.includes('cover') || p.includes('defend')) return 'defender';
    60→  return 'all-rounder';
    61→}
    62→
    63→function getPositionColor(position: string | null) {
    64→  const type = getPositionType(position);
    65→  switch (type) {
    66→    case 'raider': return { ring: 'ring-red-500', bg: 'bg-red-500', text: 'text-red-500', gradient: 'from-red-500 to-orange-500', border: 'border-red-500/30' };
    67→    case 'defender': return { ring: 'ring-blue-500', bg: 'bg-blue-500', text: 'text-blue-500', gradient: 'from-blue-500 to-cyan-500', border: 'border-blue-500/30' };
    68→    case 'all-rounder': return { ring: 'ring-amber-500', bg: 'bg-amber-500', text: 'text-amber-500', gradient: 'from-amber-500 to-yellow-400', border: 'border-amber-500/30' };
    69→  }
    70→}

    97→// ─── Animated Counter ────────────────────────────────────────────────
    98→
    99→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   100→  const [display, setDisplay] = useState(value);
   101→  const prevValueRef = useRef(value);
   102→
   103→  useEffect(() => {
   104→    const end = value;
   105→    if (end === 0) {
   106→      prevValueRef.current = 0;
   107→      return;
   108→    }
   109→    const start = prevValueRef.current;
   110→    if (start === end) return;
   111→    const increment = (end - start) / (duration / 16);
   112→    let current = start;
   113→    const timer = setInterval(() => {
   114→      current += increment;
   115→      if ((increment > 0 && current >= end) || (increment < 0 && current <= end)) {
   116→        setDisplay(end);
   117→        prevValueRef.current = end;
   118→        clearInterval(timer);
   119→      } else {
   120→        setDisplay(Math.floor(current));
   121→      }
   122→    }, 16);
   123→    return () => clearInterval(timer);
   124→  }, [value, duration]);
   125→
   126→  return <>{display}</>;
   127→}

   129→// ─── Circular Rating Indicator ───────────────────────────────────────
   130→
   131→function CircularRating({ rating, size = 48 }: { rating: number; size?: number }) {
   132→  const radius = (size - 8) / 2;
   133→  const circumference = 2 * Math.PI * radius;
   134→  const progress = Math.min(rating / 10, 1);
   135→  const offset = circumference - progress * circumference;
   136→
   137→  const getColor = (r: number) => {
   138→    if (r >= 8) return '#22c55e';
   139→    if (r >= 6) return '#eab308';
   140→    if (r >= 4) return '#f97316';
   141→    return '#ef4444';
   142→  };
   143→
   144→  return (
   145→    <div className="relative" style={{ width: size, height: size }}>
   146→      <svg width={size} height={size} className="-rotate-90">
   147→        <circle
   148→          cx={size / 2}
   149→          cy={size / 2}
   150→          r={radius}
   151→          stroke="rgba(255,255,255,0.15)"
   152→          strokeWidth="4"
   153→          fill="none"
   154→        />
   155→        <circle
   156→          cx={size / 2}
   157→          cy={size / 2}
   158→          r={radius}
   159→          stroke={getColor(rating)}
   160→          strokeWidth="4"
   161→          fill="none"
   162→          strokeLinecap="round"
   163→          strokeDasharray={circumference}
   164→          strokeDashoffset={offset}
   165→          className="transition-all duration-1000 ease-out"
   166→        />
   167→      </svg>
   168→      <div className="absolute inset-0 flex items-center justify-center">
   169→        <span className="text-xs font-black text-white">{rating.toFixed(1)}</span>
   170→      </div>
   171→    </div>
   172→  );
   173→}

   918→                    <button
   919→                      onClick={handleWebShare}
   920→                      className="w-full flex items-center gap-2.5 px-3.5 py-2.5 hover:bg-warm-200 dark:hover:bg-warm-300 transition-colors text-left"
   921→                    >
   922→                      <Share2 className="w-4 h-4 text-brand-gold" />
   923→                      <span className="text-xs font-medium text-warm-800 dark:text-warm-700">Share via...</span>
   924→                    </button>
   925→                  )}
   926→                </motion.div>
   927→              )}
   928→            </AnimatePresence>
   929→          </div>
   930→        </div>
   931→
   932→        {/* 3D Flip Container */}
   933→        <div
   934→          className="relative w-full cursor-pointer"
   935→          style={{ perspective: '1200px' }}
   936→          onClick={() => setFlipped(!flipped)}
   937→        >
   938→          <div
   939→            className="relative w-full transition-transform duration-700 ease-in-out"
   940→            style={{
   941→              transformStyle: 'preserve-3d',
   942→              transform: flipped ? 'rotateY(180deg)' : 'rotateY(0deg)',
   943→            }}
   944→          >
   945→            {/* Front */}
   946→            <div style={{ backfaceVisibility: 'hidden' }}>
   947→              {cardFront}
   948→            </div>
   949→            {/* Back */}
   950→            <div
   951→              className="absolute inset-0"
   952→              style={{ backfaceVisibility: 'hidden', transform: 'rotateY(180deg)' }}
   953→            >
   954→              {cardBack}
   955→            </div>
   956→          </div>
   957→        </div>
   958→      </div>
   959→
   960→      {/* Fullscreen overlay */}
   961→      {fullscreenView}
   962→    </>
   963→  );
   964→}
