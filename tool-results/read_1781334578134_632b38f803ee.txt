   250→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   251→  const [count, setCount] = useState(0);
   252→  const prevTarget = useRef(target);
   253→
   254→  useEffect(() => {
   255→    if (prevTarget.current === target && count !== 0) return;
   256→    prevTarget.current = target;
   257→
   258→    let start = 0;
   259→    const startTime = performance.now();
   260→
   261→    function step(currentTime: number) {
   262→      const elapsed = currentTime - startTime;
   263→      const progress = Math.min(elapsed / duration, 1);
   264→      // ease-out cubic
   265→      const eased = 1 - Math.pow(1 - progress, 3);
   266→      const current = Math.round(eased * target);
   267→      setCount(current);
   268→      if (progress < 1) {
   269→        requestAnimationFrame(step);
   270→      }
   271→    }
   272→
   273→    requestAnimationFrame(step);
   274→  }, [target, duration]);
   275→
   276→  return <span>{count}</span>;
   277→}

   281→function NumberTicker({ value }: { value: number }) {
   282→  const prevValue = useRef(value);
   283→  const [animating, setAnimating] = useState(false);
   284→  const [flash, setFlash] = useState(false);
   285→
   286→  useEffect(() => {
   287→    if (prevValue.current !== value) {
   288→      prevValue.current = value;
   289→      const t1 = setTimeout(() => setAnimating(true), 0);
   290→      const t2 = setTimeout(() => setAnimating(false), 400);
   291→      const t3 = setTimeout(() => setFlash(true), 0);
   292→      const t4 = setTimeout(() => setFlash(false), 600);
   293→      return () => { clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4); };
   294→    }
   295→  }, [value]);
   296→
   297→  return (
   298→    <span className={`${animating ? 'number-ticker' : ''} ${flash ? 'score-change-flash' : ''}`} key={`${value}-${animating}`}>
   299→      {value}
   300→    </span>
   301→  );
   302→}

   306→function ConfettiParticles({ trigger }: { trigger: number }) {
   307→  const [particles, setParticles] = useState<Array<{ id: number; x: number; color: string; delay: number }>>([]);
   308→  const prevTrigger = useRef(trigger);
   309→
   310→  useEffect(() => {
   311→    if (prevTrigger.current !== trigger) {
   312→      prevTrigger.current = trigger;
   313→      const colors = ['#DC2626', '#F59E0B', '#14B8A6', '#1E293B', '#FBBF24'];
   314→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   315→        id: Date.now() + i,
   316→        x: 30 + Math.random() * 40,
   317→        color: colors[Math.floor(Math.random() * colors.length)],
   318→        delay: i * 0.05,
   319→      }));
   320→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   321→      const t2 = setTimeout(() => setParticles([]), 900);
   322→      return () => { clearTimeout(t1); clearTimeout(t2); };
   323→    }
   324→  }, [trigger]);
   325→
   326→  return (
   327→    <>
   328→      {particles.map((p) => (
   329→        <div
   330→          key={p.id}
   331→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"
   332→          style={{
   333→            left: `${p.x}%`,
   334→            top: '20%',
   335→            backgroundColor: p.color,
   336→            animationDelay: `${p.delay}s`,

   338→        />
   339→      ))}
   340→    </>
   341→  );

   346→function AwardsConfettiSection() {
   347→  const ref = useRef<HTMLDivElement>(null);
   348→  const [shown, setShown] = useState(false);
   349→
   350→  useEffect(() => {
   351→    const el = ref.current;
   352→    if (!el) return;
   353→    const observer = new IntersectionObserver(
   354→      ([entry]) => {
   355→        if (entry.isIntersecting && !shown) {
   356→          setShown(true);
   357→        }
   358→      },
   359→      { threshold: 0.3 }
   360→    );
   361→    observer.observe(el);
   362→    return () => observer.disconnect();
   363→  }, [shown]);
   364→
   365→  const confettiColors = ['#DC2626', '#F59E0B', '#14B8A6', '#FBBF24', '#1E293B'];
   366→
   367→  return (
   368→    <div ref={ref} className="absolute inset-0 overflow-hidden pointer-events-none z-0">
   369→      {shown && [...Array(8)].map((_, i) => (
   370→        <motion.div
   371→          key={`award-confetti-${i}`}
   372→          className="absolute w-1.5 h-1.5 rounded-full"
   373→          style={{
   374→            left: `${10 + i * 12}%`,
   375→            top: '40%',
   376→            backgroundColor: confettiColors[i % confettiColors.length],
   377→          }}
   378→          initial={{ y: 0, opacity: 0, scale: 0 }}
   379→          animate={{ y: [-10, -40, -80], opacity: [0, 1, 0], scale: [0, 1.2, 0.5] }}
   380→          transition={{ duration: 1.5, delay: i * 0.1, ease: 'easeOut' }}
   381→        />
   382→      ))}
   383→    </div>
   384→  );
   385→}

   437→// ─── Countdown Timer ────────────────────────────────────────────────

   439→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   440→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   441→  const [expired, setExpired] = useState(!targetDate);
   442→
   443→  useEffect(() => {
   444→    if (!targetDate) {
   445→      return;
   446→    }
   447→
   448→    function calculate() {
   449→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   450→      if (diff === 0) {
   451→        setExpired(true);
   452→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   453→        return true;
   454→      }
   455→      setExpired(false);
   456→      setTimeLeft({
   457→        d: Math.floor(diff / 86400000),
   458→        h: Math.floor((diff % 86400000) / 3600000),
   459→        m: Math.floor((diff % 3600000) / 60000),
   460→        s: Math.floor((diff % 60000) / 1000),
   461→      });
   462→      return false;
   463→    }
   464→
   465→    const done = calculate();
   466→    if (done) return;
   467→
   468→    const id = setInterval(() => {
   469→      const finished = calculate();
   470→      if (finished) clearInterval(id);
   471→    }, 1000);
   472→
   473→    return () => clearInterval(id);
   474→  }, [targetDate]);
   475→
   476→  if (expired) {
   477→    return (
   478→      <motion.div
   479→        className="flex items-center justify-center gap-1.5 py-1"
   480→        animate={{ opacity: [1, 0.5, 1] }}
   481→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   482→      >
   483→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   484→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   485→      </motion.div>
   486→    );
   487→  }
   488→
   489→  if (!timeLeft) return null;
   490→
   491→  const units = [
   492→    { value: timeLeft.d, label: 'D' },
   493→    { value: timeLeft.h, label: 'H' },
   494→    { value: timeLeft.m, label: 'M' },
   495→    { value: timeLeft.s, label: 'S' },
   496→  ];
   497→
   498→  return (
   499→    <div className="flex items-center justify-center gap-1.5">
   500→      {units.map((unit, idx) => (
   501→        <div key={unit.label} className="flex items-center gap-1.5">
   502→          <div className="flex flex-col items-center">
   503→            <motion.div
   504→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   505→              key={`${unit.label}-${unit.value}`}
   506→              initial={{ rotateX: -90, opacity: 0 }}
   507→              animate={{ rotateX: 0, opacity: 1 }}
   508→              transition={{ duration: 0.3, ease: 'easeOut' }}
   509→            >
   510→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums countdown-flip">
   511→                {String(unit.value).padStart(2, '0')}
   512→              </span>
   513→            </motion.div>
   514→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   515→              {unit.label}
   516→            </span>
   517→          </div>
   518→          {idx < units.length - 1 && (
   519→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   520→          )}
   521→        </div>
   522→      ))}
   523→    </div>
   524→  );
   525→}
