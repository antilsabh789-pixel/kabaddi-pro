    82→// ─── Animated Value (count-up on first view) ────────────────────
    83→
    84→function AnimatedValue({ value, decimals = 0 }: { value: number; decimals?: number }) {
    85→  const [display, setDisplay] = useState(0);
    86→  const [hasAnimated, setHasAnimated] = useState(false);
    87→  const ref = useRef<HTMLSpanElement>(null);
    88→  const valueRef = useRef(value);
    89→
    90→  useEffect(() => {
    91→    valueRef.current = value;
    92→  }, [value]);
    93→
    94→  useEffect(() => {
    95→    if (hasAnimated) return;
    96→    if (value === 0) return;
    97→
    98→    const observer = new IntersectionObserver(
    99→      ([entry]) => {
   100→        if (entry.isIntersecting && !hasAnimated) {
   101→          setHasAnimated(true);
   102→          const targetVal = valueRef.current;
   103→          const duration = 1200;
   104→          const startTime = performance.now();
   105→
   106→          function step(currentTime: number) {
   107→            const elapsed = currentTime - startTime;
   108→            const progress = Math.min(elapsed / duration, 1);
   109→            const eased = 1 - Math.pow(1 - progress, 3);
   110→            const current = eased * targetVal;
   111→            setDisplay(decimals > 0 ? parseFloat(current.toFixed(decimals)) : Math.round(current));
   112→            if (progress < 1) {
   113→              requestAnimationFrame(step);
   114→            }
   115→          }
   116→
   117→          requestAnimationFrame(step);
   118→        }
   119→      },
   120→      { threshold: 0.1 }
   121→    );
   122→
   123→    if (ref.current) observer.observe(ref.current);
   124→    return () => observer.disconnect();
   125→  }, [value, decimals, hasAnimated]);
   126→
   127→  return <span ref={ref}>{hasAnimated ? (decimals > 0 ? display.toFixed(decimals) : display) : value}</span>;
   128→}

   145→// ─── Time Ago helper ────────────────────
   146→
   147→function timeAgo(dateStr: string): string {
   148→  if (!dateStr || dateStr === '—') return '';
   149→  try {
   150→    const date = new Date(dateStr);
   151→    if (isNaN(date.getTime())) return dateStr;
   152→    const now = new Date();
   153→    const diff = now.getTime() - date.getTime();
   154→    const mins = Math.floor(diff / 60000);
   155→    if (mins < 60) return mins <= 1 ? 'just now' : `${mins}m ago`;
   156→    const hours = Math.floor(mins / 60);
   157→    if (hours < 24) return `${hours}h ago`;
   158→    const days = Math.floor(hours / 24);
   159→    if (days < 7) return `${days}d ago`;
   160→    const weeks = Math.floor(days / 7);
   161→    if (weeks < 4) return `${weeks}w ago`;
   162→    return dateStr;
   163→  } catch {
   164→    return dateStr;
   165→  }
   166→}

   170→function getPlayerLevel(totalMatches: number, totalPoints: number): { label: string; color: string; icon: string; progress: number } {
   171→  const score = totalMatches * 2 + totalPoints;
   172→  if (score >= 500) return { label: 'Legend', color: 'from-amber-400 to-yellow-600', icon: '👑', progress: 100 };
   173→  if (score >= 200) return { label: 'Pro', color: 'from-red-500 to-orange-500', icon: '🔥', progress: Math.min(((score - 200) / 300) * 100, 100) };
   174→  if (score >= 50) return { label: 'Intermediate', color: 'from-teal-500 to-emerald-500', icon: '⚡', progress: Math.min(((score - 50) / 150) * 100, 100) };
   175→  return { label: 'Beginner', color: 'from-slate-400 to-slate-500', icon: '🌱', progress: Math.min((score / 50) * 100, 100) };
   176→}
