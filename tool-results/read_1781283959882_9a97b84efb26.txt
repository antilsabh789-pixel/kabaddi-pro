   110→   110→// ─── Animated Counter ─────────────────────────────────────────────
   111→   111→
   112→   112→function AnimatedCounter({ target, duration = 1000 }: { target: number; duration?: number }) {
   113→   113→  const [count, setCount] = useState(0);
   114→   114→  const prevTarget = useRef(target);
   115→   115→  const hasAnimated = useRef(false);
   116→   116→
   117→   117→  useEffect(() => {
   118→   118→    if (prevTarget.current === target && hasAnimated.current) return;
   119→   119→    prevTarget.current = target;
   120→   120→    hasAnimated.current = true;
   121→   121→
   122→   122→    let start = 0;
   123→   123→    const startTime = performance.now();
   124→   124→
   125→   125→    function step(currentTime: number) {
   126→   126→      const elapsed = currentTime - startTime;
   127→   127→      const progress = Math.min(elapsed / duration, 1);
   128→   128→      const eased = 1 - Math.pow(1 - progress, 3);
   129→   129→      const current = Math.round(eased * target);
   130→   130→      setCount(current);
   131→   131→      if (progress < 1) {
   132→   132→        requestAnimationFrame(step);
   133→   133→      }
   134→   134→    }
   135→   135→
   136→   136→    requestAnimationFrame(step);
   137→   137→  }, [target, duration]);
   138→   138→
   139→   139→  return <span>{count}</span>;
   140→   140→}

   144→   144→function MiniProgressRing({ percent, size = 40 }: { percent: number; size?: number }) {
   145→   145→  const strokeWidth = 4;
   146→   146→  const radius = (size - strokeWidth) / 2;
   147→   147→  const circumference = 2 * Math.PI * radius;
   148→   148→  const offset = circumference - (percent / 100) * circumference;
   149→   149→
   150→   150→  return (
   151→   151→    <svg width={size} height={size} className="transform -rotate-90">
   152→   152→      <circle
   153→   153→        cx={size / 2}
   154→   154→        cy={size / 2}
   155→   155→        r={radius}
   156→   156→        strokeWidth={strokeWidth}
   157→   157→        className="stroke-warm-200 dark:stroke-warm-700"
   158→   158→        fill="none"
   159→   159→      />
   160→   160→      <motion.circle
   161→   161→        cx={size / 2}

   164→   164→        strokeWidth={strokeWidth}
   165→   165→        className="stroke-emerald-500 dark:stroke-emerald-400"
   166→   166→        fill="none"
   167→   167→        strokeLinecap="round"

   169→   169→        initial={{ strokeDashoffset: circumference }}

   179→   179→function getDateGroup(dateStr: string): string {
   180→   180→  const date = new Date(dateStr);
   181→   181→  const now = new Date();
   182→   182→  const today = new Date(now.getFullYear(), now.getMonth(), now.getDate());
   183→   183→  const yesterday = new Date(today);
   184→   184→  yesterday.setDate(yesterday.getDate() - 1);
   185→   185→  const weekAgo = new Date(today);
   186→   186→  weekAgo.setDate(weekAgo.getDate() - 7);
   187→   187→
   188→   188→  const matchDate = new Date(date.getFullYear(), date.getMonth(), date.getDate());
   189→   189→
   190→   190→  if (matchDate.getTime() === today.getTime()) return 'Today';
   191→   191→  if (matchDate.getTime() === yesterday.getTime()) return 'Yesterday';
   192→   192→  if (matchDate > weekAgo) return 'This Week';
   193→   193→  return 'Earlier';
   194→   194→}

   203→   203→}

   205→   205→function getMatchResult(match: MatchItem, userId?: string): 'won' | 'lost' | 'draw' {
   206→   206→  if (match.homeScore === match.awayScore) return 'draw';
   207→   207→  const homeWon = match.homeScore > match.awayScore;
   208→   208→
   209→   209→  // If userId is a scorer, determine which side they were on
   210→   210→  if (userId) {
   211→   211→    const isScorer = match.scorers.some((s) => s.userId === userId);
   212→   212→    if (isScorer) {
   213→   213→      // Check if the user's team is home or away (based on scorer relation)
   214→   214→      // Default: assume home team if we can't determine
   215→   215→      return homeWon ? 'won' : 'lost';
   216→   216→    }
   217→   217→  }
   218→   218→
   219→   219→  // General: show from home team perspective
   220→   220→  return homeWon ? 'won' : 'lost';
   221→   221→}
   222→   222→
   223→   223→function getWinnerTeamId(match: MatchItem): string | null {
   224→   224→  if (match.homeScore > match.awayScore) return match.homeTeamId;
   225→   225→  if (match.awayScore > match.homeScore) return match.awayTeamId;
   226→   226→  return null;
   227→   227→}
