    54→// ─── Position Helpers ────────────────────────────────────────────────
    55→
    56→function getPositionType(position: string | null): 'raider' | 'defender' | 'all-rounder' {
    57→  if (!position) return 'all-rounder';
    58→  const p = position.toLowerCase();
    59→  if (p.includes('raider')) return 'raider';
    60→  if (p.includes('corner') || p.includes('cover') || p.includes('defend')) return 'defender';
    61→  return 'all-rounder';
    62→}
    63→
    64→function getPositionIcon(position: string | null) {
    65→  const type = getPositionType(position);
    66→  switch (type) {
    67→    case 'raider': return Zap;
    68→    case 'defender': return Shield;
    69→    case 'all-rounder': return Swords;
    70→  }
    71→}
    72→
    73→function getPositionColor(position: string | null) {
    74→  const type = getPositionType(position);
    75→  switch (type) {
    76→    case 'raider': return { ring: 'ring-red-500', bg: 'bg-red-500', text: 'text-red-500', gradient: 'from-red-500 to-orange-500', border: 'border-red-500/30' };
    77→    case 'defender': return { ring: 'ring-blue-500', bg: 'bg-blue-500', text: 'text-blue-500', gradient: 'from-blue-500 to-cyan-500', border: 'border-blue-500/30' };
    78→    case 'all-rounder': return { ring: 'ring-amber-500', bg: 'bg-amber-500', text: 'text-amber-500', gradient: 'from-amber-500 to-yellow-400', border: 'border-amber-500/30' };
    79→  }
    80→}

    97→// ─── Animated Counter ────────────────────────────────────────────────
    98→
    99→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   100→  const [display, setDisplay] = useState(0);
   101→
   102→  useEffect(() => {
   103→    let start = 0;
   104→    const end = value;
   105→    if (end === 0) { setDisplay(0); return; }
   106→    const increment = end / (duration / 16);
   107→    const timer = setInterval(() => {
   108→      start += increment;
   109→      if (start >= end) {
   110→        setDisplay(end);
   111→        clearInterval(timer);
   112→      } else {
   113→        setDisplay(Math.floor(start));
   114→      }
   115→    }, 16);
   116→    return () => clearInterval(timer);
   117→  }, [value, duration]);
   118→
   119→  return <>{display}</>;
   120→}

   122→// ─── Circular Rating Indicator ───────────────────────────────────────
   123→
   124→function CircularRating({ rating, size = 48 }: { rating: number; size?: number }) {
   125→  const radius = (size - 8) / 2;
   126→  const circumference = 2 * Math.PI * radius;
   127→  const progress = Math.min(rating / 10, 1);
   128→  const offset = circumference - progress * circumference;
   129→
   130→  const getColor = (r: number) => {
   131→    if (r >= 8) return '#22c55e';
   132→    if (r >= 6) return '#eab308';
   133→    if (r >= 4) return '#f97316';
   134→    return '#ef4444';
   135→  };
   136→
   137→  return (
   138→    <div className="relative" style={{ width: size, height: size }}>
   139→      <svg width={size} height={size} className="-rotate-90">
   140→        <circle
   141→          cx={size / 2}
   142→          cy={size / 2}
   143→          r={radius}
   144→          stroke="rgba(255,255,255,0.15)"
   145→          strokeWidth="4"
   146→          fill="none"
   147→        />
   148→        <circle
   149→          cx={size / 2}
   150→          cy={size / 2}
   151→          r={radius}
   152→          stroke={getColor(rating)}
   153→          strokeWidth="4"
   154→          fill="none"
   155→          strokeLinecap="round"
   156→          strokeDasharray={circumference}
   157→          strokeDashoffset={offset}
   158→          className="transition-all duration-1000 ease-out"
   159→        />
   160→      </svg>
   161→      <div className="absolute inset-0 flex items-center justify-center">
   162→        <span className="text-xs font-black text-white">{rating.toFixed(1)}</span>
   163→      </div>
   164→    </div>
   165→  );
   166→}
