   240→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   241→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   242→  let homeScore = 0;
   243→  let awayScore = 0;
   244→  let homeOutPlayers = 0;
   245→  let awayOutPlayers = 0;
   246→  let homeOutPlayerIds: string[] = [];
   247→  let awayOutPlayerIds: string[] = [];
   248→
   249→  for (const evt of events) {
   250→    const side = getTeamSide(match, evt.teamId);
   251→    const points = evt.value;
   252→    if (side === 'home') {
   253→      homeScore += points;
   254→    } else {
   255→      awayScore += points;
   256→    }
   257→
   258→    // Count-based out tracking (backward compat)
   259→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   260→      if (side === 'home') {
   261→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   262→      } else {
   263→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   264→      }
   265→    }
   266→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   267→      if (side === 'home') {
   268→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   269→      } else {
   270→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   271→      }
   272→    }
   273→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   274→    if (evt.eventType === 'self_out') {
   275→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   276→      if (side === 'home') {
   277→        // Home scored from self-out → away defender went out
   278→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   279→        // Home revives one player
   280→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   281→      } else {
   282→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   283→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   284→      }
   285→    }
   286→    if (evt.eventType === 'all_out') {
   287→      if (side === 'home') {
   288→        awayOutPlayers = 0;
   289→      } else {
   290→        homeOutPlayers = 0;
   291→      }
   292→    }
   293→
   294→    // ID-based out tracking
   295→    let details: Record<string, unknown> = {};
   296→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   297→
   298→    if (evt.eventType === 'raid_point') {
   299→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   300→      if (side === 'home') {
   301→        // Home team scored raid → away defenders go out
   302→        for (const tid of touchedIds) {
   303→          if (!awayOutPlayerIds.includes(tid)) {
   304→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   305→          }
   306→        }
   307→        // Home team revives players (first-out-first-in)
   308→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   309→        if (reviveCount > 0) {
   310→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   311→        }
   312→      } else {
   313→        // Away team scored raid → home defenders go out
   314→        for (const tid of touchedIds) {
   315→          if (!homeOutPlayerIds.includes(tid)) {
   316→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   317→          }
   318→        }
   319→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   320→        if (reviveCount > 0) {
   321→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   322→        }
   323→      }
   324→    }
   325→
   326→    if (evt.eventType === 'bonus_point') {
   327→      // One out player from scoring team revives
   328→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   329→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   330→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   331→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   332→      }
   333→    }
   334→
   335→    // Self-out: defender steps off mat during a live raid
   336→    if (evt.eventType === 'self_out') {
   337→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   338→      if (selfOutPlayerId) {
   339→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   340→        if (side === 'home') {
   341→          // Home scored → away defender self-out
   342→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   343→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   344→          }
   345→          // Home revives first-out player
   346→          if (homeOutPlayerIds.length > 0) {
   347→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   348→          }
   349→        } else {
   350→          // Away scored → home defender self-out
   351→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   352→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   353→          }
   354→          if (awayOutPlayerIds.length > 0) {
   355→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   356→          }
   357→        }
   358→      }
   359→    }
   360→
   361→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   362→      // Raider goes out. The tackle is scored by the DEFENDING team.
   363→      // So the raider is from the OPPOSING team.
   364→      const raiderId = (details.raiderId as string) || evt.playerId;
   365→      if (side === 'home' && raiderId) {
   366→        // Home team scored tackle → away raider was caught → away raider goes out
   367→        if (!awayOutPlayerIds.includes(raiderId)) {
   368→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   369→        }
   370→      } else if (side === 'away' && raiderId) {
   371→        // Away team scored tackle → home raider was caught → home raider goes out
   372→        if (!homeOutPlayerIds.includes(raiderId)) {
   373→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   374→        }
   375→      }
   376→    }
   377→
   378→    if (evt.eventType === 'all_out') {
   379→      if (side === 'home') {
   380→        awayOutPlayerIds = [];
   381→      } else {
   382→        homeOutPlayerIds = [];
   383→      }
   384→    }
   385→  }
   386→
   387→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   388→}

   390→/** Event types that constitute a complete raid and should flip the turn */
   391→const RAID_ENDING_EVENT_TYPES: string[] = [
   392→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   393→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   394→];
   395→
   396→/** Event types that do NOT flip the turn (non-raid events) */
   397→const NON_RAID_EVENT_TYPES: string[] = [
   398→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   399→];
   400→
   401→/**
   402→ * Event types where the SCORING team is the DEFENDING team.
   403→ * In these cases, the defending team scored (they made the tackle / raider failed do-or-die),
   404→ * so they should raid NEXT — the turn goes to the SAME side that scored.
   405→ *
   406→ * For all other raid-ending events (raid_point, bonus_point, super_raid, empty_raid, self_out),
   407→ * the scoring team is the RAIDING team, so the turn goes to the OPPOSITE side.
   408→ */
   409→const DEFENDING_TEAM_SCORES: Set<string> = new Set([
   410→  'tackle_point',
   411→  'super_tackle',
   412→  'do_or_die_raid',
   413→]);
   414→
   415→/** Helper: determine raid queue from last raid-ending event */
   416→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   417→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   418→  if (lastRaidEvent) {
   419→    const side = getTeamSide(match, lastRaidEvent.teamId);
   420→    // If the defending team scored (tackle/super_tackle/do_or_die_raid),
   421→    // THEY raid next — return the SAME side as the scoring team.
   422→    if (DEFENDING_TEAM_SCORES.has(lastRaidEvent.eventType)) {
   423→      return side;
   424→    }
   425→    // If the raiding team scored (touch points, bonus, super raid, etc.)
   426→    // or it was an empty raid, the DEFENDING team raids next — return the OPPOSITE side.
   427→    return side === 'home' ? 'away' : 'home';
   428→  }
   429→  return match.raidQueue;
   430→}

   590→            ...event,
   591→            id: `evt_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   592→            timestamp: Date.now(),
   593→          };
   594→          const updatedEvents = [...state.activeMatch.events, newEvent];
   595→
   596→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   597→
   598→          // Only flip raidQueue for raid-ending events.
   599→          // Non-raid events (timeout, substitution, cards, standalone all_out)
   600→          // should NOT change whose turn it is to raid.
   601→          const isRaidEndingEvent = RAID_ENDING_EVENT_TYPES.includes(event.eventType);
   602→          const raidQueue = isRaidEndingEvent
   603→            ? getRaidQueueFromEvents(state.activeMatch, updatedEvents)
   604→            : state.activeMatch.raidQueue;
   605→
   606→          return {
   607→            activeMatch: {
   608→              ...state.activeMatch,
   609→              events: updatedEvents,
   610→              ...calculated,
   611→              raidQueue,
   612→            },
   613→          };
   614→        }),
   615→
   616→      addBatchEvents: (events) =>
   617→        set((state) => {
   618→          if (!state.activeMatch || events.length === 0) return {};
   619→          const newEvents: MatchEvent[] = events.map((e, i) => ({
   620→            ...e,
   621→            id: `evt_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 7)}`,
   622→            timestamp: Date.now() + i,
   623→          }));
   624→          const updatedEvents = [...state.activeMatch.events, ...newEvents];
   625→
   626→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   627→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, updatedEvents);
   628→
   629→          return {
   630→            activeMatch: {
   631→              ...state.activeMatch,
   632→              events: updatedEvents,
   633→              ...calculated,
   634→              raidQueue,
   635→            },
   636→          };
   637→        }),
   638→
   639→      undoLastEvent: () =>
   640→        set((state) => {
   641→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   642→          const events = [...state.activeMatch.events];
   643→          events.pop();
   644→
   645→          const calculated = recalculateFromEvents(state.activeMatch, events);
   646→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   647→
   648→          return {
   649→            activeMatch: {
   650→              ...state.activeMatch,
   651→              events,
   652→              ...calculated,
   653→              raidQueue,
   654→            },
   655→          };
   656→        }),
   657→
   658→      undoLastRaid: () =>
   659→        set((state) => {
   660→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   661→
   662→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   663→          // A raid batch has events with the same second in timestamp
   664→          const events = [...state.activeMatch.events];
   665→          if (events.length === 0) return {};
   666→
   667→          // Find the timestamp of the last event
   668→          const lastTimestamp = events[events.length - 1].timestamp;
   669→          // Remove all events that have the same timestamp (within 5ms for batch events)
   670→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   671→            events.pop();
   672→          }
   673→
   674→          const calculated = recalculateFromEvents(state.activeMatch, events);
   675→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   676→
   677→          return {
   678→            activeMatch: {
   679→              ...state.activeMatch,
   680→              events,
   681→              ...calculated,
   682→              raidQueue,
   683→            },
   684→          };
   685→        }),
   686→
   687→      addPlayerToMatch: (side, player) =>
   688→        set((state) => {
   689→          if (!state.activeMatch) return {};
   690→          const lineupKey = side === 'home' ? 'homeLineup' : 'awayLineup';
   691→          return {
   692→            activeMatch: {
   693→              ...state.activeMatch,
   694→              [lineupKey]: [...state.activeMatch[lineupKey], player],
   695→            },
   696→          };
   697→        }),
   698→
   699→      switchHalf: () =>
   700→        set((state) => {
   701→          if (!state.activeMatch) return {};
   702→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   703→          return {
   704→            activeMatch: {
   705→              ...state.activeMatch,
   706→              currentHalf: newHalf,
   707→              timer: state.activeMatch.halfDuration * 60,
   708→              isDoOrDie: false,
   709→              doOrDieTeamId: null,
   710→            },
   711→          };
   712→        }),
   713→
   714→      setTimer: (time) =>
   715→        set((state) => {
   716→          if (!state.activeMatch) return {};
   717→          return { activeMatch: { ...state.activeMatch, timer: time } };
   718→        }),
   719→
   720→      setDoOrDie: (value, teamId?) =>
   721→        set((state) => {
   722→          if (!state.activeMatch) return {};
   723→          return {
   724→            activeMatch: {
   725→              ...state.activeMatch,
   726→              isDoOrDie: value,
   727→              doOrDieTeamId: value ? (teamId || null) : null,
   728→            },
   729→          };
   730→        }),
   731→
   732→      switchRaidQueue: () =>
   733→        set((state) => {
   734→          if (!state.activeMatch) return {};
   735→          return {
   736→            activeMatch: {
   737→              ...state.activeMatch,
   738→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   739→            },
   740→          };
   741→        }),
   742→
   743→      callTimeout: (team) =>
   744→        set((state) => {
   745→          if (!state.activeMatch) return {};
   746→          return {
   747→            activeMatch: {
   748→              ...state.activeMatch,
   749→              homeTimeouts:
   750→                team === 'home'
   751→                  ? state.activeMatch.homeTimeouts + 1
   752→                  : state.activeMatch.homeTimeouts,
   753→              awayTimeouts:
   754→                team === 'away'
   755→                  ? state.activeMatch.awayTimeouts + 1
   756→                  : state.activeMatch.awayTimeouts,
   757→            },
   758→          };
   759→        }),
   760→
   761→      setOutPlayers: (team, count) =>
   762→        set((state) => {
   763→          if (!state.activeMatch) return {};
   764→          return {
   765→            activeMatch: {
   766→              ...state.activeMatch,
   767→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   768→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   769→            },
   770→          };
   771→        }),
   772→
   773→      fetchHomeData: async () => {
   774→        try {
   775→          const res = await fetch('/api/stats');
   776→          if (!res.ok) throw new Error('Failed to fetch stats');
   777→
   778→          const data = await res.json();
   779→
   780→          // Cache in store
   781→          set({ homeData: data });
   782→          return data;
   783→        } catch (err) {
   784→          console.error('fetchHomeData error:', err);
   785→          return null;
   786→        }
   787→      },
   788→
   789→      // Notification actions
   790→      addNotification: (notification) =>
   791→        set((state) => ({
   792→          notifications: [
   793→            {
   794→              ...notification,
   795→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   796→              timestamp: Date.now(),
   797→              read: false,
   798→            },
   799→            ...state.notifications,
   800→          ].slice(0, 50), // Keep max 50 notifications
   801→        })),
   802→
   803→      markNotificationRead: (id) =>
   804→        set((state) => ({
   805→          notifications: state.notifications.map((n) =>
   806→            n.id === id ? { ...n, read: true } : n
   807→          ),
   808→        })),
   809→
   810→      markAllRead: () =>
   811→        set((state) => ({
   812→          notifications: state.notifications.map((n) => ({ ...n, read: true })),
   813→        })),
   814→
   815→      clearNotifications: () => set({ notifications: [] }),
   816→
   817→      setLanguage: (lang) => set({ language: lang }),
   818→
   819→      // Onboarding actions
   820→      setHasCompletedOnboarding: (value) => set({ hasCompletedOnboarding: value }),
   821→
   822→      setOnboardingProfile: (profile) =>
   823→        set((state) => ({
   824→          onboardingProfile: { ...state.onboardingProfile, ...profile },
   825→        })),
   826→
   827→      completeOnboarding: () =>
   828→        set((state) => ({
   829→          hasCompletedOnboarding: true,
   830→          currentUser: state.currentUser
   831→            ? {
   832→                ...state.currentUser,
   833→                ...(state.onboardingProfile.position && { role: state.onboardingProfile.position }),
   834→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   835→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   836→                createdAt: state.currentUser.createdAt || Date.now(),
   837→              }
   838→            : null,
   839→        })),
   840→
   841→      // Team management actions
   842→      setTeamFilter: (filter) =>
   843→        set((state) => ({
   844→          teamManagement: { ...state.teamManagement, teamFilter: filter },
   845→        })),
   846→
   847→      setTeamSearch: (search) =>
   848→        set((state) => ({
   849→          teamManagement: { ...state.teamManagement, teamSearch: search },
   850→        })),
   851→
   852→      setSelectedTeamId: (teamId) =>
   853→        set((state) => ({
   854→          teamManagement: { ...state.teamManagement, selectedTeamId: teamId },
   855→        })),
   856→
   857→      setTeamDetailOpen: (open) =>
   858→        set((state) => ({
   859→          teamManagement: { ...state.teamManagement, teamDetailOpen: open },
   860→        })),
   861→
   862→      // Match transfer / handoff — load a match from a transfer
   863→      loadMatch: (match) =>
   864→        set({
   865→          activeMatch: match,
   866→          activeTab: 'quick-score' as TabId,
   867→          showToss: false,
   868→          tossMatchConfig: null,
   869→        }),
   870→    }),
   871→    {
   872→      name: 'kabaddi-pro-storage',
   873→      storage: createJSONStorage(() => localStorage),
   874→      // Only persist auth-related fields and notifications
   875→      partialize: (state) => ({
   876→        isAuthenticated: state.isAuthenticated,
   877→        isOnboarded: state.isOnboarded,
   878→        currentUser: state.currentUser,
   879→        hasSeenSplash: state.hasSeenSplash,
   880→        notifications: state.notifications,
   881→        language: state.language,
   882→        hasCompletedOnboarding: state.hasCompletedOnboarding,
   883→        onboardingProfile: state.onboardingProfile,
   884→        coachAcademies: state.coachAcademies,
   885→      }),
   886→    }
   887→  )
   888→);

   890→export function getEventPoints(eventType: EventType): number {
   891→  switch (eventType) {
   892→    case 'raid_point':
   893→      return 1;
   894→    case 'bonus_point':
   895→      return 1;
   896→    case 'tackle_point':
   897→      return 1;
   898→    case 'super_raid':
   899→      return 1;
   900→    case 'super_tackle':
   901→      return 1;
   902→    case 'do_or_die_raid':
   903→      return 1;
   904→    case 'all_out':
   905→      return 2;
   906→    case 'self_out':
   907→      return 1;
   908→    case 'empty_raid':
   909→      return 0;
   910→    case 'substitution':
   911→    case 'timeout':
   912→    case 'yellow_card':
   913→    case 'red_card':
   914→    case 'green_card':
   915→      return 0;
   916→    default:
   917→      return 0;
   918→  }
   919→}
