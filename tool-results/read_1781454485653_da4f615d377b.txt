    14→function TossConfetti({ color, index }: { color: string; index: number }) {
    15→  const leftPos = (index * 23 + 7) % 100;
    16→  const rotateEnd = (index * 53 + 17) % 720 - 360;
    17→  const xDrift = ((index * 37) % 100) - 50;
    18→  const sizeClass = index % 3 === 0 ? 'w-3 h-3' : index % 3 === 1 ? 'w-2 h-2' : 'w-1.5 h-1.5';
    19→  const shapeClass = index % 4 === 0 ? 'rounded-full' : index % 4 === 1 ? 'rounded-sm' : index % 4 === 2 ? 'rounded-l-full' : 'rounded-t-full';
    20→
    21→  return (
    22→    <motion.div
    23→      initial={{ y: -20, x: 0, rotate: 0, opacity: 1, scale: 1 }}
    24→      animate={{ y: '100vh', x: xDrift, rotate: rotateEnd, opacity: 0, scale: 0.5 }}
    25→      transition={{ duration: 2.5 + (index % 4) * 0.5, delay: index * 0.04, ease: 'easeIn' }}
    26→      className={cn('absolute top-0', sizeClass, shapeClass)}
    27→      style={{ left: `${leftPos}%`, backgroundColor: color }}
    28→    />
    29→  );
    30→}

   307→// ─── Main TossScreen ────────────────────────────────────────────────
   308→
   309→interface TossScreenProps {
   310→  homeTeam: string;
   311→  awayTeam: string;
   312→  homeTeamColor: string;
   313→  awayTeamColor: string;
   314→  onTossComplete: (firstRaidTeam: 'home' | 'away') => void;
   315→  onBack: () => void;
   316→}
   317→
   318→export default function TossScreen({
   319→  homeTeam,
   320→  awayTeam,
   321→  homeTeamColor,
   322→  awayTeamColor,
   323→  onTossComplete,
   324→  onBack,
   325→}: TossScreenProps) {
   326→  // Phase state
   327→  const [phase, setPhase] = useState<TossPhase>('choose-caller');
   328→
   329→  // Toss data
   330→  const [callingTeam, setCallingTeam] = useState<'home' | 'away' | null>(null);
   331→  const [chosenSide, setChosenSide] = useState<'heads' | 'tails' | null>(null);
   332→  const [coinResult, setCoinResult] = useState<'heads' | 'tails'>('heads');
   333→  const [tossWinner, setTossWinner] = useState<'home' | 'away' | null>(null);
   334→  const [showConfetti, setShowConfetti] = useState(false);
   335→  const [flipKey, setFlipKey] = useState(0);
   336→
   337→  // Countdown
   338→  const [countdown, setCountdown] = useState<number | null>(null);
   339→  const countdownRef = useRef<ReturnType<typeof setInterval> | null>(null);
   340→
   341→  // Derived
   342→  const callingTeamName = callingTeam === 'home' ? homeTeam : callingTeam === 'away' ? awayTeam : '';
   343→  const callingTeamColor = callingTeam === 'home' ? homeTeamColor : callingTeam === 'away' ? awayTeamColor : '';
   344→  const winnerName = tossWinner === 'home' ? homeTeam : tossWinner === 'away' ? awayTeam : '';
   345→  const winnerColor = tossWinner === 'home' ? homeTeamColor : tossWinner === 'away' ? awayTeamColor : '';
   346→  const loserName = tossWinner === 'home' ? awayTeam : tossWinner === 'away' ? homeTeam : '';
   347→  const loserColor = tossWinner === 'home' ? awayTeamColor : tossWinner === 'away' ? homeTeamColor : '';
   348→
   349→  // Phase step number for indicator
   350→  const phaseStep: number = phase === 'choose-caller' ? 0 : phase === 'choose-side' ? 1 : phase === 'ready' ? 1 : phase === 'flipping' ? 1 : phase === 'result' ? 2 : phase === 'choose-advantage' ? 2 : 2;
   351→
   352→  // Clean up countdown on unmount
   353→  useEffect(() => {
   354→    return () => {
   355→      if (countdownRef.current) clearInterval(countdownRef.current);
   356→    };
   357→  }, []);
   358→
   359→  // ─── Handlers ────────────────────────────────────────────────────
   360→
   361→  const handleSelectCaller = (team: 'home' | 'away') => {
   362→    setCallingTeam(team);
   363→    setPhase('choose-side');
   364→  };
   365→
   366→  const handleChooseSide = (side: 'heads' | 'tails') => {
   367→    setChosenSide(side);
   368→    setPhase('ready');
   369→  };
   370→
   371→  const doToss = useCallback(() => {
   372→    setPhase('flipping');
   373→    setFlipKey((k) => k + 1);
   374→
   375→    // Determine coin result randomly
   376→    const result: 'heads' | 'tails' = Math.random() < 0.5 ? 'heads' : 'tails';
   377→    setCoinResult(result);
   378→
   379→    // The calling team wins if the coin result matches their choice
   380→    const winner: 'home' | 'away' = result === chosenSide ? callingTeam! : (callingTeam === 'home' ? 'away' : 'home');
   381→    setTossWinner(winner);
   382→
   383→    // Show confetti and move to result after animation
   384→    setTimeout(() => {
   385→      setShowConfetti(true);
   386→      setPhase('result');
   387→    }, 2200);
   388→  }, [chosenSide, callingTeam]);
   389→
   390→  const handleChooseAdvantage = (firstRaidTeam: 'home' | 'away') => {
   391→    setPhase('countdown');
   392→    setCountdown(3);
   393→
   394→    let count = 3;
   395→    countdownRef.current = setInterval(() => {
   396→      count--;
   397→      setCountdown(count);
   398→      if (count <= 0) {
   399→        if (countdownRef.current) clearInterval(countdownRef.current);
   400→        onTossComplete(firstRaidTeam);
   401→      }
   402→    }, 800);
   403→  };
   404→
   405→  // ─── Strategy text ───────────────────────────────────────────────
   406→
   407→  const raidStrategy = `${winnerName} raids first — put early pressure on ${loserName}'s defense`;
   408→  const defendStrategy = `${winnerName} defends first — choose your court side and force ${loserName} into empty raids`;
   409→
   410→  // ─── Render ──────────────────────────────────────────────────────
   411→
   412→  return (
   413→    <div className="fixed inset-0 z-50 bg-gray-950 flex flex-col items-center justify-center p-4 sm:p-6 overflow-hidden">
   414→      {/* ═══ Background — Team color spotlights ═══ */}
   415→      <div className="absolute inset-0 overflow-hidden">
   416→        {/* Home spotlight */}
   417→        <motion.div
   418→          className="absolute top-0 left-0 w-1/2 h-full"
   419→          style={{ background: `radial-gradient(ellipse at 30% 50%, ${homeTeamColor}15, transparent 70%)` }}
   420→          animate={phase === 'flipping' ? { opacity: [0.5, 0.9, 0.5] } : { opacity: 0.5 }}
   421→          transition={{ duration: 1, repeat: phase === 'flipping' ? Infinity : 0 }}
   422→        />
   423→        {/* Away spotlight */}
   424→        <motion.div
   425→          className="absolute top-0 right-0 w-1/2 h-full"
   426→          style={{ background: `radial-gradient(ellipse at 70% 50%, ${awayTeamColor}15, transparent 70%)` }}
   427→          animate={phase === 'flipping' ? { opacity: [0.5, 0.9, 0.5] } : { opacity: 0.5 }}
   428→          transition={{ duration: 1, repeat: phase === 'flipping' ? Infinity : 0, delay: 0.5 }}
   429→        />
   430→        {/* Center glow */}
   431→        <div className="absolute top-1/3 left-1/2 -translate-x-1/2 -translate-y-1/2 w-80 h-80 opacity-5 bg-yellow-400 rounded-full" />
   432→
   433→        {/* Floating particles */}
   434→        {Array.from({ length: 16 }).map((_, i) => (
   435→          <FloatingParticle
   436→            key={i}
   437→            color={i < 8 ? homeTeamColor : awayTeamColor}
   438→            delay={i * 0.35}
   439→            x={(i * 6 + 5) % 100}
   440→            size={2 + (i % 4) * 2}
   441→          />
   442→        ))}
   443→
   444→        {/* Grid pattern overlay */}
   445→        <div
   446→          className="absolute inset-0 opacity-[0.03]"
   447→          style={{
   448→            backgroundImage: 'linear-gradient(rgba(255,255,255,0.1) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,0.1) 1px, transparent 1px)',
   449→            backgroundSize: '40px 40px',
   450→          }}
   451→        />
   452→      </div>
   453→
   454→      {/* ═══ Confetti ═══ */}
   455→      <AnimatePresence>
   456→        {showConfetti && (
   457→          <div className="fixed inset-0 pointer-events-none overflow-hidden z-50">
   458→            {Array.from({ length: 50 }).map((_, i) => {
   459→              const colors = [homeTeamColor, awayTeamColor, '#FFD700', '#FF6B35', '#FFFFFF', '#E040FB', '#00E5FF'];
   460→              return <TossConfetti key={`confetti-${i}`} color={colors[i % colors.length]} index={i} />;
   461→            })}
   462→          </div>
   463→        )}
