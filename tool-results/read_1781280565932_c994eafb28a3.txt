   183→/** Format time as MM:SS */
   184→function formatTime(seconds: number): string {
   185→  const m = Math.floor(Math.max(0, seconds) / 60);
   186→  const s = Math.max(0, seconds) % 60;
   187→  return `${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
   188→}
   189→
   190→/** Format relative timestamp */
   191→function formatEventTime(timestamp: number, matchStart: number): string {
   192→  const diff = Math.max(0, Math.floor((timestamp - matchStart) / 1000));
   193→  return formatTime(diff);
   194→}

   196→// ─── Circular Timer Component ───────────────────────────────────────
   197→
   198→function CircularTimer({
   199→  seconds,
   200→  totalSeconds,
   201→  teamColor,
   202→}: {
   203→  seconds: number;
   204→  totalSeconds: number;
   205→  teamColor: string;
   206→}) {
   207→  const radius = 28;
   208→  const stroke = 3;
   209→  const normalizedRadius = radius - stroke;
   210→  const circumference = normalizedRadius * 2 * Math.PI;
   211→  const progress = totalSeconds > 0 ? Math.max(0, seconds) / totalSeconds : 0;
   212→  const strokeDashoffset = circumference * (1 - progress);
   213→
   214→  return (
   215→    <div className="relative w-16 h-16 flex items-center justify-center">
   216→      <svg height={radius * 2} width={radius * 2} className="-rotate-90">
   217→        <circle
   218→          stroke="currentColor"
   219→          className="text-gray-200 dark:text-warm-700"
   220→          fill="transparent"
   221→          strokeWidth={stroke}
   222→          r={normalizedRadius}
   223→          cx={radius}
   224→          cy={radius}
   225→        />
   226→        <circle
   227→          stroke={teamColor}
   228→          fill="transparent"
   229→          strokeWidth={stroke}
   230→          strokeDasharray={circumference + ' ' + circumference}
   231→          style={{ strokeDashoffset }}
   232→          strokeLinecap="round"
   233→          r={normalizedRadius}
   234→          cx={radius}
   235→          cy={radius}
   236→          className="transition-all duration-1000 ease-linear"
   237→        />
   238→      </svg>
   239→      <div className="absolute inset-0 flex items-center justify-center">
   240→        <span className="text-xs font-bold text-gray-800 dark:text-warm-100 tabular-nums">
   241→          {formatTime(seconds)}
   242→        </span>
   243→      </div>
   244→    </div>
   245→  );
   246→}

   250→function LiveEventCard({
   251→  event,
   252→  match,
   253→  matchStart,
   254→  index,
   255→}: {
   256→  event: MatchEventAPI;
   257→  match: MatchDataAPI;
   258→  matchStart: number;
   259→  index: number;
   260→}) {
   261→  const isHome = event.teamId === match.homeTeamId;
   262→  const teamName = isHome ? match.homeTeam : match.awayTeam;
   263→  const teamColor = isHome ? match.homeTeamColor : match.awayTeamColor;
   264→  const eventType = event.eventType as EventType;
   265→  const big = isBigEvent(eventType);
   266→  const iconColor = getEventIconColor(eventType);
   267→  const bgColor = getEventBgColor(eventType);
   268→
   269→  // Build commentary text
   270→  let commentaryText = '';
   271→  try {
   272→    let extras: CommentaryExtras = {};
   273→    if (event.details) {
   274→      const parsed = JSON.parse(event.details);
   275→      if (parsed.isSuperRaid) extras.isSuperRaid = true;
   276→      if (parsed.isSuperTackle) extras.isSuperTackle = true;
   277→      if (parsed.isDoOrDie) extras.isDoOrDie = true;
   278→      if (parsed.isAllOut) extras.isAllOut = true;
   279→      if (parsed.touchedPlayerIds) extras.defendersTouched = (parsed.touchedPlayerIds as string[]).length;
   280→    }
   281→    commentaryText = generateCommentary(event.eventType, event.playerId || 'Player', teamName, event.value, extras);
   282→  } catch {
   283→    commentaryText = `${event.playerId || 'Player'} made a play for ${teamName}`;
   284→  }
   285→
   286→  return (
   287→    <motion.div
   288→      initial={{ opacity: 0, x: 40, scale: 0.95 }}
   289→      animate={{ opacity: 1, x: 0, scale: 1 }}
   290→      transition={{ duration: 0.35, delay: index * 0.03, type: 'spring', stiffness: 200 }}
   291→    >
   292→      <div
   293→        className={cn(
   294→          'relative flex items-start gap-3 p-3 rounded-xl border transition-all',
   295→          big
   296→            ? 'border-amber-300/50 dark:border-amber-600/40 bg-amber-50/80 dark:bg-amber-900/20'
   297→            : 'border-gray-100 dark:border-warm-700 bg-white dark:bg-warm-800/60'
   298→        )}
   299→      >
   300→        {/* Big event glow */}
   301→        {big && (
   302→          <div
   303→            className="absolute inset-0 rounded-xl opacity-20 animate-pulse"
   304→            style={{ boxShadow: `0 0 20px ${teamColor}40` }}
   305→          />
   306→        )}
   307→
   308→        {/* Left color bar */}
   309→        <div
   310→          className="w-1 rounded-full min-h-[40px] flex-shrink-0 self-stretch"
   311→          style={{ backgroundColor: teamColor }}
   312→        />
   313→
   314→        {/* Icon */}
   315→        <div className={cn('w-9 h-9 rounded-lg flex items-center justify-center flex-shrink-0', bgColor)}>
   316→          <EventIconComponent type={eventType} className={cn('w-4 h-4', iconColor)} />
   317→        </div>
   318→
   319→        {/* Content */}
   320→        <div className="flex-1 min-w-0">
   321→          <div className="flex items-start justify-between gap-2">
   322→            <div>
   323→              <div className="flex items-center gap-1.5 mb-0.5">
   324→                <Badge
   325→                  className={cn(
   326→                    'text-[9px] font-bold px-1.5 py-0 h-4 border-0',
   327→                    big ? 'bg-amber-500/20 text-amber-700 dark:text-amber-400' : 'bg-gray-100 dark:bg-warm-700 text-gray-500 dark:text-warm-400'
   328→                  )}
   329→                >
   330→                  {getEventLabel(eventType)}
   331→                </Badge>
   332→                {event.half === 1 ? (
   333→                  <span className="text-[9px] text-gray-400">1H</span>
   334→                ) : (
   335→                  <span className="text-[9px] text-gray-400">2H</span>
   336→                )}
   337→              </div>
   338→              <p className="text-xs font-medium text-gray-700 dark:text-warm-200 leading-snug line-clamp-2">
   339→                {commentaryText}
   340→              </p>
   341→            </div>
   342→            {event.value > 0 && (
   343→              <motion.div
   344→                initial={{ scale: 0.5 }}
   345→                animate={{ scale: 1 }}
   346→                className="flex-shrink-0"
   347→              >
   348→                <Badge
   349→                  className="text-sm font-black px-2 py-0 h-6 border-0"
   350→                  style={{ backgroundColor: teamColor + '20', color: teamColor }}
   351→                >
   352→                  +{event.value}
   353→                </Badge>
   354→              </motion.div>
   355→            )}
   356→          </div>
   357→          <div className="flex items-center gap-2 mt-1">
   358→            <span className="text-[10px] text-gray-400 dark:text-warm-500 font-medium">
   359→              {formatEventTime(event.timestamp, matchStart)}
   360→            </span>
   361→            <span className="text-[10px] text-gray-300 dark:text-warm-600">·</span>
   362→            <span className="text-[10px] font-medium" style={{ color: teamColor }}>
   363→              {teamName}
   364→            </span>
   365→          </div>
   366→        </div>
   367→      </div>
   368→    </motion.div>
   369→  );
   370→}

   374→function StatBar({
   375→  label,
   376→  homeValue,
   377→  awayValue,
   378→  homeColor,
   379→  awayColor,
   380→  isPercentage = false,
   381→}: {
   382→  label: string;
   383→  homeValue: number;
   384→  awayValue: number;
   385→  homeColor: string;
   386→  awayColor: string;
   387→  isPercentage?: boolean;
   388→}) {
   389→  const total = homeValue + awayValue;
   390→  const homePercent = total > 0 ? (homeValue / total) * 100 : 50;
   391→  const awayPercent = total > 0 ? (awayValue / total) * 100 : 50;
   392→
   393→  return (
   394→    <div className="mb-4">
   395→      <div className="flex items-center justify-between mb-1.5">
   396→        <span className="text-sm font-bold" style={{ color: homeColor }}>
   397→          {isPercentage ? `${homeValue}%` : homeValue}
   398→        </span>
   399→        <span className="text-[10px] font-semibold text-gray-500 dark:text-warm-400 uppercase tracking-wider">
   400→          {label}
   401→        </span>
   402→        <span className="text-sm font-bold" style={{ color: awayColor }}>
   403→          {isPercentage ? `${awayValue}%` : awayValue}
   404→        </span>
   405→      </div>
   406→      <div className="flex items-center gap-1 h-2.5">
   407→        <div className="flex-1 bg-gray-100 dark:bg-warm-700 rounded-full overflow-hidden flex justify-end">
   408→          <motion.div
   409→            initial={{ width: 0 }}
   410→            animate={{ width: `${homePercent}%` }}
   411→            transition={{ duration: 0.8, ease: 'easeOut' }}
   412→            className="h-full rounded-full"
   413→            style={{ backgroundColor: homeColor }}
   414→          />
   415→        </div>

   417→          <motion.div

   559→export default function MatchDayExperience({ matchId, onClose }: MatchDayExperienceProps) {
   560→  // ─── State ───────────────────────────────────────────────────
   561→  const [matchData, setMatchData] = useState<MatchDataAPI | null>(null);
   562→  const [events, setEvents] = useState<MatchEventAPI[]>([]);
   563→  const [loading, setLoading] = useState(true);
   564→  const [activeTab, setActiveTab] = useState<TabType>('feed');
   565→  const [isFullscreen, setIsFullscreen] = useState(false);
   566→  const [isMuted, setIsMuted] = useState(false);
   567→  const [timerSeconds, setTimerSeconds] = useState(0);
   568→  const [selectedMoment, setSelectedMoment] = useState<string | null>(null);
   569→  const prevHomeScoreRef = useRef(0);
   570→  const prevAwayScoreRef = useRef(0);
   571→  const [homeScoreAnim, setHomeScoreAnim] = useState(false);
   572→  const [awayScoreAnim, setAwayScoreAnim] = useState(false);
   573→
   574→  const feedRef = useRef<HTMLDivElement>(null);
   575→  const pollIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   576→  const timerIntervalRef = useRef<ReturnType<typeof setInterval> | null>(null);
   577→
   578→  // ─── Fetch Match Data ────────────────────────────────────────
   579→  const fetchMatchData = useCallback(async () => {
   580→    try {
   581→      const res = await fetch(`/api/match-events?matchId=${matchId}`);
   582→      if (!res.ok) throw new Error('Failed to fetch');
   583→      const data: MatchEventsAPIResponse = await res.json();
   584→
   585→      setMatchData(data.match);
   586→      setEvents(data.events);
   587→
   588→      // Calculate timer from match data
   589→      if (data.match.currentHalf && data.match.halfDuration) {

   775→          id: playerId,
   776→          name: `Player #${playerId.slice(-3)}`,
   777→          tacklePts: ps.tacklePts,

   832→        className="fixed inset-0 z-50 bg-white dark:bg-warm-900 flex items-center justify-center"
