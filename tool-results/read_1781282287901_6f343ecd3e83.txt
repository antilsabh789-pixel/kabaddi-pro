   123→function formatDuration(isoStart: string | null, isoEnd: string | null): string {
   124→  if (!isoStart || !isoEnd) return '';
   125→  const ms = new Date(isoEnd).getTime() - new Date(isoStart).getTime();
   126→  const totalMins = Math.round(ms / 60000);
   127→  return `${totalMins} min`;
   128→}
   129→
   130→function formatDateTime(iso: string | null): string {
   131→  if (!iso) return '';
   132→  const d = new Date(iso);
   133→  return d.toLocaleDateString('en-IN', { day: 'numeric', month: 'short', year: 'numeric' });
   134→}

   165→// ─── Component ────────────────────────────────────────────────────────────────
   166→
   167→export default function MatchHighlightsScreen({ matchId, onClose }: MatchHighlightsScreenProps) {
   168→  const [match, setMatch] = useState<MatchData | null>(null);
   169→  const [loading, setLoading] = useState(true);
   170→  const [error, setError] = useState<string | null>(null);
   171→  const [showShare, setShowShare] = useState(false);
   172→
   173→  const { toast } = useToast();
   174→
   175→  // ── Fetch match ────────────────────────────────────────────────────────────
   176→  const fetchMatch = useCallback(async () => {
   177→    setLoading(true);
   178→    setError(null);
   179→    try {
   180→      const res = await fetch(`/api/matches?id=${encodeURIComponent(matchId)}`);
   181→      if (!res.ok) throw new Error('Match not found');
   182→      const data = await res.json();
   183→      setMatch(data.match as MatchData);
   184→    } catch (err) {
   185→      console.error('Match highlights fetch error:', err);
   186→      setError('Failed to load match highlights');
   187→    } finally {
   188→      setLoading(false);
   189→    }
   190→  }, [matchId]);
   191→
   192→  useEffect(() => {
   193→    fetchMatch();
   194→  }, [fetchMatch]);
   195→
   196→  // ── Derived data ───────────────────────────────────────────────────────────
   197→  const homeColor = match?.homeTeam?.color || '#DC2626';
   198→  const awayColor = match?.awayTeam?.color || '#1E293B';
   199→  const isHomeWin = match ? match.homeScore > match.awayScore : false;
   200→  const isAwayWin = match ? match.awayScore > match.homeScore : false;
   201→  const isDraw = match ? match.homeScore === match.awayScore : false;
   202→  const winnerSide: 'home' | 'away' | 'draw' | null = match
   203→    ? isDraw ? 'draw' : isHomeWin ? 'home' : 'away'
   204→    : null;
   205→
   206→  // Build scorer name lookup
   207→  const scorerNames = useMemo(() => {
   208→    if (!match) return {} as Record<string, string>;
   209→    const map: Record<string, string> = {};
   210→    for (const s of match.scorers) {
   211→      map[s.userId] = s.user.name;
   212→    }
   213→    return map;
   214→  }, [match]);
   215→
   216→  // ── Sorted events (chronological) ──────────────────────────────────────────
   217→  const sortedEvents = useMemo(() => {
   218→    if (!match) return [];
   219→    return [...match.events].sort(
   220→      (a, b) => new Date(a.timestamp).getTime() - new Date(b.timestamp).getTime()
   221→    );
   222→  }, [match]);
   223→
   224→  // ── Key Moments ────────────────────────────────────────────────────────────
   225→  const keyMoments = useMemo((): KeyMoment[] => {
   226→    if (!match) return [];
   227→
   228→    const moments: KeyMoment[] = [];
   229→    let homeScore = 0;
   230→    let awayScore = 0;
   231→    let leadingTeamId: string | null = null;
   232→
   233→    for (const evt of sortedEvents) {
   234→      const isHome = evt.teamId === match.homeTeamId;
   235→      const teamName = isHome ? match.homeTeam.name : match.awayTeam.name;
   236→      const teamColor = isHome ? homeColor : awayColor;
   237→      const playerName = evt.playerId ? scorerNames[evt.playerId] : undefined;
   238→
   239→      // Track score for comeback detection
   240→      if (isHome) {
   241→        homeScore += evt.value;
   242→      } else {
   243→        awayScore += evt.value;
   244→      }
   245→
   246→      // Detect lead change (comeback)
   247→      const currentLeader = homeScore > awayScore
   248→        ? match.homeTeamId
   249→        : awayScore > homeScore
   250→          ? match.awayTeamId
   251→          : null;
   252→
   253→      const isComeback = currentLeader !== null
   254→        && leadingTeamId !== null
   255→        && currentLeader !== leadingTeamId
   256→        && sortedEvents.indexOf(evt) > 0;
   257→
   258→      leadingTeamId = currentLeader;
   259→
   260→      // Filter for highlight events
   261→      if (HIGHLIGHT_EVENTS.has(evt.eventType)) {
   262→        const meta = EVENT_META[evt.eventType] || { icon: '📌', label: evt.eventType, color: 'text-warm-500' };
   263→        let description = '';
   264→
   265→        switch (evt.eventType) {
   266→          case 'super_raid':
   267→            description = `${playerName || teamName} scores 3+ points in a single raid!`;
   268→            break;
   269→          case 'super_tackle':
   270→            description = `${playerName || teamName} executes a super tackle with 3 or fewer defenders!`;
   271→            break;
   272→          case 'all_out':
   273→            description = `${teamName} inflicts an All Out — entire opposition sent off the mat!`;
   274→            break;
   275→          case 'do_or_die_raid':
   276→            description = `${playerName || teamName} delivers in the crucial Do-or-Die raid!`;
   277→            break;
   278→        }
   279→
   280→        moments.push({
   281→          id: evt.id,
   282→          eventType: evt.eventType,
   283→          label: meta.label,
   284→          icon: meta.icon,
   285→          teamId: evt.teamId,
   286→          teamName,
   287→          teamColor,
   288→          playerName,
   289→          value: evt.value,
   290→          half: evt.half,
   291→          timestamp: evt.timestamp,
   292→          description,
   293→        });
   294→      }
   295→
   296→      // Add comeback moment
   297→      if (isComeback && currentLeader !== null) {
   298→        const comebackTeam = currentLeader === match.homeTeamId ? match.homeTeam : match.awayTeam;
   299→        const comebackColor = currentLeader === match.homeTeamId ? homeColor : awayColor;
   300→        moments.push({
   301→          id: `comeback_${evt.id}`,
   302→          eventType: 'comeback',
   303→          label: 'Comeback!',
   304→          icon: '🔄',
   305→          teamId: currentLeader,
   306→          teamName: comebackTeam.name,
   307→          teamColor: comebackColor,
   308→          value: 0,
   309→          half: evt.half,
   310→          timestamp: evt.timestamp,
   311→          description: `${comebackTeam.name} takes the lead! The momentum shifts!`,
   312→        });
   313→      }
   314→    }
   315→
   316→    return moments;
   317→  }, [match, sortedEvents, homeColor, awayColor, scorerNames]);
   318→
   319→  // ── MVP Breakdown ──────────────────────────────────────────────────────────
   320→  const mvpBreakdown = useMemo(() => {
   321→    if (!match?.motmUser) return null;
   322→
   323→    let raidPoints = 0;
   324→    let tacklePoints = 0;
   325→    let bonusPoints = 0;
   326→
   327→    for (const evt of match.events) {
   328→      if (evt.playerId !== match.motmUser.id) continue;
   329→      switch (evt.eventType) {
   330→        case 'raid_point':
   331→        case 'super_raid':
   332→        case 'do_or_die_raid':
   333→          raidPoints += evt.value;
   334→          break;
   335→        case 'tackle_point':
   336→        case 'super_tackle':
   337→          tacklePoints += evt.value;
   338→          break;
   339→        case 'bonus_point':
   340→          bonusPoints += evt.value;
   341→          break;
   342→      }
   343→    }
   344→
   345→    const totalPoints = raidPoints + tacklePoints + bonusPoints;
   346→    const motmTeamId = match.events.find(e => e.playerId === match.motmUser!.id)?.teamId;
   347→    const teamName = motmTeamId === match.homeTeamId
   348→      ? match.homeTeam.name
   349→      : motmTeamId === match.awayTeamId
   350→        ? match.awayTeam.name
   351→        : '';
   352→
   353→    return {
   354→      raidPoints,
   355→      tacklePoints,
   356→      bonusPoints,
   357→      totalPoints,
   358→      teamName,
   359→    };
   360→  }, [match]);
   361→
   362→  // ── Score Progression (for momentum graph) ─────────────────────────────────
   363→  const scoreProgression = useMemo(() => {
   364→    if (!match) return [];
   365→
   366→    let homeScore = 0;
   367→    let awayScore = 0;
   368→    const progression: { index: number; homeScore: number; awayScore: number; gap: number; half: number }[] = [];
   369→
   370→    // Start at 0-0
   371→    progression.push({ index: 0, homeScore: 0, awayScore: 0, gap: 0, half: 1 });
   372→
   373→    sortedEvents.forEach((evt, idx) => {
   374→      const isHome = evt.teamId === match.homeTeamId;
   375→      if (isHome) {
   376→        homeScore += evt.value;
   377→      } else {
   378→        awayScore += evt.value;
   379→      }
   380→      progression.push({
   381→        index: idx + 1,
   382→        homeScore,
   383→        awayScore,
   384→        gap: homeScore - awayScore,
   385→        half: evt.half,
   386→      });
   387→    });
   388→
   389→    return progression;
   390→  }, [match, sortedEvents]);
   391→
   392→  // ── Turning Points ─────────────────────────────────────────────────────────
   393→  const turningPoints = useMemo((): TurningPoint[] => {
   394→    if (!match) return [];
   395→
   396→    const points: TurningPoint[] = [];
   397→    let homeScore = 0;
   398→    let awayScore = 0;
   399→    let prevLeader: 'home' | 'away' | 'draw' = 'draw';
   400→
   401→    for (const evt of sortedEvents) {
   402→      const isHome = evt.teamId === match.homeTeamId;
   403→      if (isHome) {
   404→        homeScore += evt.value;
   405→      } else {
   406→        awayScore += evt.value;
   407→      }
   408→
   409→      const currentLeader: 'home' | 'away' | 'draw' =
   410→        homeScore > awayScore ? 'home' : awayScore > homeScore ? 'away' : 'draw';
   411→
   412→      // All Out that changed lead
   413→      if (evt.eventType === 'all_out' && currentLeader !== prevLeader && currentLeader !== 'draw') {
   414→        const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;
   415→        const color = currentLeader === 'home' ? homeColor : awayColor;
   416→        points.push({
   417→          id: `tp_ao_${evt.id}`,
   418→          type: 'all_out_lead_change',
   419→          label: 'All Out Lead Change',
   420→          icon: '🔥',
   421→          description: `${team.name}'s All Out flipped the lead! ${homeScore}-${awayScore}`,
   422→          teamName: team.name,
   423→          teamColor: color,
   424→          timestamp: evt.timestamp,
   425→          half: evt.half,
   426→        });
   427→      }
   428→
   429→      // Do-or-Die raid that shifted momentum (lead change)
   430→      if (evt.eventType === 'do_or_die_raid' && currentLeader !== prevLeader && currentLeader !== 'draw') {
   431→        const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;
   432→        const color = currentLeader === 'home' ? homeColor : awayColor;
   433→        points.push({
   434→          id: `tp_dod_${evt.id}`,
   435→          type: 'do_or_die_shift',
   436→          label: 'Do-or-Die Shift',
   437→          icon: '⚡',
   438→          description: `${team.name} seized the lead in a Do-or-Die raid! ${homeScore}-${awayScore}`,
   439→          teamName: team.name,
   440→          teamColor: color,
   441→          timestamp: evt.timestamp,
   442→          half: evt.half,
   443→        });
   444→      }
   445→
   446→      // Comeback: team was losing by 3+ and takes lead
   447→      if (currentLeader !== 'draw' && currentLeader !== prevLeader) {
   448→        const prevGap = Math.abs(homeScore - awayScore);
   449→        if (prevGap >= 3) {
   450→          const team = currentLeader === 'home' ? match.homeTeam : match.awayTeam;
   451→          const color = currentLeader === 'home' ? homeColor : awayColor;
   452→          points.push({
   453→            id: `tp_cb_${evt.id}`,
   454→            type: 'comeback',
   455→            label: 'Comeback Moment',
   456→            icon: '🔄',
   457→            description: `${team.name} fights back from ${prevGap} points down to take the lead!`,
   458→            teamName: team.name,
   459→            teamColor: color,
   460→            timestamp: evt.timestamp,
   461→            half: evt.half,
   462→          });
   463→        }
   464→      }
   465→
   466→      prevLeader = currentLeader;
   467→    }
   468→
   469→    return points;
   470→  }, [match, sortedEvents, homeColor, awayColor]);
   471→
   472→  // ── Share data ─────────────────────────────────────────────────────────────
   473→  const buildShareData = useCallback(() => {
   474→    if (!match) return null;
   475→
   476→    // Top raider/defender from events
   477→    const raidMap: Record<string, { name: string; points: number }> = {};
   478→    const tackleMap: Record<string, { name: string; points: number }> = {};
   479→    for (const evt of match.events) {
   480→      if (!evt.playerId) continue;
   481→      const name = scorerNames[evt.playerId] || evt.playerId.slice(0, 6);
   482→      if (['raid_point', 'super_raid', 'do_or_die_raid'].includes(evt.eventType)) {
   483→        if (!raidMap[evt.playerId]) raidMap[evt.playerId] = { name, points: 0 };
   484→        raidMap[evt.playerId].points += evt.value;
   485→      }
   486→      if (['tackle_point', 'super_tackle'].includes(evt.eventType)) {
   487→        if (!tackleMap[evt.playerId]) tackleMap[evt.playerId] = { name, points: 0 };
   488→        tackleMap[evt.playerId].points += evt.value;
   489→      }
   490→    }
   491→
   492→    const topRaiders = Object.values(raidMap).sort((a, b) => b.points - a.points);
   493→    const topDefenders = Object.values(tackleMap).sort((a, b) => b.points - a.points);
   494→
   495→    const motm = match.motmUser
   496→      ? { name: match.motmUser.name, points: mvpBreakdown?.totalPoints || 0 }
   497→      : null;
   498→
   499→    return {
   500→      homeTeam: match.homeTeam.name,
   501→      awayTeam: match.awayTeam.name,
   502→      homeScore: match.homeScore,
   503→      awayScore: match.awayScore,
   504→      homeTeamColor: homeColor,
   505→      awayTeamColor: awayColor,
   506→      tournament: match.tournament?.name || null,
   507→      date: match.startedAt,
   508→      venue: match.venue || null,
   509→      gender: match.gender || null,
   510→      topRaider: topRaiders[0] || null,
   511→      topDefender: topDefenders[0] || null,
   512→      motm,
   513→    };
   514→  }, [match, homeColor, awayColor, scorerNames, mvpBreakdown]);
   515→
   516→  // ── Render ─────────────────────────────────────────────────────────────────
   517→  return (
   518→    <AnimatePresence>
   519→      <motion.div
   520→        initial={{ opacity: 0 }}
   521→        animate={{ opacity: 1 }}
   522→        exit={{ opacity: 0 }}
   523→        className="fixed inset-0 z-50 bg-brand-navy-dark flex flex-col"
   524→      >
   525→        {/* ── Header bar ── */}
   526→        <header className="sticky top-0 z-10 bg-brand-navy-dark/90 backdrop-blur-md border-b border-white/10">
   527→          <div className="px-4 py-3 flex items-center justify-between">
   528→            <div className="flex items-center gap-2">
   529→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-brand-gold to-brand-gold-dark flex items-center justify-center">
   530→                <Star className="w-4 h-4 text-white" />
   531→              </div>
   532→              <h1 className="text-sm font-black tracking-wider text-white">MATCH HIGHLIGHTS</h1>
   533→            </div>
   534→            <button
   535→              onClick={onClose}
   536→              className="w-8 h-8 rounded-full bg-white/10 flex items-center justify-center text-white/70 hover:bg-white/20 hover:text-white transition-colors"
   537→            >
   538→              <X className="w-4 h-4" />
   539→            </button>
   540→          </div>
   541→        </header>
   542→
   543→        {/* ── Scrollable content ── */}
   544→        <div className="flex-1 overflow-y-auto custom-scrollbar">
   545→          {loading ? (
   546→            <div className="px-4 py-8 space-y-4">
   547→              <div className="h-56 rounded-2xl bg-white/5 animate-pulse" />
   548→              <div className="h-32 rounded-2xl bg-white/5 animate-pulse" />
   549→              <div className="h-48 rounded-2xl bg-white/5 animate-pulse" />
   550→              <div className="h-64 rounded-2xl bg-white/5 animate-pulse" />
   551→            </div>
   552→          ) : error || !match ? (
   553→            <div className="flex flex-col items-center justify-center py-20 px-6">
   554→              <Shield className="w-12 h-12 text-white/20 mb-3" />
   555→              <p className="text-white/50 text-sm font-medium">{error || 'Match not found'}</p>
   556→              <Button onClick={fetchMatch} variant="outline" className="mt-4 rounded-xl border-white/20 text-white hover:bg-white/10">
   557→                Retry
