    32→

    68→  profile?: {
    69→    position?: string | null;
    70→    overallRating?: number;
    71→  } | null;

    73→

    75→  totalMatches: number;
    76→  wins: number;

    87→const TEAM_COLORS = [
    88→  '#DC2626', '#1E293B', '#14B8A6', '#475569', '#9333EA',

    91→
    92→// ─── Helpers ──────────────────────────────────────────────────────
    93→
    94→function getInitials(name: string | null | undefined): string {
    95→  if (!name) return '?';
    96→  const parts = name.trim().split(/\s+/);
    97→  if (parts.length === 1) return parts[0].charAt(0).toUpperCase();
    98→  return (parts[0].charAt(0) + parts[parts.length - 1].charAt(0)).toUpperCase();
    99→}
   100→
   101→function getDisplayName(name: string | null | undefined): string {
   102→  return name?.trim() || 'Unnamed Player';
   103→}
   104→
   105→// ─── Component ────────────────────────────────────────────────────
   106→
   107→export default function TeamManagementScreen({ onClose }: TeamManagementScreenProps) {
   108→  const currentUser = useKabaddiStore((s) => s.currentUser);
   109→  const { toast } = useToast();
   110→
   111→  // View state
   112→  const [view, setView] = useState<'list' | 'detail'>('list');
   113→  const [teams, setTeams] = useState<TeamData[]>([]);
   114→  const [selectedTeam, setSelectedTeam] = useState<TeamData | null>(null);
   115→  const [teamStats, setTeamStats] = useState<TeamStats>({
   116→    totalMatches: 0,
   117→    wins: 0,
   118→    losses: 0,
   119→    totalPoints: 0,
   120→  });
   121→
   122→  // Loading states
   123→  const [teamsLoading, setTeamsLoading] = useState(true);
   124→  const [detailLoading, setDetailLoading] = useState(false);
   125→  const [actionLoading, setActionLoading] = useState(false);
   126→
   127→  // Create Team Dialog
   128→  const [createDialogOpen, setCreateDialogOpen] = useState(false);
   129→  const [newTeamName, setNewTeamName] = useState('');
   130→  const [newTeamShortName, setNewTeamShortName] = useState('');
   131→  const [newTeamColor, setNewTeamColor] = useState(TEAM_COLORS[0]);
   132→
   133→  // Search Players Dialog
   134→  const [searchDialogOpen, setSearchDialogOpen] = useState(false);
   135→  const [searchQuery, setSearchQuery] = useState('');
   136→  const [searchResults, setSearchResults] = useState<PlayerSearchResult[]>([]);
   137→  const [searchLoading, setSearchLoading] = useState(false);
   138→
   139→  // Delete confirmation
   140→  const [deleteConfirmId, setDeleteConfirmId] = useState<string | null>(null);
   141→
   142→  // ─── Fetch teams for current user ─────────────────────────────
   143→
   144→  const fetchTeams = useCallback(async () => {
   145→    if (!currentUser) return;
   146→    setTeamsLoading(true);
   147→    try {
   148→      const res = await fetch('/api/teams');
   149→      if (!res.ok) throw new Error('Failed to fetch teams');
   150→      const data = await res.json();
   151→      // Filter teams where the current user is a member
   152→      const userTeams: TeamData[] = (data.teams || []).filter((team: TeamData) =>
   153→        team.members.some((m: TeamMemberEntry) => m.userId === currentUser.id)
   154→      );
   155→      setTeams(userTeams);
   156→    } catch (err) {
   157→      console.error('Fetch teams error:', err);
   158→      toast({
   159→        title: 'Error',
   160→        description: 'Failed to load teams',
   161→        variant: 'destructive',
   162→      });
   163→    } finally {
   164→      setTeamsLoading(false);
   165→    }
   166→  }, [currentUser, toast]);
   167→
   168→  useEffect(() => {
   169→    fetchTeams();
   170→  }, [fetchTeams]);
   171→
   172→  // ─── Fetch team detail ────────────────────────────────────────
   173→
   174→  const fetchTeamDetail = useCallback(async (teamId: string) => {
   175→    setDetailLoading(true);
   176→    try {
   177→      const res = await fetch(`/api/teams/${teamId}`);
   178→      if (!res.ok) throw new Error('Failed to fetch team detail');
   179→      const data = await res.json();
   180→      setSelectedTeam(data.team);
   181→
   182→      // Compute stats from matches (we'll approximate from what's available)
   183→      // Since there's no dedicated stats endpoint per team, we set defaults
   184→      // A real implementation would call a stats endpoint
   185→      setTeamStats({
   186→        totalMatches: 0,
   187→        wins: 0,
   188→        losses: 0,
   189→        totalPoints: 0,
   190→      });
   191→    } catch (err) {
   192→      console.error('Fetch team detail error:', err);
   193→      toast({
   194→        title: 'Error',
   195→        description: 'Failed to load team details',
   196→        variant: 'destructive',
   197→      });
   198→    } finally {
   199→      setDetailLoading(false);
   200→    }
   201→  }, [toast]);
   202→
   203→  // ─── Search players ──────────────────────────────────────────
   204→
   205→  const searchPlayers = useCallback(async (query: string) => {
   206→    if (!query.trim()) {
   207→      setSearchResults([]);
   208→      return;
   209→    }
   210→    setSearchLoading(true);
   211→    try {
   212→      const res = await fetch(`/api/players?search=${encodeURIComponent(query)}`);
   213→      if (!res.ok) throw new Error('Search failed');
   214→      const data = await res.json();
   215→      setSearchResults(data.players || []);
   216→    } catch (err) {
   217→      console.error('Search players error:', err);
   218→      setSearchResults([]);
   219→    } finally {
   220→      setSearchLoading(false);
   221→    }
   222→  }, []);
   223→
   224→  // Debounced search
   225→  useEffect(() => {
   226→    if (!searchDialogOpen) return;
   227→    const timer = setTimeout(() => {
   228→      searchPlayers(searchQuery);
   229→    }, 300);
   230→    return () => clearTimeout(timer);
   231→  }, [searchQuery, searchDialogOpen, searchPlayers]);
   232→
   233→  // ─── Create team ─────────────────────────────────────────────
   234→
   235→  const handleCreateTeam = async () => {
   236→    if (!currentUser || !newTeamName.trim()) return;
   237→    setActionLoading(true);
   238→    try {
   239→      const res = await fetch('/api/teams', {
   240→        method: 'POST',
   241→        headers: { 'Content-Type': 'application/json' },
   242→        body: JSON.stringify({
   243→          name: newTeamName.trim(),
   244→          shortName: newTeamShortName.trim().slice(0, 3).toUpperCase() || undefined,
   245→          color: newTeamColor,
   246→          memberIds: [currentUser.id],
   247→          captainId: currentUser.id,
   248→        }),
   249→      });
   250→      if (!res.ok) throw new Error('Failed to create team');
   251→      toast({
   252→        title: 'Team Created!',
   253→        description: `${newTeamName.trim()} has been created successfully.`,
   254→      });
   255→      setCreateDialogOpen(false);
   256→      setNewTeamName('');
   257→      setNewTeamShortName('');
   258→      setNewTeamColor(TEAM_COLORS[0]);
   259→      fetchTeams();
   260→    } catch (err) {
   261→      console.error('Create team error:', err);
   262→      toast({
   263→        title: 'Error',
   264→        description: 'Failed to create team',
   265→        variant: 'destructive',
   266→      });
   267→    } finally {
   268→      setActionLoading(false);
   269→    }
   270→  };
   271→
   272→  // ─── Add player to team ──────────────────────────────────────
   273→
   274→  const handleAddPlayer = async (playerId: string, playerName: string) => {
   275→    if (!selectedTeam) return;
   276→    setActionLoading(true);
   277→    try {
   278→      // Add member via PATCH — we need to create a TeamMember
   279→      // The API doesn't have a direct add-member endpoint,
   280→      // so we'll use the teams API to create a new member entry
   281→      const res = await fetch(`/api/teams/${selectedTeam.id}`, {
   282→        method: 'PATCH',
   283→        headers: { 'Content-Type': 'application/json' },
   284→        body: JSON.stringify({
   285→          addMemberId: playerId,
   286→        }),
   287→      });
   288→      if (!res.ok) throw new Error('Failed to add player');
   289→      toast({
   290→        title: 'Player Added',
   291→        description: `${playerName} has been added to the team.`,
   292→      });
   293→      setSearchDialogOpen(false);
   294→      setSearchQuery('');
   295→      setSearchResults([]);
   296→      fetchTeamDetail(selectedTeam.id);
   297→    } catch (err) {
   298→      console.error('Add player error:', err);
   299→      toast({
   300→        title: 'Error',
   301→        description: 'Failed to add player to team',
   302→        variant: 'destructive',
   303→      });
   304→    } finally {
   305→      setActionLoading(false);
   306→    }
   307→  };
   308→
   309→  // ─── Remove player from team ─────────────────────────────────
   310→
   311→  const handleRemovePlayer = async (memberId: string, playerName: string) => {
   312→    if (!selectedTeam) return;
   313→    setActionLoading(true);
   314→    try {
   315→      const res = await fetch(`/api/teams/${selectedTeam.id}`, {
   316→        method: 'PATCH',
   317→        headers: { 'Content-Type': 'application/json' },
   318→        body: JSON.stringify({
   319→          removeMemberId: memberId,
   320→        }),
   321→      });
   322→      if (!res.ok) throw new Error('Failed to remove player');
   323→      toast({
   324→        title: 'Player Removed',
   325→        description: `${playerName} has been removed from the team.`,
   326→      });
   327→      setDeleteConfirmId(null);
   328→      fetchTeamDetail(selectedTeam.id);
   329→    } catch (err) {
   330→      console.error('Remove player error:', err);
   331→      toast({
   332→        title: 'Error',
   333→        description: 'Failed to remove player',
   334→        variant: 'destructive',
   335→      });
   336→    } finally {
   337→      setActionLoading(false);
   338→    }
   339→  };
   340→
   341→  // ─── Set captain ─────────────────────────────────────────────
   342→
   343→  const handleSetCaptain = async (userId: string, playerName: string) => {
   344→    if (!selectedTeam) return;
   345→    setActionLoading(true);
   346→    try {
   347→      const res = await fetch(`/api/teams/${selectedTeam.id}`, {
   348→        method: 'PATCH',
   349→        headers: { 'Content-Type': 'application/json' },
   350→        body: JSON.stringify({ captainId: userId }),
   351→      });
   352→      if (!res.ok) throw new Error('Failed to set captain');
   353→      toast({
   354→        title: 'Captain Updated',
   355→        description: `${playerName} is now the captain.`,
   356→      });
   357→      fetchTeamDetail(selectedTeam.id);
   358→    } catch (err) {
   359→      console.error('Set captain error:', err);
   360→      toast({
   361→        title: 'Error',
   362→        description: 'Failed to set captain',
   363→        variant: 'destructive',
   364→      });
   365→    } finally {
   366→      setActionLoading(false);
   367→    }
   368→  };
   369→
   370→  // ─── Delete team ─────────────────────────────────────────────
   371→
   372→  const handleDeleteTeam = async () => {
   373→    if (!selectedTeam) return;
   374→    setActionLoading(true);
   375→    try {
   376→      const res = await fetch(`/api/teams/${selectedTeam.id}`, { method: 'DELETE' });
   377→      if (!res.ok) throw new Error('Failed to delete team');
   378→      toast({
   379→        title: 'Team Deleted',
   380→        description: `${selectedTeam.name} has been deleted.`,
   381→      });
   382→      setSelectedTeam(null);
   383→      setView('list');
   384→      fetchTeams();
   385→    } catch (err) {
   386→      console.error('Delete team error:', err);
   387→      toast({
   388→        title: 'Error',
   389→        description: 'Failed to delete team',
   390→        variant: 'destructive',
   391→      });
   392→    } finally {
   393→      setActionLoading(false);
   394→    }
   395→  };
   396→
   397→  // ─── Navigate to team detail ─────────────────────────────────
   398→
   399→  const openTeamDetail = (team: TeamData) => {
   400→    setSelectedTeam(team);
   401→    setView('detail');
   402→    fetchTeamDetail(team.id);
   403→  };
   404→
   405→  const goBackToList = () => {
   406→    setView('list');
   407→    setSelectedTeam(null);
   408→  };
   409→
   410→  // ─── Get captain name for a team ─────────────────────────────
   411→
   412→  const getCaptainName = (team: TeamData): string => {
   413→    const captain = team.members.find((m) => m.isCaptain);
   414→    return captain ? getDisplayName(captain.user.name) : 'No captain';
   415→  };
   416→
   417→  // ─── Check if player is already in team ──────────────────────
   418→
   419→  const isPlayerInTeam = (playerId: string): boolean => {
   420→    if (!selectedTeam) return false;
   421→    return selectedTeam.members.some((m) => m.userId === playerId);
   422→  };
   423→
   424→  // ─── Render ──────────────────────────────────────────────────
   425→
   426→  return (
   427→    <motion.div
   428→      initial={{ opacity: 0 }}
   429→      animate={{ opacity: 1 }}
   430→      exit={{ opacity: 0 }}
   431→      className="fixed inset-0 z-50 bg-warm-50 flex flex-col"
   432→    >
   433→      {/* ═══ Header ═══ */}
   434→      <header className="sticky top-0 z-10 bg-warm-50/90 backdrop-blur-md border-b border-warm-200/60">
   435→        <div className="px-4 py-3 flex items-center justify-between">
   436→          <div className="flex items-center gap-2">
   437→            {view === 'detail' ? (
   438→              <button
   439→                onClick={goBackToList}
   440→                className="w-8 h-8 rounded-full bg-warm-200 flex items-center justify-center text-warm-600 hover:bg-warm-300 transition-colors"
   441→              >
   442→                <ChevronLeft className="w-4 h-4" />
   443→              </button>
   444→            ) : (
   445→              <div className="w-8 h-8 rounded-lg bg-gradient-to-br from-brand-red to-brand-red-dark flex items-center justify-center">
   446→                <Shield className="w-4 h-4 text-white" />
   447→              </div>
   448→            )}
   449→            <h1 className="text-base font-black tracking-wider text-warm-800">
   450→              {view === 'detail' && selectedTeam
   451→                ? selectedTeam.name.toUpperCase()
   452→                : 'MY TEAMS'}
   453→            </h1>
   454→          </div>
   455→          <div className="flex items-center gap-2">
   456→            {view === 'list' && (
   457→              <button
   458→                onClick={() => setCreateDialogOpen(true)}
   459→                className="w-8 h-8 rounded-full bg-brand-red flex items-center justify-center text-white hover:bg-brand-red-dark transition-colors"
   460→              >
   461→                <Plus className="w-4 h-4" />
   462→              </button>
   463→            )}
   464→            {view === 'detail' && selectedTeam && (
   465→              <button
   466→                onClick={handleDeleteTeam}
   467→                disabled={actionLoading}
   468→                className="w-8 h-8 rounded-full bg-warm-200 flex items-center justify-center text-warm-500 hover:bg-red-100 hover:text-brand-red transition-colors"
   469→              >
   470→                <Trash2 className="w-4 h-4" />
   471→              </button>
   472→            )}
   473→            <button
   474→              onClick={onClose}
   475→              className="w-8 h-8 rounded-full bg-warm-200 flex items-center justify-center text-warm-600 hover:bg-warm-300 transition-colors"
   476→            >
   477→              <X className="w-4 h-4" />
   478→            </button>
   479→          </div>
   480→        </div>
   481→      </header>
   482→
   483→      {/* ═══ Content ═══ */}
   484→      <div className="flex-1 overflow-y-auto custom-scrollbar">
   485→        <AnimatePresence mode="wait">
   486→          {view === 'list' ? (
   487→            <motion.div
   488→              key="team-list"
   489→              initial={{ opacity: 0, x: -20 }}
   490→              animate={{ opacity: 1, x: 0 }}
   491→              exit={{ opacity: 0, x: -20 }}
   492→              transition={{ duration: 0.2 }}
   493→              className="px-4 py-4 space-y-3"
   494→            >
   495→              {/* ─── Team List View ─── */}
   496→              {teamsLoading ? (
   497→                <div className="flex flex-col gap-3">
   498→                  {[1, 2, 3].map((i) => (
   499→                    <div
   500→                      key={i}
   501→                      className="h-24 rounded-2xl bg-warm-100 animate-pulse"
   502→                    />
   503→                  ))}
   504→                </div>
   505→              ) : teams.length === 0 ? (
   506→                <div className="flex flex-col items-center justify-center py-16">
   507→                  <div className="w-16 h-16 rounded-full bg-warm-200 flex items-center justify-center mb-4">
   508→                    <Users className="w-8 h-8 text-warm-400" />
   509→                  </div>
   510→                  <p className="text-warm-700 font-bold text-lg">No Teams Yet</p>
   511→                  <p className="text-warm-400 text-sm mt-1 text-center max-w-[240px]">
   512→                    Create your first team and start playing Kabaddi!
   513→                  </p>
   514→                  <Button
   515→                    onClick={() => setCreateDialogOpen(true)}
   516→                    className="mt-4 bg-brand-red hover:bg-brand-red-dark text-white rounded-xl font-semibold"
   517→                  >
   518→                    <Plus className="w-4 h-4 mr-1.5" />
   519→                    Create Team
   520→                  </Button>
   521→                </div>
   522→              ) : (
   523→                teams.map((team, index) => (
   524→                  <motion.div
   525→                    key={team.id}
   526→                    initial={{ opacity: 0, y: 12 }}
   527→                    animate={{ opacity: 1, y: 0 }}
   528→                    transition={{ delay: index * 0.06 }}
   529→                  >
   530→                    <Card
   531→                      className="bg-white border-warm-200 py-0 gap-0 overflow-hidden cursor-pointer hover:shadow-md transition-shadow active:scale-[0.98]"
   532→                      onClick={() => openTeamDetail(team)}
   533→                    >
   534→                      <CardContent className="p-4">
   535→                        <div className="flex items-center gap-3">
   536→                          {/* Color badge */}
   537→                          <div
   538→                            className="w-12 h-12 rounded-xl flex items-center justify-center text-white font-black text-lg shrink-0 shadow-sm"
   539→                            style={{ backgroundColor: team.color || '#DC2626' }}
   540→                          >
   541→                            {team.shortName
   542→                              ? team.shortName.slice(0, 2)
   543→                              : team.name.charAt(0).toUpperCase()}
   544→                          </div>
   545→
   546→                          {/* Team info */}
   547→                          <div className="flex-1 min-w-0">
   548→                            <div className="flex items-center gap-2">
   549→                              <h3 className="font-bold text-warm-800 truncate">
   550→                                {team.name}
   551→                              </h3>
   552→                              {team.shortName && (
   553→                                <Badge
   554→                                  variant="secondary"
   555→                                  className="text-[10px] px-1.5 py-0 bg-warm-100 text-warm-500 shrink-0"
   556→                                >
   557→                                  {team.shortName}
   558→                                </Badge>
   559→                              )}
   560→                            </div>
   561→                            <div className="flex items-center gap-3 mt-1">
   562→                              <span className="text-xs text-warm-500 flex items-center gap-1">
   563→                                <Users className="w-3 h-3" />
   564→                                {team.members.length}
   565→                              </span>
   566→                              <span className="text-xs text-warm-500 flex items-center gap-1">
   567→                                <Crown className="w-3 h-3 text-brand-gold" />
   568→                                {getCaptainName(team)}
   569→                              </span>
   570→                            </div>
   571→                          </div>
   572→
   573→                          {/* Arrow indicator */}
   574→                          <ChevronLeft className="w-4 h-4 text-warm-400 rotate-180 shrink-0" />
   575→                        </div>
   576→                      </CardContent>
   577→                    </Card>
   578→                  </motion.div>
   579→                ))
   580→              )}
   581→            </motion.div>
   582→          ) : (
   583→            <motion.div
   584→              key="team-detail"
   585→              initial={{ opacity: 0, x: 20 }}
   586→              animate={{ opacity: 1, x: 0 }}
   587→              exit={{ opacity: 0, x: 20 }}
   588→              transition={{ duration: 0.2 }}
   589→              className="px-4 py-4 space-y-4"
   590→            >
   591→              {/* ─── Team Detail View ─── */}
   592→              {detailLoading ? (
   593→                <div className="flex flex-col gap-3">
   594→                  <div className="h-32 rounded-2xl bg-warm-100 animate-pulse" />
   595→                  <div className="h-20 rounded-2xl bg-warm-100 animate-pulse" />
   596→                  {[1, 2, 3].map((i) => (
   597→                    <div
   598→                      key={i}
