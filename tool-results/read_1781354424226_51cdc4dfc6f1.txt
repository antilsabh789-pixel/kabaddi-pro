   253→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   254→  const [count, setCount] = useState(0);
   255→  const prevTarget = useRef(target);
   256→
   257→  useEffect(() => {
   258→    if (prevTarget.current === target && count !== 0) return;
   259→    prevTarget.current = target;
   260→
   261→    let start = 0;
   262→    const startTime = performance.now();
   263→
   264→    function step(currentTime: number) {
   265→      const elapsed = currentTime - startTime;
   266→      const progress = Math.min(elapsed / duration, 1);
   267→      // ease-out cubic
   268→      const eased = 1 - Math.pow(1 - progress, 3);
   269→      const current = Math.round(eased * target);
   270→      setCount(current);
   271→      if (progress < 1) {
   272→        requestAnimationFrame(step);
   273→      }
   274→    }
   275→
   276→    requestAnimationFrame(step);
   277→  }, [target, duration]);
   278→
   279→  return <span>{count}</span>;
   280→}

   284→function NumberTicker({ value }: { value: number }) {
   285→  const prevValue = useRef(value);
   286→  const [animating, setAnimating] = useState(false);
   287→  const [flash, setFlash] = useState(false);
   288→
   289→  useEffect(() => {
   290→    if (prevValue.current !== value) {
   291→      prevValue.current = value;
   292→      const t1 = setTimeout(() => setAnimating(true), 0);
   293→      const t2 = setTimeout(() => setAnimating(false), 400);
   294→      const t3 = setTimeout(() => setFlash(true), 0);
   295→      const t4 = setTimeout(() => setFlash(false), 600);
   296→      return () => { clearTimeout(t1); clearTimeout(t2); clearTimeout(t3); clearTimeout(t4); };
   297→    }
   298→  }, [value]);
   299→
   300→  return (
   301→    <span className={`${animating ? 'number-ticker' : ''} ${flash ? 'score-change-flash' : ''}`} key={`${value}-${animating}`}>
   302→      {value}
   303→    </span>
   304→  );
   305→}

   311→  const prevTrigger = useRef(trigger);

   317→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   318→        id: Date.now() + i,
   319→        x: 30 + Math.random() * 40,
   320→        color: colors[Math.floor(Math.random() * colors.length)],
   321→        delay: i * 0.05,
   322→      }));
   323→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   324→      const t2 = setTimeout(() => setParticles([]), 900);
   325→      return () => { clearTimeout(t1); clearTimeout(t2); };
   326→    }
   327→  }, [trigger]);
   328→
   329→  return (
   330→    <>
   331→      {particles.map((p) => (
   332→        <div
   333→          key={p.id}
   334→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"

   349→function AwardsConfettiSection() {
   350→  const ref = useRef<HTMLDivElement>(null);
   351→  const [shown, setShown] = useState(false);
   352→
   353→  useEffect(() => {
   354→    const el = ref.current;
   355→    if (!el) return;
   356→    const observer = new IntersectionObserver(
   357→      ([entry]) => {
   358→        if (entry.isIntersecting && !shown) {
   359→          setShown(true);
   360→        }
   361→      },
   362→      { threshold: 0.3 }
   363→    );
   364→    observer.observe(el);
   365→    return () => observer.disconnect();
   366→  }, [shown]);
   367→
   368→  const confettiColors = ['#DC2626', '#F59E0B', '#14B8A6', '#FBBF24', '#1E293B'];
   369→
   370→  return (
   371→    <div ref={ref} className="absolute inset-0 overflow-hidden pointer-events-none z-0">
   372→      {shown && [...Array(8)].map((_, i) => (
   373→        <motion.div
   374→          key={`award-confetti-${i}`}
   375→          className="absolute w-1.5 h-1.5 rounded-full"
   376→          style={{
   377→            left: `${10 + i * 12}%`,
   378→            top: '40%',
   379→            backgroundColor: confettiColors[i % confettiColors.length],
   380→          }}
   381→          initial={{ y: 0, opacity: 0, scale: 0 }}
   382→          animate={{ y: [-10, -40, -80], opacity: [0, 1, 0], scale: [0, 1.2, 0.5] }}

   384→        />
   385→      ))}

   440→// ─── Countdown Timer ────────────────────────────────────────────────

   442→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   443→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   444→  const [expired, setExpired] = useState(!targetDate);
   445→
   446→  useEffect(() => {
   447→    if (!targetDate) {
   448→      return;
   449→    }
   450→
   451→    function calculate() {
   452→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   453→      if (diff === 0) {
   454→        setExpired(true);
   455→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   456→        return true;
   457→      }
   458→      setExpired(false);
   459→      setTimeLeft({
   460→        d: Math.floor(diff / 86400000),
   461→        h: Math.floor((diff % 86400000) / 3600000),
   462→        m: Math.floor((diff % 3600000) / 60000),
   463→        s: Math.floor((diff % 60000) / 1000),
   464→      });
   465→      return false;
   466→    }
   467→
   468→    const done = calculate();
   469→    if (done) return;
   470→
   471→    const id = setInterval(() => {
   472→      const finished = calculate();
   473→      if (finished) clearInterval(id);
   474→    }, 1000);
   475→
   476→    return () => clearInterval(id);
   477→  }, [targetDate]);
   478→
   479→  if (expired) {
   480→    return (
   481→      <motion.div
   482→        className="flex items-center justify-center gap-1.5 py-1"
   483→        animate={{ opacity: [1, 0.5, 1] }}
   484→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   485→      >
   486→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   487→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   488→      </motion.div>
   489→    );
   490→  }
   491→
   492→  if (!timeLeft) return null;
   493→
   494→  const units = [
   495→    { value: timeLeft.d, label: 'D' },
   496→    { value: timeLeft.h, label: 'H' },
   497→    { value: timeLeft.m, label: 'M' },
   498→    { value: timeLeft.s, label: 'S' },
   499→  ];
   500→
   501→  return (
   502→    <div className="flex items-center justify-center gap-1.5">
   503→      {units.map((unit, idx) => (
   504→        <div key={unit.label} className="flex items-center gap-1.5">
   505→          <div className="flex flex-col items-center">
   506→            <motion.div
   507→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   508→              key={`${unit.label}-${unit.value}`}
   509→              initial={{ rotateX: -90, opacity: 0 }}
   510→              animate={{ rotateX: 0, opacity: 1 }}
   511→              transition={{ duration: 0.3, ease: 'easeOut' }}
   512→            >
   513→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums countdown-flip">
   514→                {String(unit.value).padStart(2, '0')}
   515→              </span>
   516→            </motion.div>
   517→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   518→              {unit.label}
   519→            </span>
   520→          </div>
   521→          {idx < units.length - 1 && (
   522→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   523→          )}
   524→        </div>
   525→      ))}
   526→    </div>
   527→  );
   528→}
