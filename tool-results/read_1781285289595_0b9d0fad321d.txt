   243→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   244→  const [count, setCount] = useState(0);
   245→  const prevTarget = useRef(target);
   246→
   247→  useEffect(() => {
   248→    if (prevTarget.current === target && count !== 0) return;
   249→    prevTarget.current = target;
   250→
   251→    let start = 0;
   252→    const startTime = performance.now();
   253→
   254→    function step(currentTime: number) {
   255→      const elapsed = currentTime - startTime;
   256→      const progress = Math.min(elapsed / duration, 1);
   257→      // ease-out cubic
   258→      const eased = 1 - Math.pow(1 - progress, 3);
   259→      const current = Math.round(eased * target);
   260→      setCount(current);
   261→      if (progress < 1) {
   262→        requestAnimationFrame(step);
   263→      }
   264→    }
   265→
   266→    requestAnimationFrame(step);
   267→  }, [target, duration]);
   268→
   269→  return <span>{count}</span>;
   270→}

   274→function NumberTicker({ value }: { value: number }) {
   275→  const prevValue = useRef(value);
   276→  const [animating, setAnimating] = useState(false);
   277→
   278→  useEffect(() => {
   279→    if (prevValue.current !== value) {
   280→      prevValue.current = value;
   281→      const t1 = setTimeout(() => setAnimating(true), 0);
   282→      const t2 = setTimeout(() => setAnimating(false), 400);
   283→      return () => { clearTimeout(t1); clearTimeout(t2); };
   284→    }
   285→  }, [value]);
   286→
   287→  return (
   288→    <span className={animating ? 'number-ticker' : ''} key={`${value}-${animating}`}>
   289→      {value}
   290→    </span>
   291→  );
   292→}

   296→function ConfettiParticles({ trigger }: { trigger: number }) {
   297→  const [particles, setParticles] = useState<Array<{ id: number; x: number; color: string; delay: number }>>([]);
   298→  const prevTrigger = useRef(trigger);
   299→
   300→  useEffect(() => {
   301→    if (prevTrigger.current !== trigger) {
   302→      prevTrigger.current = trigger;
   303→      const colors = ['#DC2626', '#F59E0B', '#14B8A6', '#1E293B', '#FBBF24'];
   304→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   305→        id: Date.now() + i,
   306→        x: 30 + Math.random() * 40,
   307→        color: colors[Math.floor(Math.random() * colors.length)],
   308→        delay: i * 0.05,
   309→      }));
   310→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   311→      const t2 = setTimeout(() => setParticles([]), 900);
   312→      return () => { clearTimeout(t1); clearTimeout(t2); };
   313→    }
   314→  }, [trigger]);
   315→
   316→  return (
   317→    <>
   318→      {particles.map((p) => (
   319→        <div
   320→          key={p.id}
   321→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"
   322→          style={{
   323→            left: `${p.x}%`,
   324→            top: '20%',
   325→            backgroundColor: p.color,
   326→            animationDelay: `${p.delay}s`,
   327→          }}
   328→        />
   329→      ))}
   330→    </>
   331→  );
   332→}

   384→// ─── Countdown Timer ────────────────────────────────────────────────

   386→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   387→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   388→  const [expired, setExpired] = useState(!targetDate);
   389→
   390→  useEffect(() => {
   391→    if (!targetDate) {
   392→      return;
   393→    }
   394→
   395→    function calculate() {
   396→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   397→      if (diff === 0) {
   398→        setExpired(true);
   399→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   400→        return true;
   401→      }
   402→      setExpired(false);
   403→      setTimeLeft({
   404→        d: Math.floor(diff / 86400000),
   405→        h: Math.floor((diff % 86400000) / 3600000),
   406→        m: Math.floor((diff % 3600000) / 60000),
   407→        s: Math.floor((diff % 60000) / 1000),
   408→      });
   409→      return false;
   410→    }
   411→
   412→    const done = calculate();
   413→    if (done) return;
   414→
   415→    const id = setInterval(() => {
   416→      const finished = calculate();
   417→      if (finished) clearInterval(id);
   418→    }, 1000);
   419→
   420→    return () => clearInterval(id);
   421→  }, [targetDate]);
   422→
   423→  if (expired) {
   424→    return (
   425→      <motion.div
   426→        className="flex items-center justify-center gap-1.5 py-1"
   427→        animate={{ opacity: [1, 0.5, 1] }}
   428→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   429→      >
   430→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   431→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   432→      </motion.div>
   433→    );
   434→  }
   435→
   436→  if (!timeLeft) return null;
   437→
   438→  const units = [
   439→    { value: timeLeft.d, label: 'D' },
   440→    { value: timeLeft.h, label: 'H' },
   441→    { value: timeLeft.m, label: 'M' },
   442→    { value: timeLeft.s, label: 'S' },
   443→  ];
   444→
   445→  return (
   446→    <div className="flex items-center justify-center gap-1.5">
   447→      {units.map((unit, idx) => (
   448→        <div key={unit.label} className="flex items-center gap-1.5">
   449→          <div className="flex flex-col items-center">
   450→            <motion.div
   451→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   452→              key={`${unit.label}-${unit.value}`}
   453→              initial={{ rotateX: -90, opacity: 0 }}
   454→              animate={{ rotateX: 0, opacity: 1 }}
   455→              transition={{ duration: 0.3, ease: 'easeOut' }}
   456→            >
   457→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums countdown-flip">
   458→                {String(unit.value).padStart(2, '0')}
   459→              </span>
   460→            </motion.div>
   461→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   462→              {unit.label}
   463→            </span>
   464→          </div>
   465→          {idx < units.length - 1 && (
   466→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   467→          )}
   468→        </div>
   469→      ))}
   470→    </div>
   471→  );
   472→}
