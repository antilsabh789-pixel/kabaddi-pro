    52→// ─── Position Helpers ────────────────────────────────────────────────
    53→
    54→type PositionType = 'raider' | 'defender' | 'all-rounder';
    55→
    56→function getPositionType(position: string | null): PositionType {
    57→  if (!position) return 'all-rounder';
    58→  const p = position.toLowerCase();
    59→  if (p.includes('raider')) return 'raider';
    60→  if (p.includes('corner') || p.includes('cover') || p.includes('defend')) return 'defender';
    61→  return 'all-rounder';
    62→}
    63→
    64→function getPositionColor(position: string | null) {
    65→  const type = getPositionType(position);
    66→  switch (type) {
    67→    case 'raider': return { ring: 'ring-red-500', bg: 'bg-red-500', text: 'text-red-500', gradient: 'from-red-500 to-orange-500', border: 'border-red-500/30' };
    68→    case 'defender': return { ring: 'ring-blue-500', bg: 'bg-blue-500', text: 'text-blue-500', gradient: 'from-blue-500 to-cyan-500', border: 'border-blue-500/30' };
    69→    case 'all-rounder': return { ring: 'ring-amber-500', bg: 'bg-amber-500', text: 'text-amber-500', gradient: 'from-amber-500 to-yellow-400', border: 'border-amber-500/30' };
    70→  }
    71→}

    98→// ─── Animated Counter ────────────────────────────────────────────────
    99→
   100→function AnimatedCounter({ value, duration = 1200 }: { value: number; duration?: number }) {
   101→  const [display, setDisplay] = useState(value);
   102→  const prevValueRef = useRef(value);
   103→
   104→  useEffect(() => {
   105→    const end = value;
   106→    if (end === 0) {
   107→      prevValueRef.current = 0;
   108→      return;
   109→    }
   110→    const start = prevValueRef.current;
   111→    if (start === end) return;
   112→    const increment = (end - start) / (duration / 16);
   113→    let current = start;
   114→    const timer = setInterval(() => {
   115→      current += increment;
   116→      if ((increment > 0 && current >= end) || (increment < 0 && current <= end)) {
   117→        setDisplay(end);
   118→        prevValueRef.current = end;
   119→        clearInterval(timer);
   120→      } else {
   121→        setDisplay(Math.floor(current));
   122→      }
   123→    }, 16);
   124→    return () => clearInterval(timer);
   125→  }, [value, duration]);
   126→
   127→  return <>{display}</>;
   128→}

   130→// ─── Circular Rating Indicator ───────────────────────────────────────
   131→
   132→function CircularRating({ rating, size = 48 }: { rating: number; size?: number }) {
   133→  const radius = (size - 8) / 2;
   134→  const circumference = 2 * Math.PI * radius;
   135→  const progress = Math.min(rating / 10, 1);
   136→  const offset = circumference - progress * circumference;
   137→
   138→  const getColor = (r: number) => {
   139→    if (r >= 8) return '#22c55e';
   140→    if (r >= 6) return '#eab308';
   141→    if (r >= 4) return '#f97316';
   142→    return '#ef4444';
   143→  };
   144→
   145→  return (
   146→    <div className="relative" style={{ width: size, height: size }}>
   147→      <svg width={size} height={size} className="-rotate-90">
   148→        <circle
   149→          cx={size / 2}
   150→          cy={size / 2}
   151→          r={radius}
   152→          stroke="rgba(255,255,255,0.15)"
   153→          strokeWidth="4"
   154→          fill="none"
   155→        />
   156→        <circle
   157→          cx={size / 2}
   158→          cy={size / 2}
   159→          r={radius}
   160→          stroke={getColor(rating)}
   161→          strokeWidth="4"
   162→          fill="none"
   163→          strokeLinecap="round"
   164→          strokeDasharray={circumference}
   165→          strokeDashoffset={offset}
   166→          className="transition-all duration-1000 ease-out"
   167→        />
   168→      </svg>
   169→      <div className="absolute inset-0 flex items-center justify-center">
   170→        <span className="text-xs font-black text-white">{rating.toFixed(1)}</span>
   171→      </div>
   172→    </div>
   173→  );
   174→}

   915→            {/* Share dropdown */}
   916→            <AnimatePresence>
   917→              {showShareMenu && (
   918→                <motion.div
   919→                  initial={{ opacity: 0, scale: 0.9, y: -5 }}
   920→                  animate={{ opacity: 1, scale: 1, y: 0 }}
   921→                  exit={{ opacity: 0, scale: 0.9, y: -5 }}
   922→                  transition={{ duration: 0.15 }}
   923→                  className="absolute right-0 top-full mt-1.5 bg-warm-100 dark:bg-warm-200 rounded-xl shadow-xl border border-warm-300 dark:border-warm-300 overflow-hidden z-20 min-w-[160px]"
   924→                >
   925→                  <button
   926→                    onClick={handleShareImage}
   927→                    className="w-full flex items-center gap-2.5 px-3.5 py-2.5 hover:bg-warm-200 dark:hover:bg-warm-300 transition-colors text-left"
   928→                  >
   929→                    <Download className="w-4 h-4 text-brand-red" />
   930→                    <span className="text-xs font-medium text-warm-800 dark:text-warm-700">Download Card</span>
   931→                  </button>
   932→                  <button
   933→                    onClick={handleCopyLink}
   934→                    className="w-full flex items-center gap-2.5 px-3.5 py-2.5 hover:bg-warm-200 dark:hover:bg-warm-300 transition-colors text-left"
   935→                  >
   936→                    <Link2 className="w-4 h-4 text-brand-teal" />
   937→                    <span className="text-xs font-medium text-warm-800 dark:text-warm-700">Copy Link</span>
   938→                  </button>
   939→                  {typeof navigator !== 'undefined' && navigator.share && (
   940→                    <button
   941→                      onClick={handleWebShare}
   942→                      className="w-full flex items-center gap-2.5 px-3.5 py-2.5 hover:bg-warm-200 dark:hover:bg-warm-300 transition-colors text-left"
   943→                    >
   944→                      <Share2 className="w-4 h-4 text-brand-gold" />
   945→                      <span className="text-xs font-medium text-warm-800 dark:text-warm-700">Share via...</span>
   946→                    </button>
   947→                  )}
   948→                </motion.div>
   949→              )}
   950→            </AnimatePresence>
   951→          </div>
   952→        </div>
   953→
   954→        {/* 3D Flip Container */}
   955→        <div
   956→          className="relative w-full cursor-pointer"
   957→          style={{ perspective: '1200px' }}
   958→          onClick={() => setFlipped(!flipped)}
   959→        >
   960→          <div
   961→            className="relative w-full transition-transform duration-700 ease-in-out"
   962→            style={{
   963→              transformStyle: 'preserve-3d',
   964→              transform: flipped ? 'rotateY(180deg)' : 'rotateY(0deg)',
   965→            }}
   966→          >
   967→            {/* Front */}
   968→            <div style={{ backfaceVisibility: 'hidden' }}>
   969→              {cardFront}
   970→            </div>
   971→            {/* Back */}
   972→            <div
   973→              className="absolute inset-0"
   974→              style={{ backfaceVisibility: 'hidden', transform: 'rotateY(180deg)' }}
   975→            >
   976→              {cardBack}
   977→            </div>
   978→          </div>
   979→        </div>
   980→      </div>
   981→
   982→      {/* Fullscreen overlay */}
   983→      {fullscreenView}
   984→    </>
   985→  );
   986→}
