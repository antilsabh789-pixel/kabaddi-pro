    60→// ─── Debounce hook ──────────────────────────────────────────────────
    61→
    62→function useDebounce<T>(value: T, delay: number): T {
    63→  const [debouncedValue, setDebouncedValue] = useState(value);
    64→
    65→  useEffect(() => {
    66→    const timer = setTimeout(() => setDebouncedValue(value), delay);
    67→    return () => clearTimeout(timer);
    68→  }, [value, delay]);
    69→
    70→  return debouncedValue;
    71→}

   181→    }

   313→                  <button

   315→                    onClick={() => setActiveFilter(opt.key)}
   316→                    className={`flex items-center gap-1.5 px-3.5 py-1.5 rounded-full text-xs font-semibold transition-all whitespace-nowrap shrink-0 ${
   317→                      isActive
   318→                        ? 'bg-brand-red text-white shadow-md shadow-brand-red/25'
   319→                        : 'bg-warm-100 dark:bg-warm-800 text-warm-600 dark:text-warm-300 hover:bg-warm-200 dark:hover:bg-warm-700'
   320→                    }`}
   321→                  >
   322→                    <Icon className="w-3.5 h-3.5" />
   323→                    {opt.label}

   326→              })}

   328→

   333→          {/* Search Content */}
   334→          <div className="flex-1 overflow-y-auto custom-scrollbar px-4 py-3">

   336→            {!hasQuery && (
   337→              <div className="space-y-4">

   348→                        className="text-[10px] font-semibold text-brand-red dark:text-brand-red-light hover:underline"

   350→                        Clear All

   352→                    </div>

   356→                          key={term}
   357→                          className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-warm-100 dark:bg-warm-800 border border-warm-200 dark:border-warm-700 group hover:border-brand-red/30 dark:hover:border-brand-red-light/30 transition-colors cursor-pointer"
   358→                          onClick={() => {
   359→                            setQuery(term);
   360→                            saveRecentSearch(term);
   361→                          }}
   362→                        >
   363→                          <span className="text-xs font-medium text-warm-700 dark:text-warm-200">{term}</span>
   364→                          <button
   365→                            onClick={(e) => {
   366→                              e.stopPropagation();
   367→                              removeRecentSearch(term);
   368→                            }}

   370→                            aria-label={`Remove ${term}`}
   371→                          >
   372→                            <X className="w-3 h-3 text-warm-400" />

   374→                        </div>
   375→                      ))}
   376→                    </div>
   377→                  </div>
   378→                )}
   379→
   380→                {/* Quick Suggestions */}

   394→                        className="flex items-center gap-1.5 px-3 py-1.5 rounded-full bg-brand-red/5 dark:bg-brand-red-light/10 border border-brand-red/15 dark:border-brand-red-light/20 hover:bg-brand-red/10 dark:hover:bg-brand-red-light/15 transition-colors"

   397→                        <span className="text-xs font-medium text-brand-red dark:text-brand-red-light">{suggestion}</span>
   398→                      </button>

   401→                </div>

   403→                {/* Empty illustration area */}

   405→                  <div className="w-16 h-16 rounded-2xl bg-warm-100 dark:bg-warm-800 flex items-center justify-center mb-4">
   406→                    <Search className="w-7 h-7 text-warm-300 dark:text-warm-600" />
   407→                  </div>
   408→                  <p className="text-sm font-semibold text-warm-600 dark:text-warm-300">
   409→                    Search Kabaddi Pro
   410→                  </p>
   411→                  <p className="text-xs text-warm-400 dark:text-warm-500 mt-1 max-w-[220px]">

   451→                            saveRecentSearch(suggestion);
   452→                          }}
