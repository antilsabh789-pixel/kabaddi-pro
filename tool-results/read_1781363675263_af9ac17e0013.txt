    12→function TossConfetti({ color, index }: { color: string; index: number }) {
    13→  const leftPos = (index * 23 + 7) % 100;
    14→  const rotateEnd = (index * 53 + 17) % 720 - 360;
    15→  const xDrift = ((index * 37) % 100) - 50;
    16→  const size = index % 3 === 0 ? 'w-3 h-3' : index % 3 === 1 ? 'w-2 h-2' : 'w-1.5 h-1.5';
    17→  const shape = index % 4 === 0 ? 'rounded-full' : index % 4 === 1 ? 'rounded-sm' : index % 4 === 2 ? 'rounded-l-full' : 'rounded-t-full';
    18→
    19→  return (
    20→    <motion.div
    21→      initial={{ y: -20, x: 0, rotate: 0, opacity: 1, scale: 1 }}
    22→      animate={{ y: '100vh', x: xDrift, rotate: rotateEnd, opacity: 0, scale: 0.5 }}
    23→      transition={{ duration: 2.5 + (index % 4) * 0.5, delay: index * 0.04, ease: 'easeIn' }}
    24→      className={`absolute top-0 ${size} ${shape}`}
    25→      style={{ left: `${leftPos}%`, backgroundColor: color }}
    26→    />
    27→  );
    28→}

   203→  const [tossWinner, setTossWinner] = useState<'home' | 'away' | null>(null);
   204→  const [coinSide, setCoinSide] = useState<'heads' | 'tails'>('heads');
   205→  const [flipProgress, setFlipProgress] = useState(0);
   206→  const [showConfetti, setShowConfetti] = useState(false);
   207→  const [countdown, setCountdown] = useState<number | null>(null);
   208→  const [chosenSide, setChosenSide] = useState<'heads' | 'tails' | null>(null);
   209→  const [callingTeam, setCallingTeam] = useState<'home' | 'away'>(() =>
   210→    Math.random() < 0.5 ? 'home' : 'away'
   211→  );
   212→
   213→  const callingTeamName = callingTeam === 'home' ? homeTeam : awayTeam;
   214→  const callingTeamColor = callingTeam === 'home' ? homeTeamColor : awayTeamColor;
   215→
   216→  const handleChooseSide = (side: 'heads' | 'tails') => {
   217→    setChosenSide(side);
   218→    setPhase('ready');
   219→  };
   220→
   221→  const doToss = useCallback(() => {
   222→    setPhase('flipping');
   223→    setFlipProgress(0);
   224→
   225→    // Determine coin result randomly
   226→    const side: 'heads' | 'tails' = Math.random() < 0.5 ? 'heads' : 'tails';
   227→
   228→    // The calling team wins if the coin result matches their choice
   229→    let result: 'home' | 'away';
   230→    if (side === chosenSide) {
   231→      result = callingTeam;
   232→    } else {
   233→      result = callingTeam === 'home' ? 'away' : 'home';
   234→    }
   235→
   236→    // Animate flipping with smooth progress
   237→    let progress = 0;
   238→    const interval = setInterval(() => {
   239→      progress += 1;
   240→      setFlipProgress(progress);
   241→
   242→      // Rapid coin side switching during flip
   243→      const flipCount = Math.floor(progress / 5);
   244→      setCoinSide(flipCount % 2 === 0 ? 'heads' : 'tails');
   245→
   246→      if (progress >= 100) {
   247→        clearInterval(interval);
   248→        setCoinSide(side);
   249→        setTossWinner(result);
   250→        setShowConfetti(true);
   251→        setTimeout(() => setPhase('result'), 400);

   289→            background: `radial-gradient(ellipse at 30% 50%, ${homeTeamColor}15, transparent 70%)`,
   290→          }}
