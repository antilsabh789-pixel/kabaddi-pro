   242→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   243→  const [count, setCount] = useState(0);
   244→  const prevTarget = useRef(target);
   245→
   246→  useEffect(() => {
   247→    if (prevTarget.current === target && count !== 0) return;
   248→    prevTarget.current = target;
   249→
   250→    let start = 0;
   251→    const startTime = performance.now();
   252→
   253→    function step(currentTime: number) {
   254→      const elapsed = currentTime - startTime;
   255→      const progress = Math.min(elapsed / duration, 1);
   256→      // ease-out cubic
   257→      const eased = 1 - Math.pow(1 - progress, 3);
   258→      const current = Math.round(eased * target);
   259→      setCount(current);
   260→      if (progress < 1) {
   261→        requestAnimationFrame(step);
   262→      }
   263→    }
   264→
   265→    requestAnimationFrame(step);
   266→  }, [target, duration]);
   267→
   268→  return <span>{count}</span>;
   269→}

   273→function NumberTicker({ value }: { value: number }) {
   274→  const prevValue = useRef(value);
   275→  const [animating, setAnimating] = useState(false);
   276→
   277→  useEffect(() => {
   278→    if (prevValue.current !== value) {
   279→      prevValue.current = value;
   280→      const t1 = setTimeout(() => setAnimating(true), 0);
   281→      const t2 = setTimeout(() => setAnimating(false), 400);
   282→      return () => { clearTimeout(t1); clearTimeout(t2); };
   283→    }
   284→  }, [value]);
   285→
   286→  return (
   287→    <span className={animating ? 'number-ticker' : ''} key={`${value}-${animating}`}>
   288→      {value}
   289→    </span>
   290→  );
   291→}

   295→function ConfettiParticles({ trigger }: { trigger: number }) {
   296→  const [particles, setParticles] = useState<Array<{ id: number; x: number; color: string; delay: number }>>([]);
   297→  const prevTrigger = useRef(trigger);
   298→
   299→  useEffect(() => {
   300→    if (prevTrigger.current !== trigger) {
   301→      prevTrigger.current = trigger;
   302→      const colors = ['#DC2626', '#F59E0B', '#14B8A6', '#1E293B', '#FBBF24'];
   303→      const newParticles = Array.from({ length: 6 }, (_, i) => ({
   304→        id: Date.now() + i,
   305→        x: 30 + Math.random() * 40,
   306→        color: colors[Math.floor(Math.random() * colors.length)],
   307→        delay: i * 0.05,
   308→      }));
   309→      const t1 = setTimeout(() => setParticles(newParticles), 0);
   310→      const t2 = setTimeout(() => setParticles([]), 900);
   311→      return () => { clearTimeout(t1); clearTimeout(t2); };
   312→    }
   313→  }, [trigger]);
   314→
   315→  return (
   316→    <>
   317→      {particles.map((p) => (
   318→        <div
   319→          key={p.id}
   320→          className="absolute w-1.5 h-1.5 rounded-full confetti-particle pointer-events-none z-20"
   321→          style={{
   322→            left: `${p.x}%`,
   323→            top: '20%',
   324→            backgroundColor: p.color,
   325→            animationDelay: `${p.delay}s`,
   326→          }}
   327→        />
   328→      ))}
   329→    </>
   330→  );
   331→}

   385→function CountdownTimer({ targetDate }: { targetDate: string | null }) {
   386→  const [timeLeft, setTimeLeft] = useState<{ d: number; h: number; m: number; s: number } | null>(null);
   387→  const [expired, setExpired] = useState(!targetDate);
   388→
   389→  useEffect(() => {
   390→    if (!targetDate) {
   391→      return;
   392→    }
   393→
   394→    function calculate() {
   395→      const diff = Math.max(0, new Date(targetDate).getTime() - Date.now());
   396→      if (diff === 0) {
   397→        setExpired(true);
   398→        setTimeLeft({ d: 0, h: 0, m: 0, s: 0 });
   399→        return true;
   400→      }
   401→      setExpired(false);
   402→      setTimeLeft({
   403→        d: Math.floor(diff / 86400000),
   404→        h: Math.floor((diff % 86400000) / 3600000),
   405→        m: Math.floor((diff % 3600000) / 60000),
   406→        s: Math.floor((diff % 60000) / 1000),
   407→      });
   408→      return false;
   409→    }
   410→
   411→    const done = calculate();
   412→    if (done) return;
   413→
   414→    const id = setInterval(() => {
   415→      const finished = calculate();
   416→      if (finished) clearInterval(id);
   417→    }, 1000);
   418→
   419→    return () => clearInterval(id);
   420→  }, [targetDate]);
   421→
   422→  if (expired) {
   423→    return (
   424→      <motion.div
   425→        className="flex items-center justify-center gap-1.5 py-1"
   426→        animate={{ opacity: [1, 0.5, 1] }}
   427→        transition={{ duration: 1.5, repeat: Infinity, ease: 'easeInOut' }}
   428→      >
   429→        <span className="text-xs font-bold text-brand-red dark:text-brand-red-light">Starting Soon!</span>
   430→        <Zap className="w-3 h-3 text-brand-red dark:text-brand-red-light" />
   431→      </motion.div>
   432→    );
   433→  }
   434→
   435→  if (!timeLeft) return null;
   436→
   437→  const units = [
   438→    { value: timeLeft.d, label: 'D' },
   439→    { value: timeLeft.h, label: 'H' },
   440→    { value: timeLeft.m, label: 'M' },
   441→    { value: timeLeft.s, label: 'S' },
   442→  ];
   443→
   444→  return (
   445→    <div className="flex items-center justify-center gap-1.5">
   446→      {units.map((unit, idx) => (
   447→        <div key={unit.label} className="flex items-center gap-1.5">
   448→          <div className="flex flex-col items-center">
   449→            <motion.div
   450→              className="w-9 h-9 rounded-md bg-warm-800/80 dark:bg-warm-100/90 flex items-center justify-center shadow-sm"
   451→              key={`${unit.label}-${unit.value}`}
   452→              initial={{ rotateX: -90, opacity: 0 }}
   453→              animate={{ rotateX: 0, opacity: 1 }}
   454→              transition={{ duration: 0.3, ease: 'easeOut' }}
   455→            >
   456→              <span className="text-sm font-black text-white dark:text-warm-900 tabular-nums">
   457→                {String(unit.value).padStart(2, '0')}
   458→              </span>
   459→            </motion.div>
   460→            <span className="text-[8px] font-semibold text-warm-500 dark:text-warm-400 mt-0.5">
   461→              {unit.label}
   462→            </span>
   463→          </div>
   464→          {idx < units.length - 1 && (
   465→            <span className="text-warm-400 dark:text-warm-500 font-bold text-xs mb-3">:</span>
   466→          )}
   467→        </div>
   468→      ))}
   469→    </div>
   470→  );
   471→}
