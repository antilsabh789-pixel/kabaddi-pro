    78→function getPositionCategory(position: string | null): 'raider' | 'defender' | 'all-rounder' | 'unknown' {
    79→  if (!position) return 'unknown';
    80→  const p = position.toLowerCase();
    81→  if (p.includes('raider')) return 'raider';
    82→  if (p.includes('defend') || p.includes('corner') || p.includes('cover')) return 'defender';
    83→  if (p.includes('all')) return 'all-rounder';
    84→  return 'unknown';
    85→}

    89→function LineupValidation({ lineup, playersPerSide, teamName }: {
    90→  lineup: MatchPlayer[];
    91→  playersPerSide: number;
    92→  teamName: string;
    93→}) {
    94→  const warnings: { message: string; severity: 'error' | 'warning' | 'info' }[] = [];
    95→
    96→  if (lineup.length === 0) {
    97→    warnings.push({ message: 'No players added', severity: 'error' });
    98→  } else if (lineup.length < playersPerSide) {
    99→    warnings.push({
   100→      message: `${lineup.length}/${playersPerSide} players — need ${playersPerSide - lineup.length} more`,
   101→      severity: 'warning',
   102→    });
   103→  } else if (lineup.length > playersPerSide) {
   104→    warnings.push({
   105→      message: `${lineup.length} players — ${lineup.length - playersPerSide} extra`,
   106→      severity: 'info',
   107→    });
   108→  }
   109→
   110→  // Position balance check (only if we have position data from DB)
   111→  const balance = POSITION_BALANCE[playersPerSide];
   112→  // Note: position data might not be available for quick-added players
   113→  // We check this based on the allPlayers data which is in parent scope
   114→
   115→  if (warnings.length === 0) return null;
   116→
   117→  return (
   118→    <motion.div
   119→      initial={{ opacity: 0, y: -5 }}
   120→      animate={{ opacity: 1, y: 0 }}
   121→      className="space-y-1.5"
   122→    >
   123→      {warnings.map((w, i) => (
   124→        <div
   125→          key={i}
   126→          className={`flex items-center gap-2 px-3 py-2 rounded-lg text-xs font-medium ${
   127→            w.severity === 'error'
   128→              ? 'bg-brand-red/10 text-brand-red dark:bg-brand-red/15'
   129→              : w.severity === 'warning'
   130→                ? 'bg-amber-500/10 text-amber-600 dark:text-amber-400 dark:bg-amber-500/15'
   131→                : 'bg-blue-500/10 text-blue-600 dark:text-blue-400 dark:bg-blue-500/15'
   132→          }`}
   133→        >
   134→          <AlertTriangle className={`w-3.5 h-3.5 shrink-0 ${w.severity === 'info' ? 'hidden' : ''}`} />
   135→          <Info className={`w-3.5 h-3.5 shrink-0 ${w.severity !== 'info' ? 'hidden' : ''}`} />
   136→          <span>{teamName}: {w.message}</span>
   137→        </div>
   138→      ))}
   139→    </motion.div>
   140→  );
   141→}

   145→function PlayerStatsTooltip({ player, allPlayers }: { player: DbPlayer; allPlayers: DbPlayer[] }) {
   146→  const dbPlayer = allPlayers.find(p => p.id === player.id);
   147→  const profile = dbPlayer?.profile;
   148→
   149→  if (!profile || !profile.totalPoints) return null;
   150→
   151→  const raidSuccessRate = profile.totalRaids && profile.totalRaids > 0
   152→    ? Math.round(((profile.successfulRaids || 0) / profile.totalRaids) * 100)
   153→    : 0;
   154→
   155→  return (
   156→    <div className="absolute z-30 bottom-full left-0 mb-1 p-2 bg-white dark:bg-warm-800 border border-warm-200 dark:border-warm-600 rounded-lg shadow-lg text-[10px] min-w-[140px]">
   157→      <div className="space-y-1">
   158→        <div className="flex justify-between">
   159→          <span className="text-warm-500 dark:text-warm-400">Total Pts</span>
   160→          <span className="font-bold text-brand-navy dark:text-brand-navy-light">{profile.totalPoints}</span>
   161→        </div>
   162→        <div className="flex justify-between">
   163→          <span className="text-warm-500 dark:text-warm-400">Raid Pts</span>
   164→          <span className="font-bold text-brand-red">{profile.raidPoints || 0}</span>
   165→        </div>
   166→        <div className="flex justify-between">
   167→          <span className="text-warm-500 dark:text-warm-400">Tackle Pts</span>
   168→          <span className="font-bold text-brand-teal">{profile.tacklePoints || 0}</span>
   169→        </div>
   170→        <div className="flex justify-between">
   171→          <span className="text-warm-500 dark:text-warm-400">Raid Success</span>
   172→          <span className="font-bold text-brand-gold">{raidSuccessRate}%</span>
   173→        </div>
   174→      </div>
   175→      {/* Tooltip arrow */}
   176→      <div className="absolute -bottom-1 left-4 w-2 h-2 bg-white dark:bg-warm-800 border-r border-b border-warm-200 dark:border-warm-600 rotate-45" />
   177→    </div>
   178→  );
   179→}

   184→  lineup: MatchPlayer[];

   186→  teamName: string;

   199→        >
   200→          {teamName.charAt(0).toUpperCase()}
   201→        </div>
   202→        <p className="text-[10px] font-bold text-warm-600 dark:text-warm-300 mt-1 truncate max-w-[80px] mx-auto">
   203→          {teamName}
   204→        </p>
   205→      </div>
   206→
   207→      {/* Formation layout */}

   209→        {/* Court half line */}
   210→        <div className={`absolute top-1/2 left-0 right-0 h-px bg-warm-200 dark:bg-warm-600 ${
   211→          side === 'left' ? 'bg-gradient-to-r from-transparent to-warm-200 dark:to-warm-600' : 'bg-gradient-to-l from-transparent to-warm-200 dark:to-warm-600'
   212→        }`} />
   213→
   214→        {/* Players in formation */}
   215→        <div className="grid grid-cols-3 gap-1 place-items-center">
   216→          {displayPlayers.map((player, i) => {
   217→            // Simple formation: top row = defenders, middle = all-rounders, bottom = raiders
   218→            return (
   219→              <motion.div
   220→                key={player.id}
   221→                initial={{ scale: 0, opacity: 0 }}
   222→                animate={{ scale: 1, opacity: 1 }}
   223→                transition={{ delay: i * 0.05, type: 'spring', stiffness: 300, damping: 20 }}
   224→                className="flex flex-col items-center"
   225→              >
   226→                <div
   227→                  className="w-7 h-7 rounded-full flex items-center justify-center text-white text-[8px] font-bold shadow-sm"
   228→                  style={{ backgroundColor: teamColor }}
   229→                >
   230→                  {player.jerseyNumber || i + 1}
   231→                </div>
   232→                <span className="text-[7px] text-warm-400 dark:text-warm-500 mt-0.5 truncate max-w-[32px]">
   233→                  {player.name.split(' ')[0]}
   234→                </span>
   235→              </motion.div>
   236→            );
   237→          })}
   238→          {/* Empty slots */}
   239→          {Array.from({ length: Math.max(0, 7 - displayPlayers.length) }).map((_, i) => (
   240→            <div key={`empty-${i}`} className="flex flex-col items-center">
   241→              <div className="w-7 h-7 rounded-full border-2 border-dashed border-warm-200 dark:border-warm-600 flex items-center justify-center">
   242→                <span className="text-[8px] text-warm-300 dark:text-warm-600">?</span>
   243→              </div>
   244→              <span className="text-[7px] text-warm-300 dark:text-warm-600 mt-0.5">Empty</span>
   245→            </div>
   246→          ))}
   247→        </div>
   248→      </div>
   249→    </div>
   250→  );
   251→}
