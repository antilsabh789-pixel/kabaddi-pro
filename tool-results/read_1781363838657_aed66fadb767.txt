   230→/** Helper: determine which side a teamId belongs to in a match */
   231→function getTeamSide(match: ActiveMatch, teamId: string): 'home' | 'away' {
   232→  if (teamId === match.homeTeamId) return 'home';
   233→  return 'away';
   234→}

   236→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   237→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   238→  let homeScore = 0;
   239→  let awayScore = 0;
   240→  let homeOutPlayers = 0;
   241→  let awayOutPlayers = 0;
   242→  let homeOutPlayerIds: string[] = [];
   243→  let awayOutPlayerIds: string[] = [];
   244→
   245→  for (const evt of events) {
   246→    const side = getTeamSide(match, evt.teamId);
   247→    const points = evt.value;
   248→    if (side === 'home') {
   249→      homeScore += points;
   250→    } else {
   251→      awayScore += points;
   252→    }
   253→
   254→    // Count-based out tracking (backward compat)
   255→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   256→      if (side === 'home') {
   257→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   258→      } else {
   259→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   260→      }
   261→    }
   262→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   263→      if (side === 'home') {
   264→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   265→      } else {
   266→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   267→      }
   268→    }
   269→    if (evt.eventType === 'all_out') {
   270→      if (side === 'home') {
   271→        awayOutPlayers = 0;
   272→      } else {
   273→        homeOutPlayers = 0;
   274→      }
   275→    }
   276→
   277→    // ID-based out tracking
   278→    let details: Record<string, unknown> = {};
   279→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   280→
   281→    if (evt.eventType === 'raid_point') {
   282→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   283→      if (side === 'home') {
   284→        // Home team scored raid → away defenders go out
   285→        for (const tid of touchedIds) {
   286→          if (!awayOutPlayerIds.includes(tid)) {
   287→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   288→          }
   289→        }
   290→        // Home team revives players (first-out-first-in)
   291→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   292→        if (reviveCount > 0) {
   293→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   294→        }
   295→      } else {
   296→        // Away team scored raid → home defenders go out
   297→        for (const tid of touchedIds) {
   298→          if (!homeOutPlayerIds.includes(tid)) {
   299→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   300→          }
   301→        }
   302→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   303→        if (reviveCount > 0) {
   304→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   305→        }
   306→      }
   307→    }
   308→
   309→    if (evt.eventType === 'bonus_point') {
   310→      // One out player from scoring team revives
   311→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   312→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   313→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   314→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   315→      }
   316→    }
   317→
   318→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   319→      // Raider goes out. The tackle is scored by the DEFENDING team.
   320→      // So the raider is from the OPPOSING team.
   321→      const raiderId = (details.raiderId as string) || evt.playerId;
   322→      if (side === 'home' && raiderId) {
   323→        // Home team scored tackle → away raider was caught → away raider goes out
   324→        if (!awayOutPlayerIds.includes(raiderId)) {
   325→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   326→        }
   327→      } else if (side === 'away' && raiderId) {
   328→        // Away team scored tackle → home raider was caught → home raider goes out
   329→        if (!homeOutPlayerIds.includes(raiderId)) {
   330→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   331→        }
   332→      }
   333→    }
   334→
   335→    if (evt.eventType === 'all_out') {
   336→      if (side === 'home') {
   337→        awayOutPlayerIds = [];
   338→      } else {
   339→        homeOutPlayerIds = [];
   340→      }
   341→    }
   342→  }
   343→
   344→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   345→}

   347→/** Helper: determine raid queue from last raid event */
   348→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   349→  const raidEventTypes = ['raid_point', 'tackle_point', 'super_tackle', 'empty_raid', 'bonus_point', 'do_or_die_raid'];
   350→  const lastRaidEvent = [...events].reverse().find(e => raidEventTypes.includes(e.eventType));
   351→  if (lastRaidEvent) {
   352→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   353→  }
   354→  return match.raidQueue;
   355→}

   556→      undoLastEvent: () =>
   557→        set((state) => {
   558→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   559→          const events = [...state.activeMatch.events];
   560→          events.pop();
   561→
   562→          const calculated = recalculateFromEvents(state.activeMatch, events);
   563→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   564→
   565→          return {
   566→            activeMatch: {
   567→              ...state.activeMatch,
   568→              events,
   569→              ...calculated,
   570→              raidQueue,
   571→            },
   572→          };
   573→        }),
   574→
   575→      undoLastRaid: () =>
   576→        set((state) => {
   577→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   578→
   579→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   580→          // A raid batch has events with the same second in timestamp
   581→          const events = [...state.activeMatch.events];
   582→          if (events.length === 0) return {};
   583→
   584→          // Find the timestamp of the last event
   585→          const lastTimestamp = events[events.length - 1].timestamp;
   586→          // Remove all events that have the same timestamp (within 5ms for batch events)
   587→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   588→            events.pop();
   589→          }
   590→
   591→          const calculated = recalculateFromEvents(state.activeMatch, events);
   592→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   593→
   594→          return {
   595→            activeMatch: {
   596→              ...state.activeMatch,
   597→              events,
   598→              ...calculated,
   599→              raidQueue,
   600→            },
   601→          };
   602→        }),
   603→

   608→          return {
   609→            activeMatch: {
   610→              ...state.activeMatch,
   611→              currentHalf: newHalf,
   612→              timer: state.activeMatch.halfDuration * 60,
   613→              isDoOrDie: false,
   614→            },
   615→          };
   616→        }),
   617→
   618→      setTimer: (time) =>
   619→        set((state) => {
   620→          if (!state.activeMatch) return {};
   621→          return { activeMatch: { ...state.activeMatch, timer: time } };
   622→        }),
   623→
   624→      setDoOrDie: (value) =>

   626→          if (!state.activeMatch) return {};
   627→          return { activeMatch: { ...state.activeMatch, isDoOrDie: value } };

   630→      switchRaidQueue: () =>

   635→              ...state.activeMatch,
   636→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',

   639→        }),
   640→
   641→      callTimeout: (team) =>

   644→          return {

   647→              homeTimeouts:
   648→                team === 'home'

   650→                  : state.activeMatch.homeTimeouts,
   651→              awayTimeouts:
   652→                team === 'away'
   653→                  ? state.activeMatch.awayTimeouts + 1
   654→                  : state.activeMatch.awayTimeouts,
   655→            },
   656→          };

   659→      setOutPlayers: (team, count) =>

   666→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,

   668→          };

   670→
   671→      fetchHomeData: async () => {

   681→        } catch (err) {

   683→          return null;
   684→        }
   685→      },
   686→
   687→      // Notification actions
   688→      addNotification: (notification) =>

   691→            {
   692→              ...notification,
   693→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   694→              timestamp: Date.now(),
   695→              read: false,
   696→            },
   697→            ...state.notifications,
   698→          ].slice(0, 50), // Keep max 50 notifications
   699→        })),
   700→
   701→      markNotificationRead: (id) =>

   703→          notifications: state.notifications.map((n) =>

   732→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   733→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   734→                createdAt: state.currentUser.createdAt || Date.now(),
   735→              }
   736→            : null,
   737→        })),
   738→
   739→      // Team management actions
   740→      setTeamFilter: (filter) =>
