   231→   231→function AnimatedCounter({ target, duration = 1200 }: { target: number; duration?: number }) {
   232→   232→  const [count, setCount] = useState(0);
   233→   233→  const prevTarget = useRef(target);
   234→   234→
   235→   235→  useEffect(() => {
   236→   236→    if (prevTarget.current === target && count !== 0) return;
   237→   237→    prevTarget.current = target;
   238→   238→
   239→   239→    let start = 0;
   240→   240→    const startTime = performance.now();
   241→   241→
   242→   242→    function step(currentTime: number) {
   243→   243→      const elapsed = currentTime - startTime;
   244→   244→      const progress = Math.min(elapsed / duration, 1);
   245→   245→      // ease-out cubic
   246→   246→      const eased = 1 - Math.pow(1 - progress, 3);
   247→   247→      const current = Math.round(eased * target);
   248→   248→      setCount(current);
   249→   249→      if (progress < 1) {
   250→   250→        requestAnimationFrame(step);
   251→   251→      }
   252→   252→    }
   253→   253→
   254→   254→    requestAnimationFrame(step);
   255→   255→  }, [target, duration]);
   256→   256→
   257→   257→  return <span>{count}</span>;
   258→   258→}
