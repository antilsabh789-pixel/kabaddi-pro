   233→/** Helper: determine which side a teamId belongs to in a match */
   234→function getTeamSide(match: ActiveMatch, teamId: string): 'home' | 'away' {
   235→  if (teamId === match.homeTeamId) return 'home';

   239→/** Helper: recalculate scores, out counts, and out player IDs from a list of events */
   240→function recalculateFromEvents(match: ActiveMatch, events: MatchEvent[]) {
   241→  let homeScore = 0;
   242→  let awayScore = 0;
   243→  let homeOutPlayers = 0;
   244→  let awayOutPlayers = 0;
   245→  let homeOutPlayerIds: string[] = [];
   246→  let awayOutPlayerIds: string[] = [];
   247→
   248→  for (const evt of events) {
   249→    const side = getTeamSide(match, evt.teamId);
   250→    const points = evt.value;
   251→    if (side === 'home') {
   252→      homeScore += points;
   253→    } else {
   254→      awayScore += points;
   255→    }
   256→
   257→    // Count-based out tracking (backward compat)
   258→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   259→      if (side === 'home') {
   260→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   261→      } else {
   262→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   263→      }
   264→    }
   265→    if (evt.eventType === 'raid_point' || evt.eventType === 'bonus_point') {
   266→      if (side === 'home') {
   267→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   268→      } else {
   269→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   270→      }
   271→    }
   272→    // Self-out: defender steps off mat → +1 to raiding team, one of their out players revives
   273→    if (evt.eventType === 'self_out') {
   274→      // The self-out player is from the DEFENDING team (opposite of scoring team)
   275→      if (side === 'home') {
   276→        // Home scored from self-out → away defender went out
   277→        awayOutPlayers = Math.min(match.playersPerSide, awayOutPlayers + 1);
   278→        // Home revives one player
   279→        homeOutPlayers = Math.max(0, homeOutPlayers - 1);
   280→      } else {
   281→        homeOutPlayers = Math.min(match.playersPerSide, homeOutPlayers + 1);
   282→        awayOutPlayers = Math.max(0, awayOutPlayers - 1);
   283→      }
   284→    }
   285→    if (evt.eventType === 'all_out') {
   286→      if (side === 'home') {
   287→        awayOutPlayers = 0;
   288→      } else {
   289→        homeOutPlayers = 0;
   290→      }
   291→    }
   292→
   293→    // ID-based out tracking
   294→    let details: Record<string, unknown> = {};
   295→    try { details = evt.details ? JSON.parse(evt.details) : {}; } catch { /* skip */ }
   296→
   297→    if (evt.eventType === 'raid_point') {
   298→      const touchedIds: string[] = (details.touchedPlayerIds as string[]) || [];
   299→      if (side === 'home') {
   300→        // Home team scored raid → away defenders go out
   301→        for (const tid of touchedIds) {
   302→          if (!awayOutPlayerIds.includes(tid)) {
   303→            awayOutPlayerIds = [...awayOutPlayerIds, tid];
   304→          }
   305→        }
   306→        // Home team revives players (first-out-first-in)
   307→        const reviveCount = Math.min(evt.value, homeOutPlayerIds.length);
   308→        if (reviveCount > 0) {
   309→          homeOutPlayerIds = homeOutPlayerIds.slice(reviveCount);
   310→        }
   311→      } else {
   312→        // Away team scored raid → home defenders go out
   313→        for (const tid of touchedIds) {
   314→          if (!homeOutPlayerIds.includes(tid)) {
   315→            homeOutPlayerIds = [...homeOutPlayerIds, tid];
   316→          }
   317→        }
   318→        const reviveCount = Math.min(evt.value, awayOutPlayerIds.length);
   319→        if (reviveCount > 0) {
   320→          awayOutPlayerIds = awayOutPlayerIds.slice(reviveCount);
   321→        }
   322→      }
   323→    }
   324→
   325→    if (evt.eventType === 'bonus_point') {
   326→      // One out player from scoring team revives
   327→      if (side === 'home' && homeOutPlayerIds.length > 0) {
   328→        homeOutPlayerIds = homeOutPlayerIds.slice(1);
   329→      } else if (side === 'away' && awayOutPlayerIds.length > 0) {
   330→        awayOutPlayerIds = awayOutPlayerIds.slice(1);
   331→      }
   332→    }
   333→
   334→    // Self-out: defender steps off mat during a live raid
   335→    if (evt.eventType === 'self_out') {
   336→      const selfOutPlayerId = (details.selfOutPlayerId as string) || evt.playerId;
   337→      if (selfOutPlayerId) {
   338→        // The self-out player is from the DEFENDING team (opposite of scoring team)
   339→        if (side === 'home') {
   340→          // Home scored → away defender self-out
   341→          if (!awayOutPlayerIds.includes(selfOutPlayerId)) {
   342→            awayOutPlayerIds = [...awayOutPlayerIds, selfOutPlayerId];
   343→          }
   344→          // Home revives first-out player
   345→          if (homeOutPlayerIds.length > 0) {
   346→            homeOutPlayerIds = homeOutPlayerIds.slice(1);
   347→          }
   348→        } else {
   349→          // Away scored → home defender self-out
   350→          if (!homeOutPlayerIds.includes(selfOutPlayerId)) {
   351→            homeOutPlayerIds = [...homeOutPlayerIds, selfOutPlayerId];
   352→          }
   353→          if (awayOutPlayerIds.length > 0) {
   354→            awayOutPlayerIds = awayOutPlayerIds.slice(1);
   355→          }
   356→        }
   357→      }
   358→    }
   359→
   360→    if (evt.eventType === 'tackle_point' || evt.eventType === 'super_tackle') {
   361→      // Raider goes out. The tackle is scored by the DEFENDING team.
   362→      // So the raider is from the OPPOSING team.
   363→      const raiderId = (details.raiderId as string) || evt.playerId;
   364→      if (side === 'home' && raiderId) {
   365→        // Home team scored tackle → away raider was caught → away raider goes out
   366→        if (!awayOutPlayerIds.includes(raiderId)) {
   367→          awayOutPlayerIds = [...awayOutPlayerIds, raiderId];
   368→        }
   369→      } else if (side === 'away' && raiderId) {
   370→        // Away team scored tackle → home raider was caught → home raider goes out
   371→        if (!homeOutPlayerIds.includes(raiderId)) {
   372→          homeOutPlayerIds = [...homeOutPlayerIds, raiderId];
   373→        }
   374→      }
   375→    }
   376→
   377→    if (evt.eventType === 'all_out') {
   378→      if (side === 'home') {
   379→        awayOutPlayerIds = [];
   380→      } else {
   381→        homeOutPlayerIds = [];
   382→      }
   383→    }
   384→  }
   385→
   386→  return { homeScore, awayScore, homeOutPlayers, awayOutPlayers, homeOutPlayerIds, awayOutPlayerIds };
   387→}

   389→/** Event types that constitute a complete raid and should flip the turn */
   390→const RAID_ENDING_EVENT_TYPES: string[] = [
   391→  'raid_point', 'tackle_point', 'super_tackle', 'empty_raid',
   392→  'bonus_point', 'do_or_die_raid', 'super_raid', 'self_out',
   393→];
   394→
   395→/** Event types that do NOT flip the turn (non-raid events) */
   396→const NON_RAID_EVENT_TYPES: string[] = [
   397→  'all_out', 'substitution', 'timeout', 'yellow_card', 'red_card', 'green_card',
   398→];
   399→
   400→/** Helper: determine raid queue from last raid-ending event */
   401→function getRaidQueueFromEvents(match: ActiveMatch, events: MatchEvent[]): 'home' | 'away' {
   402→  const lastRaidEvent = [...events].reverse().find(e => RAID_ENDING_EVENT_TYPES.includes(e.eventType));
   403→  if (lastRaidEvent) {
   404→    return getTeamSide(match, lastRaidEvent.teamId) === 'home' ? 'away' : 'home';
   405→  }
   406→  return match.raidQueue;
   407→}

   596→          const newEvents: MatchEvent[] = events.map((e, i) => ({
   597→            ...e,
   598→            id: `evt_${Date.now()}_${i}_${Math.random().toString(36).slice(2, 7)}`,
   599→            timestamp: Date.now() + i,
   600→          }));
   601→          const updatedEvents = [...state.activeMatch.events, ...newEvents];
   602→
   603→          const calculated = recalculateFromEvents(state.activeMatch, updatedEvents);
   604→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, updatedEvents);
   605→
   606→          return {
   607→            activeMatch: {
   608→              ...state.activeMatch,
   609→              events: updatedEvents,
   610→              ...calculated,
   611→              raidQueue,
   612→            },
   613→          };
   614→        }),
   615→
   616→      undoLastEvent: () =>
   617→        set((state) => {
   618→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   619→          const events = [...state.activeMatch.events];
   620→          events.pop();
   621→
   622→          const calculated = recalculateFromEvents(state.activeMatch, events);
   623→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   624→
   625→          return {
   626→            activeMatch: {
   627→              ...state.activeMatch,
   628→              events,
   629→              ...calculated,
   630→              raidQueue,
   631→            },
   632→          };
   633→        }),
   634→
   635→      undoLastRaid: () =>
   636→        set((state) => {
   637→          if (!state.activeMatch || state.activeMatch.events.length === 0) return {};
   638→
   639→          // Remove events until we've removed a complete "raid" (all events with same timestamp prefix)
   640→          // A raid batch has events with the same second in timestamp
   641→          const events = [...state.activeMatch.events];
   642→          if (events.length === 0) return {};
   643→
   644→          // Find the timestamp of the last event
   645→          const lastTimestamp = events[events.length - 1].timestamp;
   646→          // Remove all events that have the same timestamp (within 5ms for batch events)
   647→          while (events.length > 0 && Math.abs(events[events.length - 1].timestamp - lastTimestamp) < 10) {
   648→            events.pop();
   649→          }
   650→
   651→          const calculated = recalculateFromEvents(state.activeMatch, events);
   652→          const raidQueue = getRaidQueueFromEvents(state.activeMatch, events);
   653→
   654→          return {
   655→            activeMatch: {
   656→              ...state.activeMatch,
   657→              events,
   658→              ...calculated,
   659→              raidQueue,
   660→            },
   661→          };
   662→        }),
   663→
   664→      switchHalf: () =>
   665→        set((state) => {
   666→          if (!state.activeMatch) return {};
   667→          const newHalf = state.activeMatch.currentHalf === 1 ? 2 : 1;
   668→          return {
   669→            activeMatch: {
   670→              ...state.activeMatch,
   671→              currentHalf: newHalf,
   672→              timer: state.activeMatch.halfDuration * 60,
   673→              isDoOrDie: false,
   674→              doOrDieTeamId: null,
   675→            },
   676→          };
   677→        }),
   678→
   679→      setTimer: (time) =>
   680→        set((state) => {
   681→          if (!state.activeMatch) return {};
   682→          return { activeMatch: { ...state.activeMatch, timer: time } };
   683→        }),
   684→
   685→      setDoOrDie: (value, teamId?) =>
   686→        set((state) => {
   687→          if (!state.activeMatch) return {};
   688→          return {
   689→            activeMatch: {
   690→              ...state.activeMatch,
   691→              isDoOrDie: value,
   692→              doOrDieTeamId: value ? (teamId || null) : null,
   693→            },
   694→          };
   695→        }),
   696→
   697→      switchRaidQueue: () =>
   698→        set((state) => {
   699→          if (!state.activeMatch) return {};
   700→          return {
   701→            activeMatch: {
   702→              ...state.activeMatch,
   703→              raidQueue: state.activeMatch.raidQueue === 'home' ? 'away' : 'home',
   704→            },
   705→          };
   706→        }),
   707→
   708→      callTimeout: (team) =>
   709→        set((state) => {
   710→          if (!state.activeMatch) return {};
   711→          return {
   712→            activeMatch: {
   713→              ...state.activeMatch,
   714→              homeTimeouts:
   715→                team === 'home'
   716→                  ? state.activeMatch.homeTimeouts + 1
   717→                  : state.activeMatch.homeTimeouts,
   718→              awayTimeouts:
   719→                team === 'away'
   720→                  ? state.activeMatch.awayTimeouts + 1
   721→                  : state.activeMatch.awayTimeouts,
   722→            },
   723→          };
   724→        }),
   725→
   726→      setOutPlayers: (team, count) =>
   727→        set((state) => {
   728→          if (!state.activeMatch) return {};
   729→          return {
   730→            activeMatch: {
   731→              ...state.activeMatch,
   732→              homeOutPlayers: team === 'home' ? count : state.activeMatch.homeOutPlayers,
   733→              awayOutPlayers: team === 'away' ? count : state.activeMatch.awayOutPlayers,
   734→            },
   735→          };
   736→        }),
   737→
   738→      fetchHomeData: async () => {
   739→        try {
   740→          const res = await fetch('/api/stats');
   741→          if (!res.ok) throw new Error('Failed to fetch stats');
   742→
   743→          const data = await res.json();
   744→
   745→          // Cache in store
   746→          set({ homeData: data });
   747→          return data;
   748→        } catch (err) {
   749→          console.error('fetchHomeData error:', err);
   750→          return null;
   751→        }
   752→      },
   753→
   754→      // Notification actions
   755→      addNotification: (notification) =>
   756→        set((state) => ({
   757→          notifications: [
   758→            {
   759→              ...notification,
   760→              id: `notif_${Date.now()}_${Math.random().toString(36).slice(2, 7)}`,
   761→              timestamp: Date.now(),
   762→              read: false,
   763→            },
   764→            ...state.notifications,
   765→          ].slice(0, 50), // Keep max 50 notifications
   766→        })),
   767→
   768→      markNotificationRead: (id) =>
   769→        set((state) => ({
   770→          notifications: state.notifications.map((n) =>
   771→            n.id === id ? { ...n, read: true } : n
   772→          ),
   773→        })),
   774→
   775→      markAllRead: () =>
   776→        set((state) => ({
   777→          notifications: state.notifications.map((n) => ({ ...n, read: true })),
   778→        })),
   779→
   780→      clearNotifications: () => set({ notifications: [] }),
   781→
   782→      setLanguage: (lang) => set({ language: lang }),
   783→
   784→      // Onboarding actions
   785→      setHasCompletedOnboarding: (value) => set({ hasCompletedOnboarding: value }),
   786→
   787→      setOnboardingProfile: (profile) =>
   788→        set((state) => ({
   789→          onboardingProfile: { ...state.onboardingProfile, ...profile },
   790→        })),
   791→
   792→      completeOnboarding: () =>
   793→        set((state) => ({
   794→          hasCompletedOnboarding: true,
   795→          currentUser: state.currentUser
   796→            ? {
   797→                ...state.currentUser,
   798→                ...(state.onboardingProfile.position && { role: state.onboardingProfile.position }),
   799→                ...(state.onboardingProfile.weightCategory && { weight: state.onboardingProfile.weightCategory }),
   800→                ...(state.onboardingProfile.experience && { experienceLevel: state.onboardingProfile.experience }),
   801→                createdAt: state.currentUser.createdAt || Date.now(),
   802→              }
   803→            : null,
   804→        })),
   805→
   806→      // Team management actions
   807→      setTeamFilter: (filter) =>
   808→        set((state) => ({
   809→          teamManagement: { ...state.teamManagement, teamFilter: filter },
   810→        })),
   811→
   812→      setTeamSearch: (search) =>
   813→        set((state) => ({
   814→          teamManagement: { ...state.teamManagement, teamSearch: search },
   815→        })),
   816→
   817→      setSelectedTeamId: (teamId) =>
   818→        set((state) => ({
   819→          teamManagement: { ...state.teamManagement, selectedTeamId: teamId },
   820→        })),
   821→
   822→      setTeamDetailOpen: (open) =>
   823→        set((state) => ({
   824→          teamManagement: { ...state.teamManagement, teamDetailOpen: open },
   825→        })),
   826→
   827→      // Match transfer / handoff — load a match from a transfer
   828→      loadMatch: (match) =>
   829→        set({
   830→          activeMatch: match,
   831→          activeTab: 'quick-score' as TabId,
   832→          showToss: false,
   833→          tossMatchConfig: null,
   834→        }),
   835→    }),
   836→    {
   837→      name: 'kabaddi-pro-storage',
   838→      storage: createJSONStorage(() => localStorage),
   839→      // Only persist auth-related fields and notifications
   840→      partialize: (state) => ({
   841→        isAuthenticated: state.isAuthenticated,
   842→        isOnboarded: state.isOnboarded,
   843→        currentUser: state.currentUser,
   844→        hasSeenSplash: state.hasSeenSplash,
   845→        notifications: state.notifications,
   846→        language: state.language,
   847→        hasCompletedOnboarding: state.hasCompletedOnboarding,
   848→        onboardingProfile: state.onboardingProfile,
   849→        coachAcademies: state.coachAcademies,
   850→      }),
   851→    }
   852→  )
   853→);

   855→export function getEventPoints(eventType: EventType): number {
   856→  switch (eventType) {
   857→    case 'raid_point':
   858→      return 1;
   859→    case 'bonus_point':
   860→      return 1;
   861→    case 'tackle_point':
   862→      return 1;
   863→    case 'super_raid':
   864→      return 1;
   865→    case 'super_tackle':
   866→      return 1;
   867→    case 'do_or_die_raid':
   868→      return 1;
   869→    case 'all_out':
   870→      return 2;
   871→    case 'self_out':
   872→      return 1;
   873→    case 'empty_raid':
   874→      return 0;
   875→    case 'substitution':
   876→    case 'timeout':
   877→    case 'yellow_card':
   878→    case 'red_card':
   879→    case 'green_card':
   880→      return 0;
   881→    default:
   882→      return 0;
   883→  }
   884→}
